1//! composer.json parser with position tracking.
2//!
3//! Parses composer.json files and extracts dependency information with precise
4//! source positions for LSP operations. Platform packages (php, ext-*, lib-*)
5//! are filtered out as they are not Packagist packages.
6
7use crate::types::{ComposerDependency, ComposerSection};
8use deps_core::Result;
9use deps_core::json_ast::{JsonAst, JsonSection};
10use deps_core::json_helpers::string_valued_entries;
11use deps_core::lsp_helpers::LineOffsetTable;
12use serde_json::Value;
13use std::any::Any;
14use tower_lsp_server::ls_types::Uri;
15
16/// Result of parsing a composer.json file.
17///
18/// Contains all non-platform dependencies found in the file with their positions.
19#[derive(Debug)]
20pub struct ComposerParseResult {
21    pub dependencies: Vec<ComposerDependency>,
22    pub uri: Uri,
23    /// Raw value of the manifest's own top-level `minimum-stability` field (e.g. `"beta"`),
24    /// if present — Composer's project-wide default stability floor, one of `dev`, `alpha`,
25    /// `beta`, `RC`, `stable` (case-insensitive). `None` when the field is absent, which
26    /// Composer itself treats as `stable` (#424).
27    ///
28    /// Consumers rank this word (`registry.rs`'s crate-private stability ranking) rather than
29    /// comparing it as a string — stored raw here rather than pre-ranked so a caller with no
30    /// interest in stability filtering (e.g. a future manifest-summary feature) is not forced
31    /// to depend on that ranking.
32    pub minimum_stability: Option<String>,
33}
34
35impl deps_core::ParseResult for ComposerParseResult {
36    fn dependencies(&self) -> Vec<&dyn deps_core::Dependency> {
37        self.dependencies
38            .iter()
39            .map(|d| d as &dyn deps_core::Dependency)
40            .collect()
41    }
42
43    fn workspace_root(&self) -> Option<&std::path::Path> {
44        None
45    }
46
47    fn uri(&self) -> &Uri {
48        &self.uri
49    }
50
51    fn as_any(&self) -> &dyn Any {
52        self
53    }
54}

56/// Returns true if the package is a platform requirement (not a Packagist package).
57///
58/// Platform packages include:
59/// - `php`, and its variants `php-64bit`, `php-ipv6`, `php-zts`, `php-debug` — PHP version
60///   requirements
61/// - `hhvm` — HHVM version requirement
62/// - `composer` — the Composer CLI version itself
63/// - `ext-*` — PHP extensions
64/// - `lib-*` — PHP libraries
65/// - `composer-plugin-api`, `composer-runtime-api` — Composer's own virtual packages
66///
67/// Mirrors Composer's `PlatformRepository` package set (#402 critique S1: an incomplete list
68/// here previously let a real platform requirement like `composer-plugin-api` — present in
69/// essentially every Composer plugin's `composer.json` — reach the Packagist-shaped
70/// `vendor/package` name validator and get flagged "Invalid package name"; M6: `composer`
71/// itself was still missing from the set).
72///
73/// Every platform package name is a single bare token with no `/` — a real Packagist package
74/// always has a `vendor/package` shape — so any name containing `/` returns `false`
75/// immediately, before the prefix checks below. Without this guard, `starts_with("php-")` (and
76/// `ext-`/`lib-`) would also match a real package merely because its *vendor* happens to start
77/// with that prefix (e.g. `php-di/php-di`, `php-amqplib/php-amqplib`, `ext-mongo/whatever`),
78/// silently dropping it from the dependency list entirely — no diagnostic, hover, inlay hint,
79/// or code lens — rather than validating it as a normal dependency (#402 critique C2).
80pub fn is_platform_package(name: &str) -> bool {
81    if name.contains('/') {
82        return false;
83    }
84    name == "php"
85        || name.starts_with("php-")
86        || name == "hhvm"
87        || name == "composer"
88        || name.starts_with("ext-")
89        || name.starts_with("lib-")
90        || name == "composer-plugin-api"
91        || name == "composer-runtime-api"
92}

94/// Parses a composer.json file and extracts all non-platform dependencies with positions.
95///
96/// Handles `require` and `require-dev` sections.
97/// Platform packages (php, ext-*, lib-*) are silently filtered out.
98///
99/// # Errors
100///
101/// Returns an error if JSON parsing fails.
102///
103/// # Examples
104///
105/// ```no_run
106/// use deps_composer::parser::parse_composer_json;
107/// use tower_lsp_server::ls_types::Uri;
108///
109/// let json = r#"{
110///   "require": {
111///     "symfony/console": "^6.0"
112///   }
113/// }"#;
114/// let uri = Uri::from_file_path("/project/composer.json").unwrap();
115///
116/// let result = parse_composer_json(json, &uri).unwrap();
117/// assert_eq!(result.dependencies.len(), 1);
118/// assert_eq!(result.dependencies[0].name, "symfony/console");
119/// ```
120pub fn parse_composer_json(content: &str, uri: &Uri) -> Result<ComposerParseResult> {
121    let root: Value = deps_core::parse_json_checked(content.as_bytes())?;
122
123    let line_table = LineOffsetTable::new(content);
124    let ast = JsonAst::parse(content);
125    if ast.is_none() {
126        tracing::warn!(
127            "jsonc-parser failed to parse composer.json content serde_json already accepted; \
128             dependency positions will default to (0,0)"
129        );
130    }
131    let mut dependencies = Vec::new();
132
133    // Parse each section, reading each entry's position directly from the AST (#613) — the
134    // AST's own `Object::properties` only ever lists a given object's *direct* children, so a
135    // name repeated across sections (e.g. also present in `require-dev`) or nested inside an
136    // unrelated value never gets confused with the real top-level occurrence, independent of
137    // `serde_json::Map` iteration order (#610).
138    const SECTIONS: [(&str, ComposerSection); 2] = [
139        ("require", ComposerSection::Require),
140        ("require-dev", ComposerSection::RequireDev),
141    ];
142
143    for (key, section) in SECTIONS {
144        if let Some(deps) = root.get(key).and_then(|v| v.as_object()) {
145            let positions = ast.as_ref().and_then(|ast| ast.section(key));
146            dependencies.extend(parse_section(
147                content,
148                deps,
149                section,
150                positions.as_ref(),
151                &line_table,
152            ));
153        }
154    }
155
156    let minimum_stability = root
157        .get("minimum-stability")
158        .and_then(|v| v.as_str())
159        .map(str::to_string);
160
161    Ok(ComposerParseResult {
162        dependencies,
163        uri: uri.clone(),
164        minimum_stability,
165    })
166}

168/// Parses a single dependency section and extracts positions, filtering platform packages.
169///
170/// `positions` is this section's own direct properties, pre-indexed by name (see
171/// [`JsonAst::section`]) — `None` when the AST parse degraded (see [`parse_composer_json`]),
172/// in which case every dependency falls back to a default, zero position rather than being
173/// dropped.
174fn parse_section(
175    content: &str,
176    deps: &serde_json::Map<String, Value>,
177    section: ComposerSection,
178    positions: Option<&JsonSection<'_>>,
179    line_table: &LineOffsetTable,
180) -> Vec<ComposerDependency> {
181    let mut result = Vec::new();
182
183    // A manifest entry whose value isn't a string (e.g. an object) is not a valid dependency
184    // declaration — `string_valued_entries` skips it rather than fabricating an entry with no
185    // `version_req` that would still be queried against the registry (#621, same bug class as
186    // npm's #619).
187    for (name, version_req) in string_valued_entries(deps) {
188        if is_platform_package(name) {
189            continue;
190        }
191
192        let (name_range, version_range) = positions
193            .and_then(|s| s.position(name, content, line_table))
194            .unwrap_or_default();
195
196        result.push(ComposerDependency {
197            name: name.into(),
198            name_range,
199            version_req: Some(version_req.into()),
200            version_range,
201            section,
202        });
203    }
204
205    result
206}

572    #[test]
573    fn test_duplicate_name_across_require_and_require_dev() {
574        // #610, same bug class as npm's #605: "vendor/pkg" appears in both `require` and
575        // `require-dev`, with keys not in source-text order relative to each section's
576        // start. The original implementation threaded a single, monotonically-advancing
577        // `search_start` cursor across `serde_json::Map` iteration, so it could skip past —
578        // or land on — the wrong occurrence whenever `Map` iteration order didn't match
579        // source-text order. Positions now come from a per-section AST index (#613), so each
580        // occurrence resolves to its own section's node directly, independent of both
581        // `Map` iteration order and source-text order entirely.
582        let json = r#"{
583  "require": {
584    "zzz/other": "^1.0",
585    "vendor/pkg": "^2.0"
586  },
587  "require-dev": {
588    "vendor/pkg": "^9.9",
589    "aaa/other": "^1.0"
590  }
591}"#;
592
593        let result = parse_composer_json(json, &test_uri()).unwrap();
594        assert_eq!(result.dependencies.len(), 4);
595
596        let require_pkg = result
597            .dependencies
598            .iter()
599            .find(|d| d.name == "vendor/pkg" && matches!(d.section, ComposerSection::Require))
600            .expect("vendor/pkg in require");
601        let dev_pkg = result
602            .dependencies
603            .iter()
604            .find(|d| d.name == "vendor/pkg" && matches!(d.section, ComposerSection::RequireDev))
605            .expect("vendor/pkg in require-dev");
606
607        assert_eq!(require_pkg.version_req, Some("^2.0".into()));
608        assert_eq!(dev_pkg.version_req, Some("^9.9".into()));
609
610        assert_eq!(require_pkg.name_range.start.line, 3);
611        assert_eq!(dev_pkg.name_range.start.line, 6);
612
613        let require_version = require_pkg
614            .version_range
615            .expect("require vendor/pkg version_range");
616        let dev_version = dev_pkg
617            .version_range
618            .expect("require-dev vendor/pkg version_range");
619        assert_eq!(require_version.start.line, 3);
620        assert_eq!(dev_version.start.line, 6);
621    }

625    /// A dependency's value can itself be a nested object containing a key with the same
626    /// name as a real top-level dependency in this section (e.g. a malformed/unusual
627    /// manifest). A text-based scanner finds the nested occurrence first; the AST only
628    /// ever indexes a section's own *direct* properties, so the real top-level occurrence's
629    /// position is never stolen by one nested inside a sibling's value. `a/b` itself has an
630    /// object value, so it is skipped entirely (#621) — only `c/d` survives.
631    #[test]
632    fn test_nested_object_value_with_colliding_key_resolves_to_top_level_position() {
633        let json = r#"{
634  "require": {
635    "a/b": {
636      "c/d": "0.0.1"
637    },
638    "c/d": "^2.0"
639  }
640}"#;
641
642        let result = parse_composer_json(json, &test_uri()).unwrap();
643        assert_eq!(result.dependencies.len(), 1);
644
645        let c_d = result
646            .dependencies
647            .iter()
648            .find(|d| d.name == "c/d")
649            .expect("c/d");
650        assert_eq!(c_d.version_req, Some("^2.0".into()));
651        // The real top-level "c/d" is on line 5, not line 3 (nested inside "a/b"'s value).
652        assert_eq!(c_d.name_range.start.line, 5);
653        let version_range = c_d.version_range.expect("c/d version_range");
654        assert_eq!(version_range.start.line, 5);
655    }

657    /// JSON permits (if unusual) a duplicate top-level key; `serde_json::Map` keeps only the
658    /// *last* occurrence's value (last-key-wins during deserialization, `preserve_order` or
659    /// not). The AST lookup must resolve the identically-named "require" key the same way —
660    /// the last one — not the first, or the surviving dependency's position silently defaults
661    /// to `Range::default()` whenever the two sections don't share every name.
662    #[test]
663    fn test_duplicate_top_level_section_key_resolves_to_last_occurrence() {
664        let json = r#"{
665  "require": {
666    "only-in-first": "^1.0"
667  },
668  "require": {
669    "vendor/pkg": "^2.0"
670  }
671}"#;
672
673        let result = parse_composer_json(json, &test_uri()).unwrap();
674        assert_eq!(result.dependencies.len(), 1);
675
676        let pkg = &result.dependencies[0];
677        assert_eq!(pkg.name, "vendor/pkg");
678        assert_eq!(pkg.version_req, Some("^2.0".into()));
679        assert_ne!(pkg.name_range, Range::default());
680        assert_eq!(pkg.name_range.start.line, 5);
681        assert!(pkg.version_range.is_some());
682    }
