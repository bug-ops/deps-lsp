18/// Composer ecosystem implementation.
19///
20/// Provides LSP functionality for composer.json files, including:
21/// - Dependency parsing with position tracking
22/// - Version information from Packagist registry
23/// - Inlay hints for latest versions
24/// - Hover tooltips with package metadata
25/// - Code actions for version updates
26/// - Diagnostics for unknown/abandoned packages
27pub struct ComposerEcosystem {
28    registry: Arc<PackagistRegistry>,
29    formatter: ComposerFormatter,
30}

132    fn fallback_completion_prefix<'a>(
133        &self,
134        content: &'a str,
135        position: Position,
136    ) -> Option<&'a str> {
137        let line = deps_core::fallback_completion::line_at(content, position)?;
138        if !is_in_dependencies_section(content, position.line as usize) {
139            return None;
140        }
141        let (prefix, _) = extract_prefix(line, position.character);
142        // A closed key or an open value string (see `extract_prefix`) both come back
143        // as an empty prefix — there is no safe text to offer at that position, so
144        // this suppresses the completion entirely (`None`, same as "no completable
145        // position at all") rather than relying on the caller's own `prefix.is_empty()`
146        // guard (#729).
147        if prefix.is_empty() {
148            return None;
149        }
150        Some(prefix)
151    }
152
153    fn fallback_completion_is_bare(&self, content: &str, position: Position) -> bool {
154        let Some(line) = deps_core::fallback_completion::line_at(content, position) else {
155            return false;
156        };
157        extract_prefix(line, position.character).1
158    }

160    fn completion_insert_text(&self, metadata: &dyn deps_core::Metadata) -> Option<String> {
161        let name = metadata.name();
162        let latest = metadata.latest_version().as_str();
163        Some(format!("\"{name}\": \"^{latest}\""))
164    }

166    fn as_any(&self) -> &dyn Any {
167        self
168    }

171/// Composer's `composer.json` dependency-like section keys.
172const DEPENDENCY_KEYS: &[&str] = &["require", "require-dev"];
173
174/// Checks if `line_number` of `content` is inside one of Composer's dependency-like
175/// sections, for `deps-lsp`'s raw-text fallback completion (parse-failure path).
176fn is_in_dependencies_section(content: &str, line_number: usize) -> bool {
177    deps_core::fallback_completion::is_in_json_dependencies(content, line_number, DEPENDENCY_KEYS)
178}
179
180/// Extracts the fallback-completion prefix on `line` up to `character`, together with
181/// whether the cursor sits inside a genuinely still-open key string (see
182/// [`deps_core::fallback_completion::strip_open_json_key`]) — used by both
183/// [`ComposerEcosystem::fallback_completion_prefix`] and
184/// [`ComposerEcosystem::fallback_completion_is_bare`].
185///
186/// A surviving `"` proves the key's own quotes are already open around the cursor —
187/// the same shape as NuGet's open-attribute case — so
188/// [`ComposerEcosystem::completion_insert_text`]'s full `"{name}": "^{latest}"` pair
189/// would duplicate that quote and produce invalid JSON if inserted there.
190/// `strip_open_json_key` only reports an open key when quote parity (escape-aware)
191/// proves the string is genuinely still open and precedes a key position, not a closed
192/// key or an open value; the closed/ambiguous/value cases return an empty prefix,
193/// which `deps-lsp`'s fallback-completion caller's `prefix.is_empty()` guard rejects
194/// before any registry search fires — suppressing rather than guessing, matching
195/// Maven's non-`artifactId`-tag discipline (#729).
196fn extract_prefix(line: &str, character: u32) -> (&str, bool) {
197    deps_core::fallback_completion::strip_open_json_key(deps_core::fallback_completion::raw_prefix(
198        line, character,
199    ))
200}
