1//! `packages.lock.json` parser.
2//!
3//! Every field except the package name is optional (S2): `"type": "Project"` and
4//! `"type": "CentralTransitive"` entries carry `requested` but no `resolved` at all — a
5//! required `resolved` field would abort deserialization of the *entire* file, which would
6//! only surface on multi-project solutions and not on the single-project fixture a test is
7//! most likely to use. Entries without `resolved` are simply skipped.
8//!
9//! `packages.<project_name>.lock.json` (per-project lock files, used when multiple projects
10//! share a directory) cannot be expressed as an exact name —
11//! [`NuGetLockParser::locate_lockfile`] falls back to that computed name (`<project_name>`
12//! being the manifest's own file stem, NuGet's convention) once the exact
13//! `packages.lock.json` name misses (D3, #451). This must be an exact match against *this*
14//! manifest's project name, not the first `packages.*.lock.json` found in the directory —
15//! a directory shared by multiple projects can hold several such files, and taking the
16//! first one silently attaches an unrelated project's resolved versions (#451 follow-up,
17//! tester-found regression).
18

29/// Mirrors `deps_core::lockfile::locate_lockfile_for_manifest`'s workspace-root search
30/// depth, so the multi-project fallback below walks exactly the same directories the
31/// exact-name search already tried.
32const MAX_WORKSPACE_DEPTH: usize = 5;
33
34/// [`LockFileProvider`] implementation for `packages.lock.json`.
35pub struct NuGetLockParser;
36
37impl NuGetLockParser {
38    const LOCKFILE_NAMES: &'static [&'static str] = &["packages.lock.json"];
39}

41/// Falls back to *this manifest's own* multi-project lock file name —
42/// `packages.<project_name>.lock.json`, where `<project_name>` is the manifest's file stem
43/// (NuGet's convention: the project name defaults to the project file's name without its
44/// extension) — once the exact `packages.lock.json` search (same directory, then up to
45/// [`MAX_WORKSPACE_DEPTH`] parent directories) has already missed. Mirrors
46/// `locate_lockfile_for_manifest`'s own directory-walk order, but against this one computed
47/// filename rather than a fixed list — an exact match, deliberately, not "the first
48/// `packages.*.lock.json` in the directory": a directory shared by multiple projects can
49/// hold several per-project lock files, and picking the wrong one would silently attach an
50/// unrelated project's resolved versions (#451 follow-up regression).
51fn locate_multi_project_lockfile(manifest_uri: &Uri) -> Option<PathBuf> {
52    let manifest_path = manifest_uri.to_file_path()?;
53    let project_name = manifest_path.file_stem()?.to_str()?;
54    if project_name.is_empty() {
55        return None;
56    }
57    let lock_filename = format!("packages.{project_name}.lock.json");
58    let manifest_dir = manifest_path.parent()?;
59
60    let mut lock_path = manifest_dir.to_path_buf();
61    lock_path.push(&lock_filename);
62    if deps_core::fs_probe::is_file(&lock_path) {
63        return Some(lock_path);
64    }
65
66    let mut current_dir = manifest_dir.parent()?;
67    for _ in 0..MAX_WORKSPACE_DEPTH {
68        lock_path = current_dir.join(&lock_filename);
69        if deps_core::fs_probe::is_file(&lock_path) {
70            return Some(lock_path);
71        }
72        current_dir = current_dir.parent()?;
73    }
74    None
75}

77#[derive(Deserialize)]
78struct PackagesLock {
79    // The top-level "version" field (1 or 2) is intentionally not modeled: both schema
80    // versions are accepted without gating on its value, and serde ignores unknown fields
81    // by default, so there is nothing to read it for.
82    #[serde(default)]
83    dependencies: HashMap<String, HashMap<String, LockEntry>>,
84}
85
86#[derive(Deserialize)]
87struct LockEntry {
88    #[serde(default)]
89    resolved: Option<String>,
90    #[serde(default, rename = "contentHash")]
91    content_hash: Option<String>,
92}
93
94impl LockFileProvider for NuGetLockParser {
95    fn locate_lockfile(&self, manifest_uri: &Uri) -> Option<PathBuf> {
96        locate_lockfile_for_manifest(manifest_uri, Self::LOCKFILE_NAMES)
97            .or_else(|| locate_multi_project_lockfile(manifest_uri))
98    }
99
100    fn parse_lockfile<'a>(
101        &'a self,
102        lockfile_path: &'a Path,
103    ) -> std::pin::Pin<Box<dyn std::future::Future<Output = Result<ResolvedPackages>> + Send + 'a>>
104    {
105        Box::pin(async move {
106            tracing::debug!("Parsing packages.lock.json: {}", lockfile_path.display());
107
108            let packages = read_and_parse_lockfile(
109                lockfile_path,
110                "packages.lock.json",
111                parse_packages_lock_json,
112            )
113            .await?;
114
115            tracing::info!(
116                "Parsed packages.lock.json: {} packages from {}",
117                packages.len(),
118                lockfile_path.display()
119            );
120
121            Ok(packages)
122        })
123    }
124}

126/// Parses `packages.lock.json` content (already read and size-capped) into resolved packages.
127///
128/// The CPU-bound half of [`NuGetLockParser::parse_lockfile`], run inside
129/// [`deps_core::lockfile::read_and_parse_lockfile`]'s `spawn_blocking`.
130fn parse_packages_lock_json(content: String) -> Result<ResolvedPackages> {
131    let lock_data: PackagesLock =
132        deps_core::parse_json_checked(content.as_bytes()).map_err(|e| DepsError::ParseError {
133            file_type: "packages.lock.json".into(),
134            source: Box::new(e),
135        })?;
136
137    // Collect every TFM's resolved version per package name, then resolve the
138    // cross-TFM tie-break with the crate's own `compare_versions` (S6) instead of
139    // `deps_core::lockfile::best_package`, whose `semver::Version::parse` fallback
140    // always fails on NuGet's 4-component versions and degrades to string
141    // comparison (e.g. "1.10.0" < "1.9.0"). Only the single winner is ever handed
142    // to `ResolvedPackages`, so that broken comparator is never reached.
143    let mut candidates: HashMap<String, Vec<(String, Option<String>)>> = HashMap::new();
144    for packages in lock_data.dependencies.into_values() {
145        for (name, entry) in packages {
146            // "type": "Project" / "CentralTransitive" entries carry no `resolved`
147            // at all — skip rather than aborting the whole file (S2).
148            if let Some(resolved) = entry.resolved {
149                candidates
150                    .entry(name)
151                    .or_default()
152                    .push((resolved, entry.content_hash));
153            }
154        }
155    }
156
157    let mut packages = ResolvedPackages::new();
158    for (name, versions) in candidates {
159        let best = versions
160            .into_iter()
161            .max_by(|a, b| crate::version::compare_versions(&a.0, &b.0));
162        if let Some((version, content_hash)) = best {
163            packages.insert(ResolvedPackage::new(
164                name,
165                version,
166                ResolvedSource::Registry {
167                    // Informational only — nothing in `deps-lsp`/`deps-core::lsp_helpers`
168                    // reads `ResolvedSource`, and this path makes no network request, so
169                    // it never routes a lockfile-resolved version against a private feed
170                    // (issue #523's config resolution intentionally stops at
171                    // `NuGetDependency::source`, not `ResolvedPackage::source`).
172                    url: crate::registry::NUGET_ORG_INDEX_URL.into(),
173                    checksum: content_hash.unwrap_or_default(),
174                },
175            ));
176        }
177    }
178
179    Ok(packages)
180}

363    /// Regression test (tester-found, #451 follow-up): with two per-project lock files in
364    /// the same directory, each manifest must resolve to *its own* lock file by matching
365    /// the `<project>` segment against the manifest's file stem — not just the first
366    /// `packages.*.lock.json` a directory scan happens to find.
367    #[test]
368    fn test_locate_lockfile_multi_project_matches_own_project_not_first_found() {
369        // Held per `deps_core::fs_probe::snapshot_guard`'s doc: `locate_lockfile` transitively
370        // touches fs_probe (via `fs_probe::is_file`), and this test runs in the same binary as
371        // `deps-nuget/src/config.rs`'s diffing test.
372        let _guard = deps_core::fs_probe::snapshot_guard();
373        let temp_dir = tempfile::tempdir().unwrap();
374        let app1_manifest = temp_dir.path().join("App1.csproj");
375        let app2_manifest = temp_dir.path().join("App2.csproj");
376        let app1_lock = temp_dir.path().join("packages.App1.lock.json");
377        let app2_lock = temp_dir.path().join("packages.App2.lock.json");
378        std::fs::write(&app1_manifest, "<Project></Project>").unwrap();
379        std::fs::write(&app2_manifest, "<Project></Project>").unwrap();
380        std::fs::write(&app1_lock, "{}").unwrap();
381        std::fs::write(&app2_lock, "{}").unwrap();
382
383        let parser = NuGetLockParser;
384        assert_eq!(
385            parser.locate_lockfile(&Uri::from_file_path(&app1_manifest).unwrap()),
386            Some(app1_lock)
387        );
388        assert_eq!(
389            parser.locate_lockfile(&Uri::from_file_path(&app2_manifest).unwrap()),
390            Some(app2_lock)
391        );
392    }
