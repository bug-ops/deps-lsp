1//! NuGet V3 registry client.
2//!
3//! NuGet base URLs are not hardcodable: the service index
4//! (`https://api.nuget.org/v3/index.json`) must be resolved first, then consulted for the
5//! flat-container ("PackageBaseAddress"), search ("SearchQueryService"), and (future,
6//! see D1 below) registration ("RegistrationsBaseUrl") resource URLs.
7

11use serde::Deserialize;
12use std::any::Any;

14use tokio::sync::OnceCell;

16const SERVICE_INDEX_URL: &str = "https://api.nuget.org/v3/index.json";

18/// Display name for NuGet used in not-found and API-response error messages.
19pub const REGISTRY: &str = "NuGet";

21#[derive(Debug, Deserialize)]
22struct ServiceIndexResponse {
23    resources: Vec<ServiceResource>,
24}
25
26#[derive(Debug, Deserialize)]
27struct ServiceResource {
28    #[serde(rename = "@id")]
29    id: String,
30    #[serde(rename = "@type")]
31    r#type: String,
32}
33
34/// Resolved base URLs from the NuGet service index.
35#[derive(Debug, Clone)]
36struct ServiceIndex {
37    /// `PackageBaseAddress/3.0.0` — flat-container version enumeration.
38    package_base_address: String,
39    /// `SearchQueryService/3.5.0` (preferred) or bare `SearchQueryService`.
40    search_query_service: String,
41}
42
43fn pick_resource(resources: &[ServiceResource], type_preference: &[&str]) -> Option<String> {
44    for want in type_preference {
45        if let Some(r) = resources.iter().find(|r| r.r#type == *want) {
46            return Some(r.id.trim_end_matches('/').to_string());
47        }
48    }
49    None
50}
51
52impl ServiceIndex {
53    fn resolve(response: &ServiceIndexResponse) -> Result<Self> {
54        let package_base_address =
55            pick_resource(&response.resources, &["PackageBaseAddress/3.0.0"]).ok_or_else(|| {
56                deps_core::DepsError::ParseError {
57                    file_type: "NuGet service index".into(),
58                    source: Box::new(std::io::Error::other(
59                        "missing PackageBaseAddress/3.0.0 resource",
60                    )),
61                }
62            })?;
63        let search_query_service = pick_resource(
64            &response.resources,
65            &["SearchQueryService/3.5.0", "SearchQueryService"],
66        )
67        .ok_or_else(|| deps_core::DepsError::ParseError {
68            file_type: "NuGet service index".into(),
69            source: Box::new(std::io::Error::other("missing SearchQueryService resource")),
70        })?;
71
72        Ok(Self {
73            package_base_address,
74            search_query_service,
75        })
76    }
77}

110#[derive(Clone)]
111pub struct NuGetRegistry {
112    cache: Arc<HttpCache>,
113    service_index_url: String,
114    service_index: Arc<OnceCell<ServiceIndex>>,
115}
116
117impl NuGetRegistry {
118    pub fn new(cache: Arc<HttpCache>) -> Self {
119        Self::with_service_index_url(cache, SERVICE_INDEX_URL.to_string())
120    }
121
122    fn with_service_index_url(cache: Arc<HttpCache>, service_index_url: String) -> Self {
123        Self {
124            cache,
125            service_index_url,
126            service_index: Arc::new(OnceCell::new()),
127        }
128    }
129
130    /// Resolves the service index once per process, retrying on the next call if
131    /// resolution failed. `get_or_try_init` (not `get_or_init`) is load-bearing: it leaves
132    /// the cell empty on `Err` so a transient failure does not permanently poison lookups,
133    /// and it serializes concurrent initializers so a cold start with many dependencies
134    /// does not stampede the index endpoint.
135    async fn service_index(&self) -> Result<&ServiceIndex> {
136        self.service_index
137            .get_or_try_init(|| async {
138                let data = self.cache.get_cached(&self.service_index_url).await?;
139                let response: ServiceIndexResponse = serde_json::from_slice(&data)?;
140                ServiceIndex::resolve(&response)
141            })
142            .await
143    }
144
145    /// Fetches all available versions for `name` from the flat-container endpoint,
146    /// sorted newest-first.
147    ///
148    /// # Errors
149    ///
150    /// Returns an error if the service index cannot be resolved or the flat-container
151    /// request fails.
152    pub async fn get_versions_typed(&self, name: &str) -> Result<Vec<NuGetVersion>> {
153        let index = self.service_index().await?;
154        let url = flat_container_url(&index.package_base_address, name);
155
156        let data = self.cache.get_cached(&url).await?;
157        parse_flat_container(&data)
158    }
159
160    /// Finds the highest version of `name` matching `req` (exact pin, interval notation,
161    /// or floating pattern). Prerelease versions are excluded unless `req` itself is
162    /// prerelease-bearing.
163    ///
164    /// # Errors
165    ///
166    /// Returns an error if the service index cannot be resolved or the flat-container
167    /// request fails.
168    pub async fn get_latest_matching_typed(
169        &self,
170        name: &str,
171        req: &str,
172    ) -> Result<Option<NuGetVersion>> {
173        let versions = self.get_versions_typed(name).await?;
174        Ok(pick_latest_matching(versions, req))
175    }
176
177    /// Searches the NuGet `SearchQueryService` for `query`, returning up to `limit` results.
178    ///
179    /// # Errors
180    ///
181    /// Returns an error if the service index cannot be resolved or the search request fails.
182    pub async fn search_typed(&self, query: &str, limit: usize) -> Result<Vec<PackageInfo>> {
183        let index = self.service_index().await?;
184        let url = search_url(&index.search_query_service, query, limit);
185
186        let data = self.cache.get_cached(&url).await?;
187        parse_search_response(&data, limit)
188    }
189}
190
191/// Builds the flat-container version-enumeration URL for `name`.
192///
193/// The package id is lowercased (NuGet ids are case-insensitive and every V3 API path
194/// segment is lowercased) and percent-encoded before being interpolated into the path.
195/// Encoding is load-bearing, not cosmetic: an unencoded id lets a crafted
196/// `PackageReference Include="..."` value inject path segments (`../../etc/passwd`
197/// collapses dot-segments) or truncate the path at `#`/`?`/control characters, making
198/// deps-lsp silently resolve and display a *different* real package's version data under
199/// an attacker-chosen name.
200pub fn flat_container_url(base: &str, name: &str) -> String {
201    let lower = name.to_lowercase();
202    format!("{base}/{}/index.json", urlencoding::encode(&lower))
203}
204
205/// Builds the `SearchQueryService` URL for `query`, limited to `limit` results.
206///
207/// `semVerLevel=2.0.0` is mandatory (spec §1) — omitting it silently hides every package
208/// whose latest version uses a dotted prerelease label.
209pub fn search_url(base: &str, query: &str, limit: usize) -> String {
210    format!(
211        "{base}?q={}&take={limit}&prerelease=false&semVerLevel=2.0.0",
212        urlencoding::encode(query),
213    )
214}

216/// Parses a flat-container `index.json` response into descending-sorted versions.
217pub fn parse_flat_container(data: &[u8]) -> Result<Vec<NuGetVersion>> {
218    let parsed: FlatContainerIndex = serde_json::from_slice(data)?;
219
220    let mut versions = parsed.versions;
221    // Sort locally, descending: the flat container's observed ascending order is not a
222    // documented contract, so relying on `.reverse()` would silently corrupt "latest" if
223    // the CDN/backend ever changes it (rev2, S3).
224    versions.sort_by(|a, b| compare_versions(b, a));
225
226    Ok(versions
227        .into_iter()
228        .map(|version| NuGetVersion { version })
229        .collect())
230}
231
232/// Picks the highest version matching `req` from an already-fetched, descending-sorted
233/// version list. `req` is treated as `"*"` when empty. Prerelease versions are excluded
234/// unless `req` itself is prerelease-bearing (contains `-`) or is a floating pattern whose
235/// own prerelease inclusion is handled by `crate::version::resolve_float`.
236fn pick_latest_matching(versions: Vec<NuGetVersion>, req: &str) -> Option<NuGetVersion> {
237    if versions.is_empty() {
238        return None;
239    }
240
241    let req = if req.is_empty() { "*" } else { req };
242
243    if req.contains('*') {
244        let strings: Vec<String> = versions.iter().map(|v| v.version.clone()).collect();
245        return crate::version::resolve_float(&strings, req).map(|v| NuGetVersion {
246            version: v.to_string(),
247        });
248    }
249
250    let req_is_prerelease_bearing = req.contains('-');
251    versions.into_iter().find(|v| {
252        crate::version::satisfies(&v.version, req)
253            && (req_is_prerelease_bearing || !crate::version::is_prerelease(&v.version))
254    })
255}
256
257fn parse_search_response(data: &[u8], limit: usize) -> Result<Vec<PackageInfo>> {
258    let response: SearchResponse = serde_json::from_slice(data)?;
259
260    Ok(response
261        .data
262        .into_iter()
263        .take(limit)
264        .map(|d| PackageInfo {
265            name: d.id,
266            description: d.description,
267            repository: d.project_url,
268            documentation: None,
269            latest_version: d.version.unwrap_or_default(),
270        })
271        .collect())
272}

279impl deps_core::Registry for NuGetRegistry {
280    fn get_versions<'a>(
281        &'a self,
282        name: &'a str,
283    ) -> deps_core::ecosystem::BoxFuture<'a, Result<Vec<Box<dyn deps_core::Version>>>> {
284        Box::pin(async move {
285            let versions = self.get_versions_typed(name).await?;
286            Ok(versions
287                .into_iter()
288                .map(|v| Box::new(v) as Box<dyn deps_core::Version>)
289                .collect())
290        })
291    }
292
293    fn get_latest_matching<'a>(
294        &'a self,
295        name: &'a str,
296        req: &'a str,
297    ) -> deps_core::ecosystem::BoxFuture<'a, Result<Option<Box<dyn deps_core::Version>>>> {
298        Box::pin(async move {
299            let version = self.get_latest_matching_typed(name, req).await?;
300            Ok(version.map(|v| Box::new(v) as Box<dyn deps_core::Version>))
301        })
302    }
303
304    fn search<'a>(
305        &'a self,
306        query: &'a str,
307        limit: usize,
308    ) -> deps_core::ecosystem::BoxFuture<'a, Result<Vec<Box<dyn deps_core::Metadata>>>> {
309        Box::pin(async move {
310            let results = self.search_typed(query, limit).await?;
311            Ok(results
312                .into_iter()
313                .map(|m| Box::new(m) as Box<dyn deps_core::Metadata>)
314                .collect())
315        })
316    }
317
318    fn package_url(&self, name: &str) -> String {
319        package_url(name)
320    }
321
322    fn as_any(&self) -> &dyn Any {
323        self
324    }
325}
