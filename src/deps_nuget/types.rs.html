6/// A single `PackageReference` / `PackageVersion` / `package` entry from a manifest.
7#[derive(Debug, Clone, PartialEq, Eq)]
8pub struct NuGetDependency {
9    pub name: String,
10    pub name_range: Range,
11    /// Absent when the manifest omits an explicit version — central package management
12    /// entries and unresolvable MSBuild property expressions (`$(...)`) both degrade to
13    /// `None` rather than a bogus or unresolved-looking requirement.
14    pub version_requirement: Option<String>,
15    pub version_range: Option<Range>,
16}
17
18deps_core::impl_dependency!(NuGetDependency {
19    name: name,
20    name_range: name_range,
21    version: version_requirement,
22    version_range: version_range,
23});

41/// A single version of a package, as returned by the NuGet flat-container endpoint.
42///
43/// Deliberately a single field. `is_yanked` always reports `false` because the flat
44/// container carries no `listed` flag — unlisted detection requires the registration
45/// hive and is deferred (spec §1, follow-up D1) to hover-only enrichment.
46///
47/// Hand-writes `deps_core::Version` instead of using `impl_version!` because the macro's
48/// `is_prerelease` cannot be overridden and the shared default (a keyword sniff for
49/// `-alpha|-beta|-rc|-dev|-pre|-snapshot|-canary|-nightly`) does not recognize standard
50/// .NET prerelease labels (`-rtm`, `-servicing.23`, `-CI-*`, `-final`, ...). This follows
51/// the same precedent as `deps-maven`'s and `deps-go`'s hand-written `Version` impls for
52/// ecosystems with a structural (non-keyword) prerelease convention.
53#[derive(Debug, Clone)]
54pub struct NuGetVersion {
55    pub version: String,
56}
57
58impl deps_core::Version for NuGetVersion {
59    fn version_string(&self) -> &str {
60        &self.version
61    }
62
63    fn is_yanked(&self) -> bool {
64        false
65    }
66
67    fn is_prerelease(&self) -> bool {
68        crate::version::is_prerelease(&self.version)
69    }
70
71    fn as_any(&self) -> &dyn Any {
72        self
73    }
74}

94#[cfg(test)]
95mod tests {
96    use super::*;
97    use tower_lsp_server::ls_types::Position;
98
99    fn test_dep() -> NuGetDependency {
100        NuGetDependency {
101            name: "Newtonsoft.Json".into(),
102            name_range: Range::new(Position::new(2, 4), Position::new(2, 20)),
103            version_requirement: Some("13.0.3".into()),
104            version_range: Some(Range::new(Position::new(2, 30), Position::new(2, 36))),
105        }
106    }
107
108    #[test]
109    fn test_dependency_trait() {
110        use deps_core::Dependency;
111
112        let dep = test_dep();
113        assert_eq!(dep.name(), "Newtonsoft.Json");
114        assert_eq!(dep.version_requirement(), Some("13.0.3"));
115        assert!(dep.features().is_empty());
116        assert!(dep.as_any().is::<NuGetDependency>());
117        assert!(matches!(
118            dep.source(),
119            deps_core::parser::DependencySource::Registry
120        ));
121    }
122
123    #[test]
124    fn test_dependency_without_version() {
125        use deps_core::Dependency;
126
127        let dep = NuGetDependency {
128            name: "MyCompany.Shared".into(),
129            name_range: Range::default(),
130            version_requirement: None,
131            version_range: None,
132        };
133        assert!(dep.version_requirement().is_none());
134        assert!(dep.version_range().is_none());
135    }
136
137    #[test]
138    fn test_parse_result_trait() {
139        use deps_core::ParseResult;
140
141        let result = NuGetParseResult {
142            dependencies: vec![test_dep()],
143            uri: deps_core::test_util::test_uri("/test/App.csproj"),
144        };
145
146        assert_eq!(result.dependencies().len(), 1);
147        assert!(result.workspace_root().is_none());
148        assert!(result.as_any().is::<NuGetParseResult>());
149    }
150
151    #[test]
152    fn test_version_trait() {
153        use deps_core::Version;
154
155        let ver = NuGetVersion {
156            version: "13.0.3".into(),
157        };
158        assert_eq!(ver.version_string(), "13.0.3");
159        assert!(!ver.is_yanked());
160        assert!(!ver.is_prerelease());
161        assert!(ver.as_any().is::<NuGetVersion>());
162    }
163
164    #[test]
165    fn test_version_prerelease_dotnet_label() {
166        use deps_core::Version;
167
168        // Real .NET label that the shared keyword-sniff default would misclassify as stable.
169        let ver = NuGetVersion {
170            version: "13.0.0-rtm".into(),
171        };
172        assert!(ver.is_prerelease());
173        assert!(!ver.is_stable());
174    }

176    #[test]

178        use deps_core::Version;
179
180        let ver = NuGetVersion {

182        };
183        assert!(!ver.is_yanked());
184    }

193            repository: None,

197        assert_eq!(info.name(), "Newtonsoft.Json");
198        assert_eq!(

202        assert_eq!(info.latest_version(), "13.0.3");
