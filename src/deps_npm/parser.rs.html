15/// Result of parsing a package.json file.
16///
17/// Contains all dependencies found in the file with their positions.
18#[non_exhaustive]
19#[derive(Debug)]
20pub struct NpmParseResult {
21    /// Dependencies found across all `package.json` sections.
22    pub dependencies: Vec<NpmDependency>,
23    /// URI of the manifest this result was parsed from.
24    pub uri: Uri,
25    /// Every `.npmrc`-resolved alternate registry this parse's dependencies reference,
26    /// deduplicated (spec FR-002–004) — fed to `NpmRegistry::register_alternate` by
27    /// `NpmEcosystem::parse_manifest`, the one place a per-document `.npmrc` resolution and
28    /// the long-lived shared router meet. Empty for a workspace declaring no `.npmrc`
29    /// (NFR-005: zero regression).
30    pub resolved_registries: Vec<NpmRegistryIndex>,
31    /// `Some((kept, total))` once the manifest declared more dependencies than
32    /// `deps_core::MAX_DEPENDENCIES_PER_DOCUMENT` (#796), read by
33    /// [`deps_core::ParseResult::dependency_truncation`]'s override below.
34    pub dependency_truncation: Option<(usize, usize)>,
35}

46/// Parses a package.json file and extracts all dependencies with positions.
47///
48/// Handles all dependency sections:
49/// - `dependencies`
50/// - `devDependencies`
51/// - `peerDependencies`
52/// - `optionalDependencies`
53///
54/// # Errors
55///
56/// Returns an error if:
57/// - JSON parsing fails
58/// - File is not a valid package.json structure
59///
60/// # Examples
61///
62/// ```no_run
63/// use deps_npm::parser::parse_package_json;
64/// use tower_lsp_server::ls_types::Uri;
65///
66/// let json = r#"{
67///   "dependencies": {
68///     "express": "^4.18.2"
69///   }
70/// }"#;
71/// let uri = Uri::from_file_path("/project/package.json").unwrap();
72///
73/// let result = parse_package_json(json, &uri).unwrap();
74/// assert_eq!(result.dependencies.len(), 1);
75/// assert_eq!(result.dependencies[0].name, "express");
76/// ```
77pub fn parse_package_json(content: &str, uri: &Uri) -> Result<NpmParseResult> {
78    parse_package_json_with_context(content, uri, &NpmParseContext::default())
79}

81/// [`parse_package_json`], but threading `ctx` through to `.npmrc` registry resolution.
82///
83/// Spec FR-002–FR-008 — the real entry point; [`parse_package_json`] delegates here with a
84/// fresh, default context, mirroring
85/// `deps_cargo::parser::parse_cargo_toml_with_context`'s pattern.
86///
87/// # Errors
88///
89/// Same as [`parse_package_json`].
90pub fn parse_package_json_with_context(
91    content: &str,
92    uri: &Uri,
93    ctx: &NpmParseContext,
94) -> Result<NpmParseResult> {
95    let root: Value = deps_core::parse_json_checked(content.as_bytes())?;
96
97    // Build line offset table once for O(log n) position lookups
98    let line_table = LineOffsetTable::new(content);
99    let ast = JsonAst::parse(content);
100    if ast.is_none() {
101        tracing::warn!(
102            "jsonc-parser failed to parse package.json content serde_json already accepted; \
103             dependency positions will default to (0,0)"
104        );
105    }
106
107    let mut dependencies = Vec::new();
108    // Shared across every section below (#796) — the ceiling is per-document, not
109    // per-section.
110    let mut budget = deps_core::DependencyBudget::new(deps_core::MAX_DEPENDENCIES_PER_DOCUMENT);
111
112    // Parse each dependency section, reading each entry's position directly from the AST
113    // (#613) — the AST's own `Object::properties` only ever lists a given object's *direct*
114    // children, so a name repeated across sections (e.g. also present in `devDependencies`)
115    // or nested inside an unrelated value (e.g. `packageExtensions`) never gets confused with
116    // the real top-level occurrence.
117    const SECTIONS: [(&str, NpmDependencySection); 4] = [
118        ("dependencies", NpmDependencySection::Dependencies),
119        ("devDependencies", NpmDependencySection::DevDependencies),
120        ("peerDependencies", NpmDependencySection::PeerDependencies),
121        (
122            "optionalDependencies",
123            NpmDependencySection::OptionalDependencies,
124        ),
125    ];
126
127    for (key, section) in SECTIONS {
128        if let Some(deps) = root.get(key).and_then(|v| v.as_object()) {
129            let positions = ast.as_ref().and_then(|ast| ast.section(key));
130            dependencies.extend(parse_dependency_section(
131                content,
132                deps,
133                section,
134                positions.as_ref(),
135                &line_table,
136                &mut budget,
137            ));
138        }
139    }
140
141    // FR-002: a non-`file:` URI (or one `Uri::to_file_path` cannot resolve) has no directory to
142    // walk `.npmrc`/pnpm-workspace discovery from — falls back to the empty `NpmConfig`, which
143    // resolves every dependency to `DependencySource::Registry` (NFR-005: byte-identical to
144    // pre-feature behavior), rather than failing the whole parse. Also the manifest directory
145    // spec 046's catalog resolution walks up from (S3: `None` here is what makes a non-`file:`
146    // URI land on `CatalogOutcome::NoWorkspaceFile` rather than skipping resolution).
147    //
148    // Implementation-critique S2: `Uri::to_file_path` does **not** check the URI's scheme (its
149    // own doc says so) — it just decodes whatever path component is present. Left unguarded,
150    // `untitled:package.json` (VS Code's untitled-buffer form, path "package.json") would
151    // resolve `manifest_dir` to a *relative* path, and `find_workspace_file`/`.npmrc` discovery
152    // would then probe the LSP server process's own CWD instead of the workspace the document
153    // notionally belongs to; a `vscode-vfs://`/`vscode-remote://` URI whose path happens to
154    // mirror a real local path could likewise resolve against an unrelated local
155    // `pnpm-workspace.yaml`/`.npmrc`. Both `.npmrc` registry resolution and the catalog gate
156    // share this one `manifest_dir`, so gating it once here closes both: require the `file`
157    // scheme (case-insensitive per RFC 3986) and an absolute resulting directory.
158    let manifest_dir = uri
159        .scheme()
160        .as_str()
161        .eq_ignore_ascii_case("file")
162        .then(|| uri.to_file_path())
163        .flatten()
164        .and_then(|path| path.parent().map(std::path::Path::to_path_buf))
165        .filter(|dir| dir.is_absolute());
166
167    let npm_config: NpmConfig = manifest_dir
168        .as_deref()
169        .map(|dir| crate::config::resolve(dir, &ctx.config_cache, &ctx.policy))
170        .unwrap_or_default();
171
172    for dep in &mut dependencies {
173        // Route by the real registry package name, not the manifest alias (issue #654 S2):
174        // a `.npmrc` scope entry (`@myorg:registry=...`) matches the package actually being
175        // installed, and routing by the alias instead can send a private scoped package's
176        // name to the public registry, or a public package to a private feed.
177        dep.source = npm_config.resolve_source_for(deps_core::Dependency::name(dep));
178    }
179
180    // Spec 046 FR-001/NFR-002: cheap fast-path — a non-pnpm manifest pays one string check per
181    // dependency and zero filesystem calls. `apply` runs unconditionally once this fires (the
182    // module's totality invariant), never short-circuited by `load` returning `None`.
183    if dependencies.iter().any(|dep| {
184        dep.version_req
185            .as_ref()
186            .is_some_and(|req| req.as_str().starts_with("catalog:"))
187    }) {
188        let workspace_config = crate::catalog::load(manifest_dir.as_deref(), &ctx.workspace_cache);
189        crate::catalog::apply(&mut dependencies, workspace_config.as_deref());
190    }
191
192    Ok(NpmParseResult {
193        dependencies,
194        uri: uri.clone(),
195        resolved_registries: npm_config.resolved_registries(),
196        dependency_truncation: budget.truncation(),
197    })
198}

200/// Parses a single dependency section and extracts positions.
201///
202/// `positions` is this section's own direct properties, pre-indexed by name (see
203/// [`JsonAst::section`]) — `None` when the AST parse degraded (see
204/// [`parse_package_json_with_context`]), in which case every dependency falls back to a
205/// default, zero position rather than being dropped.
206fn parse_dependency_section(
207    content: &str,
208    deps: &serde_json::Map<String, Value>,
209    section: NpmDependencySection,
210    positions: Option<&JsonSection<'_>>,
211    line_table: &LineOffsetTable,
212    budget: &mut deps_core::DependencyBudget,
213) -> Vec<NpmDependency> {
214    let mut result = Vec::new();
215
216    // A manifest entry whose value isn't a string (e.g. an object) is not a valid dependency
217    // declaration — `string_valued_entries` skips it rather than fabricating an entry with no
218    // `version_req` that would still be queried against the registry (#619).
219    for (name, version_req) in string_valued_entries(deps) {
220        if !budget.allow() {
221            continue;
222        }
223
224        let (name_range, version_range) = positions
225            .and_then(|s| s.position(name, content, line_table))
226            .unwrap_or_default();
227
228        let (package, version_req) = match parse_npm_alias(version_req) {
229            Some(alias) => (Some(alias.package.into()), alias.version_req),
230            None => (None, version_req.to_string()),
231        };
232
233        result.push(NpmDependency {
234            name: name.into(),
235            name_range,
236            version_req: Some(version_req.into()),
237            version_range,
238            section,
239            // Overwritten by `parse_package_json_with_context` once `.npmrc` resolution has
240            // run; `Registry` here is the correct value for a manifest with no `.npmrc` at
241            // all (NFR-005) and for `parse_dependency_section`'s own unit tests, which do
242            // not go through that resolution step.
243            source: deps_core::parser::DependencySource::Registry,
244            // Overwritten by `parse_package_json_with_context`'s catalog post-pass when the
245            // `catalog:` gate fires; `None` here is correct for both a non-pnpm manifest and
246            // for this function's own unit tests.
247            catalog: None,
248            package,
249        });
250    }
251
252    result
253}

261/// Parses an `npm:` alias specifier (issue #654): npm/pnpm/yarn let a manifest install a
262/// dependency under a different registry name than its JSON key
263/// (`"my-react": "npm:react@^18.0.0"`) — the key becomes a local import alias, and the value
264/// names the real package to resolve against the registry, with everything after the package
265/// name's own trailing `@` as its version requirement.
266///
267/// Returns `None` when `value` (after trimming surrounding whitespace — `npm`'s own
268/// `npm-package-arg` parser tolerates it, so a manifest author accidentally adding it should
269/// not silently disable the alias) doesn't start with `npm:`, when the parsed package name
270/// is empty (e.g. `"npm:"`, `"npm:@"`, `"npm:@/pkg"`, `"npm:@scope/"`), or for the
271/// pnpm-catalog combination form (`npm:<pkg>@catalog:<name>`, deliberately unhandled — see
272/// below) — the caller then falls back to the original literal value, matching
273/// pre-alias-support behavior.
274///
275/// The name/version boundary is [`deps_core::package::npm_style_name_boundary`] — shared
276/// with `deps-deno`'s `npm:`/`jsr:` specifier grammar so a scoped real package name
277/// (`@scope/pkg`) is bounded identically in both crates (issue #654 S3) rather than
278/// reimplemented here more loosely. A trailing `/` right after the name (that helper's
279/// subpath-boundary behavior, meaningful for Deno's own `npm:` import specifiers) has no
280/// equivalent in a `package.json` dependency value, so it is treated as malformed here, not
281/// silently truncated.
282///
283/// A dist-tag alias (`"npm:react@beta"`, `"npm:foo@latest"`) is legal npm syntax but not a
284/// semver range `NpmFormatter::compile_requirement`'s `node_semver::Range` can parse, which
285/// would otherwise silently match no version at all — treated the same as the
286/// missing-version case below.
287///
288/// The pnpm-catalog combination form is left to the caller's literal-value fallback rather
289/// than resolved here: `catalog.rs`'s `apply` looks up the catalog map by
290/// [`crate::types::NpmDependency::name`] (the JSON key/alias), not the real package this
291/// function would extract, so resolving the alias here would feed the catalog lookup the
292/// wrong key — see `catalog.rs`'s "Known limitations" doc, which this leaves unchanged.
293// `name_len` comes from `npm_style_name_boundary` (ASCII-find derived), so it is always a
294// char boundary.
295#[allow(clippy::string_slice)]
296fn parse_npm_alias(value: &str) -> Option<NpmAlias> {
297    let rest = value.trim().strip_prefix("npm:")?;
298
299    let Some(name_len) = deps_core::package::npm_style_name_boundary(rest) else {
300        tracing::debug!(
301            value,
302            "npm: alias has a malformed package name, using literal value"
303        );
304        return None;
305    };
306    let package = rest[..name_len].trim();
307    if package.is_empty() {
308        tracing::debug!(
309            value,
310            "npm: alias has an empty package name, using literal value"
311        );
312        return None;
313    }
314
315    let after_name = &rest[name_len..];
316    let version_req = match after_name.strip_prefix('@') {
317        Some(v) => v.trim(),
318        // Empty: no version at all (`"npm:react"`). Anything else starts with `/` — Deno's
319        // subpath syntax, not valid here (see this function's doc) — reject rather than
320        // truncate.
321        None if after_name.is_empty() => "",
322        None => {
323            tracing::debug!(
324                value,
325                "npm: alias has a subpath after the package name, using literal value"
326            );
327            return None;
328        }
329    };
330
331    if version_req.starts_with("catalog:") {
332        tracing::debug!(
333            value,
334            "npm: alias is the pnpm-catalog combination form, deferring to catalog.rs"
335        );
336        return None;
337    }
338
339    // No version requirement at all, an empty one (`"npm:react@"`), or one that isn't a
340    // semver range `node_semver::Range` accepts (a dist-tag like `"beta"`/`"latest"`) is a
341    // valid, if unusual, alias: treat it as an existence wildcard rather than fabricating an
342    // invalid semver range or dropping the dependency outright — mirrors the `"*"`
343    // existence-wildcard convention `deps_core::registry::is_existence_wildcard` already
344    // recognizes for every ecosystem's version resolution.
345    let version_req = if version_req.is_empty() || node_semver::Range::parse(version_req).is_err() {
346        "*"
347    } else {
348        version_req
349    };
350
351    Some(NpmAlias {
352        package: package.to_string(),
353        version_req: version_req.to_string(),
354    })
355}
