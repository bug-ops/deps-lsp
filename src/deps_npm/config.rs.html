1//! `.npmrc` discovery and `registry`/`@scope:registry` resolution.
2//!
3//! Resolves a dependency's npm registry the way npm itself does — a top-level `registry=`
4//! override for unscoped dependencies, or an `@scope:registry=` entry for a scoped
5//! dependency's own scope — by reading the same two-tier `.npmrc` hierarchy npm consults:
6//! the project tier (walked from the opened `package.json`'s directory up to the filesystem
7//! root, mirroring `deps-cargo`'s `.cargo/config.toml` ancestor walk) and the user tier
8//! (`~/.npmrc`). The global tier (`$PREFIX/etc/npmrc`) is out of scope (spec FR-002/FR-014).
9//!
10//! # Security model (read before touching this module)
11//!
12//! A workspace's own `.npmrc` is attacker-controlled the moment a hostile repository is
13//! cloned and opened — this LSP parses on file open, before any build ever runs. Phase 1
14//! carries **no** authentication at all (spec Out of Scope), which closes the credential
15//! half of the threat model this module would otherwise have to solve, but two things still
16//! apply:
17//!
18//! - **No auth-shaped key is ever parsed** (FR-013/NFR-001). `parse_npmrc_raw` recognizes
19//!   exactly two key shapes — `registry` and `@<scope>:registry` — and skips every other
20//!   line outright. There is no `HashMap<String, String>` "everything else" table anywhere
21//!   in this module for an `_authToken`/`_auth`/`_password`/`_authIdent`/`always-auth`/
22//!   `//host/:_*` value to land in even accidentally. This is a structural guarantee, not a
23//!   runtime filter — verified by this module's own NFR-001 test.
24//! - **A literal `user:pass@`/`user@` written directly in a `registry=`/`@scope:registry=`
25//!   value is redacted before it can leak** (M1 fix, mirroring `deps-pypi`'s own M1). No
26//!   `${VAR}` expansion is needed for this case — a hostile `.npmrc` can write the credential
27//!   straight into the raw value — so `resolve_entry` redacts it (see
28//!   [`deps_core::net_policy::redact_userinfo`]) before it ever reaches a `tracing::warn!`
29//!   call or [`InvalidEntry::raw`], which [`NpmConfig::resolve_source_for`] can surface as
30//!   [`DependencySource::CustomRegistry`]'s `url` in hover/diagnostics text.
31//! - **Internal-network reachability (SSRF-adjacent).** [`NpmRegistryIndex::new`] requires a
32//!   [`deps_core::net_policy::RegistryAccessPolicy`] and checks every candidate against it —
33//!   unlike Cargo's `$CARGO_HOME`-is-trusted split, npm's project and user tiers are
34//!   policy-symmetric (spec NFR-003(a)): phase 1 has no credential provenance to protect, so
35//!   there is no tier that is "the user's own configuration" in the way `$CARGO_HOME` is.
36//!
37//! See `specs/032-npm-npmrc-registry-support/spec.md` FR-001–FR-014 and
38//! `specs/032-npm-npmrc-registry-support/plan.md` §1/§3/§6 for the design review this module
39//! implements.

52/// Why a candidate `registry=`/`@scope:registry=` value failed [`NpmRegistryIndex::new`]'s
53/// validation, or why expansion of a `${VAR}` placeholder inside it failed (FR-007).
54#[derive(Debug, Clone, thiserror::Error, PartialEq, Eq)]
55pub enum NpmRegistryIndexError {
56    /// The value did not parse as a URL at all.
57    #[error("not a valid URL: {0}")]
58    InvalidUrl(String),
59    /// The URL's scheme is not `https` (the sole carve-out is a `cfg(test)`/`test-util`-only
60    /// `http` loopback host — see [`NpmRegistryIndex::new`]).
61    #[error("registry index must use https, got scheme {0:?}")]
62    NotHttps(String),
63    /// The URL carries a `user:pass@`/`user@` component.
64    #[error("registry index URL must not carry userinfo")]
65    UserInfoPresent,
66    /// The candidate's host is blocked by the current
67    /// [`deps_core::net_policy::WorkspaceRegistryAccess`] policy.
68    #[error("registry index host class {class} blocked by registries.workspace_registries policy")]
69    BlockedHost {
70        /// The blocked host's classification.
71        class: HostClass,
72    },
73    /// A `${VAR}` placeholder in the value names an environment variable that is not set
74    /// (FR-007) — the whole value is invalid, never fetched as the literal
75    /// `${VAR}`-containing string.
76    #[error("environment variable {0:?} referenced in registry value is not set")]
77    UndefinedEnvVar(String),
78}
79
80impl From<IndexUrlError> for NpmRegistryIndexError {
81    fn from(error: IndexUrlError) -> Self {
82        match error {
83            IndexUrlError::InvalidUrl(raw) => Self::InvalidUrl(raw),
84            IndexUrlError::NotHttps(scheme) => Self::NotHttps(scheme),
85            IndexUrlError::UserInfoPresent => Self::UserInfoPresent,
86            IndexUrlError::BlockedHost { class } => Self::BlockedHost { class },
87        }
88    }
89}

91/// A validated, normalized npm registry index URL.
92///
93/// `https`-only (the sole carve-out is a `cfg(test)`/`test-util`-only `http` loopback host),
94/// no userinfo, and `classify_host`/[`RegistryAccessPolicy`]-gated. Deliberately has no
95/// trust-tier concept (unlike `deps-cargo`'s `RegistryIndex`/`IndexTrust`) — npm phase 1
96/// attaches no credential to any request, so there is nothing for a trust tier to gate.
97///
98/// Normalized so `https://npm.pkg.github.com/` and `https://npm.pkg.github.com` produce the
99/// **same** [`Self::as_str`] output — one router entry, not two, and no doubled slash when a
100/// caller splices a package path onto it (`{base}/{name}`).
101#[derive(Debug, Clone, PartialEq, Eq, Hash)]
102pub struct NpmRegistryIndex {
103    /// The validated URL, normalized by stripping trailing `/` characters — see the type
104    /// doc. Always the exact string returned by [`Self::as_str`].
105    normalized: String,
106}
107
108impl NpmRegistryIndex {
109    /// Validates, normalizes, and wraps `raw` — an already `${VAR}`-expanded candidate
110    /// string (expansion happens in [`resolve`], before this is called).
111    ///
112    /// # Errors
113    ///
114    /// Returns [`NpmRegistryIndexError`] if `raw` does not parse as a URL, is not `https`
115    /// (outside the `cfg(test)`/`test-util` loopback carve-out), carries a userinfo
116    /// component, or resolves to a host class the current `policy` blocks.
117    ///
118    /// # Examples
119    ///
120    /// ```
121    /// use deps_core::net_policy::RegistryAccessPolicy;
122    /// use deps_npm::config::NpmRegistryIndex;
123    ///
124    /// let policy = RegistryAccessPolicy::default();
125    /// assert!(NpmRegistryIndex::new("https://npm.pkg.github.com", &policy).is_ok());
126    /// assert!(NpmRegistryIndex::new("http://npm.pkg.github.com", &policy).is_err());
127    /// assert!(NpmRegistryIndex::new("https://user:pass@npm.example", &policy).is_err());
128    /// ```
129    pub fn new(raw: &str, policy: &RegistryAccessPolicy) -> Result<Self, NpmRegistryIndexError> {
130        Self::new_for_log(raw, raw, policy)
131    }
132
133    /// Like [`Self::new`], but validates `expanded` (the candidate to actually parse) while
134    /// using `raw_for_log` — the pre-expansion `.npmrc` value — for every error payload and
135    /// `tracing::warn!` call. [`Self::new`] passes the same string for both, since it has no
136    /// separate pre-expansion form; [`resolve_entry`] passes the true `${VAR}`-expanded value
137    /// alongside the original raw one, so a rejected candidate built from `${SOME_TOKEN}`
138    /// never leaks that token's expanded value into a log line or an
139    /// [`NpmRegistryIndexError::InvalidUrl`] payload — see this module's security-model doc.
140    fn new_for_log(
141        expanded: &str,
142        raw_for_log: &str,
143        policy: &RegistryAccessPolicy,
144    ) -> Result<Self, NpmRegistryIndexError> {
145        let url = validate_index_url(expanded, raw_for_log, "npm", PolicyGate::Enforce(policy))?;
146        let normalized = url.as_str().trim_end_matches('/').to_string();
147        Ok(Self { normalized })
148    }
149
150    /// The normalized index URL — the canonical key for the router's `alternates` map and
151    /// for [`DependencySource::AlternateRegistry`]'s `index`. Never carries a trailing `/`.
152    #[must_use]
153    pub fn as_str(&self) -> &str {
154        &self.normalized
155    }
156}

158impl std::fmt::Display for NpmRegistryIndex {
159    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
160        f.write_str(self.as_str())
161    }
162}

164/// A `registry=`/`@scope:registry=` entry that was present in `.npmrc` but unusable.
165///
166/// Invalid URL, non-https, an undefined `${VAR}`, or policy-blocked. Carries the raw value
167/// as written (never the expanded form, and with any literal userinfo redacted — see
168/// [`deps_core::net_policy::redact_userinfo`]) so [`NpmConfig::resolve_source_for`] can build
169/// [`DependencySource::CustomRegistry`] and so a warning can name what the user actually
170/// wrote, never an expanded-but-rejected value that could leak an environment variable's
171/// contents into the log, and never a literal credential the user wrote directly in `.npmrc`.
172#[derive(Debug, Clone)]
173pub struct InvalidEntry {
174    /// The raw `.npmrc` value, unexpanded, with any literal userinfo redacted.
175    pub raw: String,
176    /// Why it was rejected.
177    pub reason: NpmRegistryIndexError,
178}

180/// The merged, resolved view of a workspace's `.npmrc` hierarchy (project tier overrides
181/// user tier). A plain lookup table with one resolution method — mirrors `deps-cargo`'s
182/// `CargoConfig`.
183#[derive(Debug, Default)]
184pub struct NpmConfig {
185    /// The top-level `registry=` override (FR-003), if any entry (valid or not) was found.
186    registry: Option<Result<NpmRegistryIndex, InvalidEntry>>,
187    /// `@scope:registry=` entries (FR-004), keyed by the scope **including** its leading
188    /// `@`, byte-exact as written in `.npmrc` — no case folding. This matches npm's own
189    /// literal `@${scope}:registry` key lookup: npm lowercases *package names* at publish
190    /// time but never case-folds `.npmrc` keys, so folding here would resolve a registry npm
191    /// itself would not.
192    scoped_registries: HashMap<String, Result<NpmRegistryIndex, InvalidEntry>>,
193}
194
195impl NpmConfig {
196    /// Resolves `package_name` straight to its final [`DependencySource`] (spec FR-003–006).
197    ///
198    /// - FR-004 > FR-003: a matching `@scope:registry=` entry — valid or not — wins outright
199    ///   over the top-level `registry=` override for that dependency.
200    /// - FR-005: no matching entry at all -> [`DependencySource::Registry`] (public default).
201    /// - FR-006/FR-007/FR-008: a matching entry that is present but invalid, unexpandable, or
202    ///   policy-blocked -> [`DependencySource::CustomRegistry`] — never a silent fall back to
203    ///   [`DependencySource::Registry`] (the #248 regression class this closes for npm).
204    #[must_use]
205    pub fn resolve_source_for(&self, package_name: &PackageName) -> DependencySource {
206        if let Some(scope) = scope_of(package_name.as_str())
207            && let Some(result) = self.scoped_registries.get(scope)
208        {
209            return source_from_result(result);
210        }
211        match &self.registry {
212            Some(result) => source_from_result(result),
213            None => DependencySource::Registry,
214        }
215    }
216
217    /// Every successfully resolved [`NpmRegistryIndex`] this config carries (the top-level
218    /// override plus every scoped entry), deduplicated by [`NpmRegistryIndex::as_str`] — fed
219    /// to `NpmRegistry::register_alternate` at parse time. An invalid/unresolved entry
220    /// contributes nothing here: there is no client to register for it.
221    #[must_use]
222    pub fn resolved_registries(&self) -> Vec<NpmRegistryIndex> {
223        let mut seen = std::collections::HashSet::new();
224        self.registry
225            .iter()
226            .chain(self.scoped_registries.values())
227            .filter_map(|result| result.as_ref().ok())
228            .filter(|index| seen.insert(index.as_str().to_string()))
229            .cloned()
230            .collect()
231    }
232}
233
234fn source_from_result(result: &Result<NpmRegistryIndex, InvalidEntry>) -> DependencySource {
235    match result {
236        Ok(index) => DependencySource::AlternateRegistry {
237            index: index.as_str().to_string(),
238            mirrors_crates_io: false,
239        },
240        Err(invalid) => DependencySource::CustomRegistry {
241            url: invalid.raw.clone(),
242        },
243    }
244}

246/// The scope (including its leading `@`) of a package name, e.g. `"@myorg"` for
247/// `"@myorg/pkg"`. `None` for an unscoped name.
248fn scope_of(name: &str) -> Option<&str> {
249    name.split_once('/')
250        .map(|(scope, _)| scope)
251        .filter(|scope| scope.starts_with('@'))
252}

254/// One `.npmrc` file's raw (unvalidated, unexpanded), recognized entries — see
255/// [`parse_npmrc_raw`]'s doc for exactly which keys this can ever contain.
256#[derive(Debug, Default, Clone)]
257struct RawNpmrc {
258    registry: Option<String>,
259    scoped: HashMap<String, String>,
260}
261
262/// Parses one `.npmrc` file's content into its raw, unvalidated `registry`/
263/// `@scope:registry` entries.
264///
265/// **Recognizes exactly two key shapes** — `registry` and `@<scope>:registry` — and skips
266/// every other line outright, including every auth-shaped key (`_authToken`, `_auth`,
267/// `_password`, `_authIdent`, `always-auth`, any `//<host>/:_*` scoped-credential key) and
268/// every other npm config setting. This is the structural half of FR-013/NFR-001: there is
269/// no code path in this function that can produce a value for anything but those two key
270/// shapes.
271///
272/// Grammar (FR-001): npm's own `.npmrc` INI-like format — `key=value` or `key = value` one
273/// per line, `#`/`;` as full-line comment markers, blank lines ignored. A line with no `=`
274/// (or an unrecognized key) is skipped with a `tracing::warn!`; every other valid line still
275/// applies.
276fn parse_npmrc_raw(content: &str) -> RawNpmrc {
277    let mut out = RawNpmrc::default();
278    for line in content.lines() {
279        let line = line.trim();
280        if line.is_empty() || line.starts_with('#') || line.starts_with(';') {
281            continue;
282        }
283        let Some((key, value)) = line.split_once('=') else {
284            tracing::warn!(line, "skipping malformed .npmrc line: no '='");
285            continue;
286        };
287        let key = key.trim();
288        let value = value.trim();
289
290        if key == "registry" {
291            out.registry = Some(value.to_string());
292        } else if let Some(scope) = key
293            .strip_suffix(":registry")
294            .filter(|scope| scope.starts_with('@'))
295        {
296            out.scoped.insert(scope.to_string(), value.to_string());
297        }
298        // Every other key (including every auth-shaped one) is deliberately ignored — see
299        // this function's doc.
300    }
301    out
302}

304/// Expands every `${VAR}` placeholder in `raw` from the process environment (FR-007).
305///
306/// Returns `Ok(expanded)` when every referenced variable is set (including the trivial case
307/// of no `${...}` placeholder at all), or `Err(var_name)` naming the first undefined
308/// variable encountered.
309fn expand_env_vars(raw: &str) -> Result<String, String> {
310    expand_env_vars_with(raw, |name| std::env::var(name).ok())
311}
312
313/// [`expand_env_vars`], but reading variables through `lookup` instead of
314/// [`std::env::var`] directly — lets tests inject a fake environment instead of mutating the
315/// real process environment (this workspace forbids `unsafe`, and Rust 2024 made
316/// `std::env::set_var` an `unsafe fn`, so a test cannot do that mutation at all).
317fn expand_env_vars_with(
318    raw: &str,
319    lookup: impl Fn(&str) -> Option<String>,
320) -> Result<String, String> {
321    let mut out = String::with_capacity(raw.len());
322    let mut rest = raw;
323    while let Some(start) = rest.find("${") {
324        out.push_str(&rest[..start]);
325        let after = &rest[start + 2..];
326        let Some(end) = after.find('}') else {
327            // No closing brace: keep the rest of the string literal, same as npm's own
328            // parser does for a malformed placeholder.
329            out.push_str(&rest[start..]);
330            rest = "";
331            break;
332        };
333        let var_name = &after[..end];
334        match lookup(var_name) {
335            Some(value) => out.push_str(&value),
336            None => return Err(var_name.to_string()),
337        }
338        rest = &after[end + 1..];
339    }
340    out.push_str(rest);
341    Ok(out)
342}

344/// Expands and validates one raw `.npmrc` value, producing the [`NpmConfig`] entry FR-006's
345/// fail-closed state needs on failure — a `tracing::warn!` naming the **raw**, unexpanded
346/// value either way (an expanded-but-rejected value must never leak an environment
347/// variable's contents into the log), with any literal `user:pass@`/`user@` userinfo written
348/// directly in that raw value redacted first (M1 fix — see
349/// [`deps_core::net_policy::redact_userinfo`]) — a hostile `.npmrc` can write a credential
350/// straight into `registry=`/`@scope:registry=` with no `${VAR}` expansion involved.
351fn resolve_entry(
352    raw: &str,
353    policy: &RegistryAccessPolicy,
354) -> Result<NpmRegistryIndex, InvalidEntry> {
355    match expand_env_vars(raw) {
356        Ok(expanded) => NpmRegistryIndex::new_for_log(&expanded, raw, policy).map_err(|reason| {
357            let redacted = redact_userinfo(raw);
358            tracing::warn!(raw = %redacted, %reason, "npm registry index failed validation");
359            InvalidEntry {
360                raw: redacted,
361                reason,
362            }
363        }),
364        Err(var) => {
365            let redacted = redact_userinfo(raw);
366            tracing::warn!(
367                raw = %redacted,
368                var,
369                "npm registry value references an undefined environment variable"
370            );
371            Err(InvalidEntry {
372                raw: redacted,
373                reason: NpmRegistryIndexError::UndefinedEnvVar(var),
374            })
375        }
376    }
377}

379/// Per-`.npmrc`-file-path memoization (FR-012), mirroring `deps-cargo::config::ConfigFileCache`
380/// exactly in shape.
381///
382/// Caches **raw, unvalidated** entries — `${VAR}` expansion, [`NpmRegistryIndex::new`]
383/// validation, and policy gating all re-run **per parse** against these cached entries,
384/// never cached themselves. This is what makes a `didChangeConfiguration` policy change, or
385/// an environment-variable change, take effect immediately with no cache invalidation of its
386/// own. There is no workspace-root key: npm has no workspace-root concept for config
387/// discovery (`NpmParseResult::workspace_root()` returns `None`), and neither does Cargo's
388/// config cache. A thin newtype over [`deps_core::MtimeFileCache`] — the mtime-gated caching
389/// mechanism itself lives there, shared with `deps-cargo`.
390#[derive(Debug)]
391pub struct NpmConfigCache(deps_core::MtimeFileCache<RawNpmrc>);
392
393impl Default for NpmConfigCache {
394    fn default() -> Self {
395        Self::new()
396    }
397}
398
399impl NpmConfigCache {
400    /// Creates an empty cache.
401    #[must_use]
402    pub fn new() -> Self {
403        Self(deps_core::MtimeFileCache::new(
404            deps_core::DEFAULT_MAX_CACHED_FILES,
405            "npm config",
406        ))
407    }
408
409    /// Returns `path`'s parsed contents, from cache if `path`'s mtime is unchanged, else
410    /// re-reading and re-parsing. `None` if `path` does not exist, is not a regular file, or
411    /// cannot be read.
412    fn get_or_parse(&self, path: &Path) -> Option<Arc<RawNpmrc>> {
413        self.0.get_or_parse(path, parse_npmrc_raw)
414    }
415}

431/// Resolves `manifest_dir`'s `.npmrc` hierarchy (project tier, ancestor-walked, plus the
432/// user tier) into a merged [`NpmConfig`] (spec FR-002).
433///
434/// # Examples
435///
436/// ```
437/// use deps_core::net_policy::RegistryAccessPolicy;
438/// use deps_npm::config::{NpmConfigCache, resolve};
439/// use std::path::Path;
440///
441/// let cache = NpmConfigCache::new();
442/// let policy = RegistryAccessPolicy::default();
443/// let config = resolve(Path::new("/nonexistent/workspace"), &cache, &policy);
444/// assert!(config.resolved_registries().is_empty());
445/// ```
446#[must_use]
447pub fn resolve(
448    manifest_dir: &Path,
449    config_cache: &NpmConfigCache,
450    policy: &RegistryAccessPolicy,
451) -> NpmConfig {
452    resolve_with_home(manifest_dir, config_cache, policy, dirs::home_dir())
453}
454
455/// [`resolve`], but taking the user-tier home directory explicitly instead of
456/// [`dirs::home_dir`] — lets tests inject a fixture home directory.
457fn resolve_with_home(
458    manifest_dir: &Path,
459    config_cache: &NpmConfigCache,
460    policy: &RegistryAccessPolicy,
461    home: Option<PathBuf>,
462) -> NpmConfig {
463    let user_npmrc_path = home.map(|h| h.join(".npmrc"));
464    // M9: a project living under `$HOME` has `$HOME` as an ancestor, so the project-tier
465    // walk below would otherwise find `~/.npmrc` a second time as a (wrongly
466    // outranking-itself) workspace-tier entry. Deduped by canonicalized path so a symlinked
467    // home is caught too, mirroring `deps-cargo::config::load_tiers`.
468    let user_canonical = user_npmrc_path
469        .as_deref()
470        .and_then(|p| std::fs::canonicalize(p).ok());
471
472    let mut registry_raw: Option<String> = None;
473    let mut scoped_raw: HashMap<String, String> = HashMap::new();
474
475    // FR-002: this ancestor walk is a deliberate superset of npm's own behavior (which reads
476    // only the project-root `.npmrc`, not every ancestor) — chosen for monorepo ergonomics,
477    // mirroring `deps-cargo`'s `.cargo/config.toml` discovery. Closest directory wins.
478    let mut current = Some(manifest_dir);
479    let mut depth = 0usize;
480    while let Some(dir) = current {
481        if depth >= MAX_CONFIG_ANCESTOR_DEPTH {
482            break;
483        }
484        depth += 1;
485
486        let candidate = dir.join(".npmrc");
487        let is_user_tier_duplicate =
488            std::fs::canonicalize(&candidate).ok().as_deref() == user_canonical.as_deref();
489        if !is_user_tier_duplicate && let Some(parsed) = config_cache.get_or_parse(&candidate) {
490            if registry_raw.is_none() {
491                registry_raw.clone_from(&parsed.registry);
492            }
493            for (scope, raw) in &parsed.scoped {
494                scoped_raw
495                    .entry(scope.clone())
496                    .or_insert_with(|| raw.clone());
497            }
498        }
499
500        current = dir.parent();
501    }
502
503    if let Some(user_path) = user_npmrc_path.as_deref()
504        && let Some(parsed) = config_cache.get_or_parse(user_path)
505    {
506        if registry_raw.is_none() {
507            registry_raw.clone_from(&parsed.registry);
508        }
509        for (scope, raw) in &parsed.scoped {
510            scoped_raw
511                .entry(scope.clone())
512                .or_insert_with(|| raw.clone());
513        }
514    }
515
516    NpmConfig {
517        registry: registry_raw.map(|raw| resolve_entry(&raw, policy)),
518        scoped_registries: scoped_raw
519            .into_iter()
520            .map(|(scope, raw)| (scope, resolve_entry(&raw, policy)))
521            .collect(),
522    }
523}

669    /// FR-013/NFR-001: every auth-shaped key shape is skipped entirely, never landing in the
670    /// parsed struct in any form.
671    #[test]
672    fn test_parse_npmrc_raw_skips_auth_shaped_keys() {
673        let content = concat!(
674            "_authToken=super-secret-token\n",
675            "_auth=another-secret\n",
676            "_password=hunter2\n",
677            "_authIdent=me:hunter2\n",
678            "always-auth=true\n",
679            "//registry.example.com/:_authToken=scoped-secret\n",
680            "//registry.example.com/:_password=scoped-password\n",
681            "registry=https://npm.example/\n",
682        );
683        let raw = parse_npmrc_raw(content);
684        assert_eq!(raw.registry.as_deref(), Some("https://npm.example/"));
685        assert!(raw.scoped.is_empty());
686
687        // Structural guarantee: `RawNpmrc` (and therefore `NpmConfig`) has no field capable
688        // of holding any of the values above at all — this assertion is the closest a test
689        // can get to proving that without reading the source, by confirming none of the
690        // secret strings appears anywhere in the parsed struct's debug output.
691        //
692        // The assert message deliberately names only the *label*, never the secret value
693        // itself — echoing the literal fixture string into a panic/format output is exactly
694        // the "secret in a log" pattern this test exists to rule out, and CodeQL's cleartext-
695        // logging query flags it even in test code that's asserting the value's *absence*.
696        let debug = format!("{raw:?}");
697        for (label, secret) in [
698            ("_authToken", "super-secret-token"),
699            ("_auth", "another-secret"),
700            ("_password", "hunter2"),
701            ("scoped _authToken", "scoped-secret"),
702            ("scoped _password", "scoped-password"),
703        ] {
704            assert!(
705                !debug.contains(secret),
706                "{label}'s value leaked into parsed config Debug output"
707            );
708        }
709    }

760    /// S-1 regression: a rejected entry's error must report the raw `${VAR}`-referencing
761    /// text, never the expanded value — an expanded value can carry an environment
762    /// variable's contents (e.g. a token embedded in a query string), so leaking it into
763    /// `NpmRegistryIndexError::BlockedHost`'s `tracing::warn!` or into `InvalidUrl`'s payload
764    /// would defeat the whole point of never logging the expanded form (this module's
765    /// security-model doc, `InvalidEntry`'s doc). `expand_env_vars` cannot be exercised here
766    /// without mutating the real process environment (forbidden — see
767    /// `expand_env_vars_with`'s doc), so this drives `new_for_log` directly with the two
768    /// strings `resolve_entry` would have passed it after a real `${VAR}` expansion.
769    #[test]
770    fn test_new_for_log_blocked_host_reports_raw_not_expanded() {
771        let expanded_secret = "https://127.0.0.1:9999/?token=super-secret-value";
772        let raw_placeholder = "${SECRET_REGISTRY_URL}";
773        let policy = public_only_policy();
774
775        // `BlockedHost`'s own `Display` never carries the URL at all (only the host
776        // class) — the actual leak this guards against is the `tracing::warn!` emitted
777        // from inside `new_for_log`, so this must inspect the captured log line itself,
778        // not just the returned error's rendering.
779        let log = deps_core::test_util::capture_tracing_output(|| {
780            let err = NpmRegistryIndex::new_for_log(expanded_secret, raw_placeholder, &policy)
781                .unwrap_err();
782            assert_matches!(err, NpmRegistryIndexError::BlockedHost { .. });
783        });
784
785        assert!(
786            !log.contains("super-secret-value"),
787            "leaked expanded secret into tracing output: {log:?}"
788        );
789        assert!(
790            log.contains(raw_placeholder),
791            "expected the raw placeholder in tracing output: {log:?}"
792        );
793    }

1124        std::fs::write(&path, "registry=https://npm.one.example\n").unwrap();

1128
1129        // Ensure a distinguishable mtime on filesystems with coarse timestamp resolution.
1130        let future = SystemTime::now() + std::time::Duration::from_secs(2);
1131        std::fs::write(&path, "registry=https://npm.two.example\n").unwrap();
1132        // `File::open` is read-only, which lacks `FILE_WRITE_ATTRIBUTES` on Windows and
1133        // makes `set_modified` fail with `PermissionDenied`; open for write instead.
1134        std::fs::OpenOptions::new()
1135            .write(true)
1136            .open(&path)
1137            .unwrap()
1138            .set_modified(future)
1139            .unwrap();
1140
1141        let second = cache.get_or_parse(&path).unwrap();
1142        assert_eq!(second.registry.as_deref(), Some("https://npm.two.example"));
1143        assert!(!Arc::ptr_eq(&first, &second));
