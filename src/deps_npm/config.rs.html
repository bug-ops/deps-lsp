1//! `.npmrc` discovery and `registry`/`@scope:registry` resolution.
2//!
3//! Resolves a dependency's npm registry the way npm itself does — a top-level `registry=`
4//! override for unscoped dependencies, or an `@scope:registry=` entry for a scoped
5//! dependency's own scope — by reading the same two-tier `.npmrc` hierarchy npm consults:
6//! the project tier (walked from the opened `package.json`'s directory up to the filesystem
7//! root, mirroring `deps-cargo`'s `.cargo/config.toml` ancestor walk) and the user tier
8//! (`~/.npmrc`). The global tier (`$PREFIX/etc/npmrc`) is out of scope (spec FR-002/FR-014).
9//!
10//! # Security model (read before touching this module)
11//!
12//! A workspace's own `.npmrc` is attacker-controlled the moment a hostile repository is
13//! cloned and opened — this LSP parses on file open, before any build ever runs. Phase 1
14//! carries **no** authentication at all (spec Out of Scope), which closes the credential
15//! half of the threat model this module would otherwise have to solve, but two things still
16//! apply:
17//!
18//! - **No auth-shaped key is ever parsed** (FR-013/NFR-001). `parse_npmrc_raw` recognizes
19//!   exactly two key shapes — `registry` and `@<scope>:registry` — and skips every other
20//!   line outright. There is no `HashMap<String, String>` "everything else" table anywhere
21//!   in this module for an `_authToken`/`_auth`/`_password`/`_authIdent`/`always-auth`/
22//!   `//host/:_*` value to land in even accidentally. This is a structural guarantee, not a
23//!   runtime filter — verified by this module's own NFR-001 test.
24//! - **A literal `user:pass@`/`user@` userinfo component or query-string credential written
25//!   directly in a `registry=`/`@scope:registry=` value is redacted before it can leak** (M1
26//!   fix, mirroring `deps-pypi`'s own M1, extended by #767 S2b). No `${VAR}` expansion is
27//!   needed for this case — a hostile `.npmrc` can write the credential straight into the raw
28//!   value — so `resolve_entry` redacts it (see
29//!   [`deps_core::net_policy::RedactedUrl`]) before it ever reaches a `tracing::warn!`
30//!   call or [`InvalidEntry::raw`], which [`NpmConfig::resolve_source_for`] can surface as
31//!   [`DependencySource::CustomRegistry`]'s `url` in hover/diagnostics text.
32//! - **Internal-network reachability (SSRF-adjacent).** [`NpmRegistryIndex::new`] requires a
33//!   [`deps_core::net_policy::RegistryAccessPolicy`] and checks every candidate against it —
34//!   unlike Cargo's `$CARGO_HOME`-is-trusted split, npm's project and user tiers are
35//!   policy-symmetric (spec NFR-003(a)): phase 1 has no credential provenance to protect, so
36//!   there is no tier that is "the user's own configuration" in the way `$CARGO_HOME` is.
37//!
38//! See `specs/032-npm-npmrc-registry-support/spec.md` FR-001–FR-014 and
39//! `specs/032-npm-npmrc-registry-support/plan.md` §1/§3/§6 for the design review this module
40//! implements.

52/// Why a candidate `registry=`/`@scope:registry=` value failed [`NpmRegistryIndex::new`]'s
53/// validation, or why expansion of a `${VAR}` placeholder inside it failed (FR-007).
54#[non_exhaustive]
55#[derive(Debug, Clone, thiserror::Error, PartialEq, Eq)]
56pub enum NpmRegistryIndexError {
57    /// The value did not parse as a URL at all.
58    #[error("not a valid URL: {0}")]
59    InvalidUrl(RedactedUrl),
60    /// The URL's scheme is not `https` (the sole carve-out is a `cfg(test)`/`test-util`-only
61    /// `http` loopback host — see [`NpmRegistryIndex::new`]).
62    #[error("registry index must use https, got scheme {0:?}")]
63    NotHttps(String),
64    /// The URL carries a `user:pass@`/`user@` component.
65    #[error("registry index URL must not carry userinfo")]
66    UserInfoPresent,
67    /// The candidate's host is blocked by the current
68    /// [`deps_core::net_policy::WorkspaceRegistryAccess`] policy.
69    #[error("registry index host class {class} blocked by registries.workspace_registries policy")]
70    BlockedHost {
71        /// The blocked host's classification.
72        class: HostClass,
73    },
74    /// A `${VAR}` placeholder in the value names an environment variable that is not set
75    /// (FR-007) — the whole value is invalid, never fetched as the literal
76    /// `${VAR}`-containing string.
77    #[error("environment variable {0:?} referenced in registry value is not set")]
78    UndefinedEnvVar(String),
79}
80
81impl From<IndexUrlError> for NpmRegistryIndexError {
82    fn from(error: IndexUrlError) -> Self {
83        match error {
84            // `raw` is already a `RedactedUrl` (issue #808) — no further redaction needed.
85            IndexUrlError::InvalidUrl(raw) => Self::InvalidUrl(raw),
86            IndexUrlError::NotHttps(scheme) => Self::NotHttps(scheme),
87            IndexUrlError::UserInfoPresent => Self::UserInfoPresent,
88            IndexUrlError::BlockedHost { class } => Self::BlockedHost { class },
89            // `IndexUrlError` is `#[non_exhaustive]` (issue #769): a variant added upstream
90            // and not yet mapped here still surfaces, carrying its own message, rather than
91            // failing to compile.
92            other => Self::InvalidUrl(other.to_string().into()),
93        }
94    }
95}

97/// A validated, normalized npm registry index URL.
98///
99/// `https`-only (the sole carve-out is a `cfg(test)`/`test-util`-only `http` loopback host),
100/// no userinfo, and `classify_host`/[`RegistryAccessPolicy`]-gated. Deliberately has no
101/// trust-tier concept (unlike `deps-cargo`'s `RegistryIndex`/`IndexTrust`) — npm phase 1
102/// attaches no credential to any request, so there is nothing for a trust tier to gate.
103///
104/// Normalized so `https://npm.pkg.github.com/` and `https://npm.pkg.github.com` produce the
105/// **same** [`Self::as_str`] output — one router entry, not two, and no doubled slash when a
106/// caller splices a package path onto it (`{base}/{name}`).
107#[derive(Debug, Clone, PartialEq, Eq, Hash)]
108pub struct NpmRegistryIndex {
109    /// The validated URL, normalized by stripping trailing `/` characters — see the type
110    /// doc. Always the exact string returned by [`Self::as_str`].
111    normalized: String,
112}
113
114impl NpmRegistryIndex {
115    /// Validates, normalizes, and wraps `raw` — an already `${VAR}`-expanded candidate
116    /// string (expansion happens in [`resolve`], before this is called).
117    ///
118    /// # Errors
119    ///
120    /// Returns [`NpmRegistryIndexError`] if `raw` does not parse as a URL, is not `https`
121    /// (outside the `cfg(test)`/`test-util` loopback carve-out), carries a userinfo
122    /// component, or resolves to a host class the current `policy` blocks.
123    ///
124    /// # Examples
125    ///
126    /// ```
127    /// use deps_core::net_policy::RegistryAccessPolicy;
128    /// use deps_npm::config::NpmRegistryIndex;
129    ///
130    /// let policy = RegistryAccessPolicy::default();
131    /// assert!(NpmRegistryIndex::new("https://npm.pkg.github.com", &policy).is_ok());
132    /// assert!(NpmRegistryIndex::new("http://npm.pkg.github.com", &policy).is_err());
133    /// assert!(NpmRegistryIndex::new("https://user:pass@npm.example", &policy).is_err());
134    /// ```
135    pub fn new(raw: &str, policy: &RegistryAccessPolicy) -> Result<Self, NpmRegistryIndexError> {
136        Self::new_for_log(raw, raw, policy)
137    }
138
139    /// Like [`Self::new`], but validates `expanded` (the candidate to actually parse) while
140    /// using `raw_for_log` — the pre-expansion `.npmrc` value — for every error payload and
141    /// `tracing::warn!` call. [`Self::new`] passes the same string for both, since it has no
142    /// separate pre-expansion form; [`resolve_entry`] passes the true `${VAR}`-expanded value
143    /// alongside the original raw one, so a rejected candidate built from `${SOME_TOKEN}`
144    /// never leaks that token's expanded value into a log line or an
145    /// [`NpmRegistryIndexError::InvalidUrl`] payload — see this module's security-model doc.
146    /// `raw_for_log` is always redacted the same way regardless of which caller it came from
147    /// (`validate_index_url` applies `url_for_tracing` unconditionally) — an earlier revision
148    /// tried to log the pre-expansion form more permissively, on the theory that it could only
149    /// ever spell a placeholder like `${VAR}`, never a real secret, but that assumption does
150    /// not hold: a hostile `.npmrc` can write a credential directly into the raw value with no
151    /// expansion involved at all (#767 S2b, matching this module's own M1 threat-model note
152    /// above about literal userinfo).
153    fn new_for_log(
154        expanded: &str,
155        raw_for_log: &str,
156        policy: &RegistryAccessPolicy,
157    ) -> Result<Self, NpmRegistryIndexError> {
158        let url = validate_index_url(expanded, raw_for_log, "npm", PolicyGate::Enforce(policy))?;
159        let normalized = url.as_str().trim_end_matches('/').to_string();
160        Ok(Self { normalized })
161    }
162
163    /// The normalized index URL — the canonical key for the router's `alternates` map and
164    /// for [`DependencySource::AlternateRegistry`]'s `index`. Never carries a trailing `/`.
165    #[must_use]
166    pub fn as_str(&self) -> &str {
167        &self.normalized
168    }
169}

171impl std::fmt::Display for NpmRegistryIndex {
172    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
173        f.write_str(self.as_str())
174    }
175}

177/// A `registry=`/`@scope:registry=` entry that was present in `.npmrc` but unusable.
178///
179/// Invalid URL, non-https, an undefined `${VAR}`, or policy-blocked. Carries the raw value
180/// as written (never the expanded form, and with any literal userinfo/query string
181/// redacted — see [`deps_core::net_policy::RedactedUrl`]) so
182/// [`NpmConfig::resolve_source_for`] can build [`DependencySource::CustomRegistry`] and so a
183/// warning can name what the user actually wrote, never an expanded-but-rejected value that
184/// could leak an environment variable's contents into the log, and never a literal credential
185/// the user wrote directly in `.npmrc`.
186///
187/// Output-only: constructed internally by this module's own resolution logic, never by
188/// external code — no constructor is provided.
189#[non_exhaustive]
190#[derive(Debug, Clone)]
191pub struct InvalidEntry {
192    /// The raw `.npmrc` value, unexpanded, with any literal userinfo/query string redacted.
193    pub raw: RedactedUrl,
194    /// Why it was rejected.
195    pub reason: NpmRegistryIndexError,
196}

198/// The merged, resolved view of a workspace's `.npmrc` hierarchy (project tier overrides
199/// user tier). A plain lookup table with one resolution method — mirrors `deps-cargo`'s
200/// `CargoConfig`.
201#[derive(Debug, Default)]
202pub struct NpmConfig {
203    /// The top-level `registry=` override (FR-003), if any entry (valid or not) was found.
204    registry: Option<Result<NpmRegistryIndex, InvalidEntry>>,
205    /// `@scope:registry=` entries (FR-004), keyed by the scope **including** its leading
206    /// `@`, byte-exact as written in `.npmrc` — no case folding. This matches npm's own
207    /// literal `@${scope}:registry` key lookup: npm lowercases *package names* at publish
208    /// time but never case-folds `.npmrc` keys, so folding here would resolve a registry npm
209    /// itself would not.
210    scoped_registries: HashMap<String, Result<NpmRegistryIndex, InvalidEntry>>,
211}
212
213impl NpmConfig {
214    /// Resolves `package_name` straight to its final [`DependencySource`] (spec FR-003–006).
215    ///
216    /// - FR-004 > FR-003: a matching `@scope:registry=` entry — valid or not — wins outright
217    ///   over the top-level `registry=` override for that dependency.
218    /// - FR-005: no matching entry at all -> [`DependencySource::Registry`] (public default).
219    /// - FR-006/FR-007/FR-008: a matching entry that is present but invalid, unexpandable, or
220    ///   policy-blocked -> [`DependencySource::CustomRegistry`] — never a silent fall back to
221    ///   [`DependencySource::Registry`] (the #248 regression class this closes for npm).
222    #[must_use]
223    pub fn resolve_source_for(&self, package_name: &PackageName) -> DependencySource {
224        if let Some(scope) = scope_of(package_name.as_str())
225            && let Some(result) = self.scoped_registries.get(scope)
226        {
227            return source_from_result(result);
228        }
229        match &self.registry {
230            Some(result) => source_from_result(result),
231            None => DependencySource::Registry,
232        }
233    }
234
235    /// Every successfully resolved [`NpmRegistryIndex`] this config carries (the top-level
236    /// override plus every scoped entry), deduplicated by [`NpmRegistryIndex::as_str`] — fed
237    /// to `NpmRegistry::register_alternate` at parse time. An invalid/unresolved entry
238    /// contributes nothing here: there is no client to register for it.
239    #[must_use]
240    pub fn resolved_registries(&self) -> Vec<NpmRegistryIndex> {
241        let mut seen = std::collections::HashSet::new();
242        self.registry
243            .iter()
244            .chain(self.scoped_registries.values())
245            .filter_map(|result| result.as_ref().ok())
246            .filter(|index| seen.insert(index.as_str().to_string()))
247            .cloned()
248            .collect()
249    }
250}
251
252fn source_from_result(result: &Result<NpmRegistryIndex, InvalidEntry>) -> DependencySource {
253    match result {
254        Ok(index) => DependencySource::AlternateRegistry {
255            index: index.as_str().to_string(),
256            mirrors_crates_io: false,
257        },
258        Err(invalid) => DependencySource::CustomRegistry {
259            url: invalid.raw.to_string(),
260        },
261    }
262}

264/// The scope (including its leading `@`) of a package name, e.g. `"@myorg"` for
265/// `"@myorg/pkg"`. `None` for an unscoped name.
266fn scope_of(name: &str) -> Option<&str> {
267    name.split_once('/')
268        .map(|(scope, _)| scope)
269        .filter(|scope| scope.starts_with('@'))
270}

272/// One `.npmrc` file's raw (unvalidated, unexpanded), recognized entries — see
273/// [`parse_npmrc_raw`]'s doc for exactly which keys this can ever contain.
274#[derive(Debug, Default, Clone)]
275struct RawNpmrc {
276    registry: Option<String>,
277    scoped: HashMap<String, String>,
278}
279
280/// Parses one `.npmrc` file's content into its raw, unvalidated `registry`/
281/// `@scope:registry` entries.
282///
283/// **Recognizes exactly two key shapes** — `registry` and `@<scope>:registry` — and skips
284/// every other line outright, including every auth-shaped key (`_authToken`, `_auth`,
285/// `_password`, `_authIdent`, `always-auth`, any `//<host>/:_*` scoped-credential key) and
286/// every other npm config setting. This is the structural half of FR-013/NFR-001: there is
287/// no code path in this function that can produce a value for anything but those two key
288/// shapes.
289///
290/// Grammar (FR-001): npm's own `.npmrc` INI-like format — `key=value` or `key = value` one
291/// per line, `#`/`;` as full-line comment markers, blank lines ignored. A line with no `=`
292/// (or an unrecognized key) is skipped with a `tracing::warn!`; every other valid line still
293/// applies.
294fn parse_npmrc_raw(content: &str) -> RawNpmrc {
295    let mut out = RawNpmrc::default();
296    for line in content.lines() {
297        let line = line.trim();
298        if line.is_empty() || line.starts_with('#') || line.starts_with(';') {
299            continue;
300        }
301        let Some((key, value)) = line.split_once('=') else {
302            tracing::warn!(line, "skipping malformed .npmrc line: no '='");
303            continue;
304        };
305        let key = key.trim();
306        let value = value.trim();
307
308        if key == "registry" {
309            out.registry = Some(value.to_string());
310        } else if let Some(scope) = key
311            .strip_suffix(":registry")
312            .filter(|scope| scope.starts_with('@'))
313        {
314            out.scoped.insert(scope.to_string(), value.to_string());
315        }
316        // Every other key (including every auth-shaped one) is deliberately ignored — see
317        // this function's doc.
318    }
319    out
320}

322/// Expands every `${VAR}` placeholder in `raw` from the process environment (FR-007).
323///
324/// Returns `Ok(expanded)` when every referenced variable is set (including the trivial case
325/// of no `${...}` placeholder at all), or `Err(var_name)` naming the first undefined
326/// variable encountered.
327fn expand_env_vars(raw: &str) -> Result<String, String> {
328    expand_env_vars_with(raw, |name| std::env::var(name).ok())
329}
330
331/// [`expand_env_vars`], but reading variables through `lookup` instead of
332/// [`std::env::var`] directly — lets tests inject a fake environment instead of mutating the
333/// real process environment (this workspace forbids `unsafe`, and Rust 2024 made
334/// `std::env::set_var` an `unsafe fn`, so a test cannot do that mutation at all).
335// All indices come from `find("${")`/`find('}')`, both ASCII tokens, so every slice bound
336// is always a char boundary.
337#[allow(clippy::string_slice)]
338fn expand_env_vars_with(
339    raw: &str,
340    lookup: impl Fn(&str) -> Option<String>,
341) -> Result<String, String> {
342    let mut out = String::with_capacity(raw.len());
343    let mut rest = raw;
344    while let Some(start) = rest.find("${") {
345        out.push_str(&rest[..start]);
346        let after = &rest[start + 2..];
347        let Some(end) = after.find('}') else {
348            // No closing brace: keep the rest of the string literal, same as npm's own
349            // parser does for a malformed placeholder.
350            out.push_str(&rest[start..]);
351            rest = "";
352            break;
353        };
354        let var_name = &after[..end];
355        match lookup(var_name) {
356            Some(value) => out.push_str(&value),
357            None => return Err(var_name.to_string()),
358        }
359        rest = &after[end + 1..];
360    }
361    out.push_str(rest);
362    Ok(out)
363}

365/// Expands and validates one raw `.npmrc` value, producing the [`NpmConfig`] entry FR-006's
366/// fail-closed state needs on failure — a `tracing::warn!` naming the **raw**, unexpanded
367/// value either way (an expanded-but-rejected value must never leak an environment
368/// variable's contents into the log), with any literal `user:pass@`/`user@` userinfo and any
369/// query string/fragment written directly in that raw value stripped first (M1 fix, extended
370/// by #767 S2a/S2b, structurally enforced by #789's [`deps_core::net_policy::RedactedUrl`]) —
371/// a hostile `.npmrc` can
372/// write a credential straight into `registry=`/`@scope:registry=` with no `${VAR}` expansion
373/// involved, whether as userinfo or as a query parameter.
374fn resolve_entry(
375    raw: &str,
376    policy: &RegistryAccessPolicy,
377) -> Result<NpmRegistryIndex, InvalidEntry> {
378    match expand_env_vars(raw) {
379        Ok(expanded) => NpmRegistryIndex::new_for_log(&expanded, raw, policy).map_err(|reason| {
380            let redacted = RedactedUrl::new(raw);
381            tracing::warn!(raw = %redacted, %reason, "npm registry index failed validation");
382            InvalidEntry {
383                raw: redacted,
384                reason,
385            }
386        }),
387        Err(var) => {
388            let redacted = RedactedUrl::new(raw);
389            tracing::warn!(
390                raw = %redacted,
391                var,
392                "npm registry value references an undefined environment variable"
393            );
394            Err(InvalidEntry {
395                raw: redacted,
396                reason: NpmRegistryIndexError::UndefinedEnvVar(var),
397            })
398        }
399    }
400}

402/// Per-`.npmrc`-file-path memoization (FR-012), mirroring `deps-cargo::config::ConfigFileCache`
403/// exactly in shape.
404///
405/// Caches **raw, unvalidated** entries — `${VAR}` expansion, [`NpmRegistryIndex::new`]
406/// validation, and policy gating all re-run **per parse** against these cached entries,
407/// never cached themselves. This is what makes a `didChangeConfiguration` policy change, or
408/// an environment-variable change, take effect immediately with no cache invalidation of its
409/// own. There is no workspace-root key: npm has no workspace-root concept for config
410/// discovery (`NpmParseResult::workspace_root()` returns `None`), and neither does Cargo's
411/// config cache. A thin newtype over [`deps_core::MtimeFileCache`] — the mtime-gated caching
412/// mechanism itself lives there, shared with `deps-cargo`.
413#[derive(Debug)]
414pub struct NpmConfigCache(deps_core::MtimeFileCache<RawNpmrc>);
415
416impl Default for NpmConfigCache {
417    fn default() -> Self {
418        Self::new()
419    }
420}
421
422impl NpmConfigCache {
423    /// Creates an empty cache.
424    #[must_use]
425    pub fn new() -> Self {
426        Self(deps_core::MtimeFileCache::new(
427            deps_core::DEFAULT_MAX_CACHED_FILES,
428            "npm config",
429        ))
430    }
431
432    /// Returns `path`'s parsed contents, from cache if `path`'s mtime is unchanged, else
433    /// re-reading and re-parsing. `None` if `path` does not exist, is not a regular file, or
434    /// cannot be read.
435    fn get_or_parse(&self, path: &Path) -> Option<Arc<RawNpmrc>> {
436        self.0.get_or_parse(path, parse_npmrc_raw)
437    }
438}

490/// Resolves `manifest_dir`'s `.npmrc` hierarchy (project tier, ancestor-walked, plus the
491/// user tier) into a merged [`NpmConfig`] (spec FR-002).
492///
493/// # Examples
494///
495/// ```
496/// use deps_core::net_policy::RegistryAccessPolicy;
497/// use deps_npm::config::{NpmConfigCache, resolve};
498/// use std::path::Path;
499///
500/// let cache = NpmConfigCache::new();
501/// let policy = RegistryAccessPolicy::default();
502/// let config = resolve(Path::new("/nonexistent/workspace"), &cache, &policy);
503/// assert!(config.resolved_registries().is_empty());
504/// ```
505#[must_use]
506pub fn resolve(
507    manifest_dir: &Path,
508    config_cache: &NpmConfigCache,
509    policy: &RegistryAccessPolicy,
510) -> NpmConfig {
511    resolve_with_home(manifest_dir, config_cache, policy, dirs::home_dir())
512}

514/// [`resolve`], but taking the user-tier home directory explicitly instead of
515/// [`dirs::home_dir`] — lets tests inject a fixture home directory.
516fn resolve_with_home(
517    manifest_dir: &Path,
518    config_cache: &NpmConfigCache,
519    policy: &RegistryAccessPolicy,
520    home: Option<PathBuf>,
521) -> NpmConfig {
522    let user_npmrc_path = home.map(|h| h.join(".npmrc"));
523    // M9: a project living under `$HOME` has `$HOME` as an ancestor, so the project-tier
524    // walk below would otherwise find `~/.npmrc` a second time as a (wrongly
525    // outranking-itself) workspace-tier entry. Deduped by canonicalized path so a symlinked
526    // home is caught too, mirroring `deps-cargo::config::load_tiers`.
527    let user_canonical = user_npmrc_path
528        .as_deref()
529        .and_then(|p| std::fs::canonicalize(p).ok());
530
531    let mut registry_raw: Option<String> = None;
532    let mut scoped_raw: HashMap<String, String> = HashMap::new();
533
534    // FR-002: this ancestor walk is a deliberate superset of npm's own behavior (which reads
535    // only the project-root `.npmrc`, not every ancestor) — chosen for monorepo ergonomics,
536    // mirroring `deps-cargo`'s `.cargo/config.toml` discovery. Closest directory wins.
537    for dir in deps_core::fs_probe::config_ancestors(manifest_dir) {
538        let candidate = dir.join(".npmrc");
539        let is_user_tier_duplicate =
540            std::fs::canonicalize(&candidate).ok().as_deref() == user_canonical.as_deref();
541        if !is_user_tier_duplicate && let Some(parsed) = config_cache.get_or_parse(&candidate) {
542            if registry_raw.is_none() {
543                registry_raw.clone_from(&parsed.registry);
544            }
545            for (scope, raw) in &parsed.scoped {
546                scoped_raw
547                    .entry(scope.clone())
548                    .or_insert_with(|| raw.clone());
549            }
550        }
551    }
552
553    if let Some(user_path) = user_npmrc_path.as_deref()
554        && let Some(parsed) = config_cache.get_or_parse(user_path)
555    {
556        if registry_raw.is_none() {
557            registry_raw.clone_from(&parsed.registry);
558        }
559        for (scope, raw) in &parsed.scoped {
560            scoped_raw
561                .entry(scope.clone())
562                .or_insert_with(|| raw.clone());
563        }
564    }
565
566    NpmConfig {
567        registry: registry_raw.map(|raw| resolve_entry(&raw, policy)),
568        scoped_registries: scoped_raw
569            .into_iter()
570            .map(|(scope, raw)| (scope, resolve_entry(&raw, policy)))
571            .collect(),
572    }
573}

1281    #[test]
1282    fn test_config_cache_reparses_after_mtime_change() {
1283        // Held per `deps_core::fs_probe::snapshot_guard`'s doc: `NpmConfigCache::get_or_parse`
1284        // transitively touches fs_probe, and this test runs in the same binary as this
1285        // file's own diffing test.
1286        let _guard = deps_core::fs_probe::snapshot_guard();
1287        let dir = tempfile::tempdir().unwrap();
1288        let path = dir.path().join(".npmrc");
1289        std::fs::write(&path, "registry=https://npm.one.example\n").unwrap();
1290        let cache = NpmConfigCache::new();
1291        let first = cache.get_or_parse(&path).unwrap();
1292        assert_eq!(first.registry.as_deref(), Some("https://npm.one.example"));
1293
1294        // Ensure a distinguishable mtime on filesystems with coarse timestamp resolution.
1295        let future = SystemTime::now() + std::time::Duration::from_secs(2);
1296        std::fs::write(&path, "registry=https://npm.two.example\n").unwrap();
1297        // `File::open` is read-only, which lacks `FILE_WRITE_ATTRIBUTES` on Windows and
1298        // makes `set_modified` fail with `PermissionDenied`; open for write instead.
1299        std::fs::OpenOptions::new()
1300            .write(true)
1301            .open(&path)
1302            .unwrap()
1303            .set_modified(future)
1304            .unwrap();
1305
1306        let second = cache.get_or_parse(&path).unwrap();
1307        assert_eq!(second.registry.as_deref(), Some("https://npm.two.example"));
1308        assert!(!Arc::ptr_eq(&first, &second));
1309    }
