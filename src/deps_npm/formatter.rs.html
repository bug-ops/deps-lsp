7/// Precise npm semver range matcher, compiled once per dependency by
8/// [`NpmFormatter::compile_requirement`].
9struct NodeSemverMatcher(node_semver::Range);
10
11impl RequirementMatcher for NodeSemverMatcher {
12    fn matches(&self, version: &ConcreteVersion) -> Option<bool> {
13        let version = version.as_str();
14        node_semver::Version::parse(version)
15            .ok()
16            .map(|v| self.0.satisfies(&v))
17    }
18}

20/// Maximum name length npm's registry accepts.
21///
22/// npm counts UTF-16 code units; this counts Unicode scalar values
23/// (`str::chars().count()`) instead, which undercounts names containing
24/// characters outside the Basic Multilingual Plane. Still strictly more
25/// accurate than a byte-length check for the common case of non-ASCII names
26/// within the BMP.
27const MAX_NAME_LENGTH: usize = 214;
28
29/// Names npm's own validator hard-rejects regardless of character content.
30const BLOCKED_NAMES: [&str; 2] = ["node_modules", "favicon.ico"];
31
32/// Reports whether every character of `segment` is in npm's unreserved set —
33/// the set `encodeURIComponent` leaves untouched (`A-Za-z0-9` plus
34/// `! ' ( ) * - . _ ~`). This mirrors npm's actual
35/// `encodeURIComponent(segment) === segment` check: any other ASCII
36/// punctuation or any non-ASCII character fails it.
37fn is_url_friendly_segment(segment: &str) -> bool {
38    segment.chars().all(|c| {
39        c.is_ascii_alphanumeric()
40            || matches!(c, '!' | '\'' | '(' | ')' | '*' | '-' | '.' | '_' | '~')
41    })
42}

46impl PackageNaming for NpmFormatter {
47    /// Lints `name` against npm's own `validate-npm-package-name` rules.
48    ///
49    /// Deliberately permissive beyond what npm hard-rejects: uppercase letters are
50    /// allowed (npm only warns for legacy packages, never rejects), and any
51    /// character in npm's unreserved set (`! ' ( ) * - . _ ~` plus alphanumerics)
52    /// is accepted, matching npm's `encodeURIComponent(name) === name` check
53    /// exactly rather than an approximation of it.
54    ///
55    /// # Errors
56    ///
57    /// Returns [`InvalidPackageName`] if `name` is empty, exceeds 214 characters,
58    /// starts with `.` or `_`, is a reserved name (`node_modules`, `favicon.ico`),
59    /// has a malformed `@scope/name` structure, or contains a character outside
60    /// npm's unreserved set.
61    fn validate_package_name(&self, name: &str) -> Result<(), InvalidPackageName> {
62        if name.is_empty() {
63            return Err(InvalidPackageName::new("name cannot be empty"));
64        }
65        if name.chars().count() > MAX_NAME_LENGTH {
66            return Err(InvalidPackageName::new(format!(
67                "name cannot exceed {MAX_NAME_LENGTH} characters"
68            )));
69        }
70        if name.starts_with('.') {
71            return Err(InvalidPackageName::new("name cannot start with a period"));
72        }
73        if name.starts_with('_') {
74            return Err(InvalidPackageName::new(
75                "name cannot start with an underscore",
76            ));
77        }
78        if BLOCKED_NAMES
79            .iter()
80            .any(|blocked| name.eq_ignore_ascii_case(blocked))
81        {
82            return Err(InvalidPackageName::new(format!(
83                "'{name}' is a reserved name"
84            )));
85        }
86
87        // Scoped names are `@scope/name`; anything else with a '/' is invalid.
88        let (scope, pkg_name) = match name.split_once('/') {
89            Some((scope, pkg_name)) => {
90                let Some(scope) = scope.strip_prefix('@') else {
91                    return Err(InvalidPackageName::new("unscoped name cannot contain '/'"));
92                };
93                (Some(scope), pkg_name)
94            }
95            None => (None, name),
96        };
97
98        if let Some(scope) = scope {
99            if scope.is_empty() {
100                return Err(InvalidPackageName::new("scope cannot be empty"));
101            }
102            if !is_url_friendly_segment(scope) {
103                return Err(InvalidPackageName::new(
104                    "scope contains characters that are not URL-friendly",
105                ));
106            }
107        }
108
109        if pkg_name.is_empty() {
110            return Err(InvalidPackageName::new("name cannot be empty"));
111        }
112        if pkg_name.contains('/') {
113            return Err(InvalidPackageName::new(
114                "name cannot contain more than one '/'",
115            ));
116        }
117        if !is_url_friendly_segment(pkg_name) {
118            return Err(InvalidPackageName::new(
119                "name contains characters that are not URL-friendly",
120            ));
121        }
122
123        Ok(())
124    }
125}

148impl RequirementResolution for NpmFormatter {
149    /// Compiles `requirement` via `node_semver::Range`, the same crate `deps-npm`'s
150    /// registry uses for matching — precise npm semver range semantics, unlike the
151    /// default `version_satisfies_requirement` heuristic this method deliberately does
152    /// not reuse (see that method's docs).
153    fn compile_requirement(&self, requirement: &VersionReq) -> Option<Box<dyn RequirementMatcher>> {
154        node_semver::Range::parse(requirement.as_str())
155            .ok()
156            .map(|req| Box::new(NodeSemverMatcher(req)) as Box<dyn RequirementMatcher>)
157    }
158}

170impl DiagnosticPolicy for NpmFormatter {
171    /// `node_semver::Range::satisfies` excludes pre-releases unless `requirement` itself pins
172    /// to the same `X.Y.Z` tuple with a pre-release tag — strict SemVer 2.0.0 semantics (#299).
173    fn strict_semver_prerelease_exclusion(&self) -> bool {
174        true
175    }
176
177    /// Disables the manifest-requirement-level "requirement satisfiable only by a yanked
178    /// version" diagnostic entirely (#436, follow-up to #205's plan.md §6): unconditionally
179    /// `false`, for every requirement shape, not only ranges.
180    ///
181    /// npm's `Version::removal_status()` is genuinely per-version data (`npm deprecate` can
182    /// target one version), but `npm deprecate` is routinely applied to *every* published
183    /// version of a package at once (live-verified: the `request` package has all 126/126
184    /// versions marked deprecated) — common enough that this diagnostic would frequently
185    /// duplicate the dedicated package-level deprecation diagnostic
186    /// ([`deps_core::lsp_helpers::DiagnosticMessages::deprecated_message`], issue #205), including for an exact-pin
187    /// requirement (the case this hook used to still allow through). This does not touch
188    /// [`Registry::reports_yanked`](deps_core::Registry::reports_yanked), which npm keeps at
189    /// its default `true`: the independent in-use-version yanked check (#263,
190    /// `crates/deps-core/src/lsp_helpers/diagnostics.rs`) reads real per-version data and
191    /// stays live — e.g. a lockfile-pinned old version flagged by `npm deprecate pkg@"<1.2.3"`
192    /// while `latest` is clean still surfaces its own "yanked" diagnostic.
193    fn yanked_diagnostic_applies_to(
194        &self,
195        _dep: &dyn Dependency,
196        _requirement: &VersionReq,
197    ) -> bool {
198        false
199    }
200}

202impl SourcePolicy for NpmFormatter {
203    /// FR-009: a `.npmrc`-resolved `AlternateRegistry` is resolvable, alongside the default
204    /// public `Registry` — gates hover/diagnostics/code-actions onto the router's per-source
205    /// dispatch (`NpmRegistry::get_versions_from`) instead of the source-blind path.
206    /// `CustomRegistry` (FR-006's fail-closed state) keeps the trait default's `false`.
207    fn can_resolve_source(&self, source: &deps_core::DependencySource) -> bool {
208        matches!(
209            source,
210            deps_core::DependencySource::Registry
211                | deps_core::DependencySource::AlternateRegistry { .. }
212        )
213    }

408    fn test_version_satisfies_requirement() {
409        let formatter = NpmFormatter;
410
411        // Exact match
412        assert!(formatter.version_satisfies_requirement(&ConcreteVersion::new("1.2.3"), "1.2.3"));
413
414        // Partial versions
415        assert!(formatter.version_satisfies_requirement(&ConcreteVersion::new("1.2.3"), "1"));
416        assert!(formatter.version_satisfies_requirement(&ConcreteVersion::new("1.2.3"), "1.2"));
417
418        // Caret - allows any version with same major (for major > 0)
419        assert!(formatter.version_satisfies_requirement(&ConcreteVersion::new("1.2.3"), "^1.2"));
420        assert!(formatter.version_satisfies_requirement(&ConcreteVersion::new("1.2.3"), "^1.0"));
421        assert!(formatter.version_satisfies_requirement(&ConcreteVersion::new("1.5.0"), "^1.2.3"));
422        assert!(
423            formatter.version_satisfies_requirement(&ConcreteVersion::new("10.1.3"), "^10.1.3")
424        ); // Same version
425        assert!(
426            formatter.version_satisfies_requirement(&ConcreteVersion::new("10.2.0"), "^10.1.3")
427        ); // Higher minor
428
429        // Tilde - allows patch changes
430        assert!(formatter.version_satisfies_requirement(&ConcreteVersion::new("1.2.3"), "~1.2"));
431        assert!(formatter.version_satisfies_requirement(&ConcreteVersion::new("1.2.5"), "~1.2"));
432
433        // Should not match
434        assert!(!formatter.version_satisfies_requirement(&ConcreteVersion::new("1.2.3"), "2.0.0"));
435        assert!(!formatter.version_satisfies_requirement(&ConcreteVersion::new("1.2.3"), "1.3"));
436        assert!(!formatter.version_satisfies_requirement(&ConcreteVersion::new("2.0.0"), "^1.2.3")); // Different major
437    }
