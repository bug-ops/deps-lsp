23/// Whether `gha_dep`'s ref is diagnosable as a tag — either because
24/// [`crate::parser::classify_uses_value`] already classified it as [`PinStyle::Tag`] from
25/// its text shape, or because `tag_index`'s live `/tags` fetch confirms the ref is a
26/// literal member of the repository's tag list even though it doesn't *look*
27/// tag-shaped (issue #551, e.g. `taiki-e/install-action@cargo-deny`).
28///
29/// [`crate::parser::is_tag_shaped`] is a pure, registry-blind heuristic — it cannot tell
30/// a literal tool-name tag from a genuinely moving branch, since both fail the same
31/// "starts with `v`/a digit" test. Once the registry has actually answered (`tag_index`
32/// carries this repository's entry), the real answer is available and takes priority
33/// over the static guess; before that (`tag_index` cache miss), a [`PinStyle::Branch`]
34/// step stays classified as the "honest unknown" — exactly the pre-#551 behavior — until
35/// a fetch resolves it one way or the other.
36fn is_registry_confirmed_tag(
37    gha_dep: &GithubActionsDependency,
38    tag_index: &DashMap<PackageName, Arc<TagIndex>>,
39) -> bool {
40    match &gha_dep.pin {
41        Some(PinStyle::Tag) => true,
42        Some(PinStyle::Branch) => gha_dep
43            .version_req
44            .as_ref()
45            .map(deps_core::VersionReq::as_str)
46            .is_some_and(|ref_text| {
47                tag_index
48                    .get(&gha_dep.name)
49                    .is_some_and(|index| index.tag_to_sha.contains_key(ref_text))
50            }),
51        Some(PinStyle::Sha { .. }) | None => false,
52    }
53}

237    /// One documented NFR-004 divergence (S3): appends a `**Resolved**` line naming the
238    /// tag a SHA pin's commit actually corresponds to, per
239    /// [`crate::registry::GithubActionsRegistry`]'s [`crate::registry::TagIndex`].
240    ///
241    /// Necessary, not merely additive: `versions.resolved` (the shared helper's
242    /// `**Current**` source) is keyed by package name and is unconditionally empty for
243    /// GHA (no lockfile provider), so it cannot express per-occurrence resolution when
244    /// the same action is pinned at two different SHAs in one workflow. The splice also
245    /// makes a stale or hand-edited `# vX.Y.Z` comment visible for free (M4): the tag
246    /// shown here comes from `TagIndex.sha_to_tag`, not from trusting the comment text.
247    ///
248    /// Scoped to [`PinStyle::Sha`] only — the SHA is the one pin form GitHub itself does
249    /// not render as a readable version, so it is the only form where naming the tag it
250    /// resolves to adds information; a `PinStyle::Tag` pin already shows the tag text
251    /// directly. Guarded on `dep.version_range().is_some()` (N3): a non-resolvable
252    /// dependency (a reusable-workflow call, `./local`, `docker://…`) still matches the
253    /// shared helper's own hover-target predicate and must not have a `**Resolved**` line
254    /// spliced onto it.
255    fn generate_hover<'a>(
256        &'a self,
257        parse_result: &'a dyn ParseResultTrait,
258        position: Position,
259        versions: deps_core::VersionData<'a>,
260        freshness: deps_core::FreshnessSettings,
261    ) -> deps_core::ecosystem::BoxFuture<'a, Option<Hover>> {
262        Box::pin(async move {
263            let registry = self.registry.clone();
264            let base_hover = deps_core::lsp_generate_hover(
265                parse_result,
266                position,
267                versions,
268                registry.as_ref(),
269                self.formatter(),
270                freshness,
271                deps_core::PublishTime::now(),
272            )
273            .await;
274            let mut hover = base_hover?;
275
276            let dep = parse_result.dependencies().into_iter().find(|d| {
277                deps_core::position_in_range(position, d.name_range())
278                    || d.version_range()
279                        .is_some_and(|r| deps_core::position_in_range(position, r))
280            });
281            let Some(dep) = dep else {
282                return Some(hover);
283            };
284            if dep.version_range().is_none() {
285                return Some(hover);
286            }
287            let Some(gha_dep) = dep.as_any().downcast_ref::<GithubActionsDependency>() else {
288                return Some(hover);
289            };
290
291            // #501 gap (tester finding), widened by critic finding C1 (#550): the shared
292            // `has_offline_actionable_data`/footer gate in `deps_core::generate_hover` only
293            // sees `VersionData` and cannot see `formatter.tag_index` — GHA's "Pin to commit
294            // SHA" quickfix (`build_sha_pin_action`, wired into `generate_code_actions`
295            // above) is driven entirely by that separately-populated index, independent of
296            // `versions`. So a `PinStyle::Tag` step with a warm `TagIndex` entry has a real
297            // `Cmd+.` action even when the shared gate suppressed the footer for lack of any
298            // `VersionData` signal — originally observed offline (#501), but #550's
299            // empty-live-fetch footer gate reintroduced the identical gap *online* too: a
300            // bare-major tag (`@v4`) whose repo tags entirely fail the full-semver filter
301            // (`tags_to_versions`) makes `available_versions == Some([])`, which the shared
302            // gate now (correctly, for the general case) treats as "nothing to update" — but
303            // `populate_tag_index` still indexes bare-major tags (`v4 -> sha`) independent of
304            // that filter, so the quickfix is still genuinely available. Restores the footer
305            // post-hoc in both modes using the shared `CMD_DOT_FOOTER` constant (not a
306            // hand-copied literal) so the two can never drift apart; the
307            // `!content.value.contains(CMD_DOT_FOOTER)` guard below makes this idempotent
308            // when the shared gate already rendered it, so dropping the online/offline split
309            // cannot double-append.
310            // `is_plain_scalar` mirrors `build_sha_pin_action`'s own guard (FR-010, spec
311            // 031): for a quoted `uses:` scalar, `version_range` sits inside the quotes, so
312            // the quickfix withholds itself rather than corrupt the value — the footer must
313            // not advertise an action that was never actually offered.
314            //
315            // Deliberately keyed on the raw `PinStyle::Tag` check, not
316            // `is_registry_confirmed_tag` (#551 plan): this footer only ever advertises
317            // `build_sha_pin_action`, which itself stays on the stricter, pre-#551 guard
318            // (see that function's doc comment) — the footer must never promise an action
319            // that withholds itself.
320            if gha_dep.pin == Some(PinStyle::Tag)
321                && gha_dep.is_plain_scalar
322                && let Some(tag) = gha_dep
323                    .version_req
324                    .as_ref()
325                    .map(deps_core::VersionReq::as_str)
326                && self
327                    .formatter
328                    .sha_pin_replacement_for(&gha_dep.name, tag)
329                    .is_some()
330                && let HoverContents::Markup(content) = &mut hover.contents
331                && !content
332                    .value
333                    .contains(deps_core::lsp_helpers::CMD_DOT_FOOTER)
334            {
335                content
336                    .value
337                    .push_str(deps_core::lsp_helpers::CMD_DOT_FOOTER);
338            }
339
340            let Some(PinStyle::Sha { comment_tag }) = &gha_dep.pin else {
341                return Some(hover);
342            };
343
344            let sha = match comment_tag {
345                // Whitespace-token split, not `split_once(" # ")`: the parser's
346                // comment-tag rule (B3) only requires the `#` to be *preceded* by
347                // whitespace, so `<sha>  # v4.2.0` (two spaces) or `<sha>\t# v4.2.0`
348                // are both valid literals that a single-space exact match would miss
349                // (critic M2).
350                Some(_) => gha_dep
351                    .version_literal
352                    .as_deref()
353                    .and_then(|lit| lit.split_whitespace().next()),
354                None => gha_dep
355                    .version_req
356                    .as_ref()
357                    .map(deps_core::VersionReq::as_str),
358            };
359            let Some(sha) = sha else {
360                return Some(hover);
361            };
362
363            let Some(resolved_tag) = self
364                .formatter
365                .tag_index
366                .get(dep.name())
367                .and_then(|index| index.sha_to_tag.get(sha).cloned())
368            else {
369                return Some(hover);
370            };
371
372            if let HoverContents::Markup(content) = &mut hover.contents {
373                content.value = splice_resolved_line(&content.value, &resolved_tag, sha);
374            }
375
376            Some(hover)
377        })
378    }

440/// Whether line `line_number` of `content` is a workflow `uses:` step key (`uses:
441/// owner/repo@ref` or, as a sequence item, `- uses: owner/repo@ref`), for
442/// `deps-lsp`'s raw-text fallback completion (parse-failure path).
443///
444/// A `uses:` step key can appear at any nesting depth in a workflow file
445/// (`jobs.*.steps[].uses`, `jobs.<id>.uses`), so unlike TOML/JSON/other-YAML
446/// ecosystems there is no enclosing section header to track — the target line itself
447/// is the only signal needed.

449    content
450        .lines()
451        .nth(line_number)
452        .map(str::trim_start)
453        .is_some_and(|trimmed| trimmed.starts_with("uses:") || trimmed.starts_with("- uses:"))
454}

462/// Inserts a `**Resolved**: `tag` (`sha…`)` line immediately after the shared hover's
463/// `**Current**`/`**Requirement**` line (whichever is present), falling back to append
464/// only if neither anchor is found.
465// `pos`/`rel_end`/`insert_at` come from `find` of ASCII anchors (`"**Current**: "`,
466// `"\n\n"`), so all are always char boundaries.
467#[allow(clippy::string_slice)]
468fn splice_resolved_line(markdown: &str, resolved_tag: &str, sha: &str) -> String {
469    // `sha` is expected to be a validated, pure-ASCII full hex SHA by the time it
470    // reaches here (`TagIndex` entries are filtered in `populate_tag_index`, security
471    // S-3), so a byte slice is normally safe — `get(..7)` is a char-boundary-safe
472    // belt-and-braces guard rather than a raw index (security S-4), falling back to
473    // the whole string on anything unexpected instead of panicking.
474    let short_sha = sha.get(..7).unwrap_or(sha);
475    let line = format!(
476        "**Resolved**: {} ({})\n\n",
477        markdown_code_span(resolved_tag),
478        markdown_code_span(&format!("{short_sha}…"))
479    );
480
481    for anchor in ["**Current**: ", "**Requirement**: "] {
482        if let Some(pos) = markdown.find(anchor)
483            && let Some(rel_end) = markdown[pos..].find("\n\n")
484        {
485            let insert_at = pos + rel_end + 2;
486            let mut out = String::with_capacity(markdown.len() + line.len());
487            out.push_str(&markdown[..insert_at]);
488            out.push_str(&line);
489            out.push_str(&markdown[insert_at..]);
490            return out;
491        }
492    }
493    format!("{markdown}{line}")
494}

496/// Builds one mutable-ref-pin [`Diagnostic`] (issue #473) per diagnosable-as-tag step in
497/// `parse_result` — every `PinStyle::Tag` step, plus a `PinStyle::Branch` step
498/// `tag_index` confirms is actually a real tag (issue #551, e.g.
499/// `taiki-e/install-action@cargo-deny`: see [`is_registry_confirmed_tag`]).
500/// `PinStyle::Sha` and a `PinStyle::Branch` `tag_index` cannot (yet) confirm produce no
501/// diagnostic (FR-003).
502/// Maximum character count of `mutable_ref_pin_diagnostics`' interpolated `name`/`tag`
503/// values before truncation (security audit finding). Mirrors
504/// `deps_core::lsp_helpers::diagnostics`' `MAX_BLOCKED_REGISTRY_MESSAGE_VALUE_CHARS`
505/// precedent: nothing upstream caps a workflow file's `owner/repo` or ref text length, so
506/// this is the last chokepoint before either renders inline in the editor, re-sent on
507/// every `publishDiagnostics`.
508const MAX_MUTABLE_REF_PIN_MESSAGE_VALUE_CHARS: usize = 128;
509
510fn mutable_ref_pin_diagnostics(
511    parse_result: &dyn ParseResultTrait,
512    severity: tower_lsp_server::ls_types::DiagnosticSeverity,
513    tag_index: &DashMap<PackageName, Arc<TagIndex>>,
514) -> Vec<Diagnostic> {
515    parse_result
516        .dependencies()
517        .into_iter()
518        .filter_map(|dep| {
519            let gha_dep = dep.as_any().downcast_ref::<GithubActionsDependency>()?;
520            if !is_registry_confirmed_tag(gha_dep, tag_index) {
521                return None;
522            }
523            let range = gha_dep.version_range?;
524            let tag = gha_dep
525                .version_req
526                .as_ref()
527                .map(deps_core::VersionReq::as_str)?;
528            let name = deps_core::lsp_helpers::truncate_for_diagnostic(
529                gha_dep.name.as_str(),
530                MAX_MUTABLE_REF_PIN_MESSAGE_VALUE_CHARS,
531            );
532            let tag = deps_core::lsp_helpers::truncate_for_diagnostic(
533                tag,
534                MAX_MUTABLE_REF_PIN_MESSAGE_VALUE_CHARS,
535            );
536            // Critic finding C2 (#551): `build_sha_pin_action` deliberately stays
537            // restricted to a statically-classified `PinStyle::Tag` (FR-005/plan §11 —
538            // see that function's doc comment on the branch/tag name-collision risk), so
539            // a registry-confirmed-but-`PinStyle::Branch` ref has no automated fix behind
540            // `Cmd+.` here. The message must say so rather than imply one is a keystroke
541            // away, matching every other resolvable-but-unactionable case in this
542            // codebase (e.g. `deps_core`'s own "Registry lookup failed" wording never
543            // promises a quickfix it can't offer).
544            let message = if gha_dep.pin == Some(PinStyle::Tag) {
545                format!(
546                    "{name} is pinned to the mutable tag ref `{tag}`; pin to a full commit \
547                     SHA to guard against tag mutation"
548                )
549            } else {
550                format!(
551                    "{name} is pinned to the mutable tag ref `{tag}`; pin to a full commit \
552                     SHA to guard against tag mutation (manual edit — no automated fix \
553                     available for this ref)"
554                )
555            };
556            Some(Diagnostic {

570/// Builds the "Pin `{name}` to commit SHA" quickfix (issue #473, US-002) for the
571/// `PinStyle::Tag` dependency at `position`, if [`GithubActionsFormatter::sha_pin_replacement_for`]
572/// resolves its current tag against the shared `TagIndex`.
573///
574/// Returns `None` (no destructive/no-op edit, FR-005) when the dependency at `position`
575/// is not `PinStyle::Tag`, has no `version_range`, or the `TagIndex` lookup misses (cache
576/// miss — e.g. the document was opened before the registry fetch completed).
577///
578/// Deliberately **not** widened to [`is_registry_confirmed_tag`]'s `PinStyle::Branch`
579/// case the way [`mutable_ref_pin_diagnostics`] is (#551 plan): FR-005/plan §11 (see
580/// `test_build_sha_pin_action_no_quickfix_for_branch_pin`) already forbids this
581/// quickfix for a `PinStyle::Branch` step even when a same-named `TagIndex` entry
582/// exists, since git permits a branch and a tag to share one name and GitHub's own
583/// `uses:` ref resolution for that collision is undocumented — an *automated edit*
584/// that silently pins to the tag's commit could pin to a different commit than the
585/// ref actually resolves to at run time. A diagnostic's advisory text carries no such
586/// risk (pinning to *some* SHA is safer than a moving ref either way), but this
587/// destructive edit keeps the stricter, pre-#551 guard.
588fn build_sha_pin_action(
589    parse_result: &dyn ParseResultTrait,
590    position: Position,
591    uri: &Uri,
592    formatter: &GithubActionsFormatter,
593) -> Option<CodeAction> {
594    // Same lookup convention every other deps-lsp code action goes through (critic S2) —
595    // not a hand-rolled position check. The default (`version_range` only, GHA does not
596    // override it) is exactly right here: the diagnostic and the edit both anchor on
597    // `version_range`, never `name_range`.
598    let dep = parse_result
599        .dependencies()
600        .into_iter()
601        .find(|d| formatter.is_position_on_dependency(*d, position))?;
602
603    let gha_dep = dep.as_any().downcast_ref::<GithubActionsDependency>()?;
604    let text_edit = sha_pin_text_edit_for(dep, formatter)?;
605    let diagnostic_range = text_edit.range;
606
607    let mut changes = std::collections::HashMap::new();
608    changes.insert(uri.clone(), vec![text_edit]);
609
610    Some(CodeAction {
611        title: format!("Pin {} to commit SHA", gha_dep.name),
612        kind: Some(CodeActionKind::QUICKFIX),
613        edit: Some(WorkspaceEdit {
614            changes: Some(changes),
615            ..Default::default()
616        }),
617        data: Some(serde_json::json!({
618            "diagnostic_codes": [MUTABLE_REF_PIN_DIAGNOSTIC_CODE],
619            "diagnostic_range": diagnostic_range,
620        })),
621        ..Default::default()
622    })
623}

625/// Builds the `{sha} # {tag}` [`TextEdit`] for a single `PinStyle::Tag` step, shared by
626/// [`build_sha_pin_action`] (wraps it into a per-position quickfix) and
627/// [`collect_pin_all_to_sha_edits`] (the bulk "Pin all to SHA" code lens, issue #633).
628///
629/// `None` for anything but a plain-scalar `PinStyle::Tag` step with a resolvable
630/// `TagIndex` entry — the same withholding guards `build_sha_pin_action`'s doc comment
631/// describes (FR-010's quoted-scalar guard, and a `TagIndex` cache miss).
632fn sha_pin_text_edit_for(
633    dep: &dyn deps_core::Dependency,
634    formatter: &GithubActionsFormatter,
635) -> Option<TextEdit> {
636    let gha_dep = dep.as_any().downcast_ref::<GithubActionsDependency>()?;
637    if gha_dep.pin != Some(PinStyle::Tag) {
638        return None;
639    }
640    // FR-010: for a quoted `uses:` scalar, `version_range` sits inside the quotes —
641    // writing `{sha} # {tag}` there corrupts the value instead of adding a YAML comment
642    // (security audit finding, spec 031 FR-010). Withhold rather than risk that edit.
643    if !gha_dep.is_plain_scalar {
644        return None;
645    }
646    // Security audit finding (issue #633): for a `uses:` step written in YAML flow
647    // style (`{uses: actions/checkout@v4, with: {node: 20}}`), appending `# <tag>`
648    // right after the ref comments out the rest of the flow collection, producing
649    // invalid (unterminated) YAML instead of merely leaving the step unpinned.
650    // Withhold rather than risk corrupting the workflow.
651    if !gha_dep.is_last_on_line {
652        return None;
653    }
654    let version_range = gha_dep.version_range?;
655    let tag = gha_dep
656        .version_req
657        .as_ref()
658        .map(deps_core::VersionReq::as_str)?;
659    let new_text = formatter.sha_pin_replacement_for(&gha_dep.name, tag)?;
660    Some(TextEdit {
661        range: version_range,
662        new_text,
663    })
664}

666/// Builds one [`TextEdit`] per `PinStyle::Tag` step in `parse_result` resolvable to a
667/// commit SHA via `formatter`'s `TagIndex` — the bulk counterpart to
668/// [`build_sha_pin_action`]'s single-step quickfix (issue #633). A step with no
669/// resolvable `TagIndex` entry (cache miss) is silently skipped, exactly like that
670/// quickfix's own withholding behavior — never blocking on, or triggering, a fetch.
671fn collect_pin_all_to_sha_edits(
672    parse_result: &dyn ParseResultTrait,
673    formatter: &GithubActionsFormatter,
674) -> Vec<TextEdit> {
675    let edits: Vec<TextEdit> = parse_result
676        .dependencies()
677        .into_iter()
678        .filter_map(|dep| sha_pin_text_edit_for(dep, formatter))
679        .collect();
680    deps_core::lsp_helpers::dedup_overlapping_edits(edits, "collect_pin_all_to_sha_edits")
681}
