23/// Whether `gha_dep`'s ref is diagnosable as a tag — either because
24/// [`crate::parser::classify_uses_value`] already classified it as [`PinStyle::Tag`] from
25/// its text shape, or because `tag_index`'s live `/tags` fetch confirms the ref is a
26/// literal member of the repository's tag list even though it doesn't *look*
27/// tag-shaped (issue #551, e.g. `taiki-e/install-action@cargo-deny`).
28///
29/// [`crate::parser::is_tag_shaped`] is a pure, registry-blind heuristic — it cannot tell
30/// a literal tool-name tag from a genuinely moving branch, since both fail the same
31/// "starts with `v`/a digit" test. Once the registry has actually answered (`tag_index`
32/// carries this repository's entry), the real answer is available and takes priority
33/// over the static guess; before that (`tag_index` cache miss), a [`PinStyle::Branch`]
34/// step stays classified as the "honest unknown" — exactly the pre-#551 behavior — until
35/// a fetch resolves it one way or the other.
36fn is_registry_confirmed_tag(
37    gha_dep: &GithubActionsDependency,
38    tag_index: &DashMap<PackageName, Arc<TagIndex>>,
39) -> bool {
40    match &gha_dep.pin {
41        Some(PinStyle::Tag) => true,
42        Some(PinStyle::Branch) => gha_dep
43            .version_req
44            .as_ref()
45            .map(deps_core::VersionReq::as_str)
46            .is_some_and(|ref_text| {
47                tag_index
48                    .get(&gha_dep.name)
49                    .is_some_and(|index| index.tag_to_sha.contains_key(ref_text))
50            }),
51        Some(PinStyle::Sha { .. }) | None => false,
52    }
53}

286    /// One documented NFR-004 divergence (S3): appends a `**Resolved**` line naming the
287    /// tag a SHA pin's commit actually corresponds to, per
288    /// [`crate::registry::GithubActionsRegistry`]'s [`crate::registry::TagIndex`].
289    ///
290    /// Necessary, not merely additive: `versions.resolved` (the shared helper's
291    /// `**Current**` source) is keyed by package name and is unconditionally empty for
292    /// GHA (no lockfile provider), so it cannot express per-occurrence resolution when
293    /// the same action is pinned at two different SHAs in one workflow. The splice also
294    /// makes a stale or hand-edited `# vX.Y.Z` comment visible for free (M4): the tag
295    /// shown here comes from `TagIndex.sha_to_tag`, not from trusting the comment text.
296    ///
297    /// Scoped to [`PinStyle::Sha`] only — the SHA is the one pin form GitHub itself does
298    /// not render as a readable version, so it is the only form where naming the tag it
299    /// resolves to adds information; a `PinStyle::Tag` pin already shows the tag text
300    /// directly. Guarded on `dep.version_range().is_some()` (N3): a non-resolvable
301    /// dependency (a reusable-workflow call, `./local`, `docker://…`) still matches the
302    /// shared helper's own hover-target predicate and must not have a `**Resolved**` line
303    /// spliced onto it.
304    fn generate_hover<'a>(
305        &'a self,
306        parse_result: &'a dyn ParseResultTrait,
307        position: Position,
308        versions: deps_core::VersionData<'a>,
309        freshness: deps_core::FreshnessSettings,
310    ) -> deps_core::ecosystem::BoxFuture<'a, Option<Hover>> {
311        Box::pin(async move {
312            let registry = self.registry.clone();
313            let base_hover = deps_core::lsp_generate_hover(
314                parse_result,
315                position,
316                versions,
317                registry.as_ref(),
318                self.formatter(),
319                freshness,
320                deps_core::PublishTime::now(),
321            )
322            .await;
323            let mut hover = base_hover?;
324
325            let dep = parse_result.dependencies().into_iter().find(|d| {
326                deps_core::position_in_range(position, d.name_range())
327                    || d.version_range()
328                        .is_some_and(|r| deps_core::position_in_range(position, r))
329            });
330            let Some(dep) = dep else {
331                return Some(hover);
332            };
333            if dep.version_range().is_none() {
334                return Some(hover);
335            }
336            let Some(gha_dep) = dep.as_any().downcast_ref::<GithubActionsDependency>() else {
337                return Some(hover);
338            };
339
340            // #501 gap (tester finding), widened by critic finding C1 (#550): the shared
341            // `has_offline_actionable_data`/footer gate in `deps_core::generate_hover` only
342            // sees `VersionData` and cannot see `formatter.tag_index` — GHA's "Pin to commit
343            // SHA" quickfix (`build_sha_pin_action`, wired into `generate_code_actions`
344            // above) is driven entirely by that separately-populated index, independent of
345            // `versions`. So a `PinStyle::Tag` step with a warm `TagIndex` entry has a real
346            // `Cmd+.` action even when the shared gate suppressed the footer for lack of any
347            // `VersionData` signal — originally observed offline (#501), but #550's
348            // empty-live-fetch footer gate reintroduced the identical gap *online* too: a
349            // bare-major tag (`@v4`) whose repo tags entirely fail the full-semver filter
350            // (`tags_to_versions`) makes `available_versions == Some([])`, which the shared
351            // gate now (correctly, for the general case) treats as "nothing to update" — but
352            // `populate_tag_index` still indexes bare-major tags (`v4 -> sha`) independent of
353            // that filter, so the quickfix is still genuinely available. Restores the footer
354            // post-hoc in both modes using the shared `CMD_DOT_FOOTER` constant (not a
355            // hand-copied literal) so the two can never drift apart; the
356            // `!content.value.contains(CMD_DOT_FOOTER)` guard below makes this idempotent
357            // when the shared gate already rendered it, so dropping the online/offline split
358            // cannot double-append.
359            // `is_plain_scalar` mirrors `build_sha_pin_action`'s own guard (FR-010, spec
360            // 031): for a quoted `uses:` scalar, `version_range` sits inside the quotes, so
361            // the quickfix withholds itself rather than corrupt the value — the footer must
362            // not advertise an action that was never actually offered.
363            //
364            // Deliberately keyed on the raw `PinStyle::Tag` check, not
365            // `is_registry_confirmed_tag` (#551 plan): this footer only ever advertises
366            // `build_sha_pin_action`, which itself stays on the stricter, pre-#551 guard
367            // (see that function's doc comment) — the footer must never promise an action
368            // that withholds itself.
369            if gha_dep.pin == Some(PinStyle::Tag)
370                && gha_dep.is_plain_scalar
371                && let Some(tag) = gha_dep
372                    .version_req
373                    .as_ref()
374                    .map(deps_core::VersionReq::as_str)
375                && self
376                    .formatter
377                    .sha_pin_replacement_for(&gha_dep.name, tag)
378                    .is_some()
379                && let HoverContents::Markup(content) = &mut hover.contents
380                && !content
381                    .value
382                    .contains(deps_core::lsp_helpers::CMD_DOT_FOOTER)
383            {
384                content
385                    .value
386                    .push_str(deps_core::lsp_helpers::CMD_DOT_FOOTER);
387            }
388
389            let Some(PinStyle::Sha { comment_tag }) = &gha_dep.pin else {
390                return Some(hover);
391            };
392
393            let sha = match comment_tag {
394                // Whitespace-token split, not `split_once(" # ")`: the parser's
395                // comment-tag rule (B3) only requires the `#` to be *preceded* by
396                // whitespace, so `<sha>  # v4.2.0` (two spaces) or `<sha>\t# v4.2.0`
397                // are both valid literals that a single-space exact match would miss
398                // (critic M2).
399                Some(_) => gha_dep
400                    .version_literal
401                    .as_deref()
402                    .and_then(|lit| lit.split_whitespace().next()),
403                None => gha_dep
404                    .version_req
405                    .as_ref()
406                    .map(deps_core::VersionReq::as_str),
407            };
408            let Some(sha) = sha else {
409                return Some(hover);
410            };
411
412            let Some(resolved_tag) = self
413                .formatter
414                .tag_index
415                .get(dep.name())
416                .and_then(|index| index.sha_to_tag.get(sha).cloned())
417            else {
418                return Some(hover);
419            };
420
421            if let HoverContents::Markup(content) = &mut hover.contents {
422                content.value = splice_resolved_line(&content.value, &resolved_tag, sha);
423            }
424
425            Some(hover)
426        })
427    }

434/// Inserts a `**Resolved**: `tag` (`sha…`)` line immediately after the shared hover's
435/// `**Current**`/`**Requirement**` line (whichever is present), falling back to append
436/// only if neither anchor is found.
437fn splice_resolved_line(markdown: &str, resolved_tag: &str, sha: &str) -> String {
438    // `sha` is expected to be a validated, pure-ASCII full hex SHA by the time it
439    // reaches here (`TagIndex` entries are filtered in `populate_tag_index`, security
440    // S-3), so a byte slice is normally safe — `get(..7)` is a char-boundary-safe
441    // belt-and-braces guard rather than a raw index (security S-4), falling back to
442    // the whole string on anything unexpected instead of panicking.
443    let short_sha = sha.get(..7).unwrap_or(sha);
444    let line = format!(
445        "**Resolved**: {} ({})\n\n",
446        markdown_code_span(resolved_tag),
447        markdown_code_span(&format!("{short_sha}…"))
448    );
449
450    for anchor in ["**Current**: ", "**Requirement**: "] {
451        if let Some(pos) = markdown.find(anchor)
452            && let Some(rel_end) = markdown[pos..].find("\n\n")
453        {
454            let insert_at = pos + rel_end + 2;
455            let mut out = String::with_capacity(markdown.len() + line.len());
456            out.push_str(&markdown[..insert_at]);
457            out.push_str(&line);
458            out.push_str(&markdown[insert_at..]);
459            return out;
460        }
461    }
462    format!("{markdown}{line}")
463}

465/// Builds one mutable-ref-pin [`Diagnostic`] (issue #473) per diagnosable-as-tag step in
466/// `parse_result` — every `PinStyle::Tag` step, plus a `PinStyle::Branch` step
467/// `tag_index` confirms is actually a real tag (issue #551, e.g.
468/// `taiki-e/install-action@cargo-deny`: see [`is_registry_confirmed_tag`]).
469/// `PinStyle::Sha` and a `PinStyle::Branch` `tag_index` cannot (yet) confirm produce no
470/// diagnostic (FR-003).
471/// Maximum character count of `mutable_ref_pin_diagnostics`' interpolated `name`/`tag`
472/// values before truncation (security audit finding). Mirrors
473/// `deps_core::lsp_helpers::diagnostics`' `MAX_BLOCKED_REGISTRY_MESSAGE_VALUE_CHARS`
474/// precedent: nothing upstream caps a workflow file's `owner/repo` or ref text length, so
475/// this is the last chokepoint before either renders inline in the editor, re-sent on
476/// every `publishDiagnostics`.
477const MAX_MUTABLE_REF_PIN_MESSAGE_VALUE_CHARS: usize = 128;
478
479fn mutable_ref_pin_diagnostics(
480    parse_result: &dyn ParseResultTrait,
481    severity: tower_lsp_server::ls_types::DiagnosticSeverity,
482    tag_index: &DashMap<PackageName, Arc<TagIndex>>,
483) -> Vec<Diagnostic> {
484    parse_result
485        .dependencies()
486        .into_iter()
487        .filter_map(|dep| {
488            let gha_dep = dep.as_any().downcast_ref::<GithubActionsDependency>()?;
489            if !is_registry_confirmed_tag(gha_dep, tag_index) {
490                return None;
491            }
492            let range = gha_dep.version_range?;
493            let tag = gha_dep
494                .version_req
495                .as_ref()
496                .map(deps_core::VersionReq::as_str)?;
497            let name = deps_core::lsp_helpers::truncate_for_diagnostic(
498                gha_dep.name.as_str(),
499                MAX_MUTABLE_REF_PIN_MESSAGE_VALUE_CHARS,
500            );
501            let tag = deps_core::lsp_helpers::truncate_for_diagnostic(
502                tag,
503                MAX_MUTABLE_REF_PIN_MESSAGE_VALUE_CHARS,
504            );
505            // Critic finding C2 (#551): `build_sha_pin_action` deliberately stays
506            // restricted to a statically-classified `PinStyle::Tag` (FR-005/plan §11 —
507            // see that function's doc comment on the branch/tag name-collision risk), so
508            // a registry-confirmed-but-`PinStyle::Branch` ref has no automated fix behind
509            // `Cmd+.` here. The message must say so rather than imply one is a keystroke
510            // away, matching every other resolvable-but-unactionable case in this
511            // codebase (e.g. `deps_core`'s own "Registry lookup failed" wording never
512            // promises a quickfix it can't offer).
513            let message = if gha_dep.pin == Some(PinStyle::Tag) {
514                format!(
515                    "{name} is pinned to the mutable tag ref `{tag}`; pin to a full commit \
516                     SHA to guard against tag mutation"
517                )
518            } else {
519                format!(
520                    "{name} is pinned to the mutable tag ref `{tag}`; pin to a full commit \
521                     SHA to guard against tag mutation (manual edit — no automated fix \
522                     available for this ref)"
523                )
524            };
525            Some(Diagnostic {
526                range,
527                severity: Some(severity),
528                message,
529                code: Some(NumberOrString::String(
530                    MUTABLE_REF_PIN_DIAGNOSTIC_CODE.into(),
531                )),
532                source: Some("deps-lsp".into()),
533                ..Default::default()
534            })
535        })
536        .collect()
537}

539/// Builds the "Pin `{name}` to commit SHA" quickfix (issue #473, US-002) for the
540/// `PinStyle::Tag` dependency at `position`, if [`GithubActionsFormatter::sha_pin_replacement_for`]
541/// resolves its current tag against the shared `TagIndex`.
542///
543/// Returns `None` (no destructive/no-op edit, FR-005) when the dependency at `position`
544/// is not `PinStyle::Tag`, has no `version_range`, or the `TagIndex` lookup misses (cache
545/// miss — e.g. the document was opened before the registry fetch completed).
546///
547/// Deliberately **not** widened to [`is_registry_confirmed_tag`]'s `PinStyle::Branch`
548/// case the way [`mutable_ref_pin_diagnostics`] is (#551 plan): FR-005/plan §11 (see
549/// `test_build_sha_pin_action_no_quickfix_for_branch_pin`) already forbids this
550/// quickfix for a `PinStyle::Branch` step even when a same-named `TagIndex` entry
551/// exists, since git permits a branch and a tag to share one name and GitHub's own
552/// `uses:` ref resolution for that collision is undocumented — an *automated edit*
553/// that silently pins to the tag's commit could pin to a different commit than the
554/// ref actually resolves to at run time. A diagnostic's advisory text carries no such
555/// risk (pinning to *some* SHA is safer than a moving ref either way), but this
556/// destructive edit keeps the stricter, pre-#551 guard.
557fn build_sha_pin_action(
558    parse_result: &dyn ParseResultTrait,
559    position: Position,
560    uri: &Uri,
561    formatter: &GithubActionsFormatter,
562) -> Option<CodeAction> {
563    // Same lookup convention every other deps-lsp code action goes through (critic S2) —
564    // not a hand-rolled position check. The default (`version_range` only, GHA does not
565    // override it) is exactly right here: the diagnostic and the edit both anchor on
566    // `version_range`, never `name_range`.
567    let dep = parse_result

579    Some(CodeAction {
580        title: format!("Pin {} to commit SHA", gha_dep.name),
581        kind: Some(CodeActionKind::QUICKFIX),
582        edit: Some(WorkspaceEdit {
583            changes: Some(changes),
584            ..Default::default()
585        }),

587            "diagnostic_codes": [MUTABLE_REF_PIN_DIAGNOSTIC_CODE],

594/// Builds the `{sha} # {tag}` [`TextEdit`] for a single `PinStyle::Tag` step, shared by
595/// [`build_sha_pin_action`] (wraps it into a per-position quickfix) and
596/// [`collect_pin_all_to_sha_edits`] (the bulk "Pin all to SHA" code lens, issue #633).
597///
598/// `None` for anything but a plain-scalar `PinStyle::Tag` step with a resolvable
599/// `TagIndex` entry — the same withholding guards `build_sha_pin_action`'s doc comment
600/// describes (FR-010's quoted-scalar guard, and a `TagIndex` cache miss).
601fn sha_pin_text_edit_for(
602    dep: &dyn deps_core::Dependency,
603    formatter: &GithubActionsFormatter,
604) -> Option<TextEdit> {
605    let gha_dep = dep.as_any().downcast_ref::<GithubActionsDependency>()?;
606    if gha_dep.pin != Some(PinStyle::Tag) {
607        return None;
608    }
609    // FR-010: for a quoted `uses:` scalar, `version_range` sits inside the quotes —
610    // writing `{sha} # {tag}` there corrupts the value instead of adding a YAML comment
611    // (security audit finding, spec 031 FR-010). Withhold rather than risk that edit.
612    if !gha_dep.is_plain_scalar {
613        return None;
614    }
615    // Security audit finding (issue #633): for a `uses:` step written in YAML flow
616    // style (`{uses: actions/checkout@v4, with: {node: 20}}`), appending `# <tag>`
617    // right after the ref comments out the rest of the flow collection, producing
618    // invalid (unterminated) YAML instead of merely leaving the step unpinned.
619    // Withhold rather than risk corrupting the workflow.
620    if !gha_dep.is_last_on_line {
621        return None;
622    }
623    let version_range = gha_dep.version_range?;
624    let tag = gha_dep
625        .version_req
626        .as_ref()
627        .map(deps_core::VersionReq::as_str)?;
628    let new_text = formatter.sha_pin_replacement_for(&gha_dep.name, tag)?;
629    Some(TextEdit {
630        range: version_range,
631        new_text,
632    })
633}

635/// Builds one [`TextEdit`] per `PinStyle::Tag` step in `parse_result` resolvable to a
636/// commit SHA via `formatter`'s `TagIndex` — the bulk counterpart to
637/// [`build_sha_pin_action`]'s single-step quickfix (issue #633). A step with no
638/// resolvable `TagIndex` entry (cache miss) is silently skipped, exactly like that
639/// quickfix's own withholding behavior — never blocking on, or triggering, a fetch.
640fn collect_pin_all_to_sha_edits(
641    parse_result: &dyn ParseResultTrait,
642    formatter: &GithubActionsFormatter,
643) -> Vec<TextEdit> {
644    let edits: Vec<TextEdit> = parse_result
645        .dependencies()
646        .into_iter()
647        .filter_map(|dep| sha_pin_text_edit_for(dep, formatter))
648        .collect();
649    deps_core::lsp_helpers::dedup_overlapping_edits(edits, "collect_pin_all_to_sha_edits")
650}

652/// Zero or one lens for "Pin N actions to commit SHA" (issue #633), the bulk counterpart
653/// of [`build_sha_pin_action`]'s per-step quickfix. Mirrors
654/// `deps_core::lsp_helpers::generate_code_lenses`'s shape: no lens when there is nothing
655/// to pin, so an all-SHA-pinned (or empty) workflow gets no permanent line-0 annotation.
656fn build_pin_all_to_sha_lens(
657    parse_result: &dyn ParseResultTrait,
658    uri: &Uri,
659    formatter: &GithubActionsFormatter,
660) -> Option<CodeLens> {
661    let edits = collect_pin_all_to_sha_edits(parse_result, formatter);
662    if edits.is_empty() {
663        return None;
664    }
665
666    let count = edits.len();
667    let title = if count == 1 {
668        "Pin 1 action to commit SHA".to_string()
669    } else {
670        format!("Pin {count} actions to commit SHA")
671    };
672
673    Some(CodeLens {
674        range: Range::new(Position::new(0, 0), Position::new(0, 0)),
675        command: Some(Command {
676            title,
677            command: crate::PIN_ALL_TO_SHA_COMMAND_ID.to_string(),
678            arguments: Some(vec![serde_json::json!({ "uri": uri })]),
679        }),
680        data: None,
681    })
682}
