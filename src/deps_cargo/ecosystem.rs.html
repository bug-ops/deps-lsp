20/// Cargo ecosystem implementation.
21///
22/// Provides LSP functionality for Cargo.toml files, including:
23/// - Dependency parsing with position tracking
24/// - Version information from crates.io
25/// - Inlay hints for latest versions
26/// - Hover tooltips with package metadata
27/// - Code actions for version updates
28/// - Diagnostics for unknown/yanked packages
29pub struct CargoEcosystem {
30    registry: Arc<CargoRegistry>,
31    formatter: CargoFormatter,
32    /// The reachability policy (spec #443) and `.cargo/config.toml` memoization cache (spec
33    /// NFR-005) every `parse_manifest` call threads through to
34    /// [`crate::parser::parse_cargo_toml_with_context`]. Defaulted by [`Self::new`]; set
35    /// explicitly by [`Self::with_context`] so `crate::lib::register_ecosystems` can share
36    /// one process-wide policy handle with `ServerState`.
37    context: CargoParseContext,
38}
39
40/// The source(s) a `CompletionContext::Version`/`Feature`'s bare `package_name` joins back
41/// to within a manifest's already-parsed dependencies (spec FR-012).
42enum CompletionSource {
43    /// No dependency in the manifest has this exact name yet — most commonly because the
44    /// user is still typing a brand-new dependency line, with `registry`/`registry-index`
45    /// not yet present for the parser to classify. Callers fall back to the pre-existing
46    /// crates.io-only behavior, unchanged.
47    NotInManifest,
48    /// Every occurrence of this name in the manifest agrees on one resolved source.
49    Resolved(DependencySource),
50    /// Two or more occurrences of this name resolve to different sources (the same
51    /// ambiguity FR-011 covers for the background fetch) — callers must offer no
52    /// completions at all rather than picking one arbitrarily.
53    Ambiguous,
54}
55
56/// Joins `package_name` back to `parse_result.dependencies()` by name (spec FR-012).
57fn resolve_completion_source(
58    parse_result: &dyn ParseResultTrait,
59    package_name: &deps_core::PackageName,
60) -> CompletionSource {
61    let mut sources = parse_result
62        .dependencies()
63        .into_iter()
64        .filter(|d| d.name() == package_name)
65        .map(deps_core::Dependency::source);
66
67    let Some(first) = sources.next() else {
68        return CompletionSource::NotInManifest;
69    };
70    if sources.all(|s| s == first) {
71        CompletionSource::Resolved(first)
72    } else {
73        tracing::warn!(
74            package = %package_name,
75            "ambiguous dependency source for version/feature completion; offering none"
76        );
77        CompletionSource::Ambiguous
78    }
79}

81impl CargoEcosystem {
82    /// Creates a new Cargo ecosystem with the given HTTP cache, using a fresh, default
83    /// [`CargoParseContext`] — an all-`PublicOnly`-policy, empty-cache context private to
84    /// this ecosystem instance. Production use goes through [`Self::with_context`] instead,
85    /// so the policy handle is shared with `ServerState` and live-updatable via
86    /// `workspace/didChangeConfiguration`.
87    pub fn new(cache: Arc<deps_core::HttpCache>) -> Self {
88        Self::with_context(cache, CargoParseContext::default())
89    }
90
91    /// Creates a new Cargo ecosystem sharing `ctx`'s reachability policy and config-file
92    /// cache — the production constructor (plan-1b §1.6), used by
93    /// `crate::lib::register_ecosystems` so `initialize`/`workspace/didChangeConfiguration`
94    /// can update the same `Arc<RegistryAccessPolicy>` this ecosystem's every parse reads.
95    pub fn with_context(cache: Arc<deps_core::HttpCache>, ctx: CargoParseContext) -> Self {
96        Self {
97            registry: Arc::new(CargoRegistry::new(cache)),
98            formatter: CargoFormatter,
99            context: ctx,
100        }
101    }
102
103    async fn complete_package_names(&self, prefix: &str, range: Range) -> Vec<CompletionItem> {
104        // Package-name search is crates.io-only unconditionally (spec Out of Scope: the
105        // sparse index protocol has no search endpoint), so this never needs source
106        // awareness — `self.registry`'s source-blind `Registry::search` already means
107        // crates.io by construction (`CargoRegistry::search`).
108        deps_core::completion::complete_package_names_generic(
109            self.registry.as_ref(),
110            prefix,
111            20,
112            range,
113        )
114        .await
115    }
116
117    /// Completes version requirements for the dependency at `position`, resolved by cursor
118    /// position rather than by name (issue #593) — delegates to
119    /// [`deps_core::completion::complete_versions_at_position`], which mirrors
120    /// `deps_gitlab_ci::ecosystem::GitLabCiEcosystem::generate_completions`'s reference
121    /// pattern. Position-based lookup also fixes a residual gap in the old name-based
122    /// [`resolve_completion_source`] routing: two dependencies sharing one `PackageName` but
123    /// resolving to different sources (e.g. two `[[registries]]`-scoped Cargo entries) used to
124    /// collapse into `CompletionSource::Ambiguous` and offer no completions for either
125    /// occurrence, even though the cursor position unambiguously identifies which one the user
126    /// is editing.
127    ///
128    /// The shared helper's `can_resolve_source` gate keeps `Registry::get_versions_from`'s
129    /// permissive catch-all (anything it doesn't explicitly recognize — Git, Path, an
130    /// unresolved `CustomRegistry`, ...) from leaking a private/non-registry dependency's name
131    /// to crates.io on every keystroke (#248). One source it does *not* reject —
132    /// `AlternateRegistry { mirrors_crates_io: true, .. }` left unregistered — deliberately
133    /// degrades to crates.io inside `CargoRegistry::get_versions_for_source`
134    /// (`registry.rs`'s `mirrors_crates_io` arm): safe, since Cargo verifies per-version
135    /// checksum equality against crates.io for a `[source.crates-io] replace-with` mirror, and
136    /// matches hover's identical degrade-to-public behavior for the same flag.
137    async fn complete_versions(
138        &self,
139        parse_result: &dyn ParseResultTrait,
140        position: Position,
141        prefix: &str,
142        freshness: deps_core::FreshnessSettings,
143    ) -> Vec<CompletionItem> {
144        deps_core::completion::complete_versions_at_position(
145            self.registry.as_ref(),
146            &self.formatter,
147            parse_result,
148            position,
149            prefix,
150            &['^', '~', '=', '<', '>'],
151            freshness,
152        )
153        .await
154    }
155
156    /// Completes feature flags for a specific package.
157    ///
158    /// Fetches features from the latest stable version, routed by the source `package_name`
159    /// resolves to in `parse_result` by name (spec FR-012) — unlike [`Self::complete_versions`]
160    /// (issue #593, position-based), this still joins by name via
161    /// [`resolve_completion_source`]/[`CompletionSource`], so it keeps the same residual
162    /// same-name-different-source `Ambiguous` gap #593 fixed for versions (not itself in
163    /// #593's scope: `features_range`-based position routing for this method is a follow-up,
164    /// not done here).
165    async fn complete_features(
166        &self,
167        parse_result: &dyn ParseResultTrait,
168        package_name: &deps_core::PackageName,
169        prefix: &str,
170    ) -> Vec<CompletionItem> {
171        use deps_core::completion::build_feature_completion;
172
173        let versions_result: Result<Vec<Box<dyn Version>>> =
174            match resolve_completion_source(parse_result, package_name) {
175                CompletionSource::Ambiguous => return vec![],
176                CompletionSource::NotInManifest
177                | CompletionSource::Resolved(DependencySource::Registry) => {
178                    Registry::get_versions(self.registry.as_ref(), package_name).await
179                }
180                CompletionSource::Resolved(DependencySource::AlternateRegistry {
181                    index, ..
182                }) => match self.registry.alternate_client(&index) {
183                    Some(client) => Registry::get_versions(client.as_ref(), package_name).await,
184                    None => return vec![],
185                },
186                CompletionSource::Resolved(_) => return vec![],
187            };
188
189        let versions = match versions_result {
190            Ok(v) => v,
191            Err(e) => {
192                tracing::warn!("Failed to fetch versions for '{}': {}", package_name, e);
193                return vec![];
194            }
195        };
196
197        let latest = match versions.iter().find(|v| v.is_stable()) {
198            Some(v) => v,
199            None => {
200                tracing::warn!("No stable version found for '{}'", package_name);
201                return vec![];
202            }
203        };
204
205        // Get features and filter by prefix
206        let features = latest.features();
207        features
208            .into_iter()
209            .filter(|f| f.starts_with(prefix))
210            .map(|feature| build_feature_completion(&feature, package_name, None))
211            .collect()
212    }
213}

227        &["Cargo.toml"]
228    }
229
230    fn lockfile_filenames(&self) -> &[&'static str] {
231        &["Cargo.lock"]
232    }
233
234    fn parse_manifest<'a>(
235        &'a self,
236        content: &'a str,
237        uri: &'a Uri,
238    ) -> deps_core::ecosystem::BoxFuture<'a, Result<Box<dyn ParseResultTrait>>> {
239        Box::pin(async move {
240            let result = crate::parser::parse_cargo_toml_with_context(content, uri, &self.context)?;
241            // Registers every alternate index this parse resolved (spec FR-002) into the
242            // shared router, including its credential (if any) — the only point in the
243            // whole pipeline where a `.cargo/config.toml`/`$CARGO_HOME` resolution and the
244            // long-lived `CargoRegistry` this ecosystem shares across every document ever
245            // meet. See `crate::parser::ParseResult::resolved_registries`'s docs.
246            for (index, auth) in result.resolved_registries.clone() {
247                self.registry.register_alternate(index, auth);
248            }
249            Ok(Box::new(result) as Box<dyn ParseResultTrait>)
250        })
251    }
252
253    fn registry(&self) -> Arc<dyn Registry> {
254        self.registry.clone() as Arc<dyn Registry>
255    }
256
257    fn lockfile_provider(&self) -> Option<Arc<dyn deps_core::lockfile::LockFileProvider>> {
258        Some(Arc::new(crate::lockfile::CargoLockParser))
259    }
260
261    fn formatter(&self) -> &dyn EcosystemFormatter {
262        &self.formatter
263    }
264
265    fn generate_completions<'a>(
266        &'a self,
267        parse_result: &'a dyn ParseResultTrait,
268        position: Position,
269        content: &'a str,
270        freshness: deps_core::FreshnessSettings,
271    ) -> deps_core::ecosystem::BoxFuture<'a, Completions> {
272        Box::pin(async move {
273            use deps_core::completion::{CompletionContext, detect_completion_context};
274
275            let context = detect_completion_context(parse_result, position, content);
276
277            match context {
278                CompletionContext::PackageName { prefix, range } => {
279                    self.complete_package_names(&prefix, range).await
280                }
281                CompletionContext::Version { prefix, .. } => {
282                    self.complete_versions(parse_result, position, &prefix, freshness)
283                        .await
284                }
285                CompletionContext::Feature {
286                    package_name,
287                    prefix,
288                } => {
289                    self.complete_features(parse_result, &package_name, &prefix)
290                        .await
291                }
292                CompletionContext::None => vec![],
293            }
294            .into()
295        })
296    }
297
298    fn as_any(&self) -> &dyn Any {
299        self
300    }
301}

764    /// Issue #593: two dependencies sharing one `PackageName` but resolving to different
765    /// sources no longer collapse into the old name-based `CompletionSource::Ambiguous`
766    /// "offer nothing for either" result (review finding #6) — cursor position now
767    /// identifies exactly one dependency, so each occurrence routes independently.
768    #[tokio::test]
769    async fn test_complete_versions_same_name_different_sources_routes_by_position() {
770        let cache = Arc::new(deps_core::HttpCache::new());
771        let ecosystem = CargoEcosystem::new(cache);
772
773        let mut registry_dep = mock_dependency("shared-name", Some("1.0"), 0, 0);
774        registry_dep.source = DependencySource::Registry;
775        let mut alternate_dep = mock_dependency("shared-name", Some("1.0"), 1, 1);
776        alternate_dep.source = DependencySource::AlternateRegistry {
777            index: "https://index.mycorp.dev/never-registered".into(),
778            mirrors_crates_io: false,
779        };
780        let alternate_position = alternate_dep.version_range.unwrap().start;
781        let parse_result = MockParseResult {
782            dependencies: vec![registry_dep, alternate_dep],
783        };
784
785        // The alternate occurrence resolves deterministically without network: its index
786        // was never registered, so `CargoRegistry::alternate_client` returns `None` and the
787        // fetch fails closed with `PackageNotFound` before any HTTP call — proving its own
788        // source, not the co-occurring `Registry`-sourced entry, drove the routing.
789        let results = ecosystem
790            .complete_versions(
791                &parse_result,
792                alternate_position,
793                "1",
794                deps_core::FreshnessSettings::default(),
795            )
796            .await;
797        assert!(
798            results.is_empty(),
799            "unregistered alternate index must offer no completions"
800        );
801    }

803    /// Issue #593 critic finding M5: the test above only proves the *empty* case, which a
804    /// totally broken position lookup would also satisfy. This proves position-based routing
805    /// actually selects the right source's data — a *registered* alternate index's own client
806    /// is hit and its versions come back — mirroring `deps-go`'s/`deps-nuget`'s equivalent
807    /// `..._routes_to_registered_alternate_client` tests.
808    #[tokio::test]
809    async fn test_complete_versions_routes_to_registered_alternate_client() {
810        let mut server = mockito::Server::new_async().await;
811        let _mock = server
812            .mock("GET", "/se/rd/serde")
813            .with_status(200)
814            .with_body(
815                "{\"name\":\"serde\",\"vers\":\"1.0.0\",\"yanked\":false,\"features\":{},\"deps\":[]}\n\
816                 {\"name\":\"serde\",\"vers\":\"1.5.0\",\"yanked\":false,\"features\":{},\"deps\":[]}\n",
817            )
818            .create_async()
819            .await;
820
821        let cache = Arc::new(deps_core::HttpCache::new());
822        let ecosystem = CargoEcosystem::new(cache);
823
824        let policy = deps_core::net_policy::RegistryAccessPolicy::default();
825        let registry_index = crate::config::RegistryIndex::new(
826            &server.url(),
827            crate::config::IndexTrust::Trusted,
828            &policy,
829        )
830        .unwrap();
831        let index_key = registry_index.as_str().to_string();
832        ecosystem.registry.register_alternate(registry_index, None);
833
834        let mut dep = mock_dependency("serde", Some("1.0"), 0, 0);
835        dep.source = DependencySource::AlternateRegistry {
836            index: index_key,
837            mirrors_crates_io: false,
838        };
839        let position = dep.version_range.unwrap().start;
840        let parse_result = MockParseResult {
841            dependencies: vec![dep],
842        };
843
844        let results = ecosystem
845            .complete_versions(
846                &parse_result,
847                position,
848                "1.",
849                deps_core::FreshnessSettings::default(),
850            )
851            .await;
852        assert!(
853            !results.is_empty(),
854            "a registered alternate index must route completions through its own client"
855        );
