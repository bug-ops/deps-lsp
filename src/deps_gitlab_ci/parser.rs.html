1//! `.gitlab-ci.yml`-syntax parser using `yaml-rust2`'s event-driven
2//! (`MarkedEventReceiver`) API.
3//!
4//! Tracks the top-level `include:` key's value and, within it, each entry's
5//! `project:`/`ref:`/`component:`/`template:`/`remote:`/`local:` keys — `template:`,
6//! `remote:` and a `local:` key (or a bare-string `include:` list item, GitLab's `local:`
7//! shorthand) are recognized and skipped gracefully (spec FR-003); `image:`/`services:`
8//! entries are never visited at all, since they never appear under `include:` (FR-016
9//! falls out for free).
10//!
11//! # Multi-document input
12//!
13//! A GitLab CI **component** file uses the `spec:` header form (`spec: … \n--- \n job:`).
14//! Per-document parser state is reset at each `Event::DocumentStart`/`DocumentEnd`, so
15//! document 1's nesting never mis-scopes document 2's top-level `include:`.
16//!
17//! # Host resolution and the per-document fan-out cap
18//!
19//! Every dependency's host is resolved here (not deferred to the registry), since the
20//! per-document cap on distinct literal `component:` hosts (spec plan §4.6) must be
21//! enforced at the point a host string first turns into a fetch target. A `project:`
22//! include (which never carries a host segment) and a `$CI_SERVER_FQDN`-relative
23//! `component:` include resolve against `registries.gitlab_instance_host` when set, or are
24//! left [`crate::types::HostRef::Unresolved`] otherwise (spec FR-011a/FR-012).
25

45/// Bound on distinct literal `component:` hosts admitted per document (spec plan §4.6) —
46/// the bound that matters for a `didOpen` burst, protecting `HttpCache`'s unbounded
47/// per-origin transport pool from a `.gitlab-ci.yml` naming N distinct hosts on one parse.
48const MAX_HOSTS_PER_DOCUMENT: usize = 8;

52const CI_SERVER_FQDN: &str = "$CI_SERVER_FQDN";

54/// Which container kind a [`Frame`] represents.
55#[derive(Clone, Copy, PartialEq, Eq)]
56enum FrameKind {
57    Mapping,
58    Sequence,
59}
60
61/// What a [`Frame`] means for include-entry extraction purposes.
62#[derive(Clone, Copy, PartialEq, Eq)]
63enum FrameRole {
64    /// The document's root mapping — scanned only for a top-level `include:` key.
65    Root,
66    /// The value of a top-level `include:` key, when it is a sequence of entries.
67    IncludeValue,
68    /// One include entry mapping — either a `include:`'s single-mapping form, or one
69    /// mapping item inside an [`FrameRole::IncludeValue`] sequence. Its direct scalar keys
70    /// (`project`/`ref`/`component`/`template`/`remote`/`local`) are captured into
71    /// [`Frame::entry`].
72    IncludeEntry,
73    /// Anything else — a job body, `inputs:`, `rules:`, or any other structure this parser
74    /// does not need to look inside.
75    Irrelevant,
76}
77
78/// Which key (if any) a [`FrameRole::Root`] or [`FrameRole::IncludeEntry`] mapping frame is
79/// currently awaiting the value for.
80#[derive(Clone, Copy, PartialEq, Eq)]
81enum PendingKey {
82    None,
83    Include,
84    Project,
85    Ref,
86    Component,
87    Template,
88    Remote,
89    Local,
90}
91
92fn key_for(role: FrameRole, text: &str) -> PendingKey {
93    match role {
94        FrameRole::Root => {
95            if text == "include" {
96                PendingKey::Include
97            } else {
98                PendingKey::None
99            }
100        }
101        FrameRole::IncludeEntry => match text {
102            "project" => PendingKey::Project,
103            "ref" => PendingKey::Ref,
104            "component" => PendingKey::Component,
105            "template" => PendingKey::Template,
106            "remote" => PendingKey::Remote,
107            "local" => PendingKey::Local,
108            _ => PendingKey::None,
109        },
110        FrameRole::IncludeValue | FrameRole::Irrelevant => PendingKey::None,
111    }
112}

114/// One raw scalar value captured from an include entry: its text, YAML scalar style, and
115/// `yaml-rust2` marker line (1-indexed) and column (0-indexed char count) — for span
116/// re-derivation after parsing completes via [`marker_byte_offset`], not `Marker::index()`
117/// (#879).
118type RawField = (String, TScalarStyle, usize, usize);
119
120/// One `include:` entry's raw, not-yet-classified field values, collected during the event
121/// stream and finalized into a [`GitlabCiDependency`] after parsing completes.
122#[derive(Default)]
123struct RawEntry {
124    project: Option<RawField>,
125    ref_field: Option<RawField>,
126    component: Option<RawField>,
127    has_template: bool,
128    has_remote: bool,
129    has_local: bool,
130}

132struct Frame {
133    kind: FrameKind,
134    role: FrameRole,
135    /// Only meaningful for `kind == Mapping`.
136    awaiting_key: bool,
137    /// Only meaningful for `kind == Mapping`.
138    pending_key: PendingKey,
139    /// Only meaningful for `role == IncludeEntry`.
140    entry: RawEntry,
141}
142
143/// Collects every `include:` entry's raw field values, gated to exactly the top-level
144/// `include:` key's subtree.
145struct GitlabCiReceiver {
146    stack: Vec<Frame>,
147    entries: Vec<RawEntry>,
148}
149
150impl GitlabCiReceiver {
151    fn new() -> Self {
152        Self {
153            stack: Vec::new(),
154            entries: Vec::new(),
155        }
156    }
157
158    fn push_container(&mut self, kind: FrameKind) {
159        let role = match self.stack.last() {
160            None => {
161                if kind == FrameKind::Mapping {
162                    FrameRole::Root
163                } else {
164                    FrameRole::Irrelevant
165                }
166            }
167            Some(parent) if parent.kind == FrameKind::Sequence => {
168                if parent.role == FrameRole::IncludeValue && kind == FrameKind::Mapping {
169                    FrameRole::IncludeEntry
170                } else {
171                    FrameRole::Irrelevant
172                }
173            }
174            Some(parent) => match (parent.role, parent.pending_key, kind) {
175                (FrameRole::Root, PendingKey::Include, FrameKind::Sequence) => {
176                    FrameRole::IncludeValue
177                }
178                (FrameRole::Root, PendingKey::Include, FrameKind::Mapping) => {
179                    FrameRole::IncludeEntry
180                }
181                _ => FrameRole::Irrelevant,
182            },
183        };
184        self.consume_pending_value();
185        self.stack.push(Frame {
186            kind,
187            role,
188            awaiting_key: true,
189            pending_key: PendingKey::None,
190            entry: RawEntry::default(),
191        });
192    }
193
194    fn consume_pending_value(&mut self) {
195        if let Some(top) = self.stack.last_mut()
196            && top.kind == FrameKind::Mapping
197        {
198            top.awaiting_key = true;
199            top.pending_key = PendingKey::None;
200        }
201    }
202
203    fn pop_container(&mut self) {
204        if let Some(frame) = self.stack.pop()
205            && frame.role == FrameRole::IncludeEntry
206        {
207            self.entries.push(frame.entry);
208        }
209        self.consume_pending_value();
210    }
211}
212
213impl MarkedEventReceiver for GitlabCiReceiver {
214    fn on_event(&mut self, event: Event, marker: Marker) {
215        match event {
216            Event::MappingStart(..) => self.push_container(FrameKind::Mapping),
217            Event::SequenceStart(..) => self.push_container(FrameKind::Sequence),
218            Event::MappingEnd | Event::SequenceEnd => self.pop_container(),
219            Event::Scalar(value, style, _anchor, _tag) => {
220                let Some(frame) = self.stack.last_mut() else {
221                    return;
222                };
223                if frame.kind != FrameKind::Mapping {
224                    // A bare scalar sequence item (e.g. `include: - templates/x.yml`, the
225                    // `local:` shorthand) carries nothing to record; irrelevant items are
226                    // ignored the same way.
227                    return;
228                }
229                if frame.awaiting_key {
230                    frame.pending_key = key_for(frame.role, &value);
231                    frame.awaiting_key = false;
232                } else {
233                    if frame.role == FrameRole::IncludeEntry {
234                        match frame.pending_key {
235                            PendingKey::Project => {
236                                frame.entry.project =
237                                    Some((value, style, marker.line(), marker.col()));
238                            }
239                            PendingKey::Ref => {
240                                frame.entry.ref_field =
241                                    Some((value, style, marker.line(), marker.col()));
242                            }
243                            PendingKey::Component => {
244                                frame.entry.component =
245                                    Some((value, style, marker.line(), marker.col()));
246                            }
247                            PendingKey::Template => frame.entry.has_template = true,
248                            PendingKey::Remote => frame.entry.has_remote = true,
249                            PendingKey::Local => frame.entry.has_local = true,
250                            PendingKey::None | PendingKey::Include => {}
251                        }
252                    }
253                    frame.awaiting_key = true;
254                    frame.pending_key = PendingKey::None;
255                }
256            }
257            // A `*anchor` alias value must still clear the pending-key slot, mirroring
258            // `deps-github-actions`'s identical fix — GitLab CI files use YAML anchors
259            // heavily for job reuse, so this is a real, not merely defensive, case here.
260            Event::Alias(_) => self.consume_pending_value(),
261            Event::DocumentStart | Event::DocumentEnd => {
262                // A GitLab CI **component** file's `spec:` header form is multi-document
263                // (`spec: … \n--- \n job:`); resetting here stops document 1's nesting from
264                // mis-scoping document 2's top-level `include:` in either direction.
265                self.stack.clear();
266            }
267            Event::Nothing | Event::StreamStart | Event::StreamEnd => {}
268        }
269    }
270}

272/// Whether `origin` is already admitted, or can be newly admitted under
273/// [`MAX_HOSTS_PER_DOCUMENT`]. Mutates `admitted` on a successful new admission.
274fn admit_origin(admitted: &mut HashSet<String>, origin: &str) -> bool {
275    if admitted.contains(origin) {
276        return true;
277    }
278    if admitted.len() >= MAX_HOSTS_PER_DOCUMENT {
279        return false;
280    }
281    admitted.insert(origin.to_string());
282    true
283}

285/// Classifies a `project:` include's `ref:` text — the same two-way SHA/tag-shape test
286/// `deps-github-actions` uses, with no third "confirmed by the registry" state (this crate
287/// has no per-repository tag/SHA cross-reference the way GHA's `TagIndex` does).
288fn classify_project_pin(ref_text: &str) -> PinStyle {
289    if is_full_sha(ref_text) {
290        PinStyle::Sha
291    } else if is_tag_shaped(ref_text) {
292        PinStyle::Tag
293    } else {
294        PinStyle::Branch
295    }
296}

298fn host_qualified_name(host: &HostRef, project_path: &str, component_name: Option<&str>) -> String {
299    let base = match host {
300        HostRef::Literal(h) => format!("{}/{project_path}", h.host()),
301        HostRef::Unresolved(_) | HostRef::CapacityRefused(_) => project_path.to_string(),
302    };
303    match component_name {
304        Some(name) => format!("{base}/{name}"),
305        None => base,
306    }
307}

309fn build_source_and_route(
310    host: &HostRef,
311    endpoint: EndpointKind,
312) -> (DependencySource, Option<(String, GitlabRoute)>) {
313    match host {
314        HostRef::Literal(h) => {
315            let route_key =
316                deps_core::hash_routing_key("gitlab", [h.origin(), endpoint.as_str()].into_iter());
317            let route = GitlabRoute {
318                origin: h.origin().to_string(),
319                endpoint,
320            };
321            (
322                DependencySource::AlternateRegistry {
323                    index: route_key.clone(),
324                    mirrors_crates_io: false,
325                },
326                Some((route_key, route)),
327            )
328        }
329        HostRef::Unresolved(raw) => (DependencySource::CustomRegistry { url: raw.clone() }, None),
330        HostRef::CapacityRefused(origin) => (
331            DependencySource::CustomRegistry {
332                url: origin.clone(),
333            },
334            None,
335        ),
336    }
337}

339fn resolve_project_host(instance_host: &GitlabInstanceHost) -> HostRef {
340    instance_host.get().map_or_else(
341        || HostRef::Unresolved(CI_SERVER_FQDN.to_string()),
342        HostRef::Literal,
343    )

346fn resolve_component_host(
347    host_expr: &str,
348    policy: &RegistryAccessPolicy,
349    instance_host: &GitlabInstanceHost,
350    admitted_origins: &mut HashSet<String>,
351) -> HostRef {
352    if host_expr.starts_with('$') {
353        return instance_host.get().map_or_else(
354            || HostRef::Unresolved(host_expr.to_string()),
355            HostRef::Literal,
356        );
357    }
358    match GitlabHost::parse(host_expr, policy) {
359        Ok(host) if admit_origin(admitted_origins, host.origin()) => HostRef::Literal(host),
360        // M-a (#466 review): the host itself validated fine — only the per-document cap
361        // refused it — so this is `CapacityRefused`, not `Unresolved`: its origin is a known,
362        // usable value, unlike a genuinely unresolvable host.
363        Ok(host) => {
364            warn_rejected_value(
365                "admit_origin",
366                "gitlab-ci component: host (per-document cap)",
367                host_expr,
368            );
369            HostRef::CapacityRefused(host.origin().to_string())
370        }
371        Err(_) => HostRef::Unresolved(host_expr.to_string()),
372    }
373}

382fn build_project_dependency(
383    content: &str,
384    line_table: &LineOffsetTable,
385    instance_host: &GitlabInstanceHost,
386    project_field: RawField,
387    ref_field: Option<RawField>,
388) -> Option<(GitlabCiDependency, Option<(String, GitlabRoute)>)> {
389    let (raw_project, project_style, project_line, project_col) = project_field;
390    if !is_valid_gitlab_coordinate(&raw_project) {
391        warn_rejected_value(
392            "is_valid_gitlab_coordinate",
393            "gitlab-ci project: value",
394            &raw_project,
395        );
396        return None;
397    }
398
399    let value_start = marker_byte_offset(content, line_table, project_line, project_col);
400    let (raw_start, raw_end) = locate_value_span(content, value_start, &raw_project)?;
401    let name_range = make_range(line_table, content, raw_start, raw_end);
402
403    let host = resolve_project_host(instance_host);
404    let name = host_qualified_name(&host, &raw_project, None);
405
406    let (version_req, version_range, pin, is_plain_scalar) = match ref_field {
407        Some((ref_text, ref_style, ref_line, ref_col)) => {
408            let ref_value_start = marker_byte_offset(content, line_table, ref_line, ref_col);
409            let (rs, re) = locate_value_span(content, ref_value_start, &ref_text)?;
410            let range = make_range(line_table, content, rs, re);
411            let pin = classify_project_pin(&ref_text);
412            let plain = ref_style == TScalarStyle::Plain;
413            (Some(ref_text.into()), Some(range), Some(pin), plain)
414        }
415        None => (None, None, None, project_style == TScalarStyle::Plain),
416    };
417
418    let (source, route) = build_source_and_route(&host, EndpointKind::Tags);
419
420    Some((
421        GitlabCiDependency {
422            name: name.into(),
423            name_range,
424            version_req,
425            version_range,
426            version_literal: None,
427            source,
428            is_plain_scalar,
429            kind: IncludeKind::Project,
430            host,
431            pin,
432            project_path: raw_project,
433        },
434        route,
435    ))
436}

438fn build_component_dependency(
439    content: &str,
440    line_table: &LineOffsetTable,
441    policy: &RegistryAccessPolicy,
442    instance_host: &GitlabInstanceHost,
443    admitted_origins: &mut HashSet<String>,
444    component_field: RawField,
445) -> Option<(GitlabCiDependency, Option<(String, GitlabRoute)>)> {
446    let (raw_component, style, comp_line, comp_col) = component_field;
447    let Some((prefix, ref_text)) = raw_component.split_once('@') else {
448        warn_rejected_value(
449            "classify_component_value",
450            "gitlab-ci component: value (missing @version)",
451            &raw_component,
452        );
453        return None;
454    };
455    if ref_text.is_empty() {
456        warn_rejected_value(
457            "classify_component_value",
458            "gitlab-ci component: value (empty version)",
459            &raw_component,
460        );
461        return None;
462    }
463
464    // `<fqdn>/<full-path-to-component-directory>@<version>`: the last path segment before
465    // `@` is the component name (GitLab's own documented shape — see this crate's `lib.rs`
466    // pin-contract table); everything between the host and the component name is the
467    // project path. Minimum 4 segments: host + at least a 2-segment project path + the
468    // component name.
469    let segments: Vec<&str> = prefix.split('/').collect();
470    if segments.len() < 4 {
471        warn_rejected_value(
472            "classify_component_value",
473            "gitlab-ci component: value (too few path segments)",
474            &raw_component,
475        );
476        return None;
477    }
478    let [host_expr, project_segments @ .., component_name] = segments.as_slice() else {
479        return None;
480    };
481    let (host_expr, component_name) = (*host_expr, *component_name);
482    let project_path = project_segments.join("/");
483    if !is_valid_gitlab_coordinate(&project_path) || !is_valid_path_segment(component_name) {
484        warn_rejected_value(
485            "classify_component_value",
486            "gitlab-ci component: value (malformed path)",
487            &raw_component,
488        );
489        return None;
490    }
491
492    let value_start = marker_byte_offset(content, line_table, comp_line, comp_col);
493    let (raw_start, raw_end) = locate_value_span(content, value_start, &raw_component)?;
494    let name_end = raw_start + prefix.len();
495    let ref_start = name_end + 1; // skip '@'
496    let name_range = make_range(line_table, content, raw_start, name_end);
497    let version_range = make_range(line_table, content, ref_start, raw_end);
498
499    let host = resolve_component_host(host_expr, policy, instance_host, admitted_origins);
500    let name = host_qualified_name(&host, &project_path, Some(component_name));
501    let pin = crate::component::classify_component_pin_style(ref_text);
502    let (source, route) = build_source_and_route(&host, EndpointKind::Releases);
503
504    Some((
505        GitlabCiDependency {
506            name: name.into(),
507            name_range,
508            version_req: Some(ref_text.into()),
509            version_range: Some(version_range),
510            version_literal: None,
511            source,
512            is_plain_scalar: style == TScalarStyle::Plain,
513            kind: IncludeKind::Component,
514            host,
515            pin: Some(pin),
516            project_path,
517        },
518        route,
519    ))
520}

522fn build_dependency(
523    content: &str,
524    line_table: &LineOffsetTable,
525    policy: &RegistryAccessPolicy,
526    instance_host: &GitlabInstanceHost,
527    admitted_origins: &mut HashSet<String>,
528    entry: RawEntry,
529) -> Option<(GitlabCiDependency, Option<(String, GitlabRoute)>)> {
530    // `template:`/`remote:`/a `local:` key are recognized and skipped gracefully — not
531    // version-pinnable (spec FR-003).
532    if entry.has_template || entry.has_remote || entry.has_local {
533        return None;
534    }
535    if let Some(component_field) = entry.component {
536        return build_component_dependency(
537            content,
538            line_table,
539            policy,
540            instance_host,
541            admitted_origins,
542            component_field,
543        );
544    }
545    if let Some(project_field) = entry.project {
546        return build_project_dependency(
547            content,
548            line_table,
549            instance_host,
550            project_field,
551            entry.ref_field,
552        );
553    }
554    tracing::debug!("gitlab-ci include entry has neither project: nor component:; skipping");
555    None
556}

558/// Parses a `.gitlab-ci.yml`-syntax file and returns every `include:` dependency found,
559/// with LSP position tracking, host resolution, and routing.
560///
561/// Gated first by [`deps_core::check_yaml_nesting_depth`]/[`deps_core::check_yaml_expansion`],
562/// which return a real [`DepsError::ParseError`]. A downstream YAML syntax error degrades to
563/// an **empty** [`GitlabCiParseResult`] (logged at `debug`) rather than propagating.
564///
565/// # Errors
566///
567/// Returns [`DepsError::ParseError`] only when `content` exceeds the shared YAML
568/// nesting-depth or expansion-size gate.
569///
570/// # Examples
571///
572/// ```
573/// use deps_core::Dependency;
574/// use deps_core::net_policy::RegistryAccessPolicy;
575/// use deps_gitlab_ci::GitlabInstanceHost;
576/// use deps_gitlab_ci::parse_gitlab_ci_yaml;
577/// use std::sync::{Arc, RwLock};
578///
579/// let content = "include:\n  - project: org/proj\n    ref: v1.0.0\n";
580/// let uri = deps_core::test_util::test_uri("/repo/.gitlab-ci.yml");
581/// let policy = RegistryAccessPolicy::default();
582/// let instance_host = GitlabInstanceHost::new(Arc::new(RwLock::new(None)), Arc::new(policy));
583///
584/// let policy = RegistryAccessPolicy::default();
585/// let result = parse_gitlab_ci_yaml(content, &uri, &policy, &instance_host).unwrap();
586/// assert_eq!(result.dependencies.len(), 1);
587/// ```
588pub fn parse_gitlab_ci_yaml(
589    content: &str,
590    uri: &Uri,
591    policy: &RegistryAccessPolicy,
592    instance_host: &GitlabInstanceHost,
593) -> Result<GitlabCiParseResult> {
594    if let Err(depth) =
595        deps_core::check_yaml_nesting_depth(content, deps_core::MAX_YAML_NESTING_DEPTH)
596    {
597        return Err(DepsError::ParseError {
598            file_type: "gitlab-ci.yml".into(),
599            source: Box::new(std::io::Error::other(format!(
600                "YAML nesting depth {depth} exceeds maximum of {}",
601                deps_core::MAX_YAML_NESTING_DEPTH
602            ))),
603        });
604    }
605    if let Err(bytes) = deps_core::check_yaml_expansion(content, deps_core::MAX_YAML_EXPANDED_BYTES)
606    {
607        return Err(DepsError::ParseError {
608            file_type: "gitlab-ci.yml".into(),
609            source: Box::new(std::io::Error::other(format!(
610                "YAML expansion {bytes} bytes exceeds maximum of {} bytes",
611                deps_core::MAX_YAML_EXPANDED_BYTES
612            ))),
613        });
614    }
615
616    let mut receiver = GitlabCiReceiver::new();
617    let mut parser = Parser::new_from_str(content);
618    // `multi: true` — unlike `deps-github-actions`'s workflow files, a GitLab CI
619    // **component** file's `spec:` header form is genuinely multi-document
620    // (`spec: … \n--- \n job:`); `Parser::load(_, false)` stops after the first document.
621    if let Err(e) = parser.load(&mut receiver, true) {
622        tracing::debug!(error = %e, "failed to parse GitLab CI YAML, treating as empty");
623        return Ok(GitlabCiParseResult {
624            dependencies: Vec::new(),
625            routes: Vec::new(),
626            uri: uri.clone(),
627            dependency_truncation: None,
628        });
629    }
630
631    let line_table = LineOffsetTable::new(content);
632    let mut admitted_origins = HashSet::new();
633    let mut seen_route_keys = HashSet::new();
634    let mut routes = Vec::new();
635    let mut dependencies = Vec::new();
636    // #796: checked before `build_dependency` (the expensive step — policy/host
637    // resolution, range computation) rather than after, so an entry beyond the ceiling
638    // never reaches it.
639    let mut budget = deps_core::DependencyBudget::new(deps_core::MAX_DEPENDENCIES_PER_DOCUMENT);
640
641    for entry in receiver.entries {
642        if !budget.allow() {
643            continue;
644        }
645        let Some((dep, route)) = build_dependency(
646            content,
647            &line_table,
648            policy,
649            instance_host,
650            &mut admitted_origins,
651            entry,
652        ) else {
653            continue;
654        };
655        if let Some((key, route_value)) = route
656            && seen_route_keys.insert(key.clone())
657        {
658            routes.push((key, route_value));
659        }
660        dependencies.push(dep);
661    }
662
663    Ok(GitlabCiParseResult {
664        dependencies,
665        routes,
666        uri: uri.clone(),
667        dependency_truncation: budget.truncation(),
668    })
669}
