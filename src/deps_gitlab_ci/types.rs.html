8/// Which `include:` form a dependency came from.
9#[derive(Debug, Clone, Copy, PartialEq, Eq)]
10pub enum IncludeKind {
11    /// `include: - project: org/proj` + `ref:`.
12    Project,
13    /// `include: - component: host/org/proj/name@ref`.
14    Component,
15}
16
17impl IncludeKind {
18    /// The [`EndpointKind`] an include of this kind always resolves against — the fixed,
19    /// 1:1 correspondence `crate::parser::build_project_dependency`/
20    /// `crate::parser::build_component_dependency` bake in at parse time.
21    ///
22    /// Used to key [`crate::registry::TagIndex`] lookups by `(EndpointKind, PackageName)`
23    /// rather than by `PackageName` alone (validation finding S2): a `component:`'s
24    /// host-qualified name can textually collide with an unrelated `project:` include's own
25    /// name (spec §3.1's documented residual collision is same-project only; this is the
26    /// cross-project case it does not cover), and without the endpoint in the key the two
27    /// would share one `TagIndex` entry, letting a quickfix resolve a SHA from the wrong
28    /// repository.
29    #[must_use]
30    pub const fn endpoint(self) -> EndpointKind {
31        match self {
32            Self::Project => EndpointKind::Tags,
33            Self::Component => EndpointKind::Releases,
34        }
35    }
36}
37
38/// Which GitLab REST endpoint a [`crate::registry::GitlabCiRegistry`] route resolves
39/// against.
40#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
41pub enum EndpointKind {
42    /// `GET /projects/:id/repository/tags` — backs [`IncludeKind::Project`].
43    Tags,
44    /// `GET /projects/:id/releases` — backs [`IncludeKind::Component`] (spec FR-004/FR-007:
45    /// a component version *is* a project Release; a tag with no release is not one).
46    Releases,
47}
48
49impl EndpointKind {
50    /// A stable string discriminator, used as one part of the route's hashed routing key.
51    #[must_use]
52    pub const fn as_str(self) -> &'static str {
53        match self {
54            Self::Tags => "tags",
55            Self::Releases => "releases",
56        }
57    }
58}

60/// A dependency's resolved (or not-yet-resolvable) host.
61#[derive(Debug, Clone, PartialEq, Eq)]
62pub enum HostRef {
63    /// A validated, policy-gated host — from a `component:` prefix, or from
64    /// `registries.gitlab_instance_host` (spec FR-011a).
65    Literal(GitlabHost),
66    /// `$CI_SERVER_FQDN` (or another unresolved CI-time variable), or a `project:` include
67    /// with the instance-host setting unset — carries the raw, unresolved expression for
68    /// display only (spec FR-012).
69    Unresolved(String),
70    /// A host that validated successfully but whose route/admission was refused purely by a
71    /// capacity limit (spec §4.6's per-document host cap, or the registry's process-wide
72    /// `MAX_GITLAB_ROUTES` cap) — carries the host's normalized origin. Deliberately distinct
73    /// from [`Self::Unresolved`] (#466 review M-a): the host genuinely *is* determinable, so
74    /// the diagnostic this produces must never suggest `registries.gitlab_instance_host` as
75    /// the fix — a capacity refusal needs fewer distinct hosts/includes, not that setting.
76    CapacityRefused(String),
77}

79/// The `(host, endpoint)` pair a dependency resolves against, registered at parse time
80/// under an opaque routing key carried in `DependencySource::AlternateRegistry.index`.
81#[derive(Debug, Clone, PartialEq, Eq)]
82pub struct GitlabRoute {
83    /// Normalized, ASCII-serialized origin (`https://{host}`).
84    pub origin: String,
85    /// Which endpoint this route resolves against.
86    pub endpoint: EndpointKind,
87}

89/// How a pin (a `project:` ref, or a `component:` version) is classified.
90#[derive(Debug, Clone, PartialEq, Eq)]
91pub enum PinStyle {
92    /// A 40-character commit SHA.
93    Sha,
94    /// An exact published tag (`project:`) or release name (`component:`).
95    Tag,
96    /// Honest-unknown: not a SHA, not an exact tag/release, not `~latest`, not
97    /// partial-semver-shaped. Also covers a git branch ref.
98    Branch,
99    /// Literal `~latest` (`component:` only) — highest published non-prerelease semver
100    /// release.
101    Latest,
102    /// A partial semantic version, e.g. `1.2` or `1` (`component:` only).
103    Partial,
104}

106/// Parsed `include:` dependency from a `.gitlab-ci.yml`-syntax file, with position
107/// tracking.
108#[derive(Debug, Clone, PartialEq, Eq)]
109pub struct GitlabCiDependency {
110    /// Host-qualified when the host is known: `{host}/{project_path}` for [`IncludeKind::Project`],
111    /// `{host}/{project_path}/{component_name}` for [`IncludeKind::Component`]. The bare
112    /// path alone (no host prefix) when [`Self::host`] is [`HostRef::Unresolved`] (spec
113    /// §3.1 — this also means every name-keyed structure is automatically per-instance).
114    pub name: deps_core::PackageName,
115    /// LSP range of the `project:`/`component:` value text.
116    pub name_range: Range,
117    /// Normalized version requirement: the ref/pin text, or `None` for a `project:`
118    /// include with no `ref:` at all (GitLab defaults that to the project's default
119    /// branch, which this crate cannot resolve to a concrete version).
120    pub version_req: Option<deps_core::VersionReq>,
121    /// LSP range of the ref/pin text.
122    pub version_range: Option<Range>,
123    /// The raw literal text, when it differs from `version_req` (unused today — no
124    /// GitLab CI pin form carries a comment-derived requirement the way GitHub Actions'
125    /// SHA-with-comment form does; kept for [`deps_core::ecosystem::Dependency`] parity).
126    pub version_literal: Option<String>,
127    /// Dependency source: [`DependencySource::AlternateRegistry`] when [`Self::host`] is
128    /// known and its route was registered; [`DependencySource::CustomRegistry`] otherwise
129    /// (unresolved host, or a route the process-wide cap refused) — see spec §3.2.
130    pub source: DependencySource,
131    /// Whether the whole include-entry value was written as a plain (unquoted) YAML
132    /// scalar, mirroring `deps-github-actions`'s identical field.
133    pub is_plain_scalar: bool,
134    /// Which `include:` form this dependency came from.
135    pub kind: IncludeKind,
136    /// This dependency's resolved (or not-yet-resolvable) host.
137    pub host: HostRef,
138    /// How the ref/pin is classified; `None` only for a hostless-ref `project:` include
139    /// (no `ref:` key at all).
140    pub pin: Option<PinStyle>,
141    /// The bare `org/sub/proj[/component]` path, without a host prefix — kept for URL
142    /// construction and the registry's own fetch-path use.
143    pub project_path: String,
144}

176/// Version information for a GitLab CI dependency: a repository tag (`project:`) or a
177/// project release (`component:`).
178#[derive(Debug, Clone)]
179pub struct GitlabCiVersion {
180    /// The tag/release name as published, `v` prefix (or lack of one) kept as-is.
181    pub version: deps_core::ConcreteVersion,
182    /// The commit SHA this tag/release points at.
183    pub sha: String,
184    /// Whether the semver `pre` component is non-empty.
185    pub prerelease: bool,
186    /// `Some(released_at)` for the releases endpoint (free — same response); `None` for
187    /// tags, since a tag's only date is its *commit* date, not a publish date, and using
188    /// it would misreport freshness.
189    pub published_at: Option<deps_core::PublishTime>,
190}
191
192// GitLab exposes no yank/deprecation signal for either endpoint, so `status` is
193// unconditionally `Available` (mirrors `deps-github-actions`'s `GithubActionsVersion`).
194deps_core::impl_version!(GitlabCiVersion {
195    version: version,
196    status: |_v: &GitlabCiVersion| deps_core::RemovalStatus::Available,
197    published_at: published_at,
198    prerelease: |v: &GitlabCiVersion| v.prerelease,
199});

201/// Result of parsing a `.gitlab-ci.yml`-syntax file.
202#[derive(Debug)]
203pub struct GitlabCiParseResult {
204    /// Every `include:` dependency found, including ones with an unresolved host (their
205    /// consumers filter on `source()`/hover-visible `HostRef` as usual).
206    pub dependencies: Vec<GitlabCiDependency>,
207    /// Distinct `(route_key, route)` pairs this parse produced, to be registered into the
208    /// shared [`crate::registry::GitlabCiRegistry`] by `GitlabCiEcosystem::parse_manifest`
209    /// before this result is returned (spec §3.2/§4.6's downgrade pass).
210    pub routes: Vec<(String, GitlabRoute)>,
211    /// URI of the parsed file.
212    pub uri: Uri,
213}
214
215impl deps_core::ParseResult for GitlabCiParseResult {
216    fn dependencies(&self) -> Vec<&dyn deps_core::Dependency> {
217        self.dependencies
218            .iter()
219            .map(|d| d as &dyn deps_core::Dependency)
220            .collect()
221    }
222
223    fn workspace_root(&self) -> Option<&std::path::Path> {
224        None
225    }
226
227    fn uri(&self) -> &Uri {
228        &self.uri
229    }
230
231    fn as_any(&self) -> &dyn std::any::Any {
232        self
233    }
234}

281            DependencySource::AlternateRegistry { .. }
