40/// Whether `gl_dep`'s pin is diagnosable as a mutable tag ref — either because it was
41/// already classified [`PinStyle::Tag`] from its text shape, or because `tag_index`'s live
42/// fetch confirms the ref/pin text is a literal member of the project's published
43/// tags/releases even though it doesn't *look* tag-shaped (mirrors
44/// `deps_github_actions::ecosystem::is_registry_confirmed_tag`, issue #551's lesson: a
45/// [`PinStyle::Branch`] classification is a registry-blind, honest-unknown guess, not
46/// proof the ref is actually a moving branch).
47///
48/// Keyed on `(gl_dep.kind.endpoint(), gl_dep.name)` — validation finding S2: a
49/// `PackageName` alone can collide between an unrelated `project:` and `component:`
50/// include (spec §3.1's documented residual collision is same-project only; a
51/// same-*text*-different-*project* collision is not covered by it), so the endpoint must
52/// be part of the key to keep the two `TagIndex` entries from ever being read as one.
53fn is_registry_confirmed_tag(
54    gl_dep: &GitlabCiDependency,
55    tag_index: &DashMap<(EndpointKind, PackageName), Arc<TagIndex>>,
56) -> bool {
57    match &gl_dep.pin {
58        Some(PinStyle::Tag) => true,
59        Some(PinStyle::Branch) => gl_dep
60            .version_req
61            .as_ref()
62            .map(deps_core::VersionReq::as_str)
63            .is_some_and(|ref_text| {
64                tag_index
65                    .get(&(gl_dep.kind.endpoint(), gl_dep.name.clone()))
66                    .is_some_and(|index| index.tag_to_sha.contains_key(ref_text))
67            }),
68        Some(PinStyle::Sha | PinStyle::Latest | PinStyle::Partial) | None => false,
69    }
70}

100    /// Creates a GitLab CI ecosystem sharing live `policy`/`gitlab_instance_host` handles —
101    /// the production wiring path (`deps-lsp`'s `register_ecosystems`), mirroring
102    /// `NuGetEcosystem`/`PypiEcosystem`'s identical `with_context` precedent.
103    ///
104    /// `gitlab_instance_host_raw` is the feature-agnostic `Arc<RwLock<Option<String>>>` cell
105    /// `deps-lsp`'s `EcosystemRuntime` owns (spec §4.5's revision-3 note); this constructor
106    /// is the one place it becomes a crate-local [`GitlabInstanceHost`].
107    #[must_use]
108    pub fn with_context(
109        cache: Arc<HttpCache>,
110        policy: Arc<RegistryAccessPolicy>,
111        gitlab_instance_host_raw: Arc<RwLock<Option<String>>>,
112    ) -> Self {
113        let instance_host = Arc::new(GitlabInstanceHost::new(
114            gitlab_instance_host_raw,
115            Arc::clone(&policy),
116        ));
117        let client = Arc::new(GitlabApiClient::new(cache, Arc::clone(&instance_host)));
118        let registry = Arc::new(GitlabCiRegistry::new(client));
119        let formatter = GitlabCiFormatter::new(registry.routes(), registry.tag_index());
120        Self {
121            registry,
122            formatter,
123            policy,
124            instance_host,
125        }
126    }

157    /// Parses the manifest, then registers its routes into the shared registry and
158    /// downgrades any dependency whose route the process-wide cap refused to
159    /// `CustomRegistry` + [`HostRef::CapacityRefused`] (spec §3.2/§4.6) before returning —
160    /// the same downgrade shape (#466 review M-a) [`crate::parser`]'s per-document host cap
161    /// already produces, so the two capacity-refusal paths agree.
162    fn parse_manifest<'a>(
163        &'a self,
164        content: &'a str,
165        uri: &'a Uri,
166    ) -> deps_core::ecosystem::BoxFuture<'a, Result<Box<dyn ParseResultTrait>>> {
167        Box::pin(async move {
168            let mut result = crate::parser::parse_gitlab_ci_yaml(
169                content,
170                uri,
171                &self.policy,
172                &self.instance_host,
173            )?;
174            let refused = self.registry.register_routes(&result.routes);
175            if !refused.is_empty() {
176                for dep in &mut result.dependencies {
177                    if let deps_core::parser::DependencySource::AlternateRegistry { index, .. } =
178                        &dep.source
179                        && refused.contains(index)
180                    {
181                        let origin = match &dep.host {
182                            HostRef::Literal(host) => host.origin().to_string(),
183                            HostRef::Unresolved(raw) | HostRef::CapacityRefused(raw) => raw.clone(),
184                        };
185                        dep.source = deps_core::parser::DependencySource::CustomRegistry {
186                            url: origin.clone(),
187                        };
188                        dep.host = HostRef::CapacityRefused(origin);
189                    }
190                }
191            }
192            Ok(Box::new(result) as Box<dyn ParseResultTrait>)
193        })
194    }

204    /// Version completions only, resolved through the source-aware
205    /// [`deps_core::completion::complete_versions_generic_from`] (spec §7a.1) — the
206    /// source-unaware default would return nothing, since this crate's `Registry` never
207    /// resolves an unsourced fetch. The dependency's `source` is resolved **by position**,
208    /// not by name: a `project:` and a `component:` include of the same project can share
209    /// one `PackageName` (spec §3.1's documented residual collision), and a by-name lookup
210    /// would risk picking the wrong one's source.
211    fn generate_completions<'a>(
212        &'a self,
213        parse_result: &'a dyn ParseResultTrait,
214        position: Position,
215        content: &'a str,
216        freshness: deps_core::FreshnessSettings,
217    ) -> deps_core::ecosystem::BoxFuture<'a, Completions> {
218        Box::pin(async move {
219            use deps_core::completion::{
220                CompletionContext, complete_versions_generic_from, detect_completion_context,
221            };
222
223            let CompletionContext::Version { prefix, .. } =
224                detect_completion_context(parse_result, position, content)
225            else {
226                return Completions::default();
227            };
228            let Some(dep) = parse_result.dependencies().into_iter().find(|d| {
229                d.version_range()
230                    .is_some_and(|r| deps_core::position_in_range(position, r))
231            }) else {
232                return Completions::default();
233            };
234
235            complete_versions_generic_from(
236                self.registry.as_ref(),
237                dep.name(),
238                &dep.source(),
239                &prefix,
240                &[],
241                freshness,
242            )
243            .await
244            .into()
245        })
246    }

335    /// Splices a `**Resolved**` line for a SHA pin (via the shared tag index) and, for a
336    /// `component:` include only, a `**Project**` link line — the one NFR-004 hover
337    /// divergence this ecosystem has (spec §8.1/§8.2).
338    fn generate_hover<'a>(
339        &'a self,
340        parse_result: &'a dyn ParseResultTrait,
341        position: Position,
342        versions: deps_core::VersionData<'a>,
343        freshness: deps_core::FreshnessSettings,
344    ) -> deps_core::ecosystem::BoxFuture<'a, Option<Hover>> {
345        Box::pin(async move {
346            let registry = self.registry.clone();
347            let base_hover = deps_core::lsp_generate_hover(
348                parse_result,
349                position,
350                versions,
351                registry.as_ref(),
352                self.formatter(),
353                freshness,
354                deps_core::PublishTime::now(),
355            )
356            .await;
357            let mut hover = base_hover?;
358
359            let dep = parse_result.dependencies().into_iter().find(|d| {
360                deps_core::position_in_range(position, d.name_range())
361                    || d.version_range()
362                        .is_some_and(|r| deps_core::position_in_range(position, r))
363            });
364            let Some(dep) = dep else {
365                return Some(hover);
366            };
367            let Some(gl_dep) = dep.as_any().downcast_ref::<GitlabCiDependency>() else {
368                return Some(hover);
369            };
370
371            if gl_dep.kind == IncludeKind::Component
372                && let HostRef::Literal(host) = &gl_dep.host
373                && is_valid_gitlab_coordinate(&gl_dep.project_path)
374                && let HoverContents::Markup(content) = &mut hover.contents
375            {
376                let url = format!("https://{}/{}", host.host(), gl_dep.project_path);
377                content.value = splice_project_line(&content.value, &url);
378            }
379
380            if gl_dep.pin == Some(PinStyle::Sha)
381                && let Some(sha) = gl_dep
382                    .version_req
383                    .as_ref()
384                    .map(deps_core::VersionReq::as_str)
385                && let Some(resolved_tag) =
386                    self.formatter
387                        .resolved_tag_for_sha(gl_dep.kind.endpoint(), dep.name(), sha)
388                && let HoverContents::Markup(content) = &mut hover.contents
389            {
390                content.value = splice_resolved_line(&content.value, &resolved_tag, sha);
391            }
392
393            // FR-007 (H1, #466 review): a `component:` `Latest`/`Partial` pin names no
394            // concrete version by itself — unlike `Sha` (resolved above via the tag index,
395            // no extra fetch needed) or `Tag`/`Branch` (whose text either is or isn't the
396            // version). Resolving it needs the priority ladder run against the project's
397            // published releases.
398            if gl_dep.kind == IncludeKind::Component
399                && let Some(pin @ (PinStyle::Latest | PinStyle::Partial)) = &gl_dep.pin
400                && let deps_core::parser::DependencySource::AlternateRegistry { index, .. } =
401                    dep.source()
402                && let Some(route) = registry.routes().get(&index).map(|r| r.clone())
403                && let Some(raw) = gl_dep
404                    .version_req
405                    .as_ref()
406                    .map(deps_core::VersionReq::as_str)
407            {
408                let outcome = tokio::time::timeout(
409                    COMPONENT_PIN_RESOLUTION_TIMEOUT,
410                    registry.resolve_component_pin(dep.name(), &route, pin, raw),
411                )
412                .await;
413                match outcome {
414                    Ok(Ok(Some(resolved))) => {
415                        if let HoverContents::Markup(content) = &mut hover.contents {
416                            content.value = splice_resolved_line(
417                                &content.value,
418                                resolved.version.as_str(),
419                                &resolved.sha,
420                            );
421                        }
422                    }
423                    Ok(Ok(None)) => {}
424                    Ok(Err(error)) => {
425                        tracing::warn!(package = %dep.name(), %error, "FR-007 component pin resolution failed");
426                    }
427                    Err(_) => {
428                        tracing::warn!(package = %dep.name(), "FR-007 component pin resolution timed out");
429                    }
430                }
431            }
432
433            Some(hover)
434        })
435    }

442fn unresolved_host_diagnostics(parse_result: &dyn ParseResultTrait) -> Vec<Diagnostic> {
443    parse_result
444        .dependencies()
445        .into_iter()
446        .filter_map(|dep| {
447            let gl_dep = dep.as_any().downcast_ref::<GitlabCiDependency>()?;
448            // M-a (#466 review): a capacity refusal gets its own message — the host itself
449            // was perfectly determinable, so telling the user to set
450            // `registries.gitlab_instance_host` (a fix for `Unresolved`, not this) would be
451            // actively wrong.
452            let message = match &gl_dep.host {
453                HostRef::Unresolved(raw) => {
454                    let raw = truncate_for_diagnostic(raw, MAX_UNRESOLVED_HOST_MESSAGE_VALUE_CHARS);
455                    format!(
456                        "Cannot determine the GitLab instance host for '{raw}'. Set the \
457                         `registries.gitlab_instance_host` setting to enable version resolution."
458                    )
459                }
460                HostRef::CapacityRefused(origin) => {
461                    let origin =
462                        truncate_for_diagnostic(origin, MAX_UNRESOLVED_HOST_MESSAGE_VALUE_CHARS);
463                    format!(
464                        "'{origin}' was not registered for version resolution because a \
465                         GitLab CI host/route capacity limit was reached. Reduce the number of \
466                         distinct GitLab hosts or includes referenced in this workspace \
467                         (unrelated to the `registries.gitlab_instance_host` setting)."
468                    )
469                }
470                HostRef::Literal(_) => return None,
471            };
472            Some(Diagnostic {
473                range: gl_dep.name_range,
474                severity: Some(DiagnosticSeverity::INFORMATION),
475                message,
476                code: Some(NumberOrString::String(
477                    UNRESOLVED_HOST_DIAGNOSTIC_CODE.into(),
478                )),
479                source: Some("deps-lsp".into()),
480                ..Default::default()
481            })
482        })
483        .collect()
484}

486/// Builds one mutable-ref-pin [`Diagnostic`] (issue #634) per diagnosable dependency in
487/// `parse_result`:
488/// - every `PinStyle::Tag` include (quickfix available);
489/// - a `PinStyle::Branch` include `tag_index` confirms is actually a real published
490///   tag/release (issue #551's lesson, see [`is_registry_confirmed_tag`]);
491/// - every `PinStyle::Latest`/`PinStyle::Partial` `component:` include — both always
492///   resolve to whichever release currently matches, so they are mutable by construction,
493///   not merely by absence of registry confirmation (validation finding C3/#634 follow-up);
494/// - a ref-less `project:` include (`pin: None`) — GitLab defaults an omitted `ref:` to the
495///   project's default branch, which is exactly as mutable as an explicit branch ref, so
496///   this is not the "nothing to say yet" case its `None` might suggest (same finding).
497///
498/// None of the last three get the "Pin to commit SHA" quickfix — [`build_sha_pin_action`]
499/// stays restricted to `PinStyle::Tag` — so their message says so explicitly. Only
500/// `PinStyle::Sha` and an *unconfirmed* `PinStyle::Branch` produce no diagnostic at all.
501fn mutable_ref_pin_diagnostics(
502    parse_result: &dyn ParseResultTrait,
503    severity: DiagnosticSeverity,
504    tag_index: &DashMap<(EndpointKind, PackageName), Arc<TagIndex>>,
505) -> Vec<Diagnostic> {
506    parse_result
507        .dependencies()
508        .into_iter()
509        .filter_map(|dep| {
510            let gl_dep = dep.as_any().downcast_ref::<GitlabCiDependency>()?;
511            let name = truncate_for_diagnostic(
512                gl_dep.name.as_str(),
513                MAX_MUTABLE_REF_PIN_MESSAGE_VALUE_CHARS,
514            );
515            let noun = match gl_dep.kind {
516                IncludeKind::Project => "project",
517                IncludeKind::Component => "component",
518            };
519
520            let Some(pin) = &gl_dep.pin else {
521                // Ref-less `project:` include: no `ref:` key at all, so there is no
522                // version span to anchor on or edit — anchor on the `project:` value
523                // itself, mirroring the FR-012 unresolved-host diagnostic's convention.
524                return Some(Diagnostic {
525                    range: gl_dep.name_range,
526                    severity: Some(severity),
527                    message: format!(
528                        "{name} project has no `ref:`; GitLab CI defaults to the project's \
529                         default branch, which is mutable — add an explicit `ref:` pinned to \
530                         a tag or commit SHA (manual edit — no automated fix available)"
531                    ),
532                    code: Some(NumberOrString::String(
533                        MUTABLE_REF_PIN_DIAGNOSTIC_CODE.into(),
534                    )),
535                    source: Some("deps-lsp".into()),
536                    ..Default::default()
537                });
538            };
539
540            let diagnosable = match pin {
541                PinStyle::Tag | PinStyle::Latest | PinStyle::Partial => true,
542                PinStyle::Branch => is_registry_confirmed_tag(gl_dep, tag_index),
543                PinStyle::Sha => false,
544            };
545            if !diagnosable {
546                return None;
547            }
548
549            let range = gl_dep.version_range?;
550            let tag = gl_dep
551                .version_req
552                .as_ref()
553                .map(deps_core::VersionReq::as_str)?;
554            let tag = truncate_for_diagnostic(tag, MAX_MUTABLE_REF_PIN_MESSAGE_VALUE_CHARS);
555
556            // Mirrors `deps_github_actions::ecosystem::mutable_ref_pin_diagnostics`'s C2
557            // (#551) discipline: `build_sha_pin_action` deliberately stays restricted to a
558            // statically-classified `PinStyle::Tag` (see that function's doc comment on
559            // the branch/tag name-collision risk), so every other diagnosable form has no
560            // automated fix behind `Cmd+.` here — the message must say so rather than
561            // imply one is a keystroke away.
562            let message = if matches!(pin, PinStyle::Tag) {
563                format!(
564                    "{name} {noun} is pinned to the mutable ref `{tag}`; pin to a full commit \
565                     SHA to guard against ref mutation"
566                )
567            } else if matches!(pin, PinStyle::Latest | PinStyle::Partial) {
568                format!(
569                    "{name} {noun} is pinned to `{tag}`, which always resolves to whichever \
570                     release currently matches rather than a fixed version; pin to an exact \
571                     release and commit SHA to guard against ref mutation (manual edit — no \
572                     automated fix available for this ref)"
573                )
574            } else {
575                format!(
576                    "{name} {noun} is pinned to the mutable ref `{tag}`; pin to a full commit \
577                     SHA to guard against ref mutation (manual edit — no automated fix \
578                     available for this ref)"
579                )
580            };
581            Some(Diagnostic {
582                range,
583                severity: Some(severity),
584                message,
585                code: Some(NumberOrString::String(

595/// Builds the "Pin `{name}` to commit SHA" quickfix (issue #634) for the `PinStyle::Tag`
596/// dependency at `position`, if [`GitlabCiFormatter::sha_pin_replacement_for`] resolves its
597/// current tag/release name against the shared `TagIndex`.
598///
599/// Returns `None` (no destructive/no-op edit) when the dependency at `position` is not
600/// `PinStyle::Tag`, has no `version_range`, or the `TagIndex` lookup misses (cache miss —
601/// e.g. the document was opened before the registry fetch completed).
602///
603/// Deliberately **not** widened to [`is_registry_confirmed_tag`]'s `PinStyle::Branch` case
604/// the way [`mutable_ref_pin_diagnostics`] is, mirroring
605/// `deps_github_actions::ecosystem::build_sha_pin_action`'s identical pre-#551 guard: a
606/// `PinStyle::Branch` include can share its ref/pin text with an unrelated tag of the same
607/// name, and GitLab's own ref resolution for that collision is undocumented — an
608/// *automated edit* that silently pins to the tag's commit could pin to a different commit
609/// than the ref actually resolves to at run time. A diagnostic's advisory text carries no
610/// such risk, but this destructive edit keeps the stricter guard.
611fn build_sha_pin_action(
612    parse_result: &dyn ParseResultTrait,
613    position: Position,
614    uri: &Uri,
615    formatter: &GitlabCiFormatter,
616) -> Option<CodeAction> {
617    // Same lookup convention every other deps-lsp code action goes through (critic S2) —
618    // not a hand-rolled position check.
619    let dep = parse_result
620        .dependencies()
621        .into_iter()
622        .find(|d| formatter.is_position_on_dependency(*d, position))?;
623
624    let gl_dep = dep.as_any().downcast_ref::<GitlabCiDependency>()?;
625    if gl_dep.pin != Some(PinStyle::Tag) {
626        return None;
627    }
628    let version_range = gl_dep.version_range?;
629    let tag = gl_dep
630        .version_req
631        .as_ref()
632        .map(deps_core::VersionReq::as_str)?;
633    let new_text = formatter.sha_pin_replacement_for(gl_dep.kind.endpoint(), &gl_dep.name, tag)?;
634
635    let mut changes = std::collections::HashMap::new();
636    changes.insert(
637        uri.clone(),
638        vec![TextEdit {
639            range: version_range,
640            new_text,
641        }],
642    );
643
644    Some(CodeAction {
645        title: format!("Pin {} to commit SHA", gl_dep.name),
646        kind: Some(CodeActionKind::QUICKFIX),
647        edit: Some(WorkspaceEdit {
648            changes: Some(changes),
649            ..Default::default()
650        }),
651        data: Some(serde_json::json!({
652            "diagnostic_codes": [MUTABLE_REF_PIN_DIAGNOSTIC_CODE],
653            "diagnostic_range": version_range,
654        })),
655        ..Default::default()
656    })
657}

659/// Builds the "Pin `{name}` to commit SHA" quickfix (validation follow-up C2/S2) for a
660/// `component:` include pinned via `PinStyle::Latest`/`PinStyle::Partial` at `position`.
661///
662/// Unlike [`build_sha_pin_action`] (a synchronous `TagIndex` lookup only, since a
663/// `PinStyle::Tag` pin's own text already names the version), neither `Latest` nor
664/// `Partial` names a concrete version by itself — resolving one needs the FR-007 priority
665/// ladder run against the project's published releases, exactly the live fetch
666/// `generate_hover`'s `**Resolved**` splice already drives for the same pin forms. Bounded
667/// by [`COMPONENT_PIN_RESOLUTION_TIMEOUT`], mirroring that call site's identical
668/// degrade-to-nothing-on-timeout discipline: a failure or timeout here withholds the
669/// quickfix rather than blocking the rest of `generate_code_actions`.
670///
671/// Returns `None` when the dependency at `position` is not a `component:` include pinned
672/// via `Latest`/`Partial`, has no registered route, or the live resolution misses/fails/
673/// times out.
674async fn build_dynamic_component_pin_action(
675    parse_result: &dyn ParseResultTrait,
676    position: Position,
677    uri: &Uri,
678    registry: &GitlabCiRegistry,
679) -> Option<CodeAction> {
680    let dep = parse_result.dependencies().into_iter().find(|d| {
681        d.version_range()
682            .is_some_and(|r| deps_core::position_in_range(position, r))
683    })?;
684    let gl_dep = dep.as_any().downcast_ref::<GitlabCiDependency>()?;
685    if gl_dep.kind != IncludeKind::Component {
686        return None;
687    }
688    let pin @ (PinStyle::Latest | PinStyle::Partial) = gl_dep.pin.as_ref()? else {
689        return None;
690    };
691    let version_range = gl_dep.version_range?;
692    let raw = gl_dep
693        .version_req
694        .as_ref()
695        .map(deps_core::VersionReq::as_str)?;
696    let deps_core::parser::DependencySource::AlternateRegistry { index, .. } = dep.source() else {
697        return None;
698    };
699    let route = registry.routes().get(&index).map(|r| r.clone())?;
700
701    let outcome = tokio::time::timeout(
702        COMPONENT_PIN_RESOLUTION_TIMEOUT,
703        registry.resolve_component_pin(dep.name(), &route, pin, raw),
704    )
705    .await;
706    let resolved = match outcome {
707        Ok(Ok(Some(resolved))) => resolved,
708        Ok(Ok(None)) => return None,
709        Ok(Err(error)) => {
710            tracing::warn!(package = %dep.name(), %error, "C2 component pin quickfix resolution failed");
711            return None;
712        }
713        Err(_) => {
714            tracing::warn!(package = %dep.name(), "C2 component pin quickfix resolution timed out");
715            return None;
716        }
717    };
718
719    let mut changes = std::collections::HashMap::new();
720    changes.insert(
721        uri.clone(),
722        vec![TextEdit {
723            range: version_range,
724            new_text: resolved.sha,
725        }],
726    );
727
728    Some(CodeAction {
729        title: format!("Pin {} to commit SHA", gl_dep.name),
730        kind: Some(CodeActionKind::QUICKFIX),
731        edit: Some(WorkspaceEdit {
732            changes: Some(changes),
733            ..Default::default()
734        }),
735        data: Some(serde_json::json!({
736            "diagnostic_codes": [MUTABLE_REF_PIN_DIAGNOSTIC_CODE],
737            "diagnostic_range": version_range,
738        })),
739        ..Default::default()
740    })
741}

743/// Inserts a `**Project**: [name](url)` line immediately after the hover heading, for a
744/// `component:` include whose heading link is suppressed (spec §8.2).
745fn splice_project_line(markdown: &str, url: &str) -> String {
746    let line = format!("**Project**: [{url}]({url})\n\n");
747    if let Some(pos) = markdown.find("\n\n") {
748        let insert_at = pos + 2;
749        let mut out = String::with_capacity(markdown.len() + line.len());
750        out.push_str(&markdown[..insert_at]);
751        out.push_str(&line);
752        out.push_str(&markdown[insert_at..]);
753        out
754    } else {
755        format!("{markdown}\n\n{line}")
756    }
757}

759/// Inserts a `**Resolved**: `tag` (`sha…`)` line immediately after the shared hover's
760/// `**Current**`/`**Requirement**` line, mirroring
761/// `deps_github_actions::ecosystem::splice_resolved_line` exactly.
762fn splice_resolved_line(markdown: &str, resolved_tag: &str, sha: &str) -> String {
763    let short_sha = sha.get(..7).unwrap_or(sha);
764    let line = format!(
765        "**Resolved**: {} ({})\n\n",
766        markdown_code_span(resolved_tag),
767        markdown_code_span(&format!("{short_sha}…"))
768    );
769    for anchor in ["**Current**: ", "**Requirement**: "] {
770        if let Some(pos) = markdown.find(anchor)
771            && let Some(rel_end) = markdown[pos..].find("\n\n")
772        {
773            let insert_at = pos + rel_end + 2;
774            let mut out = String::with_capacity(markdown.len() + line.len());
775            out.push_str(&markdown[..insert_at]);
776            out.push_str(&line);
777            out.push_str(&markdown[insert_at..]);
778            return out;
779        }
780    }
781    format!("{markdown}{line}")
782}
