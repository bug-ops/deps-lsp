40/// Whether `gl_dep`'s pin is diagnosable as a mutable tag ref — either because it was
41/// already classified [`PinStyle::Tag`] from its text shape, or because `tag_index`'s live
42/// fetch confirms the ref/pin text is a literal member of the project's published
43/// tags/releases even though it doesn't *look* tag-shaped (mirrors
44/// `deps_github_actions::ecosystem::is_registry_confirmed_tag`, issue #551's lesson: a
45/// [`PinStyle::Branch`] classification is a registry-blind, honest-unknown guess, not
46/// proof the ref is actually a moving branch).
47///
48/// Keyed on `(gl_dep.kind.endpoint(), gl_dep.name)` — validation finding S2: a
49/// `PackageName` alone can collide between an unrelated `project:` and `component:`
50/// include (spec §3.1's documented residual collision is same-project only; a
51/// same-*text*-different-*project* collision is not covered by it), so the endpoint must
52/// be part of the key to keep the two `TagIndex` entries from ever being read as one.
53fn is_registry_confirmed_tag(gl_dep: &GitlabCiDependency, formatter: &GitlabCiFormatter) -> bool {
54    match &gl_dep.pin {
55        Some(PinStyle::Tag) => true,
56        Some(PinStyle::Branch) => gl_dep
57            .version_req
58            .as_ref()
59            .map(deps_core::VersionReq::as_str)
60            .is_some_and(|ref_text| {
61                formatter
62                    .tag_index
63                    .get(&(gl_dep.kind.endpoint(), gl_dep.name.clone()))
64                    .is_some_and(|index| index.tag_to_sha.contains_key(ref_text))
65            }),
66        Some(PinStyle::Sha | PinStyle::Latest | PinStyle::Partial) | None => false,
67    }
68}

70/// Which "Pin to commit SHA" quickfix a diagnosable pin resolves to, if any — the single
71/// source of truth for both [`mutable_ref_pin_diagnostics`]'s message text (issue #643)
72/// and `generate_code_actions`'s dispatch, so the two can never independently drift about
73/// whether a quickfix is actually available.
74enum ShaPinQuickfixKind {
75    /// A `PinStyle::Tag` pin, resolved synchronously against the shared `TagIndex` — see
76    /// [`build_sha_pin_action`].
77    StaticTagIndex,
78    /// A `component:` include pinned via `PinStyle::Latest`/`PinStyle::Partial`, resolved
79    /// against the project's published releases — see
80    /// [`build_dynamic_component_pin_action`]. Only when the dependency's route was
81    /// actually registered (mirrors that function's own guard exactly, spec 048 FR-001):
82    /// an `Unresolved`/`CapacityRefused` host has no route and therefore no quickfix.
83    DynamicComponentPin,
84}
85
86/// Classifies `dep`/`gl_dep`'s pin against the two quickfix builders' own guards, so
87/// message construction and quickfix dispatch read from one place instead of maintaining
88/// two independent guards that can drift (issue #643).
89fn sha_pin_quickfix_kind(
90    dep: &dyn deps_core::Dependency,
91    gl_dep: &GitlabCiDependency,
92    formatter: &GitlabCiFormatter,
93) -> Option<ShaPinQuickfixKind> {
94    match &gl_dep.pin {
95        Some(PinStyle::Tag) => Some(ShaPinQuickfixKind::StaticTagIndex),
96        Some(PinStyle::Latest | PinStyle::Partial) if gl_dep.kind == IncludeKind::Component => {
97            let deps_core::parser::DependencySource::AlternateRegistry { index, .. } = dep.source()
98            else {
99                return None;
100            };
101            formatter
102                .routes
103                .contains_key(&index)
104                .then_some(ShaPinQuickfixKind::DynamicComponentPin)
105        }
106        _ => None,
107    }
108}

139    /// the production wiring path (`deps-lsp`'s `register_ecosystems`), mirroring
140    /// `NuGetEcosystem`/`PypiEcosystem`'s identical `with_context` precedent.
141    ///
142    /// `gitlab_instance_host_raw` is the feature-agnostic `Arc<RwLock<Option<String>>>` cell

145    #[must_use]
146    pub fn with_context(
147        cache: Arc<HttpCache>,
148        policy: Arc<RegistryAccessPolicy>,
149        gitlab_instance_host_raw: Arc<RwLock<Option<String>>>,
150    ) -> Self {
151        let instance_host = Arc::new(GitlabInstanceHost::new(
152            gitlab_instance_host_raw,

154        ));
155        let client = Arc::new(GitlabApiClient::new(cache, Arc::clone(&instance_host)));
156        let registry = Arc::new(GitlabCiRegistry::new(client));
157        let formatter = GitlabCiFormatter::new(registry.routes(), registry.tag_index());
158        Self {
159            registry,
160            formatter,
161            policy,
162            instance_host,
163        }
164    }

195    /// Parses the manifest, then registers its routes into the shared registry and
196    /// downgrades any dependency whose route the process-wide cap refused to
197    /// `CustomRegistry` + [`HostRef::CapacityRefused`] (spec §3.2/§4.6) before returning —
198    /// the same downgrade shape (#466 review M-a) [`crate::parser`]'s per-document host cap
199    /// already produces, so the two capacity-refusal paths agree.
200    fn parse_manifest<'a>(
201        &'a self,
202        content: &'a str,
203        uri: &'a Uri,
204    ) -> deps_core::ecosystem::BoxFuture<'a, Result<Box<dyn ParseResultTrait>>> {
205        Box::pin(async move {
206            let mut result = crate::parser::parse_gitlab_ci_yaml(
207                content,
208                uri,
209                &self.policy,
210                &self.instance_host,
211            )?;
212            let refused = self.registry.register_alternate(&result.routes);
213            if !refused.is_empty() {
214                for dep in &mut result.dependencies {
215                    if let deps_core::parser::DependencySource::AlternateRegistry { index, .. } =
216                        &dep.source
217                        && refused.contains(index)
218                    {
219                        let origin = match &dep.host {
220                            HostRef::Literal(host) => host.origin().to_string(),
221                            HostRef::Unresolved(raw) | HostRef::CapacityRefused(raw) => raw.clone(),
222                        };
223                        dep.source = deps_core::parser::DependencySource::CustomRegistry {
224                            url: origin.clone(),
225                        };
226                        dep.host = HostRef::CapacityRefused(origin);
227                    }
228                }
229            }
230            Ok(Box::new(result) as Box<dyn ParseResultTrait>)
231        })
232    }

369    /// Splices a `**Resolved**` line for a SHA pin (via the shared tag index) and, for a
370    /// `component:` include only, a `**Project**` link line — the one NFR-004 hover
371    /// divergence this ecosystem has (spec §8.1/§8.2).
372    fn generate_hover<'a>(
373        &'a self,
374        parse_result: &'a dyn ParseResultTrait,
375        position: Position,
376        versions: deps_core::VersionData<'a>,
377        freshness: deps_core::FreshnessSettings,
378    ) -> deps_core::ecosystem::BoxFuture<'a, Option<Hover>> {
379        Box::pin(async move {
380            let registry = self.registry.clone();
381            let base_hover = deps_core::lsp_generate_hover(
382                parse_result,
383                position,
384                versions,
385                registry.as_ref(),
386                self.formatter(),
387                freshness,
388                deps_core::PublishTime::now(),
389            )
390            .await;
391            let mut hover = base_hover?;
392
393            let dep = parse_result.dependencies().into_iter().find(|d| {
394                deps_core::position_in_range(position, d.name_range())
395                    || d.version_range()
396                        .is_some_and(|r| deps_core::position_in_range(position, r))
397            });
398            let Some(dep) = dep else {
399                return Some(hover);
400            };
401            let Some(gl_dep) = dep.as_any().downcast_ref::<GitlabCiDependency>() else {
402                return Some(hover);
403            };
404
405            if gl_dep.kind == IncludeKind::Component
406                && let HostRef::Literal(host) = &gl_dep.host
407                && is_valid_gitlab_coordinate(&gl_dep.project_path)
408                && let HoverContents::Markup(content) = &mut hover.contents
409            {
410                let url = format!("https://{}/{}", host.host(), gl_dep.project_path);
411                content.value = splice_project_line(&content.value, &url);
412            }
413
414            if gl_dep.pin == Some(PinStyle::Sha)
415                && let Some(sha) = gl_dep
416                    .version_req
417                    .as_ref()
418                    .map(deps_core::VersionReq::as_str)
419                && let Some(resolved_tag) =
420                    self.formatter
421                        .resolved_tag_for_sha(gl_dep.kind.endpoint(), dep.name(), sha)
422                && let HoverContents::Markup(content) = &mut hover.contents
423            {
424                content.value = splice_resolved_line(&content.value, &resolved_tag, sha);
425            }
426
427            // FR-007 (H1, #466 review): a `component:` `Latest`/`Partial` pin names no
428            // concrete version by itself — unlike `Sha` (resolved above via the tag index,
429            // no extra fetch needed) or `Tag`/`Branch` (whose text either is or isn't the
430            // version). Resolving it needs the priority ladder run against the project's
431            // published releases.
432            if gl_dep.kind == IncludeKind::Component
433                && let Some(pin @ (PinStyle::Latest | PinStyle::Partial)) = &gl_dep.pin
434                && let deps_core::parser::DependencySource::AlternateRegistry { index, .. } =
435                    dep.source()
436                && let Some(route) = registry.routes().get(&index).map(|r| r.clone())
437                && let Some(raw) = gl_dep
438                    .version_req
439                    .as_ref()
440                    .map(deps_core::VersionReq::as_str)
441            {
442                let outcome = tokio::time::timeout(
443                    COMPONENT_PIN_RESOLUTION_TIMEOUT,
444                    registry.resolve_component_pin(dep.name(), &route, pin, raw),
445                )
446                .await;
447                match outcome {
448                    Ok(Ok(Some(resolved))) => {
449                        if let HoverContents::Markup(content) = &mut hover.contents {
450                            content.value = splice_resolved_line(
451                                &content.value,
452                                resolved.version.as_str(),
453                                &resolved.sha,
454                            );
455                        }
456                    }
457                    Ok(Ok(None)) => {}
458                    Ok(Err(error)) => {
459                        tracing::warn!(package = %dep.name(), %error, "FR-007 component pin resolution failed");
460                    }
461                    Err(_) => {
462                        tracing::warn!(package = %dep.name(), "FR-007 component pin resolution timed out");
463                    }
464                }
465            }
466
467            Some(hover)
468        })
469    }

471    fn completion_insert_text(&self, metadata: &dyn deps_core::Metadata) -> Option<String> {
472        let name = metadata.name();
473        let latest = metadata.latest_version().as_str();
474        // `search()` always returns `Ok(vec![])` (spec NFR-002 — no cheap GitLab

480        if latest.is_empty() {
481            Some(name.to_string())
482        } else {
483            Some(format!("{name}@{latest}"))
484        }

512fn unresolved_host_diagnostics(parse_result: &dyn ParseResultTrait) -> Vec<Diagnostic> {
513    parse_result
514        .dependencies()
515        .into_iter()
516        .filter_map(|dep| {
517            let gl_dep = dep.as_any().downcast_ref::<GitlabCiDependency>()?;
518            // M-a (#466 review): a capacity refusal gets its own message — the host itself
519            // was perfectly determinable, so telling the user to set
520            // `registries.gitlab_instance_host` (a fix for `Unresolved`, not this) would be
521            // actively wrong.
522            let message = match &gl_dep.host {
523                HostRef::Unresolved(raw) => {
524                    let raw = truncate_for_diagnostic(raw, MAX_UNRESOLVED_HOST_MESSAGE_VALUE_CHARS);
525                    format!(
526                        "Cannot determine the GitLab instance host for '{raw}'. Set the \
527                         `registries.gitlab_instance_host` setting to enable version resolution."
528                    )
529                }
530                HostRef::CapacityRefused(origin) => {
531                    let origin =
532                        truncate_for_diagnostic(origin, MAX_UNRESOLVED_HOST_MESSAGE_VALUE_CHARS);
533                    format!(
534                        "'{origin}' was not registered for version resolution because a \
535                         GitLab CI host/route capacity limit was reached. Reduce the number of \
536                         distinct GitLab hosts or includes referenced in this workspace \
537                         (unrelated to the `registries.gitlab_instance_host` setting)."
538                    )
539                }
540                HostRef::Literal(_) => return None,
541            };
542            Some(Diagnostic {
543                range: gl_dep.name_range,
544                severity: Some(DiagnosticSeverity::INFORMATION),
545                message,
546                code: Some(NumberOrString::String(
547                    UNRESOLVED_HOST_DIAGNOSTIC_CODE.into(),
548                )),
549                source: Some("deps-lsp".into()),
550                ..Default::default()
551            })
552        })
553        .collect()
554}

556/// Builds one mutable-ref-pin [`Diagnostic`] (issue #634) per diagnosable dependency in
557/// `parse_result`:
558/// - every `PinStyle::Tag` include (quickfix available);
559/// - a `PinStyle::Branch` include `formatter`'s `TagIndex` confirms is actually a real
560///   published tag/release (issue #551's lesson, see [`is_registry_confirmed_tag`]);
561/// - every `PinStyle::Latest`/`PinStyle::Partial` `component:` include — both always
562///   resolve to whichever release currently matches, so they are mutable by construction,
563///   not merely by absence of registry confirmation (validation finding C3/#634 follow-up);
564/// - a ref-less `project:` include (`pin: None`) — GitLab defaults an omitted `ref:` to the
565///   project's default branch, which is exactly as mutable as an explicit branch ref, so
566///   this is not the "nothing to say yet" case its `None` might suggest (same finding).
567///
568/// Whether a diagnosable form's message carries the "manual edit — no automated fix
569/// available" suffix is decided by [`sha_pin_quickfix_kind`] — the single source of truth
570/// for whether `generate_code_actions` actually offers a quickfix for this dependency, so
571/// this message and that dispatch cannot independently drift apart again (issue #643).
572/// `PinStyle::Sha` and an *unconfirmed* `PinStyle::Branch` produce no diagnostic at all.
573fn mutable_ref_pin_diagnostics(
574    parse_result: &dyn ParseResultTrait,
575    severity: DiagnosticSeverity,
576    formatter: &GitlabCiFormatter,
577) -> Vec<Diagnostic> {
578    parse_result
579        .dependencies()
580        .into_iter()
581        .filter_map(|dep| {
582            let gl_dep = dep.as_any().downcast_ref::<GitlabCiDependency>()?;
583            let name = truncate_for_diagnostic(
584                gl_dep.name.as_str(),
585                MAX_MUTABLE_REF_PIN_MESSAGE_VALUE_CHARS,
586            );
587            let noun = match gl_dep.kind {
588                IncludeKind::Project => "project",
589                IncludeKind::Component => "component",
590            };
591
592            let Some(pin) = &gl_dep.pin else {
593                // Ref-less `project:` include: no `ref:` key at all, so there is no
594                // version span to anchor on or edit — anchor on the `project:` value
595                // itself, mirroring the FR-012 unresolved-host diagnostic's convention.
596                return Some(Diagnostic {
597                    range: gl_dep.name_range,
598                    severity: Some(severity),
599                    message: format!(
600                        "{name} project has no `ref:`; GitLab CI defaults to the project's \
601                         default branch, which is mutable — add an explicit `ref:` pinned to \
602                         a tag or commit SHA (manual edit — no automated fix available)"
603                    ),
604                    code: Some(NumberOrString::String(
605                        MUTABLE_REF_PIN_DIAGNOSTIC_CODE.into(),
606                    )),
607                    source: Some("deps-lsp".into()),
608                    ..Default::default()
609                });
610            };
611
612            let diagnosable = match pin {
613                PinStyle::Tag | PinStyle::Latest | PinStyle::Partial => true,
614                PinStyle::Branch => is_registry_confirmed_tag(gl_dep, formatter),
615                PinStyle::Sha => false,
616            };
617            if !diagnosable {
618                return None;
619            }
620
621            let range = gl_dep.version_range?;
622            let tag = gl_dep
623                .version_req
624                .as_ref()
625                .map(deps_core::VersionReq::as_str)?;
626            let tag = truncate_for_diagnostic(tag, MAX_MUTABLE_REF_PIN_MESSAGE_VALUE_CHARS);
627
628            // Issue #643/S1,S2: `sha_pin_quickfix_kind` is the single source of truth for
629            // whether a quickfix is actually available for this dependency — the same
630            // predicate `build_sha_pin_action`/`build_dynamic_component_pin_action` consult
631            // for their own guards — so the suffix decision below can never independently
632            // drift from what `generate_code_actions` actually offers. `PinStyle::Tag`
633            // always classifies `Some(StaticTagIndex)` (even on a cold `TagIndex` miss —
634            // the *diagnostic* doesn't need a live cache hit the way the quickfix build
635            // does, so `Tag`'s message never carries the suffix regardless of cache state,
636            // by design); a registry-confirmed `PinStyle::Branch` always classifies `None`
637            // (`build_sha_pin_action` stays restricted to `Tag` only, C2/#551 branch/tag
638            // name-collision risk), so its message always carries the suffix.
639            let has_quickfix = sha_pin_quickfix_kind(dep, gl_dep, formatter).is_some();
640            let manual_edit_suffix = if has_quickfix {
641                ""
642            } else {
643                " (manual edit — no automated fix available for this ref)"
644            };
645
646            let message = if matches!(pin, PinStyle::Tag) {
647                format!(
648                    "{name} {noun} is pinned to the mutable ref `{tag}`; pin to a full commit \
649                     SHA to guard against ref mutation{manual_edit_suffix}"
650                )
651            } else if matches!(pin, PinStyle::Latest | PinStyle::Partial) {
652                format!(
653                    "{name} {noun} is pinned to `{tag}`, which always resolves to whichever \
654                     release currently matches rather than a fixed version; pin to an exact \
655                     release and commit SHA to guard against ref mutation{manual_edit_suffix}"
656                )
657            } else {
658                format!(
659                    "{name} {noun} is pinned to the mutable ref `{tag}`; pin to a full commit \
660                     SHA to guard against ref mutation{manual_edit_suffix}"
661                )
662            };
663            Some(Diagnostic {
664                range,
665                severity: Some(severity),
666                message,
667                code: Some(NumberOrString::String(
668                    MUTABLE_REF_PIN_DIAGNOSTIC_CODE.into(),
669                )),
670                source: Some("deps-lsp".into()),
671                ..Default::default()
672            })
673        })
674        .collect()
675}

677/// Builds the "Pin `{name}` to commit SHA" quickfix (issue #634) for the `PinStyle::Tag`
678/// dependency at `position`, if [`GitlabCiFormatter::sha_pin_replacement_for`] resolves its
679/// current tag/release name against the shared `TagIndex`.
680///
681/// Returns `None` (no destructive/no-op edit) when the dependency at `position` is not
682/// `PinStyle::Tag`, has no `version_range`, or the `TagIndex` lookup misses (cache miss —
683/// e.g. the document was opened before the registry fetch completed).
684///
685/// Deliberately **not** widened to [`is_registry_confirmed_tag`]'s `PinStyle::Branch` case
686/// the way [`mutable_ref_pin_diagnostics`] is, mirroring
687/// `deps_github_actions::ecosystem::build_sha_pin_action`'s identical pre-#551 guard: a
688/// `PinStyle::Branch` include can share its ref/pin text with an unrelated tag of the same
689/// name, and GitLab's own ref resolution for that collision is undocumented — an
690/// *automated edit* that silently pins to the tag's commit could pin to a different commit
691/// than the ref actually resolves to at run time. A diagnostic's advisory text carries no
692/// such risk, but this destructive edit keeps the stricter guard.
693fn build_sha_pin_action(
694    parse_result: &dyn ParseResultTrait,
695    position: Position,
696    uri: &Uri,
697    formatter: &GitlabCiFormatter,
698) -> Option<CodeAction> {
699    // Same lookup convention every other deps-lsp code action goes through (critic S2) —
700    // not a hand-rolled position check.
701    let dep = parse_result
702        .dependencies()
703        .into_iter()
704        .find(|d| formatter.is_position_on_dependency(*d, position))?;
705
706    let gl_dep = dep.as_any().downcast_ref::<GitlabCiDependency>()?;
707    if !matches!(
708        sha_pin_quickfix_kind(dep, gl_dep, formatter),
709        Some(ShaPinQuickfixKind::StaticTagIndex)
710    ) {
711        return None;
712    }
713    let version_range = gl_dep.version_range?;
714    let tag = gl_dep
715        .version_req
716        .as_ref()
717        .map(deps_core::VersionReq::as_str)?;
718    let new_text = formatter.sha_pin_replacement_for(gl_dep.kind.endpoint(), &gl_dep.name, tag)?;
719
720    let mut changes = std::collections::HashMap::new();
721    changes.insert(
722        uri.clone(),
723        vec![TextEdit {
724            range: version_range,
725            new_text,
726        }],
727    );
728
729    Some(CodeAction {
730        title: format!("Pin {} to commit SHA", gl_dep.name),
731        kind: Some(CodeActionKind::QUICKFIX),
732        edit: Some(WorkspaceEdit {
733            changes: Some(changes),
734            ..Default::default()
735        }),
736        data: Some(serde_json::json!({
737            "diagnostic_codes": [MUTABLE_REF_PIN_DIAGNOSTIC_CODE],
738            "diagnostic_range": version_range,
739        })),
740        ..Default::default()
741    })
742}

744/// Builds the "Pin `{name}` to commit SHA" quickfix (validation follow-up C2/S2) for a
745/// `component:` include pinned via `PinStyle::Latest`/`PinStyle::Partial` at `position`.
746///
747/// Unlike [`build_sha_pin_action`] (a synchronous `TagIndex` lookup only, since a
748/// `PinStyle::Tag` pin's own text already names the version), neither `Latest` nor
749/// `Partial` names a concrete version by itself — resolving one needs the FR-007 priority
750/// ladder run against the project's published releases, exactly the live fetch
751/// `generate_hover`'s `**Resolved**` splice already drives for the same pin forms. Bounded
752/// by [`COMPONENT_PIN_RESOLUTION_TIMEOUT`], mirroring that call site's identical
753/// degrade-to-nothing-on-timeout discipline: a failure or timeout here withholds the
754/// quickfix rather than blocking the rest of `generate_code_actions`.
755///
756/// Returns `None` when the dependency at `position` is not a `component:` include pinned
757/// via `Latest`/`Partial`, has no registered route, or the live resolution misses/fails/
758/// times out. The eligibility guard (kind/pin/route) is [`sha_pin_quickfix_kind`] (issue
759/// #643) — the single source of truth this and `mutable_ref_pin_diagnostics`'s message
760/// both consult, so they cannot independently drift about whether a quickfix exists.
761async fn build_dynamic_component_pin_action(
762    parse_result: &dyn ParseResultTrait,
763    position: Position,
764    uri: &Uri,
765    formatter: &GitlabCiFormatter,
766    registry: &GitlabCiRegistry,
767) -> Option<CodeAction> {
768    // M2 (#640): the same `formatter.is_position_on_dependency` lookup
769    // `build_sha_pin_action` uses, rather than a bare `version_range` check — a
770    // consistency unification, not a behavior change for this pin shape.
771    let dep = parse_result
772        .dependencies()
773        .into_iter()
774        .find(|d| formatter.is_position_on_dependency(*d, position))?;
775    let gl_dep = dep.as_any().downcast_ref::<GitlabCiDependency>()?;
776    if !matches!(
777        sha_pin_quickfix_kind(dep, gl_dep, formatter),
778        Some(ShaPinQuickfixKind::DynamicComponentPin)
779    ) {
780        return None;
781    }
782    let pin = gl_dep.pin.as_ref()?;
783    let version_range = gl_dep.version_range?;
784    let raw = gl_dep
785        .version_req
786        .as_ref()
787        .map(deps_core::VersionReq::as_str)?;
788    let deps_core::parser::DependencySource::AlternateRegistry { index, .. } = dep.source() else {
789        return None;
790    };
791    let route = registry.routes().get(&index).map(|r| r.clone())?;
792
793    let outcome = tokio::time::timeout(
794        COMPONENT_PIN_RESOLUTION_TIMEOUT,
795        registry.resolve_component_pin(dep.name(), &route, pin, raw),
796    )
797    .await;
798    let resolved = match outcome {
799        Ok(Ok(Some(resolved))) => resolved,
800        Ok(Ok(None)) => return None,
801        Ok(Err(error)) => {
802            tracing::warn!(package = %dep.name(), %error, "C2 component pin quickfix resolution failed");
803            return None;
804        }
805        Err(_) => {
806            tracing::warn!(package = %dep.name(), "C2 component pin quickfix resolution timed out");
807            return None;
808        }
809    };
810
811    let mut changes = std::collections::HashMap::new();
812    changes.insert(
813        uri.clone(),
814        vec![TextEdit {
815            range: version_range,
816            new_text: resolved.sha,
817        }],
818    );
819
820    Some(CodeAction {
821        title: format!("Pin {} to commit SHA", gl_dep.name),
822        kind: Some(CodeActionKind::QUICKFIX),
823        edit: Some(WorkspaceEdit {
824            changes: Some(changes),
825            ..Default::default()
826        }),
827        data: Some(serde_json::json!({
828            "diagnostic_codes": [MUTABLE_REF_PIN_DIAGNOSTIC_CODE],
829            "diagnostic_range": version_range,
830        })),
831        ..Default::default()
832    })
833}

836/// commit SHA from already-in-hand data (issue #640) — the bulk counterpart of
837/// [`build_sha_pin_action`] and [`build_dynamic_component_pin_action`]'s per-position

845/// to the unmodified [`crate::component::resolve_component_pin`] ladder — see
846/// [`reconstitute_component_releases`] for why an unresolved-SHA entry is kept as a

855        .into_iter()

857        .collect();
858    deps_core::lsp_helpers::dedup_overlapping_edits(edits, "collect_pin_all_to_sha_edits")

861/// Builds the bulk edit for one dependency, dispatching on [`sha_pin_quickfix_kind`]
862/// exactly like [`build_sha_pin_action`]/[`build_dynamic_component_pin_action`] do for a
863/// single position. `None` when the dependency is not diagnosable via either path, has no
864/// `version_range`, or (`DynamicComponentPin` only) the ladder can't resolve it from the
865/// caller's already-fetched `versions`.
866fn bulk_sha_pin_text_edit_for(
867    dep: &dyn deps_core::Dependency,
868    formatter: &GitlabCiFormatter,
869    versions: deps_core::VersionData<'_>,
870) -> Option<TextEdit> {
871    let gl_dep = dep.as_any().downcast_ref::<GitlabCiDependency>()?;
872    let version_range = gl_dep.version_range?;
873
874    match sha_pin_quickfix_kind(dep, gl_dep, formatter)? {
875        ShaPinQuickfixKind::StaticTagIndex => {
876            let tag = gl_dep
877                .version_req
878                .as_ref()
879                .map(deps_core::VersionReq::as_str)?;
880            let new_text =
881                formatter.sha_pin_replacement_for(gl_dep.kind.endpoint(), &gl_dep.name, tag)?;
882            Some(TextEdit {
883                range: version_range,
884                new_text,
885            })
886        }
887        ShaPinQuickfixKind::DynamicComponentPin => {
888            let pin = gl_dep.pin.as_ref()?;
889            let raw = gl_dep
890                .version_req
891                .as_ref()
892                .map(deps_core::VersionReq::as_str)?;
893            let releases =
894                reconstitute_component_releases(dep.name(), gl_dep, formatter, versions)?;
895            let resolved = crate::component::resolve_component_pin(pin, raw, &releases)?;
896            // The reconstituted list may carry empty-SHA placeholders (see
897            // `reconstitute_component_releases`); only a winner with a genuine full SHA
898            // is safe to splice into a `TextEdit`.
899            if !deps_core::lsp_helpers::is_full_sha(&resolved.sha) {
900                return None;
901            }
902            Some(TextEdit {
903                range: version_range,
904                new_text: resolved.sha,
905            })
906        }
907    }
908}

910/// Reconstitutes the release list [`crate::component::resolve_component_pin`]'s ladder
911/// needs for `name`, from `versions.cached` (already-fetched, never re-fetched here) and
912/// `formatter`'s `TagIndex` (for each release's SHA) — the cache-only alternative to
913/// [`crate::registry::GitlabCiRegistry::resolve_component_pin`]'s live fetch, since a bulk
914/// code lens must never itself trigger one.
915///
916/// **Never drops an entry whose SHA is unknown in the `TagIndex`** — keeps it with an
917/// empty-string `sha` placeholder instead. This is not a per-release SHA gap guard (the
918/// registry's own `releases_to_versions` already drops any release with no valid SHA
919/// before it ever reaches `versions.cached`); it guards a *lifetime* mismatch between two
920/// caches populated at different times: `TagIndex` is capacity-bounded and evictable,
921/// while `versions.cached` is not, so a `tag_to_sha` miss for a release still present in
922/// `versions.cached` is a live possibility, not a hypothetical one. Dropping such an entry
923/// would silently shift what `Latest`/`Partial` resolves to onto a *different* release —
924/// and since `resolve_component_pin`'s `max_by` returns the **last** maximum on a tie
925/// (two releases normalizing to the same semver), preserving `versions.cached`'s exact
926/// order (itself the registry's own fetch/sort order) matters as much as preserving every
927/// entry. The placeholder is unreachable by the ladder's only SHA-matching arm
928/// (`PinStyle::Sha`, which [`sha_pin_quickfix_kind`] never routes to this path) — the
929/// caller still verifies the *winning* release's SHA independently
930/// (`bulk_sha_pin_text_edit_for`'s [`deps_core::lsp_helpers::is_full_sha`] check) before
931/// splicing it into a `TextEdit`.
932fn reconstitute_component_releases(
933    name: &PackageName,
934    gl_dep: &GitlabCiDependency,
935    formatter: &GitlabCiFormatter,
936    versions: deps_core::VersionData<'_>,
937) -> Option<Vec<crate::types::GitlabCiVersion>> {
938    let normalized_name = formatter.normalize_package_name(name);
939    let available = versions
940        .cached
941        .get(normalized_name.as_str())
942        .or_else(|| versions.cached.get(name))
943        .map(|v| &v.available)?;
944    let endpoint = gl_dep.kind.endpoint();
945    let tag_index = formatter.tag_index.get(&(endpoint, gl_dep.name.clone()));
946
947    Some(
948        available
949            .iter()
950            .map(|version| {
951                let sha = tag_index
952                    .as_ref()
953                    .and_then(|index| index.tag_to_sha.get(version.as_str()).cloned())
954                    .unwrap_or_default();
955                let prerelease =
956                    semver::Version::parse(deps_core::github::normalize_tag(version.as_str()))
957                        .is_ok_and(|parsed| !parsed.pre.is_empty());
958                crate::types::GitlabCiVersion {
959                    version: version.clone(),
960                    sha,
961                    prerelease,
962                    published_at: None,
963                }
964            })
965            .collect(),
966    )
967}

969/// Inserts a `**Project**: [name](url)` line immediately after the hover heading, for a
970/// `component:` include whose heading link is suppressed (spec §8.2).
971// `pos`/`insert_at` come from `find("\n\n")`, an ASCII token, so both are always char
972// boundaries.
973#[allow(clippy::string_slice)]
974fn splice_project_line(markdown: &str, url: &str) -> String {
975    let line = format!("**Project**: [{url}]({url})\n\n");
976    if let Some(pos) = markdown.find("\n\n") {
977        let insert_at = pos + 2;
978        let mut out = String::with_capacity(markdown.len() + line.len());
979        out.push_str(&markdown[..insert_at]);
980        out.push_str(&line);
981        out.push_str(&markdown[insert_at..]);
982        out
983    } else {
984        format!("{markdown}\n\n{line}")
985    }
986}

988/// Inserts a `**Resolved**: `tag` (`sha…`)` line immediately after the shared hover's
989/// `**Current**`/`**Requirement**` line, mirroring
990/// `deps_github_actions::ecosystem::splice_resolved_line` exactly.
991// `pos`/`rel_end`/`insert_at` come from `find` of ASCII anchors (`"**Current**: "`,
992// `"\n\n"`), so all are always char boundaries.
993#[allow(clippy::string_slice)]
994fn splice_resolved_line(markdown: &str, resolved_tag: &str, sha: &str) -> String {
995    let short_sha = sha.get(..7).unwrap_or(sha);
996    let line = format!(
997        "**Resolved**: {} ({})\n\n",
998        markdown_code_span(resolved_tag),
999        markdown_code_span(&format!("{short_sha}…"))
1000    );
1001    for anchor in ["**Current**: ", "**Requirement**: "] {
1002        if let Some(pos) = markdown.find(anchor)
1003            && let Some(rel_end) = markdown[pos..].find("\n\n")
1004        {
1005            let insert_at = pos + rel_end + 2;
1006            let mut out = String::with_capacity(markdown.len() + line.len());
1007            out.push_str(&markdown[..insert_at]);
1008            out.push_str(&line);
1009            out.push_str(&markdown[insert_at..]);
1010            return out;
1011        }
1012    }
1013    format!("{markdown}{line}")
1014}
