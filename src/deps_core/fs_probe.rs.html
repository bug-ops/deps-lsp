1//! Counted filesystem probe funnel, shared by every ecosystem crate's config-file cache.
2//!
3//! [`crate::mtime_cache::MtimeFileCache`] claims "one stat, zero reads on a cache hit" — a
4//! claim `Arc::ptr_eq` on the returned value cannot verify, since that only proves the
5//! parsed *value* was reused, not that no syscall ran to get there. Counting actual
6//! `stat`/`read` calls needs a single chokepoint every cache implementation routes through,
7//! so the count is trustworthy across crate boundaries.
8//!
9//! Most wrappers below are a bare passthrough to their `std::fs` equivalent in a shipped
10//! build (the exception is [`read_to_string_capped`], which adds a real size bound on top of
11//! `File::open`/`Read::take`); the counters are compiled out entirely unless this crate is
12//! built for its own tests or with the `test-util` feature, so counting one function's calls
13//! costs nothing in production. `cfg(test)` alone cannot gate the public `snapshot` function,
14//! because
15//! `deps-core` is an ordinary (non-dev) dependency of `deps-cargo`/`deps-npm` — it is never
16//! compiled with `cfg(test)` when a downstream crate's own tests build, so the `test-util`
17//! feature is what those crates enable in their `dev-dependencies` instead.
18//!
19//! Any test in `deps-core`, `deps-gradle`, `deps-cargo`, `deps-npm`, `deps-nuget`, or
20//! `deps-lsp` that touches `fs_probe` (directly, through [`crate::mtime_cache::MtimeFileCache`],
21//! through an ecosystem's own config-file cache, or through `Ecosystem::parse_manifest`) must
22//! hold [`snapshot_guard`]/[`snapshot_guard_async`] for the duration of that touch — see
23//! [`snapshot_guard`]'s doc for why. Nothing enforces this at compile time; run
24//! `scripts/check-fs-probe-race.sh` after adding a new fs_probe-touching test in one of those
25//! six crates to catch a missing guard before it reaches CI's i686 cross-test leg (issue #806).
26

30/// Shared upper bound on how many ancestor directories a config-file discovery walk
31/// climbs, independent of whether the filesystem root has been reached.
32///
33/// `deps-gradle`, `deps-cargo`, `deps-npm`, and `deps-nuget` all import this canonical
34/// definition directly for their own workspace-root / config-file ancestor walks, rather
35/// than each declaring their own duplicate. 64 directories up is not a realistic project
36/// layout — a pathologically deep or hostile tree hits this cap instead of doing unbounded
37/// work per parse (CWE-400).
38pub const MAX_CONFIG_ANCESTOR_DEPTH: usize = 64;
39
40/// Ancestor directories of `start_dir`, inclusive, capped at [`MAX_CONFIG_ANCESTOR_DEPTH`].
41///
42/// Yields `start_dir` itself first, then each successive parent, stopping at whichever
43/// comes first: the filesystem root, or [`MAX_CONFIG_ANCESTOR_DEPTH`] directories visited.
44/// This is the single canonical replacement for the hand-written
45/// `while let Some(dir) = current { if depth >= MAX_CONFIG_ANCESTOR_DEPTH { break; } ... }`
46/// loop every ancestor-config-file walk in this workspace used to duplicate — the depth
47/// bound is enforced by the iterator itself, so a caller cannot forget it.
48///
49/// # Examples
50///
51/// ```
52/// use deps_core::fs_probe::config_ancestors;
53/// use std::path::Path;
54///
55/// let dirs: Vec<_> = config_ancestors(Path::new("/a/b/c")).collect();
56/// assert_eq!(
57///     dirs,
58///     vec![Path::new("/a/b/c"), Path::new("/a/b"), Path::new("/a"), Path::new("/")]
59/// );
60/// ```
61pub fn config_ancestors(start_dir: &Path) -> impl Iterator<Item = &Path> {
62    std::iter::successors(Some(start_dir), |d| d.parent()).take(MAX_CONFIG_ANCESTOR_DEPTH)
63}

73/// Counted wrapper around [`std::fs::metadata`].
74///
75/// # Errors
76///
77/// Returns an error under the same conditions as [`std::fs::metadata`] — most commonly,
78/// `path` does not exist or is not accessible.
79pub fn metadata(path: &Path) -> std::io::Result<std::fs::Metadata> {
80    #[cfg(any(test, feature = "test-util"))]
81    STAT_COUNT.fetch_add(1, Ordering::Relaxed);
82    std::fs::metadata(path)
83}

85/// Counted, size-bounded wrapper around [`std::fs::File::open`] + [`Read::take`].
86///
87/// Reads at most `max_bytes + 1` bytes and returns `Ok(None)` if that read produced more than
88/// `max_bytes` — the one extra byte is what distinguishes "exactly `max_bytes` long" from
89/// "longer than `max_bytes`" without reading the whole (potentially huge) file. Unlike a
90/// `stat`-then-`read_to_string` sequence, this bound is enforced by the read call itself, so
91/// it holds even if the file grows, or is swapped via a symlink, between a caller's earlier
92/// `stat` and this call.
93///
94/// # Errors
95///
96/// Returns an error under the same conditions as [`std::fs::read_to_string`] — most commonly,
97/// `path` does not exist, is not accessible, or the bounded content is not valid UTF-8.
98pub fn read_to_string_capped(path: &Path, max_bytes: u64) -> std::io::Result<Option<String>> {
99    #[cfg(any(test, feature = "test-util"))]
100    READ_COUNT.fetch_add(1, Ordering::Relaxed);
101    let file = std::fs::File::open(path)?;
102    let mut buf = Vec::new();
103    file.take(max_bytes.saturating_add(1))
104        .read_to_end(&mut buf)?;
105    if buf.len() as u64 > max_bytes {
106        return Ok(None);
107    }
108    String::from_utf8(buf)
109        .map(Some)
110        .map_err(|e| std::io::Error::new(std::io::ErrorKind::InvalidData, e.utf8_error()))
111}

113/// Whether `path` exists and is a regular file — `false` on any error, including a missing
114/// path.
115///
116/// Deliberately rejects a FIFO, socket, character device, or directory at `path`: unlike a
117/// regular file, reading one of those can **block the calling thread indefinitely**, which
118/// would stall the parse. [`std::fs::metadata`] follows symlinks, so a symlinked regular
119/// file still resolves as a file here.
120///
121/// Callers that already hold a [`std::fs::Metadata`] from [`metadata`] (as
122/// [`crate::mtime_cache::MtimeFileCache`] does) should call `.is_file()` on it directly
123/// rather than re-probing through here.
124///
125/// # Examples
126///
127/// ```
128/// use deps_core::fs_probe::is_file;
129/// use std::path::Path;
130///
131/// assert!(!is_file(Path::new("/nonexistent/path/to/nowhere")));
132/// ```
133#[must_use]
134pub fn is_file(path: &Path) -> bool {
135    metadata(path).is_ok_and(|m| m.is_file())
136}
137
138/// Whether `path` exists — `false` on any error, including a missing path.
139#[must_use]
140pub fn exists(path: &Path) -> bool {
141    metadata(path).is_ok()
142}

144/// The current `(stat_count, read_count)` totals.
145///
146/// For a test to snapshot before an operation and diff against afterward — never a global
147/// "reset to zero", since `cargo nextest` gives each test its own process but a bare
148/// count-from-zero would still race a hypothetical future multi-threaded runner.
149#[cfg(feature = "test-util")]
150#[must_use]
151pub fn snapshot() -> (usize, usize) {
152    (
153        STAT_COUNT.load(Ordering::Relaxed),
154        READ_COUNT.load(Ordering::Relaxed),
155    )
156}

158/// Backing lock for [`snapshot_guard`]/[`snapshot_guard_async`] — shared so a sync and an
159/// async caller serialize against each other, not just against callers of the same function.
160#[cfg(feature = "test-util")]
161static SNAPSHOT_LOCK: tokio::sync::Mutex<()> = tokio::sync::Mutex::const_new(());
162
163/// Serializes a `snapshot`-before/operation/`snapshot`-after/assert section, run from a plain
164/// (non-async) test, against every other test doing the same.
165///
166/// [`snapshot`] reads the process-global `STAT_COUNT`/`READ_COUNT` counters, so a
167/// before/after diff is only race-free when nothing else touches `fs_probe` between the two
168/// calls. `cargo nextest run` gives each test its own process, which happens to make that
169/// true — but plain `cargo test` runs every test in one multi-threaded process, where a
170/// concurrently running `fs_probe`-touching test can bump the same counters mid-section and
171/// produce a spurious failure. Hold this guard for the whole section under measurement
172/// (both `snapshot()` calls and everything between them), not just around the calls
173/// themselves, so no other snapshot-based test can interleave.
174///
175/// This is not limited to tests that themselves diff a snapshot: **every** test in the same
176/// binary that calls an `fs_probe`-touching function (directly or transitively, e.g. through
177/// [`crate::mtime_cache::MtimeFileCache`] or an ecosystem's own config-file cache) must also
178/// acquire this guard for the duration of that call — otherwise it can still bump the
179/// counters mid-diff even though it never calls [`snapshot`] itself. Verified empirically:
180/// guarding only the diffing test left `deps-gradle`'s and `deps-core`'s own snapshot-diff
181/// tests flaky under plain `cargo test` (siblings in the same file that call the same
182/// instrumented function without diffing were still free to interleave) until every
183/// fs_probe-touching test in those files took the guard too.
184///
185/// Use [`snapshot_guard_async`] instead from an `async fn` test.
186///
187/// # Panics
188///
189/// Panics if called from inside an async execution context (e.g. from within a
190/// `#[tokio::test] async fn` — use [`snapshot_guard_async`] there instead).
191///
192/// # Examples
193///
194/// ```
195/// use deps_core::fs_probe::{snapshot, snapshot_guard};
196///
197/// let _guard = snapshot_guard();
198/// let (stats_before, _) = snapshot();
199/// // ... perform the operation under test ...
200/// let (stats_after, _) = snapshot();
201/// assert!(stats_after >= stats_before);
202/// ```
203#[cfg(feature = "test-util")]
204#[must_use = "dropping the guard immediately releases it, guarding nothing — bind it to a \
205              named variable (e.g. `let _guard = snapshot_guard();`) held for the whole \
206              measured section"]
207pub fn snapshot_guard() -> tokio::sync::MutexGuard<'static, ()> {
208    SNAPSHOT_LOCK.blocking_lock()
209}
210
211/// Async counterpart of [`snapshot_guard`], for a test whose measured section spans an
212/// `.await` point (e.g. an async `load_document_from_disk` call).
213///
214/// A `std::sync::MutexGuard` held across an `.await` would stall the runtime for the
215/// duration of the wait — this returns a `tokio::sync::MutexGuard` instead, which is safe to
216/// hold across `.await`.
217///
218/// # Panics
219///
220/// Never panics — unlike [`snapshot_guard`], this is always safe to call from an async
221/// execution context, since it yields to the runtime instead of blocking the thread.
222///
223/// # Examples
224///
225/// ```
226/// use deps_core::fs_probe::{snapshot, snapshot_guard_async};
227///
228/// # #[tokio::main]
229/// # async fn main() {
230/// let _guard = snapshot_guard_async().await;
231/// let (stats_before, _) = snapshot();
232/// // ... perform the async operation under test ...
233/// let (stats_after, _) = snapshot();
234/// assert!(stats_after >= stats_before);
235/// # }
236/// ```
237#[cfg(feature = "test-util")]
238#[must_use = "dropping the guard immediately releases it, guarding nothing — bind it to a \
239              named variable (e.g. `let _guard = snapshot_guard_async().await;`) held for the \
240              whole measured section"]
241pub async fn snapshot_guard_async() -> tokio::sync::MutexGuard<'static, ()> {
242    SNAPSHOT_LOCK.lock().await
243}

249    #[test]
250    fn read_to_string_capped_returns_content_under_cap() {
251        // Held per `snapshot_guard`'s doc: this module's own tests call the very functions
252        // instrumented for counting, and run in the same test binary as other modules'
253        // (e.g. `mtime_cache`'s) diffing tests.
254        let _guard = snapshot_guard();
255        let dir = tempfile::tempdir().unwrap();
256        let path = dir.path().join("small.txt");
257        std::fs::write(&path, "hello").unwrap();
258
259        assert_eq!(
260            read_to_string_capped(&path, 1024).unwrap().as_deref(),
261            Some("hello")
262        );
263    }

269    fn read_to_string_capped_accepts_content_exactly_at_cap() {
