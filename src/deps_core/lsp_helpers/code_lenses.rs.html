33/// Manifest edits bringing every safely-editable outdated dependency to `latest`.
34///
35/// A dependency is included when all of the following hold:
36/// - it declares a `version_range` (a span to rewrite exists);
37/// - a `latest` version is known in `versions.cached` (normalized name first, then raw —
38///   mirroring [`crate::lsp_helpers::generate_diagnostics_from_cache`]);
39/// - `formatter.is_requirement_up_to_date` reports the declared requirement as *not*
40///   satisfying `latest` — the same predicate diagnostics use, so on a fixture where the
41///   guard below is a no-op, `collect_update_all_edits(..).len()` equals the number of
42///   `generate_diagnostics_from_cache` "Newer version available" diagnostics;
43/// - the **literal-span guard** (`literal_span_matches`): `content` sliced over
44///   `version_range` must still be (up to whitespace and NuGet's bracket wrap) the
45///   literal text — [`Dependency::version_literal`](crate::Dependency::version_literal)
46///   when the ecosystem provides one (e.g. `deps-swift`, whose synthesized comparator
47///   requirement string diverges from the bare literal `version_range` spans), falling
48///   back to the declared requirement text otherwise. Some ecosystems point
49///   `version_range` at something that is not a version literal at all — a Maven
50///   `${property}` reference or a Gradle DSL variable/version-catalog alias — and
51///   rewriting those spans would corrupt the manifest instead of fixing it. A dependency
52///   that fails the guard is skipped entirely: neither counted nor edited.
53///
54/// Accepted edits are sorted by start position; a later edit whose start falls before the
55/// previous edit's end (an overlap — a `WorkspaceEdit` protocol violation) is dropped with
56/// a `tracing::warn!`. No current parser produces overlapping `version_range`s, so this is
57/// a guard against future parser changes, not an expected code path.
58///
59/// `content` is the manifest source, needed for the literal-span guard above — the same
60/// parameter [`Ecosystem::generate_completions`](crate::Ecosystem::generate_completions)
61/// already threads through for a similar reason.
62///
63/// # Examples
64///
65/// ```
66/// use deps_core::lsp_helpers::{
67///     collect_update_all_edits, DiagnosticMessages, DiagnosticPolicy, OsvNaming, PackageNaming,
68///     PackageRendering, PackageVersions, RequirementResolution, SourcePolicy, VersionData,
69/// };
70/// use deps_core::{ConcreteVersion, Dependency, ParseResult, PackageName, VersionReq};
71/// use std::any::Any;
72/// use std::collections::HashMap;
73/// use tower_lsp_server::ls_types::{Position, Range, Uri};
74///
75/// struct MockFormatter;
76/// impl PackageNaming for MockFormatter {}
77/// impl PackageRendering for MockFormatter {
78///     fn format_version_for_text_edit(&self, version: &ConcreteVersion) -> String {
79///         version.to_string()
80///     }
81///     fn package_url(&self, name: &PackageName) -> String {
82///         format!("https://example.com/{name}")
83///     }
84/// }
85/// impl RequirementResolution for MockFormatter {}
86/// impl DiagnosticMessages for MockFormatter {}
87/// impl DiagnosticPolicy for MockFormatter {}
88/// impl SourcePolicy for MockFormatter {}
89/// impl OsvNaming for MockFormatter {}
90///
91/// struct MockDep {
92///     name: PackageName,
93///     version_req: VersionReq,
94///     version_range: Range,
95///     name_range: Range,
96/// }
97/// impl Dependency for MockDep {
98///     fn name(&self) -> &PackageName { &self.name }
99///     fn name_range(&self) -> Range { self.name_range }
100///     fn version_requirement(&self) -> Option<&VersionReq> { Some(&self.version_req) }
101///     fn version_range(&self) -> Option<Range> { Some(self.version_range) }
102///     fn source(&self) -> deps_core::parser::DependencySource {
103///         deps_core::parser::DependencySource::Registry
104///     }
105///     fn as_any(&self) -> &dyn Any { self }

109/// impl ParseResult for MockParseResult {
110///     fn dependencies(&self) -> Vec<&dyn Dependency> {
111///         self.deps.iter().map(|d| d as &dyn Dependency).collect()

113///     fn workspace_root(&self) -> Option<&std::path::Path> { None }
114///     fn uri(&self) -> &Uri { &self.uri }
115///     fn as_any(&self) -> &dyn Any { self }

122///         version_req: VersionReq::new("1.0.0"),
123///         version_range: Range::new(Position::new(0, 9), Position::new(0, 14)),

140/// assert_eq!(edits.len(), 1);
141/// assert_eq!(edits[0].new_text, "1.2.0");

143pub fn collect_update_all_edits(
144    parse_result: &dyn ParseResult,
145    content: &str,
146    versions: VersionData<'_>,
147    formatter: &dyn EcosystemFormatter,
148) -> Vec<TextEdit> {
149    let deps = parse_result.dependencies();
150    let mut edits: Vec<TextEdit> = Vec::with_capacity(deps.len());
151    // Built once and reused for every dependency below — content is fixed for the
152    // whole call, so re-scanning it per dependency would be O(n²) in dependency count.
153    let line_offsets = LineOffsetTable::new(content);
154
155    for dep in deps {
156        let Some(version_range) = dep.version_range() else {
157            continue;
158        };
159
160        let normalized_name = formatter.normalize_package_name(dep.name());
161        let Some(latest) = versions
162            .cached
163            .get(normalized_name.as_str())
164            .or_else(|| versions.cached.get(dep.name()))
165            .map(|v| &v.latest)
166        else {
167            continue;
168        };
169        if !is_safe_version_string(latest.as_str()) {
170            warn_rejected_value(
171                "is_safe_version_string",
172                "update-all code lens edit",
173                latest.as_str(),
174            );
175            continue;
176        }
177
178        let Some(version_req) = dep.version_requirement() else {
179            continue;
180        };
181        if version_req.as_str().is_empty() {
182            // Defense-in-depth: an empty requirement would trivially satisfy the guard
183            // below (both sides normalize to ""), so without this, a future formatter
184            // whose `is_requirement_up_to_date` doesn't treat "" as up to date could
185            // emit an edit anchored on a span that was never a version literal.
186            continue;
187        }
188        if formatter.is_requirement_up_to_date(version_req, latest) {
189            continue;
190        }
191
192        let slice = slice_for_range(content, &line_offsets, version_range);
193        let literal_target = dep
194            .version_literal()
195            .unwrap_or_else(|| version_req.as_str());
196        if !literal_span_matches(slice, literal_target) {
197            continue;
198        }
199
200        let new_text = formatter.format_version_replacing_for(dep, latest, version_req.as_str());
201        // No-op guard, mirroring the REFACTOR-loop dedup and vulnerability-fix N1
202        // guard in `code_actions`: a formatter can decide a declared
203        // requirement has no single unambiguous rewrite (e.g. `deps-gradle`'s
204        // `{strictly}!!{preferred}` shorthand, left unchanged rather than risking a
205        // destructive or misleading edit) and return it unchanged. Without this
206        // check, such a dependency would still count toward — and appear fixed
207        // by — the "Update N outdated dependencies" lens while its click applies
208        // nothing. Compares against `literal_target` (not `version_req`) for the same
209        // reason the literal-span guard above does: for `deps-swift`, `version_req` is a
210        // synthesized comparator that never equals the bare-literal formatted text even
211        // when the edit genuinely is a no-op.
212        if strip_whitespace(&new_text) == strip_whitespace(literal_target) {
213            continue;
214        }
215
216        edits.push(TextEdit {
217            range: version_range,
218            new_text,
219        });
220    }
221
222    dedup_overlapping_edits(edits, "collect_update_all_edits")
223}

225/// Sorts `edits` by start position and drops any edit whose start falls before the
226/// previous (surviving) edit's end — a `WorkspaceEdit` protocol violation no LSP client
227/// can apply.
228///
229/// `caller` names the collector in the `tracing::warn!` emitted for each dropped edit, so
230/// a log line is traceable back to which bulk command produced it.
231///
232/// Shared by every "collect edits for a bulk `WorkspaceEdit`" aggregator in the
233/// workspace (issue #633 critic M3: a sibling collector omitting this pass — while every
234/// current parser happens not to produce overlapping spans — is exactly the kind of
235/// divergence the project's cross-ecosystem-consistency rule exists to catch) —
236/// currently [`collect_update_all_edits`] and
237/// `deps_github_actions::ecosystem::collect_pin_all_to_sha_edits`.
238pub fn dedup_overlapping_edits(mut edits: Vec<TextEdit>, caller: &str) -> Vec<TextEdit> {
239    edits.sort_by_key(|edit| (edit.range.start.line, edit.range.start.character));
240
241    let mut non_overlapping: Vec<TextEdit> = Vec::with_capacity(edits.len());
242    for edit in edits {
243        let overlaps_prev = non_overlapping.last().is_some_and(|prev: &TextEdit| {
244            (edit.range.start.line, edit.range.start.character)
245                < (prev.range.end.line, prev.range.end.character)
246        });
247        if overlaps_prev {
248            tracing::warn!(
249                range = ?edit.range,
250                caller,
251                "dropping overlapping TextEdit"
252            );
253            continue;
254        }
255        non_overlapping.push(edit);
256    }
257
258    non_overlapping
259}

261/// Zero or one lens for the document, bound to `command_id`.
262///
263/// Delegates to [`collect_update_all_edits`] for the count in the lens title — the same
264/// call the command handler makes to produce the edits it applies, so `title N == edits
265/// applied` holds by construction rather than by two implementations agreeing. Returns no
266/// lens when there is nothing to update: a permanent line-0 annotation on every
267/// up-to-date manifest would be noise.
268///
269/// # Examples
270///
271/// ```
272/// use deps_core::lsp_helpers::{
273///     generate_code_lenses, DiagnosticMessages, DiagnosticPolicy, OsvNaming, PackageNaming,
274///     PackageRendering, RequirementResolution, SourcePolicy, VersionData,
275/// };
276/// use deps_core::{ConcreteVersion, PackageName, ParseResult};
277/// use std::collections::HashMap;
278///
279/// struct MockFormatter;
280/// impl PackageNaming for MockFormatter {}
281/// impl PackageRendering for MockFormatter {
282///     fn format_version_for_text_edit(&self, version: &ConcreteVersion) -> String {
283///         version.to_string()
284///     }
285///     fn package_url(&self, name: &PackageName) -> String {
286///         format!("https://example.com/{name}")
287///     }
288/// }
289/// impl RequirementResolution for MockFormatter {}
290/// impl DiagnosticMessages for MockFormatter {}
291/// impl DiagnosticPolicy for MockFormatter {}
292/// impl SourcePolicy for MockFormatter {}
293/// impl OsvNaming for MockFormatter {}
294///
295/// // An empty parse result yields no outdated dependencies, so no lens is generated.
296/// # struct EmptyParseResult { uri: tower_lsp_server::ls_types::Uri }
297/// # impl deps_core::ParseResult for EmptyParseResult {
298/// #     fn dependencies(&self) -> Vec<&dyn deps_core::Dependency> { vec![] }
299/// #     fn workspace_root(&self) -> Option<&std::path::Path> { None }
300/// #     fn uri(&self) -> &tower_lsp_server::ls_types::Uri { &self.uri }
301/// #     fn as_any(&self) -> &dyn std::any::Any { self }
302/// # }
303/// let parse_result = EmptyParseResult { uri: deps_core::test_util::test_uri("/test/Cargo.toml") };
304/// let cached = HashMap::new();
305/// let resolved = HashMap::new();
306///
307/// let lenses = generate_code_lenses(
308///     &parse_result,
309///     "",
310///     VersionData::new(&cached, &resolved),
311///     &MockFormatter,
312///     parse_result.uri(),
313///     "deps-lsp.updateAllOutdated",
314/// );
315///
316/// assert!(lenses.is_empty());
317/// ```
318pub fn generate_code_lenses(
319    parse_result: &dyn ParseResult,
320    content: &str,
321    versions: VersionData<'_>,
322    formatter: &dyn EcosystemFormatter,
323    uri: &Uri,
324    command_id: &str,
325) -> Vec<CodeLens> {
326    let edits = collect_update_all_edits(parse_result, content, versions, formatter);
327    if edits.is_empty() {
328        return Vec::new();
329    }
330
331    let count = edits.len();
332    let title = if count == 1 {
333        "Update 1 outdated dependency".to_string()
334    } else {
335        format!("Update {count} outdated dependencies")
336    };
337
338    vec![CodeLens {
339        range: Range::new(Position::new(0, 0), Position::new(0, 0)),
340        command: Some(Command {
341            title,
342            command: command_id.to_string(),
343            arguments: Some(vec![serde_json::json!({ "uri": uri })]),
344        }),
345        data: None,
346    }]
347}

349/// Zero or one lens for the bulk "Pin N {noun} to commit SHA" command, from an
350/// already-computed edit `count`.
351///
352/// Issue #633, generalized cross-ecosystem in #640 — the caller (`deps-lsp`'s
353/// `handlers::code_lens`) takes `count = ecosystem.collect_pin_all_to_sha_edits(..).len()`
354/// itself, so the lens title and the edits the `deps-lsp.pinAllToSha` command applies can
355/// never disagree about how many there are. Returns no lens when `count` is zero — a
356/// permanent line-0 annotation on a manifest with nothing to pin would be noise.
357///
358/// # Examples
359///
360/// ```
361/// use deps_core::lsp_helpers::{PinNoun, build_pin_all_to_sha_lens};
362///
363/// let uri = deps_core::test_util::test_uri("/repo/.github/workflows/ci.yml");
364/// let noun = PinNoun { singular: "action", plural: "actions" };
365///
366/// assert!(build_pin_all_to_sha_lens(0, noun, &uri).is_none());
367///
368/// let lens = build_pin_all_to_sha_lens(1, noun, &uri).unwrap();
369/// let command = lens.command.unwrap();
370/// assert_eq!(command.title, "Pin 1 action to commit SHA");
371/// assert_eq!(command.command, deps_core::lsp_helpers::PIN_ALL_TO_SHA_COMMAND_ID);
372///
373/// let lens = build_pin_all_to_sha_lens(3, noun, &uri).unwrap();
374/// assert_eq!(lens.command.unwrap().title, "Pin 3 actions to commit SHA");
375/// ```
376#[must_use]
377pub fn build_pin_all_to_sha_lens(count: usize, noun: PinNoun, uri: &Uri) -> Option<CodeLens> {
378    if count == 0 {
379        return None;
380    }
381
382    let title = if count == 1 {
383        format!("Pin 1 {} to commit SHA", noun.singular)
384    } else {
385        format!("Pin {count} {} to commit SHA", noun.plural)
386    };
387
388    Some(CodeLens {
389        range: Range::new(Position::new(0, 0), Position::new(0, 0)),
390        command: Some(Command {
391            title,
392            command: PIN_ALL_TO_SHA_COMMAND_ID.to_string(),
393            arguments: Some(vec![serde_json::json!({ "uri": uri })]),
394        }),
395        data: None,
396    })
397}
