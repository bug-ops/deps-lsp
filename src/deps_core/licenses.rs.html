1//! SPDX license-policy matching (issue #661, spec 010 Phase 2).
2//!
3//! Deliberately minimal: exact, case-insensitive, top-level SPDX identifier
4//! set-membership matching only — no `AND`/`OR`/`WITH` expression-operator
5//! parsing and no new SPDX-parsing dependency (spec 010 `plan.md`'s explicit
6//! scope decision). A dependency's license is already a flat set of discrete
7//! identifiers by the time it reaches this module — [`crate::Version::license`]
8//! returns `Vec<String>`, one entry per declared identifier (a dual-licensed
9//! package's `"MIT OR Apache-2.0"` SPDX expression is reported by every source
10//! this feature uses as two separate strings, not one raw expression) — so
11//! plain set membership needs no parser.
12//!
13//! [`normalize_pom_license_name`]/[`normalize_pom_license_names`] (issue #679) bring a
14//! Gradle dependency's Maven Central POM free-text `<license><name>` into this same
15//! flat-identifier shape before evaluation. Free text like `"GNU General Public License
16//! v3"` never disambiguates the deprecated bare SPDX id (`GPL-3.0`) from the current
17//! `-only`/`-or-later` split, so the GPL/LGPL/AGPL families normalize to *all three*
18//! forms for one POM entry — the same "any entry matches" set-membership semantics
19//! `evaluate` already applies to a genuinely multi-licensed dependency (module docs
20//! above), so a `license_policy.deny`/`.allow` list written in either convention still
21//! matches. Dropping an entry the table doesn't recognize is fail-closed against a
22//! *false* violation (an unmatched entry can never manufacture a `Denied` or
23//! `NotAllowed` result it doesn't deserve), but it is **not** fail-closed with respect
24//! to policy *enforcement* — an unrecognized copyleft license on a `deny` list silently
25//! escapes rather than being flagged. Extending `KNOWN_POM_LICENSE_NAMES`'s coverage
26//! is the mitigation; there is no way to make an unrecognized string safe to guess at.
27
28use std::fmt;
29
30/// Maximum length of a single SPDX identifier accepted into a
31/// [`LicensePolicy`] — mirrors `deps-core::lsp_helpers::hover`'s
32/// `MAX_LICENSE_ID_CHARS` cap on the untrusted-length side of this same data
33/// (registry-declared license strings), applied here to user-supplied policy
34/// config instead.
35const MAX_SPDX_ID_CHARS: usize = 128;
36
37/// Cap on how many declared license entries [`evaluate`] joins into a
38/// [`LicenseViolation::license`] message for [`ViolationReason::NotAllowed`] — defense in
39/// depth against a compromised/malicious registry response declaring an excessive number of
40/// license entries for one dependency (issue #660/#661 critic security P2), mirroring
41/// `deps-core::lsp_helpers::hover`'s own `MAX_LICENSE_ENTRIES_RENDERED` cap for the same
42/// untrusted data.
43const MAX_LICENSE_ENTRIES_JOINED: usize = 8;
44
45/// Whether `id` is syntactically plausible as a single SPDX license
46/// identifier (e.g. `MIT`, `Apache-2.0`, `GPL-3.0-or-later`,
47/// `LicenseRef-Acme-Custom`) — a lightweight charset/length sanity check, not
48/// validation against the canonical SPDX License List (which would need a
49/// new dependency or an embedded list this feature's scope deliberately
50/// avoids, see the module docs).
51///
52/// Rejects anything containing whitespace or characters outside
53/// `[A-Za-z0-9.+-]`, which also rejects a raw multi-token SPDX *expression*
54/// (`"MIT OR Apache-2.0"`) — this module only ever matches single
55/// identifiers, so an expression could never match anyway and is better
56/// dropped with a warning than kept as a policy entry that silently never
57/// matches.
58fn is_syntactically_valid_spdx_id(id: &str) -> bool {
59    !id.is_empty()
60        && id.len() <= MAX_SPDX_ID_CHARS
61        && id
62            .chars()
63            .all(|c| c.is_ascii_alphanumeric() || matches!(c, '.' | '-' | '+'))
64}
65
66/// Filters `ids` down to entries `is_syntactically_valid_spdx_id` accepts,
67/// logging one `tracing::warn!` per dropped entry.
68///
69/// Shared by [`LicensePolicy::new`] and `deps-lsp`'s
70/// `initializationOptions.license_policy` config deserializer, so an invalid
71/// identifier is only ever validated (and warned about) once, at config-load
72/// time, regardless of how many times a [`LicensePolicy`] is subsequently
73/// constructed from the already-cleaned lists.
74///
75/// # Examples
76///
77/// ```
78/// use deps_core::licenses::filter_valid_spdx_ids;
79///
80/// let cleaned = filter_valid_spdx_ids(vec![
81///     "MIT".to_string(),
82///     "not a license".to_string(),
83/// ]);
84/// assert_eq!(cleaned, vec!["MIT".to_string()]);
85/// ```
86#[must_use]
87pub fn filter_valid_spdx_ids(ids: Vec<String>) -> Vec<String> {
88    ids.into_iter()
89        .filter(|id| {
90            let valid = is_syntactically_valid_spdx_id(id);
91            if !valid {
92                tracing::warn!(
93                    identifier = %id,
94                    "license policy: dropping invalid SPDX identifier"
95                );
96            }
97            valid
98        })
99        .collect()
100}

102/// An allow-list/deny-list SPDX license policy (issue #661), configured via
103/// `initializationOptions.license_policy: { allow?: string[], deny?: string[] }`.
104///
105/// Both lists are independently optional; an empty policy (`is_empty()`)
106/// matches nothing and [`evaluate`] always returns `None` for it. See
107/// [`evaluate`] for matching and precedence rules.
108#[non_exhaustive]
109#[derive(Debug, Clone, Default, PartialEq, Eq)]
110pub struct LicensePolicy {
111    /// SPDX identifiers a dependency's license must include at least one of,
112    /// when non-empty.
113    pub allow: Vec<String>,
114    /// SPDX identifiers a dependency's license must not include any of.
115    pub deny: Vec<String>,
116}
117
118impl LicensePolicy {
119    /// Builds a policy from raw, user-supplied SPDX identifier lists,
120    /// dropping any entry [`filter_valid_spdx_ids`] rejects.
121    ///
122    /// Never fails and never panics (NFR-003): an
123    /// `initializationOptions.license_policy` payload has no document URI to
124    /// anchor an LSP diagnostic to, so the `tracing::warn!` logged by
125    /// [`filter_valid_spdx_ids`] for a dropped entry is the only feedback
126    /// channel available for a typo'd identifier.
127    ///
128    /// # Examples
129    ///
130    /// ```
131    /// use deps_core::licenses::LicensePolicy;
132    ///
133    /// let policy = LicensePolicy::new(
134    ///     vec!["MIT".to_string(), "not a license".to_string()],
135    ///     vec!["GPL-3.0".to_string()],
136    /// );
137    /// assert_eq!(policy.allow, vec!["MIT".to_string()]);
138    /// assert_eq!(policy.deny, vec!["GPL-3.0".to_string()]);
139    /// ```
140    #[must_use]
141    pub fn new(allow: Vec<String>, deny: Vec<String>) -> Self {
142        Self {
143            allow: filter_valid_spdx_ids(allow),
144            deny: filter_valid_spdx_ids(deny),
145        }
146    }
147
148    /// Whether this policy has neither an allow-list nor a deny-list entry —
149    /// [`evaluate`] never produces a violation against an empty policy.
150    ///
151    /// # Examples
152    ///
153    /// ```
154    /// use deps_core::licenses::LicensePolicy;
155    ///
156    /// assert!(LicensePolicy::default().is_empty());
157    /// assert!(!LicensePolicy::new(vec!["MIT".to_string()], vec![]).is_empty());
158    /// ```
159    #[must_use]
160    pub fn is_empty(&self) -> bool {
161        self.allow.is_empty() && self.deny.is_empty()
162    }
163}

165/// Why a dependency's license violates a [`LicensePolicy`].
166#[non_exhaustive]
167#[derive(Debug, Clone, Copy, PartialEq, Eq)]
168pub enum ViolationReason {
169    /// The license matches an entry on the policy's deny-list. Deny always
170    /// wins over allow when a license matches both (spec 010 `plan.md`'s
171    /// "Allow vs deny precedence" decision — matches `cargo deny licenses`'
172    /// own precedence convention).
173    Denied,
174    /// The policy has a non-empty allow-list and none of the dependency's
175    /// declared licenses matches an entry on it.
176    NotAllowed,
177}
178
179impl fmt::Display for ViolationReason {
180    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
181        f.write_str(match self {
182            Self::Denied => "denied by policy",
183            Self::NotAllowed => "not on the allowed license list",
184        })
185    }
186}

188/// A policy violation for one dependency, ready to render as an LSP
189/// diagnostic (issue #661 FR-007).
190///
191/// Output-only: constructed internally by [`evaluate`], never by external code — no
192/// constructor is provided.
193#[non_exhaustive]
194#[derive(Debug, Clone, PartialEq, Eq)]
195pub struct LicenseViolation {
196    /// The specific declared license that triggered the violation, for
197    /// [`ViolationReason::Denied`]; every declared license joined with
198    /// `", "`, for [`ViolationReason::NotAllowed`] — no single entry alone
199    /// caused that violation.
200    pub license: String,
201    /// Why this dependency's license violates the policy.
202    pub reason: ViolationReason,
203}
204
205/// Checks `license` — a dependency's declared SPDX identifiers, e.g. from
206/// [`crate::Version::license`] — against `policy`, returning the violation
207/// if any.
208///
209/// Matching is exact, case-insensitive, top-level SPDX identifier
210/// set-membership (module docs) — no expression-operator parsing.
211///
212/// A dependency with no known license (`license.is_empty()`) never violates
213/// the policy: there is nothing to check, and treating "unknown" as "denied"
214/// would turn every dependency this feature hasn't fetched license data for
215/// into a false positive (NFR-003 graceful degradation).
216///
217/// A multi-licensed dependency (more than one entry in `license`, e.g. a
218/// dual-licensed `"MIT OR Apache-2.0"` reported as two identifiers) is
219/// denied if **any** entry matches the deny-list — the `Vec` license shape
220/// carries no `AND`/`OR` structure to know whether picking a different
221/// declared license would avoid the denied one, so a compliance gate errs
222/// toward flagging for manual review. It is allowed if **any** entry matches
223/// a non-empty allow-list — the permissive reading, since a multi-licensed
224/// dependency lets the consumer pick whichever license they comply with.
225///
226/// # Examples
227///
228/// ```
229/// use deps_core::licenses::{LicensePolicy, ViolationReason, evaluate};
230///
231/// let policy = LicensePolicy::new(
232///     vec!["MIT".to_string(), "Apache-2.0".to_string()],
233///     vec!["GPL-3.0".to_string()],
234/// );
235///
236/// // Denied wins even if also on the allow-list.
237/// let denied = evaluate(&["GPL-3.0".to_string()], &policy).unwrap();
238/// assert_eq!(denied.reason, ViolationReason::Denied);
239///
240/// // Not on the allow-list.
241/// let not_allowed = evaluate(&["ISC".to_string()], &policy).unwrap();
242/// assert_eq!(not_allowed.reason, ViolationReason::NotAllowed);
243///
244/// // Compliant.
245/// assert!(evaluate(&["MIT".to_string()], &policy).is_none());
246///
247/// // Unknown license never violates.
248/// assert!(evaluate(&[], &policy).is_none());
249/// ```
250#[must_use]
251pub fn evaluate(license: &[String], policy: &LicensePolicy) -> Option<LicenseViolation> {
252    if license.is_empty() {
253        return None;
254    }
255
256    if let Some(denied) = license.iter().find(|l| contains_ci(&policy.deny, l)) {
257        return Some(LicenseViolation {
258            license: denied.clone(),
259            reason: ViolationReason::Denied,
260        });
261    }
262
263    if !policy.allow.is_empty() && !license.iter().any(|l| contains_ci(&policy.allow, l)) {
264        return Some(LicenseViolation {
265            license: join_capped(license, MAX_LICENSE_ENTRIES_JOINED),
266            reason: ViolationReason::NotAllowed,
267        });
268    }
269
270    None
271}

273/// Maximum length of a raw free-text `<license><name>` value considered for normalization.
274///
275/// Mirrors `deps-core::lsp_helpers::hover`'s `MAX_LICENSE_ID_CHARS` (issue #660/#661
276/// critic security P2 precedent). The longest `KNOWN_POM_LICENSE_NAMES` key is under 60
277/// characters, so this rejects no real match; it exists purely to stop
278/// `collapse_whitespace`'s `split_whitespace().collect::<Vec<_>>().join(...)` from
279/// re-allocating a large token vector for an oversized, whitespace-dense value on every
280/// diagnostics pass (issue #679 security P2: an unbounded `<name>` gave ~10x memory
281/// amplification per dependency).
282///
283/// `pub` (#702, L1) so `deps-gradle::license`'s POM-parsing loop — which caps the raw
284/// `<license><name>` text node it allocates *before* it ever reaches this module's
285/// normalization step — shares this exact bound instead of redeclaring its own copy of
286/// the literal `128`.
287pub const MAX_POM_LICENSE_NAME_RAW_CHARS: usize = 128;

503/// Collapses runs of ASCII/Unicode whitespace in `s` down to a single space, so e.g.
504/// `"Apache  License,\nVersion 2.0"` (a POM's raw, occasionally multi-line free text)
505/// still matches the single-space-normalized [`KNOWN_POM_LICENSE_NAMES`] entries.
506fn collapse_whitespace(s: &str) -> String {
507    s.split_whitespace().collect::<Vec<_>>().join(" ")
508}

510/// Maps a Maven Central POM's free-text `<license><name>` to its canonical SPDX
511/// identifier(s) (issue #679).
512///
513/// Looks `raw` up (case-insensitive, whitespace-collapsed, never fuzzy/substring)
514/// against the explicit variant list in `KNOWN_POM_LICENSE_NAMES`, e.g. mapping
515/// `"The Apache Software License, Version 2.0"` to `["Apache-2.0"]` — see that
516/// constant's doc comment for why some families expand to more than one id. A `raw`
517/// value longer than `MAX_POM_LICENSE_NAME_RAW_CHARS` or that matches nothing in the
518/// table falls through to one more check: if it is already syntactically shaped like an
519/// SPDX id (`is_syntactically_valid_spdx_id`) *and* contains at least one ASCII digit,
520/// it is accepted as-is — a POM sometimes uses the SPDX id itself as the free-text name
521/// (e.g. `"AGPL-3.0"`, `"GPL-2.0-with-classpath-exception"`). The digit requirement is
522/// deliberate: it excludes genuinely ambiguous bare terms real POMs also use verbatim
523/// (`"GPL"`, `"BSD"`, `"Apache License"` with no version), which would otherwise pass
524/// the same charset check and produce a false match. Anything else returns an empty
525/// `Vec` rather than a guessed identifier, so a caller feeding the result into
526/// [`evaluate`] never risks a false allow/deny match on text this function doesn't
527/// explicitly recognize.
528///
529/// # Examples
530///
531/// ```
532/// use deps_core::licenses::normalize_pom_license_name;
533///
534/// assert_eq!(
535///     normalize_pom_license_name("The Apache Software License, Version 2.0"),
536///     vec!["Apache-2.0".to_string()]
537/// );
538/// assert_eq!(
539///     normalize_pom_license_name("GNU General Public License v3"),
540///     vec!["GPL-3.0".to_string(), "GPL-3.0-only".to_string(), "GPL-3.0-or-later".to_string()]
541/// );
542/// // Already SPDX-shaped free text passes through as-is.
543/// assert_eq!(normalize_pom_license_name("SSPL-1.0"), vec!["SSPL-1.0".to_string()]);
544/// // Genuinely ambiguous (no version) — never guessed at.
545/// assert!(normalize_pom_license_name("GPL").is_empty());
546/// assert!(normalize_pom_license_name("Some Bespoke Corporate License").is_empty());
547/// ```
548#[must_use]
549pub fn normalize_pom_license_name(raw: &str) -> Vec<String> {
550    if raw.len() > MAX_POM_LICENSE_NAME_RAW_CHARS {
551        return Vec::new();
552    }
553    let canonical = collapse_whitespace(raw.trim());
554    if canonical.is_empty() {
555        return Vec::new();
556    }
557    if let Some((_, ids)) = KNOWN_POM_LICENSE_NAMES
558        .iter()
559        .find(|(variant, _)| variant.eq_ignore_ascii_case(&canonical))
560    {
561        return ids.iter().map(|id| (*id).to_string()).collect();
562    }
563    if is_syntactically_valid_spdx_id(&canonical) && canonical.bytes().any(|b| b.is_ascii_digit()) {
564        return vec![canonical];
565    }
566    Vec::new()
567}

569/// Applies `normalize_pom_license_name` to every entry in `raw`, merging the results.
570///
571/// A thin wrapper around [`normalize_pom_license_names_checked`] that drops the
572/// "did every entry normalize" flag — use the checked variant directly instead of
573/// calling both in the same place, to avoid normalizing each entry twice (issue #679
574/// review nitpick).
575///
576/// # Examples
577///
578/// ```
579/// use deps_core::licenses::normalize_pom_license_names;
580///
581/// let normalized = normalize_pom_license_names(&[
582///     "The Apache Software License, Version 2.0".to_string(),
583///     "Some Bespoke Corporate License".to_string(),
584/// ]);
585/// assert_eq!(normalized, vec!["Apache-2.0".to_string()]);
586/// ```
587#[must_use]
588pub fn normalize_pom_license_names(raw: &[String]) -> Vec<String> {
589    normalize_pom_license_names_checked(raw).0
590}

592/// Applies `normalize_pom_license_name` to every entry in `raw`, merging the results
593/// and reporting whether every entry normalized successfully.
594///
595/// Deduplicates the combined ids and logs (`tracing::debug!`) each entry that doesn't
596/// match a known variant (issue #679 fail-closed contract) rather than keeping the free
597/// text or guessing — `debug!`, not `warn!`, since this runs on every diagnostics pass
598/// for ordinary registry data, not a one-time user config load like
599/// [`filter_valid_spdx_ids`]'s equivalent per-dropped-entry logging.
600///
601/// Used to bring a Gradle dependency's Maven Central POM license names into
602/// [`evaluate`]'s expected SPDX-identifier shape before license-policy evaluation. A
603/// dual-licensed POM with one recognized and one unrecognized entry still evaluates the
604/// recognized one — only the unmatched entry is excluded from the returned `Vec`, since
605/// `evaluate` requires no `AND`/`OR` structure between entries (module docs). The `bool`
606/// is `false` when at least one entry was dropped — a caller needs this to decide
607/// whether to trust a resulting `NotAllowed` evaluation, since the surviving ids are
608/// then incomplete evidence (issue #679 critic S1): "no entry matches allow" can't be
609/// trusted when an entry was silently excluded, but a `Denied` match is still valid
610/// regardless, since a recognized entry matching `deny` is real evidence either way.
611///
612/// # Examples
613///
614/// ```
615/// use deps_core::licenses::normalize_pom_license_names_checked;
616///
617/// let (ids, all_matched) = normalize_pom_license_names_checked(&["MIT License".to_string()]);
618/// assert_eq!(ids, vec!["MIT".to_string()]);
619/// assert!(all_matched);
620///
621/// let (ids, all_matched) = normalize_pom_license_names_checked(&[
622///     "MIT License".to_string(),
623///     "Some Bespoke Corporate License".to_string(),
624/// ]);
625/// assert_eq!(ids, vec!["MIT".to_string()]);
626/// assert!(!all_matched);
627/// ```
628#[must_use]
629pub fn normalize_pom_license_names_checked(raw: &[String]) -> (Vec<String>, bool) {
630    let mut normalized = Vec::new();
631    let mut all_matched = true;
632    for name in raw {
633        let ids = normalize_pom_license_name(name);
634        if ids.is_empty() {
635            all_matched = false;
636            tracing::debug!(
637                license = %name,
638                "license policy: gradle POM license name did not normalize to a known SPDX identifier"
639            );
640        }
641        for id in ids {
642            if !normalized.contains(&id) {
643                normalized.push(id);
644            }
645        }
646    }
647    (normalized, all_matched)
648}

650/// Resolves `raw` tier-3 pre-fetched license text into policy-evaluation-ready SPDX
651/// identifiers for `source` (issue #687/#688).
652///
653/// See [`resolve_license_entries_for_display`] for the sibling used to *render* the same
654/// data in hover. Applies [`normalize_pom_license_names_checked`] for [`crate::LicenseSource::PomFreeText`]
655/// and passes every other source through unchanged. The single normalization call site
656/// `generate_diagnostics_from_cache`'s license-policy rule reads through — before this,
657/// it re-derived its own Gradle-only `==` branch inline.
658///
659/// Deliberately fail-closed, unlike [`resolve_license_entries_for_display`]: an entry the
660/// normalization table doesn't recognize is dropped rather than guessed at (never
661/// fabricate a policy violation), and a POM entry that is genuinely one declared license
662/// but ambiguous between SPDX conventions (e.g. `"GNU General Public License v3"`)
663/// expands to every synonym (`GPL-3.0`/`-only`/`-or-later`) so a `deny`/`allow` list
664/// written in any of them still matches — both are correct for policy evaluation and
665/// wrong for a human-facing hover line, which is exactly why display uses a different
666/// function.
667///
668/// The second element mirrors [`normalize_pom_license_names_checked`]'s "did every entry
669/// normalize" flag — always `true` for a source with nothing to normalize, so a caller
670/// that uses it to gate a policy conclusion (see that function's doc) behaves identically
671/// for both a genuinely non-`PomFreeText` source and a `PomFreeText` source whose entries
672/// all recognized.
673///
674/// # Examples
675///
676/// ```
677/// use deps_core::LicenseSource;
678/// use deps_core::licenses::resolve_license_entries;
679///
680/// let (ids, all_matched) = resolve_license_entries(
681///     LicenseSource::PomFreeText,
682///     &["The Apache Software License, Version 2.0".to_string()],
683/// );
684/// assert_eq!(ids, vec!["Apache-2.0".to_string()]);
685/// assert!(all_matched);
686///
687/// let (ids, all_matched) =
688///     resolve_license_entries(LicenseSource::RegistryDeclaredSpdx, &["MIT".to_string()]);
689/// assert_eq!(ids, vec!["MIT".to_string()]);
690/// assert!(all_matched);
691/// ```
692#[must_use]
693pub fn resolve_license_entries(
694    source: crate::LicenseSource,
695    raw: &[String],
696) -> (Vec<String>, bool) {
697    match source {
698        crate::LicenseSource::PomFreeText => normalize_pom_license_names_checked(raw),
699        crate::LicenseSource::RegistryDeclaredSpdx
700        | crate::LicenseSource::FetchedDeclaredSpdx
701        | crate::LicenseSource::DetectedSpdx => (raw.to_vec(), true),
702    }
703}

705/// Resolves `raw` tier-3 pre-fetched license text for *display* (hover), for `source`
706/// (issue #687 critic S1/S2).
707///
708/// See [`resolve_license_entries`] for the sibling used to *evaluate* the same data
709/// against a [`LicensePolicy`]. Differs from it in exactly the two ways display needs:
710/// - **Never drops an entry.** [`resolve_license_entries`] fails closed and drops an
711///   entry [`normalize_pom_license_name`] doesn't recognize (correct for policy
712///   evaluation: never fabricate a violation from guessed data) — but the
713///   `KNOWN_POM_LICENSE_NAMES` table is deliberately non-exhaustive (see the module
714///   docs above), so dropping is the *designed-for* path, not a rare edge case, and
715///   applying it to hover made the license line silently vanish for any unrecognized
716///   POM name. Falls back to the raw text instead, since showing the author's declared
717///   string is strictly more useful to a reader than showing nothing.
718/// - **One id per entry when the ids are synonyms, not a policy-matching slice.** A POM
719///   name ambiguous between SPDX conventions (e.g. the GPL family) normalizes to three
720///   synonymous ids for [`resolve_license_entries`] so a `deny`/`allow` list written in
721///   any of them still matches — correct for policy, but printing `` `GPL-3.0`,
722///   `GPL-3.0-only`, `GPL-3.0-or-later` `` in hover for what is genuinely one declared
723///   license reads as three licenses. Keeps only the first (canonical) id in that case —
724///   but a genuinely disjunctive entry (`"CDDL + GPLv2 with classpath exception"` names
725///   two unrelated licenses, not synonyms of one) still shows every id, or a
726///   dual-licensed dependency would misleadingly read as single-licensed. See this
727///   module's private `ids_are_synonyms` helper for how the two shapes are told apart.
728///
729/// Entries are deduplicated (preserving first-seen order), same as
730/// [`normalize_pom_license_names_checked`].
731///
732/// # Examples
733///
734/// ```
735/// use deps_core::LicenseSource;
736/// use deps_core::licenses::resolve_license_entries_for_display;
737///
738/// // Recognized: normalizes to its canonical id, like `resolve_license_entries`.
739/// let ids = resolve_license_entries_for_display(
740///     LicenseSource::PomFreeText,
741///     &["The Apache Software License, Version 2.0".to_string()],
742/// );
743/// assert_eq!(ids, vec!["Apache-2.0".to_string()]);
744///
745/// // Unrecognized: falls back to the raw text instead of vanishing (issue #687 S1).
746/// let ids = resolve_license_entries_for_display(
747///     LicenseSource::PomFreeText,
748///     &["Some Bespoke Corporate License".to_string()],
749/// );
750/// assert_eq!(ids, vec!["Some Bespoke Corporate License".to_string()]);
751///
752/// // Ambiguous SPDX convention: one canonical id, not the full policy-matching slice
753/// // (issue #687 S2).
754/// let ids = resolve_license_entries_for_display(
755///     LicenseSource::PomFreeText,
756///     &["GNU General Public License v3".to_string()],
757/// );
758/// assert_eq!(ids, vec!["GPL-3.0".to_string()]);
759///
760/// // Genuinely disjunctive (two unrelated licenses, not synonyms of one): both ids
761/// // are kept, unlike the GPL-family case above (code-review must-fix).
762/// let ids = resolve_license_entries_for_display(
763///     LicenseSource::PomFreeText,
764///     &["CDDL + GPLv2 with classpath exception".to_string()],
765/// );
766/// assert_eq!(
767///     ids,
768///     vec!["CDDL-1.1".to_string(), "GPL-2.0-with-classpath-exception".to_string()]
769/// );
770/// ```
771#[must_use]
772pub fn resolve_license_entries_for_display(
773    source: crate::LicenseSource,
774    raw: &[String],
775) -> Vec<String> {
776    match source {
777        crate::LicenseSource::PomFreeText => {
778            let mut resolved = Vec::new();
779            for name in raw {
780                let ids = normalize_pom_license_name(name);
781                let display_ids: Vec<String> = if ids.is_empty() {
782                    vec![name.clone()]
783                } else if ids_are_synonyms(&ids) {
784                    // Safe to collapse: every id names the same declared license,
785                    // just a different SPDX-convention spelling of it.
786                    ids.into_iter().take(1).collect()
787                } else {
788                    // A genuine multi-license OR (e.g. `"CDDL + GPLv2 with classpath
789                    // exception"`) — every id names a different license, so all must
790                    // be shown or a dual-licensed dependency reads as single-licensed.
791                    ids
792                };
793                for id in display_ids {
794                    if !resolved.contains(&id) {
795                        resolved.push(id);
796                    }
797                }
798            }
799            resolved
800        }
801        crate::LicenseSource::RegistryDeclaredSpdx
802        | crate::LicenseSource::FetchedDeclaredSpdx
803        | crate::LicenseSource::DetectedSpdx => raw.to_vec(),
804    }
805}

807/// Whether `ids` — one [`KNOWN_POM_LICENSE_NAMES`] entry's normalized SPDX id list —
808/// represents interchangeable synonyms of a single declared license (safe for
809/// [`resolve_license_entries_for_display`] to collapse to one representative id)
810/// rather than a genuine multi-license OR (every id must be shown, code-review
811/// must-fix: collapsing `"CDDL + GPLv2 with classpath exception"` to one id
812/// misrepresented a dual-licensed dependency as solely CDDL-licensed).
813///
814/// Every synonym-set entry in the table is formed by SPDX's own `-only`/`-or-later`
815/// suffix convention over one shared base id (e.g. `GPL-3.0`, `GPL-3.0-only`,
816/// `GPL-3.0-or-later` — see the table's own doc comment on why free text can't
817/// disambiguate the convention), so every id after the first is that first id with a
818/// suffix appended. The table's one genuinely disjunctive entry names two unrelated
819/// license families with no such shared prefix, so a plain prefix check tells the two
820/// shapes apart without a separate marker on every one of the table's ~90 rows. A
821/// single-id (or empty) list is trivially "synonyms" — there is nothing to collapse.
822fn ids_are_synonyms(ids: &[String]) -> bool {
823    match ids.split_first() {
824        Some((first, rest)) => rest.iter().all(|id| id.starts_with(first.as_str())),
825        None => true,
826    }
827}

829/// Case-insensitive `haystack.contains(needle)` for SPDX identifiers —
830/// registry-declared and user-configured license strings are free text, not
831/// a normalized enum, mirroring `deps-core::lsp_helpers::hover`'s
832/// `license_sets_differ` precedent for the same data.
833fn contains_ci(haystack: &[String], needle: &str) -> bool {
834    haystack.iter().any(|h| h.eq_ignore_ascii_case(needle))
835}

837/// Joins `licenses` with `", "`, capping the number of entries rendered at `max_entries`
838/// and collapsing the remainder into a `"(+N more)"` suffix (issue #660/#661 critic
839/// security P2) — mirrors `lsp_helpers::hover::format_license_list`'s shape for the same
840/// registry-controlled, unbounded-length data.
841fn join_capped(licenses: &[String], max_entries: usize) -> String {
842    let shown = licenses.len().min(max_entries);
843    // `shown <= licenses.len()` by construction (the `.min` above), so `.get(..shown)`
844    // never actually falls back — `unwrap_or(licenses)` just satisfies
845    // `clippy::indexing_slicing` (issue #678 hardening) without panicking if that
846    // invariant is ever violated.
847    let mut joined = licenses.get(..shown).unwrap_or(licenses).join(", ");
848    let remaining = licenses.len() - shown;
849    if remaining > 0 {
850        joined.push_str(&format!(" (+{remaining} more)"));
851    }
852    joined
853}

1333    /// Code-review follow-up: `ids_are_synonyms` assumes every synonym-set entry in
1334    /// `KNOWN_POM_LICENSE_NAMES` lists its base SPDX id first, with every later id
1335    /// formed by appending an `-only`/`-or-later` suffix to it —
1336    /// `resolve_license_entries_for_display` relies on that ordering to collapse a
1337    /// synonym set to one id for hover (issue #687 S2) without also collapsing the
1338    /// table's genuinely disjunctive entry. This walks the whole table and locks the
1339    /// classification in, so a future entry added base-id-last (or a new disjunctive
1340    /// entry not added to `DISJUNCTIVE_KEYS` below) fails loudly here instead of
1341    /// silently breaking hover's collapse/no-collapse decision.
1342    #[test]
1343    fn known_pom_license_names_synonym_entries_list_base_id_first() {
1344        // The table's only entry naming two genuinely unrelated licenses (not
1345        // convention-ambiguous synonyms of one) — see that entry's own comment.
1346        const DISJUNCTIVE_KEYS: &[&str] = &["CDDL + GPLv2 with classpath exception"];
1347
1348        for &(key, ids) in KNOWN_POM_LICENSE_NAMES {
1349            let ids_owned: Vec<String> = ids.iter().map(|id| (*id).to_string()).collect();
1350            let expected_disjunctive = DISJUNCTIVE_KEYS.contains(&key);
1351            assert_eq!(
1352                ids_are_synonyms(&ids_owned),
1353                !expected_disjunctive,
1354                "{key:?} -> {ids:?}: expected {}, ids_are_synonyms said {}",
1355                if expected_disjunctive {
1356                    "disjunctive (base id not first, or a new genuine OR)"
1357                } else {
1358                    "synonyms (base id first)"
1359                },
1360                ids_are_synonyms(&ids_owned)
1361            );
1362        }
1363    }
