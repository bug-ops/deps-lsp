1//! Shared scan-budget constants for a bounded `quick-xml` event loop over a remote,
2//! registry-generated XML response (#698).
3//!
4//! `deps-gradle::license`'s POM-parsing loop already bounds itself with a private
5//! entry-count + `buffer_position()` pair (issue #690/#694); this module exists so a
6//! second remote-XML consumer — `deps-maven::registry::parse_metadata_xml` — shares the
7//! same budget shape instead of inventing its own. It deliberately does not wrap
8//! `quick_xml::Reader` itself: the workspace's XML consumers split between
9//! `Reader::from_str` + `read_event()` and `Reader::from_reader` + `read_event_into(&mut
10//! buf)`, and at least one (`deps-nuget::parser`) relies on the zero-copy borrow that
11//! `from_str` gives deliberately — a shared wrapper would have to fight that for no
12//! security gain. Each consumer keeps its own reader setup and calls [`exhausted`] as a
13//! loop-top guard.
14
15/// Entry-count threshold at which a bounded XML scan is rejected as exhausted.
16///
17/// Once `<version>` (or equivalent per-entry) elements retained by the scan reach this
18/// count, [`exhausted`] returns `true` and the caller returns `Err` — so the largest
19/// document actually *accepted* holds `MAX_METADATA_VERSIONS - 1` entries, not this count
20/// itself. Real Maven Central artifacts top out around 3-4k versions for even the
21/// longest-lived groups, so this leaves generous headroom above any legitimate document.
22pub const MAX_METADATA_VERSIONS: usize = 8192;
23
24/// Hard cap, in bytes of `reader.buffer_position()`, on how far into a document a bounded XML scan will read.
25///
26/// A real `maven-metadata.xml` with a few thousand versions is on the order of 100 KB, so
27/// this leaves generous headroom while still bounding a hostile or pathological response.
28pub const MAX_METADATA_BYTES_SCANNED: usize = 8 * 1024 * 1024;
29
30/// Whether a bounded XML scan has exceeded its budget and must stop.
31///
32/// Checked at the top of the read loop, before processing the next event — mirrors the
33/// shape-independent guard `deps-gradle::license::parse_pom_licenses` already uses: it
34/// advances on every event regardless of what it is, so it cannot be starved by a
35/// document that simply omits the element type a narrower, shape-keyed counter would be
36/// watching for.
37///
38/// Thin wrapper around [`exhausted_with`] using this module's own
39/// [`MAX_METADATA_VERSIONS`]/[`MAX_METADATA_BYTES_SCANNED`] budget — use `exhausted_with`
40/// directly for a consumer with its own, differently-sized budget (e.g.
41/// `deps-gradle::license`'s POM scan, issue #725).
42///
43/// # Examples
44///
45/// ```
46/// use deps_core::xml_bounds::{exhausted, MAX_METADATA_VERSIONS};
47///
48/// assert!(!exhausted(0, 0));
49/// assert!(exhausted(MAX_METADATA_VERSIONS, 0));
50/// ```
51#[must_use]
52pub fn exhausted(entries: usize, buffer_position: u64) -> bool {
53    exhausted_with(
54        entries,
55        buffer_position,
56        MAX_METADATA_VERSIONS,
57        MAX_METADATA_BYTES_SCANNED,
58    )
59}
60
61/// Whether a bounded XML scan has exceeded a caller-supplied entry/byte budget.
62///
63/// Same shape-independent check as [`exhausted`], parameterized so a consumer with its
64/// own budget (e.g. a POM's license-entry cap, distinct from a `maven-metadata.xml`
65/// version-list cap) can share this conversion instead of keeping a private, lossy `as
66/// usize` copy of it (issue #725). The `usize::try_from(..).unwrap_or(usize::MAX)`
67/// fail-closed path only differs from a lossy `as usize` cast on a target where `usize`
68/// is narrower than `u64` — i.e. a 32-bit target; every target this workspace's CI
69/// actually builds for (see `.github/workflows/ci.yml`'s `cross-check` matrix) is
70/// 64-bit, where the two are equivalent. This is 32-bit-target hardening and DRY
71/// consolidation of the budget check, not a fix for a bug reachable on a supported
72/// target today.
73///
74/// # Examples
75///
76/// ```
77/// use deps_core::xml_bounds::exhausted_with;
78///
79/// assert!(!exhausted_with(0, 0, 64, 1024));
80/// assert!(exhausted_with(64, 0, 64, 1024));
81/// assert!(exhausted_with(0, 2048, 64, 1024));
82/// ```
83#[must_use]
84pub fn exhausted_with(
85    entries: usize,
86    buffer_position: u64,
87    max_entries: usize,
88    max_bytes: usize,
89) -> bool {
90    entries >= max_entries
91        // Fails closed: a `buffer_position` too large to fit `usize` (only possible on a
92        // 32-bit target) is treated as exceeding the cap rather than silently wrapping.
93        || usize::try_from(buffer_position).unwrap_or(usize::MAX) >= max_bytes
94}

96#[cfg(test)]
97mod tests {
98    use super::{MAX_METADATA_BYTES_SCANNED, MAX_METADATA_VERSIONS, exhausted, exhausted_with};
99
100    #[test]
101    fn exhausted_delegates_to_exhausted_with_using_module_constants() {
102        assert!(!exhausted(0, 0));
103        assert!(exhausted(MAX_METADATA_VERSIONS, 0));
104        assert!(exhausted(0, MAX_METADATA_BYTES_SCANNED as u64));
105    }
106
107    #[test]
108    fn exhausted_with_respects_caller_supplied_budget() {
109        assert!(!exhausted_with(0, 0, 64, 1024));
110        assert!(exhausted_with(64, 0, 64, 1024));
111        assert!(exhausted_with(0, 1024, 64, 1024));
112        assert!(!exhausted_with(0, 1023, 64, 1024));
113    }
114
115    /// A `buffer_position` that overflows a 32-bit `usize` must fail closed — treated as
116    /// exceeding the cap — rather than wrapping via a lossy `as usize` cast (the pre-#725
117    /// bug). `u32::MAX + 1` is the smallest such value, and only fails to fit `usize` on
118    /// a genuine 32-bit target, so this test is gated to actually distinguish the fix
119    /// from the bug it replaces — asserting it on 64-bit (where the conversion always
120    /// succeeds and the two implementations agree) would be a vacuous, always-passing
121    /// test that proves nothing about the fix.
122    #[test]
123    #[cfg(target_pointer_width = "32")]
124    fn exhausted_with_fails_closed_on_oversized_buffer_position() {
125        let overflows_32_bit_usize = u64::from(u32::MAX) + 1;
126        assert!(exhausted_with(
127            0,
128            overflows_32_bit_usize,
129            usize::MAX,
130            usize::MAX
131        ));
132    }
133}
