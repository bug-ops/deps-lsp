1//! Shared `serde_json`-side helpers for leniently interpreting untrusted JSON value shapes
2//! from a manifest or registry response (#624, #662).
3//!
4//! `deps-npm` and `deps-composer` each parse a dependency section as a raw
5//! `serde_json::Map<String, Value>` and apply the same "is this a valid dependency
6//! declaration" rule to every entry; `deps-composer` and `deps-nuget` each accept a field
7//! that may be a bare string or an array of strings — this module is where those rules
8//! live, so every ecosystem crate that needs them shares one implementation and one set of
9//! tests instead of duplicating both.
10//!
11//! This is a different concern from [`crate::json_ast`], which recovers a dependency's
12//! *position* in the source text from a separate jsonc-AST parse of the same content — this
13//! module never touches that AST, only the `serde_json::Value` tree callers already hold.
14
15/// Iterates a JSON object's entries, yielding only those whose value is a plain string.
16///
17/// A manifest entry whose value isn't a string (e.g. an object, number, bool, null, or array)
18/// is not a valid dependency declaration — callers skip it rather than fabricating an entry
19/// with no version requirement that would still be queried against the registry (`deps-npm`
20/// #619, `deps-composer` #621).
21///
22/// Yields entries in `entries`' own iteration order, which `serde_json::Map` derives from
23/// either `BTreeMap` (sorted by key) or `IndexMap` (insertion order), depending on this
24/// build's `serde_json/preserve_order` feature unification — do not write a test against a
25/// specific order without accounting for that.
26///
27/// # Examples
28///
29/// ```
30/// use deps_core::json_helpers::string_valued_entries;
31/// use serde_json::json;
32///
33/// let deps = json!({"express": "^4.18.2", "nested-shadow": {"express": "0.0.1"}});
34/// let deps = deps.as_object().unwrap();
35///
36/// let entries: Vec<_> = string_valued_entries(deps).collect();
37/// assert_eq!(entries, vec![("express", "^4.18.2")]);
38/// ```
39pub fn string_valued_entries(
40    entries: &serde_json::Map<String, serde_json::Value>,
41) -> impl Iterator<Item = (&str, &str)> {
42    entries
43        .iter()
44        .filter_map(|(name, value)| Some((name.as_str(), value.as_str()?)))
45}

47/// Accepts a JSON value that is either a single string or an array of strings, and
48/// normalizes it to `Option<Vec<String>>` — never erroring on an unexpected shape.
49///
50/// #662, consolidated from `deps-composer`'s `deserialize_license` (#204) and
51/// `deps-nuget`'s `deserialize_type_list` (#523).
52///
53/// Input classes and their result:
54///
55/// - a bare string (`"MIT"`) → `Some(vec!["MIT"])`
56/// - an array of strings (`["MIT", "Apache-2.0"]`) → `Some(vec!["MIT", "Apache-2.0"])`
57/// - an empty array (`[]`) → `Some(vec![])`
58/// - a mixed array with some non-string elements (`["MIT", 7]`) → `Some(vec!["MIT"])`,
59///   dropping only the non-string elements
60/// - a non-empty array whose elements are *all* non-string (`[7, 8]`), or any other JSON
61///   shape (a number, bool, object, `null`, or an absent field) → `None`
62///
63/// The general contract this establishes: `None` means "this value was not meaningfully
64/// present" and `Some(vec![])` means "this value was present and is explicitly empty" — a
65/// caller with no such absent-vs-empty distinction to make can collapse both to an empty
66/// list via `.unwrap_or_default()`. The `[7, 8]` → `None` case (rather than `Some(vec![])`)
67/// exists so that a caller treating `Some(_)` as an explicit override never sees a
68/// fabricated empty override manufactured from malformed input data — see
69/// `deps-composer`'s `MinifiedVersion::license` doc for a concrete case where this
70/// distinction is load-bearing.
71///
72/// # Errors
73///
74/// Returns an error only if the underlying `deserializer` cannot produce a
75/// `serde_json::Value` at all (e.g. malformed input at the transport level) — no JSON
76/// *shape* this function receives is itself treated as an error.
77///
78/// # Examples
79///
80/// ```
81/// use deps_core::json_helpers::deserialize_string_or_string_array;
82/// use serde_json::json;
83///
84/// assert_eq!(
85///     deserialize_string_or_string_array(json!("MIT")).unwrap(),
86///     Some(vec!["MIT".to_string()])
87/// );
88/// assert_eq!(
89///     deserialize_string_or_string_array(json!(["MIT", 7])).unwrap(),
90///     Some(vec!["MIT".to_string()])
91/// );
92/// assert_eq!(deserialize_string_or_string_array(json!([7, 8])).unwrap(), None);
93/// assert_eq!(deserialize_string_or_string_array(json!(null)).unwrap(), None);
94/// ```
95pub fn deserialize_string_or_string_array<'de, D>(
96    deserializer: D,
97) -> std::result::Result<Option<Vec<String>>, D::Error>
98where
99    D: serde::Deserializer<'de>,
100{
101    use serde::Deserialize;
102
103    let value = serde_json::Value::deserialize(deserializer)?;
104    Ok(match value {
105        serde_json::Value::String(s) => Some(vec![s]),
106        serde_json::Value::Array(entries) => {
107            // A non-empty raw array whose every element is a non-string (e.g. `[7, 8]`)
108            // must degrade to `None` ("not meaningfully present"), not `Some(vec![])`
109            // ("explicitly empty") — see this function's doc for why that distinction
110            // matters to a caller like `deps-composer`'s inheritance chain.
111            let had_entries = !entries.is_empty();
112            let strings: Vec<String> = entries
113                .into_iter()
114                .filter_map(|entry| match entry {
115                    serde_json::Value::String(s) => Some(s),
116                    _ => None,
117                })
118                .collect();
119            if had_entries && strings.is_empty() {
120                None
121            } else {
122                Some(strings)
123            }
124        }
125        _ => None,
126    })
127}

129#[cfg(test)]
130mod tests {
131    use super::*;

133    // --- `string_valued_entries` (#624: consolidated from deps-npm #619 / deps-composer #621) ---
134
135    #[test]
136    fn test_string_valued_entries_skips_every_non_string_value_kind() {
137        let deps = serde_json::json!({
138            "bad-object": {"nested": "0.0.1"},
139            "bad-number": 1,
140            "bad-bool": true,
141            "bad-null": null,
142            "bad-array": ["1.0.0"],
143            "express": "^4.18.2",
144        });
145        let deps = deps.as_object().unwrap();
146
147        let mut entries: Vec<_> = string_valued_entries(deps).collect();
148        entries.sort_unstable();
149        assert_eq!(entries, vec![("express", "^4.18.2")]);
150    }
151
152    #[test]
153    fn test_string_valued_entries_all_invalid_yields_empty() {
154        let deps = serde_json::json!({
155            "bad-object": {"nested": "0.0.1"},
156            "bad-number": 1,
157        });
158        let deps = deps.as_object().unwrap();
159
160        assert_eq!(string_valued_entries(deps).count(), 0);
161    }
162
163    #[test]
164    fn test_string_valued_entries_empty_object_yields_empty() {
165        let deps = serde_json::json!({});
166        let deps = deps.as_object().unwrap();
167
168        assert_eq!(string_valued_entries(deps).count(), 0);
169    }

174    #[test]
175    fn test_deserialize_string_or_string_array_bare_string() {
176        assert_eq!(
177            deserialize_string_or_string_array(serde_json::json!("MIT")).unwrap(),
178            Some(vec!["MIT".to_string()])
179        );
180    }

182    #[test]
183    fn test_deserialize_string_or_string_array_of_strings() {
184        assert_eq!(
185            deserialize_string_or_string_array(serde_json::json!(["MIT", "Apache-2.0"])).unwrap(),
186            Some(vec!["MIT".to_string(), "Apache-2.0".to_string()])
187        );
188    }
189
190    #[test]
191    fn test_deserialize_string_or_string_array_empty_array() {
192        assert_eq!(
193            deserialize_string_or_string_array(serde_json::json!([])).unwrap(),
194            Some(vec![])
195        );
196    }
197
198    #[test]
199    fn test_deserialize_string_or_string_array_mixed_array_keeps_only_strings() {
200        assert_eq!(
201            deserialize_string_or_string_array(serde_json::json!(["MIT", 7, "Apache-2.0"]))
202                .unwrap(),
203            Some(vec!["MIT".to_string(), "Apache-2.0".to_string()])
204        );
205    }
206
207    /// Non-empty array with no string element must degrade to `None` ("not present"), not
208    /// `Some(vec![])` ("explicitly empty") — a caller like `deps-composer`'s minified-entry
209    /// inheritance chain would otherwise treat a malformed array as an explicit override
210    /// that silently wipes out (and then propagates) a real inherited value.
211    #[test]
212    fn test_deserialize_string_or_string_array_all_non_string_array_is_none() {
213        assert_eq!(
214            deserialize_string_or_string_array(serde_json::json!([7, 8])).unwrap(),
215            None
216        );
217    }
218
219    #[test]
220    fn test_deserialize_string_or_string_array_other_shapes_are_none() {
221        for value in [
222            serde_json::json!(42),
223            serde_json::json!(true),
224            serde_json::json!({"not": "a list"}),
225            serde_json::json!(null),
226        ] {
227            assert_eq!(deserialize_string_or_string_array(value).unwrap(), None);
228        }
229    }
