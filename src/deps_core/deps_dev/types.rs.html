1//! Wire types for deps.dev API v3 responses, plus the public,
2//! ecosystem-agnostic types the hover-formatting layer consumes.
3//!
4//! Wire types declare only the fields this spec actually reads — serde
5//! ignores unknown fields by default, so deps.dev's much larger response
6//! shape (`licenses`, `isDeprecated`, `advisoryKeys`, `checks[]`,
7//! `scorecard.version`, ...) never needs a struct field here (spec §5,
8//! Out of Scope).
9
10use serde::Deserialize;
11
12/// Parsed subset of `GET /v3/systems/{system}/packages/{name}/versions/{version}`.
13#[derive(Deserialize)]
14#[serde(rename_all = "camelCase")]
15pub(super) struct DepsDevVersionInfo {
16    #[serde(default)]
17    pub(super) slsa_provenances: Vec<ProvenanceEntry>,
18    #[serde(default)]
19    pub(super) attestations: Vec<ProvenanceEntry>,
20    #[serde(default)]
21    pub(super) related_projects: Vec<RelatedProject>,
22}
23
24/// One `slsaProvenances[]`/`attestations[]` entry.
25#[derive(Deserialize)]
26#[serde(rename_all = "camelCase")]
27pub(super) struct ProvenanceEntry {
28    #[serde(default)]
29    pub(super) verified: bool,
30}

32/// One `relatedProjects[]` entry.
33#[derive(Deserialize)]
34#[serde(rename_all = "camelCase")]
35pub(super) struct RelatedProject {
36    pub(super) project_key: ProjectKey,
37    pub(super) relation_type: String,
38    #[serde(default)]
39    pub(super) relation_provenance: String,
40}

42/// The nested `projectKey` object carrying the project's id
43/// (e.g. `github.com/expressjs/express`).
44#[derive(Deserialize)]
45pub(super) struct ProjectKey {
46    pub(super) id: String,
47}

49/// Parsed subset of `GET /v3/projects/{project-key}`.
50#[derive(Deserialize)]
51pub(super) struct DepsDevProject {
52    pub(super) scorecard: Option<DepsDevScorecardWire>,
53}
54
55/// The `scorecard` object's consumed field.
56///
57/// `overall_score` is `Option<f32>` with **no** `#[serde(default)]`: serde
58/// already treats a missing or `null` key as `None` for an `Option` field,
59/// but this must stay `Option`, never a defaulted `f32`, so an absent score
60/// can never be conflated with a real `0.0` (a defaulted zero would itself be
61/// a false trust claim — a project deps.dev has no Scorecard for would render
62/// as maximally damning instead of simply omitted).
63#[derive(Deserialize)]
64#[serde(rename_all = "camelCase")]
65pub(super) struct DepsDevScorecardWire {
66    pub(super) overall_score: Option<f32>,
67}
68
69/// The three-state SLSA/attestation provenance verdict for one resolved version (FR-004).
70///
71/// [`Self::Verified`] and [`Self::Unverified`] are both only reachable when the
72/// version-level query itself succeeded, so a caller can distinguish "we checked and
73/// found nothing" from "we didn't check" via `Option<ProvenanceStatus>` at the
74/// [`SupplyChainTrustSignal`] level.
75#[derive(Debug, Clone, Copy, PartialEq, Eq)]
76pub enum ProvenanceStatus {
77    /// At least one `slsaProvenances[]`/`attestations[]` entry has `verified == true`.
78    Verified,
79    /// Both arrays are non-empty, but no entry has `verified == true`.
80    Unverified,
81    /// Both arrays are empty.
82    None,
83}
84
85/// The hover-facing OpenSSF Scorecard summary for a package's linked source
86/// repository.
87#[derive(Debug, Clone, PartialEq)]
88pub struct ScorecardSummary {
89    /// deps.dev's `scorecard.overallScore`, already validated to be a finite
90    /// value in `0.0..=10.0` — never a defaulted zero.
91    pub overall_score: f32,
92    /// `true` when the `SOURCE_REPO` relation this score was fetched for was
93    /// only `UNVERIFIED_METADATA` (package-self-reported), not
94    /// `SLSA_ATTESTATION` — see [`crate::deps_dev`] module docs for why this
95    /// matters and how it is rendered.
96    pub self_reported: bool,
97}
98
99/// The ecosystem-agnostic aggregate assembled from deps.dev's two calls,
100/// consumed by the hover-formatting layer.
101///
102/// Every field being `None` means "nothing to render". `trust_signal`'s
103/// success path never actually produces that all-`None` shape today
104/// (`provenance` is always `Some` once the version call itself succeeds),
105/// so `push_trust_signal_hover_section`'s own `scorecard.is_none() &&
106/// provenance.is_none()` check is defensive, not dead. Modelling both
107/// fields as independently `Option` still matters: see
108/// [`ProvenanceStatus`]'s docs for why `provenance` in particular stays
109/// `Option` rather than collapsing into `scorecard`'s shape.
110#[derive(Debug, Clone, Default, PartialEq)]
111pub struct SupplyChainTrustSignal {
112    /// The linked source repository's OpenSSF Scorecard, when one could be
113    /// resolved and fetched (FR-002/FR-003/FR-005).
114    pub scorecard: Option<ScorecardSummary>,
115    /// This version's SLSA/attestation provenance status, when the
116    /// version-level query itself succeeded (FR-004).
117    pub provenance: Option<ProvenanceStatus>,
118}
