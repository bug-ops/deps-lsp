1//! Wire types for the OSV.dev batch and single-query APIs, and the
2//! `deps-lsp`-facing types derived from them.
3//!
4//! The wire types deliberately mirror OSV's schema sparsely: every optional
5//! field uses `#[serde(default)]` because OSV records are sparse and the
6//! schema evolves, and a missing field must never fail the whole response
7//! (see `architecture.md` §6).

16/// One dependency to query against OSV.
17///
18/// Four distinct strings, deliberately: `key` is this project's internal
19/// lookup key (`EcosystemFormatter::normalize_package_name`), `osv_name` is
20/// OSV's canonical spelling (`EcosystemFormatter::osv_package_name`). The
21/// transform from `key` to `osv_name` is not round-trippable (Swift
22/// `owner/repo` -> `github.com/owner/repo`; NuGet raw vs Composer lowercased),
23/// so the client cannot reconstruct the map key from what it sends on the
24/// wire — both must be carried alongside each other.
25///
26/// `version` and `display_version` are the same split, one level down: `version`
27/// is what gets sent on the wire (`EcosystemFormatter::osv_version`), while
28/// `display_version` is the ecosystem-native spelling a caller should surface
29/// back to the user (e.g. in [`crate::osv::UpgradeStatus`]). They coincide for
30/// every ecosystem except Go, where `osv_version` strips the mandatory `v`
31/// prefix — a caller that echoed `version` in an upgrade suggestion would show
32/// `1.2.3` instead of the `go.mod`-native `v1.2.3`.
33///
34/// # Examples
35///
36/// ```
37/// use deps_core::osv::ScanTarget;
38///
39/// let target = ScanTarget {
40///     key: "time".to_string(),
41///     osv_name: "time".to_string(),
42///     version: "0.1.43".to_string(),
43///     display_version: "0.1.43".to_string(),
44/// };
45/// assert_eq!(target.key, target.osv_name);
46/// ```
47#[derive(Debug, Clone, PartialEq, Eq)]
48pub struct ScanTarget {
49    /// This project's internal lookup key — used to key [`VulnerabilityMap`].
50    pub key: String,
51    /// OSV's canonical package name for this ecosystem — sent on the wire.
52    pub osv_name: String,
53    /// Concrete version to query, resolved per the version-selection policy
54    /// and rewritten to OSV's wire spelling via
55    /// `EcosystemFormatter::osv_version`. Never surface this to the user —
56    /// use [`Self::display_version`] instead.
57    pub version: String,
58    /// The same version in the ecosystem's native spelling (pre-`osv_version`
59    /// rewrite), for callers that need to display it back to the user rather
60    /// than send it to OSV.
61    pub display_version: String,
62}

64/// Severity bucket derived from an OSV advisory record.
65///
66/// See `architecture.md` §6 for the precedence rules used to derive this
67/// from a raw record, and [`crate::osv::diagnostic_severity_for`] for the mapping to
68/// [`tower_lsp_server::ls_types::DiagnosticSeverity`].
69#[derive(Debug, Clone, Copy, PartialEq, Eq)]
70pub enum VulnSeverity {
71    /// `database_specific.severity` or `ecosystem_specific.severity` reported `CRITICAL`.
72    Critical,
73    /// Reported `HIGH`.
74    High,
75    /// Reported `MODERATE` or `MEDIUM`.
76    Medium,
77    /// Reported `LOW`.
78    Low,
79    /// No severity field was present or recognized on the record.
80    Unknown,
81}

83/// A single vulnerability advisory, converted from OSV's wire format at the
84/// crate boundary.
85///
86/// # Examples
87///
88/// ```
89/// use deps_core::osv::{Advisory, VulnSeverity};
90///
91/// let advisory = Advisory {
92///     id: "RUSTSEC-2020-0071".to_string(),
93///     modified: "2023-01-01T00:00:00Z".to_string(),
94///     summary: Some("Potential segfault in the time crate".to_string()),
95///     aliases: vec!["CVE-2020-26235".to_string()],
96///     severity: VulnSeverity::High,
97///     cvss_vector: None,
98///     fixed_versions: vec!["0.2.23".to_string()],
99///     url: "https://osv.dev/vulnerability/RUSTSEC-2020-0071".to_string(),
100/// };
101/// assert_eq!(advisory.fixed_versions.last(), Some(&"0.2.23".to_string()));
102/// ```
103#[derive(Debug, Clone, PartialEq, Eq)]
104pub struct Advisory {
105    /// Advisory identifier (e.g. `"RUSTSEC-2020-0071"`, `"GHSA-..."`).
106    pub id: String,
107    /// RFC3339 last-modified timestamp — the [`crate::osv::OsvClient`] record-cache validator.
108    pub modified: String,
109    /// Human-readable one-line summary, if OSV provided one.
110    pub summary: Option<String>,
111    /// Alternate identifiers (CVE, GHSA, ...).
112    pub aliases: Vec<String>,
113    /// Derived severity bucket.
114    pub severity: VulnSeverity,
115    /// Raw CVSS vector string, shown verbatim in hover but never parsed.
116    pub cvss_vector: Option<String>,
117    /// All `fixed` events found in the record's ranges, ascending. May be
118    /// empty if OSV recorded no fix. The highest entry is the one to surface
119    /// as "the fix" — see `architecture.md` §6 for why the *first* one is not.
120    pub fixed_versions: Vec<String>,
121    /// `https://osv.dev/vulnerability/{id}`.
122    pub url: String,
123}

125/// A list that may have been truncated when it was produced, paired with the
126/// true count of items that existed at the source.
127///
128/// Exists so a truncated list can never be silently read as complete: items are
129/// only reachable through [`Capped::items`] and the real count only through
130/// [`Capped::total`], so `items().len()` is never mistaken for "everything there
131/// is". Both OSV lists capped at [`crate::osv::ADVISORY_DISPLAY_CAP`] use it —
132/// [`DependencyVulnerabilities::advisories`] and [`UpgradeStatus::CandidateVulnerable`].
133///
134/// # Examples
135///
136/// ```
137/// use deps_core::osv::Capped;
138///
139/// let truncated = Capped::new(vec!["A1".to_string(), "A2".to_string()], 5);
140/// assert_eq!(truncated.items().len(), 2);
141/// assert_eq!(truncated.total(), 5);
142/// assert!(!truncated.is_complete());
143/// assert_eq!(truncated.remaining(), 3);
144///
145/// let complete = Capped::new(vec!["A1".to_string()], 1);
146/// assert!(complete.is_complete());
147/// assert_eq!(complete.remaining(), 0);
148/// ```
149#[derive(Debug, Clone, PartialEq, Eq)]
150pub struct Capped<T> {
151    items: Vec<T>,
152    total: usize,
153}
154
155impl<T> Capped<T> {
156    /// Pairs an already-truncated `items` list with the `total` the source
157    /// reported, which may exceed `items.len()`.
158    #[must_use]
159    pub fn new(items: Vec<T>, total: usize) -> Self {
160        Self { items, total }
161    }
162
163    /// The items actually retained — **not necessarily all [`Self::total`] of them**.
164    #[must_use]
165    pub fn items(&self) -> &[T] {
166        &self.items
167    }
168
169    /// The count the source reported, independent of how many items were retained.
170    #[must_use]
171    pub const fn total(&self) -> usize {
172        self.total
173    }
174
175    /// Whether [`Self::items`] holds everything the source reported.
176    #[must_use]
177    pub fn is_complete(&self) -> bool {
178        self.items.len() >= self.total
179    }
180
181    /// How many items were dropped — the render layer's "+N more" count.
182    #[must_use]
183    pub fn remaining(&self) -> usize {
184        self.total.saturating_sub(self.items.len())
185    }
186}

231/// Result of checking whether a recommended upgrade target is itself affected.
232///
233/// Populated by [`crate::osv::OsvClient::check_candidates`] (phase B), which only runs for
234/// dependencies phase A already flagged as [`ScanOutcome::Vulnerable`]. Used for both the
235/// registry's "latest" candidate ([`DependencyVulnerabilities::upgrade_status`]) and the
236/// independently-verified fix target F ([`DependencyVulnerabilities::fix_target_status`]) —
237/// see the latter's doc for why F needs its own verification result distinct from latest's.
238#[derive(Debug, Clone, PartialEq, Eq)]
239pub enum UpgradeStatus {
240    /// Phase B has not run for this dependency (phase A found nothing, or
241    /// phase B has not completed yet).
242    NotChecked,
243    /// The candidate upgrade version is not itself affected by any known advisory.
244    CandidateClean {
245        /// The version that was checked.
246        version: String,
247    },
248    /// The candidate upgrade version is itself affected.
249    CandidateVulnerable {
250        /// The version that was checked.
251        version: String,
252        /// Advisory IDs that still apply to the candidate version, capped at
253        /// [`crate::osv::ADVISORY_DISPLAY_CAP`] the same way
254        /// [`DependencyVulnerabilities::advisories`] is (#462 critic M1) —
255        /// **not necessarily exhaustive**. Check [`Capped::is_complete`] before
256        /// treating it as the complete set of advisories still affecting this
257        /// candidate; an incomplete list means some are missing, not that none
258        /// exist.
259        advisory_ids: Capped<String>,
260    },
261}

263/// Vulnerability data for one dependency that OSV reported as non-clean.
264#[derive(Debug, Clone)]
265pub struct DependencyVulnerabilities {
266    /// Advisories fetched in full, capped at [`crate::osv::ADVISORY_DISPLAY_CAP`] (invariant 3
267    /// in `architecture.md` §8: the fetch itself is capped, not only the render) — the total
268    /// advisory count OSV reported is carried alongside via [`Capped::total`], independent of
269    /// how many were actually fetched, and is the source of the render layer's "+N more
270    /// advisories" count (`architecture.md` §7/§8 invariant 3).
271    pub advisories: Capped<Arc<Advisory>>,
272    /// Result of phase B's "latest" check, if it has run for this dependency.
273    pub upgrade_status: UpgradeStatus,
274    /// Independent verification of [`Self::recommended_fix`]'s target version F, if F
275    /// differs from the "latest" candidate `upgrade_status` already covers. Left at
276    /// [`UpgradeStatus::NotChecked`] until [`Self::recommended_fix`] has been computed and F's
277    /// status resolved — either reused from `upgrade_status` when F equals latest, or checked
278    /// live via [`crate::osv::OsvClient::check_candidates`] otherwise (always live-checked when
279    /// F differs from latest: a data-derived shortcut was tried and rejected — see git history
280    /// on this field and #462's critique — because it degenerates into checking F against
281    /// exactly the advisories it was computed from, proving nothing about an advisory phase
282    /// A never fetched at all, which is the actual gap #462 closes). See
283    /// `run_osv_phase_b_and_commit` in `deps-lsp` for the resolution order.
284    ///
285    /// A caller must not treat a bare [`UpgradeStatus::CandidateClean`] check as the only valid
286    /// "verified" state: [`UpgradeStatus::CandidateVulnerable`] can also be a legitimate,
287    /// presentable fix when every reported id is an advisory [`Self::recommended_fix`] already
288    /// declined to claim (excluded via `still_applying`, or never had a known fix) — see
289    /// `deps-core`'s `lsp_helpers::code_actions::fix_target_is_verified` (the actual gate
290    /// `generate_code_actions` uses) for the full contract, rather than re-deriving it ad hoc.
291    pub fix_target_status: UpgradeStatus,
292}

294/// A single upgrade target recommended by [`DependencyVulnerabilities::recommended_fix`].
295///
296/// `version` is in OSV's version namespace (see
297/// [`crate::lsp_helpers::OsvNaming::osv_version_to_native`] for the
298/// conversion callers must apply before using it in a manifest edit or a
299/// registry lookup).
300#[derive(Debug, Clone, PartialEq, Eq)]
301pub struct FixRecommendation {
302    /// The highest [`Advisory::fixed_versions`] entry across the advisories
303    /// named in `advisory_ids` — the lowest version that resolves everything
304    /// this recommendation actually claims to fix.
305    pub version: String,
306    /// Advisory ids this recommendation actually resolves, sorted by
307    /// severity descending (worst first) and tied by id — the order a
308    /// title should list them in.
309    pub advisory_ids: Vec<String>,
310}
311
312/// Numeric ranking used only to sort [`FixRecommendation::advisory_ids`],
313/// worst severity first.
314const fn severity_rank(severity: VulnSeverity) -> u8 {
315    match severity {
316        VulnSeverity::Critical => 4,
317        VulnSeverity::High => 3,
318        VulnSeverity::Medium => 2,
319        VulnSeverity::Low => 1,
320        VulnSeverity::Unknown => 0,
321    }
322}
323
324impl DependencyVulnerabilities {
325    /// Recommends a single upgrade target that resolves as many of this
326    /// dependency's known advisories as possible.
327    ///
328    /// `advisory_ids` is computed first: every advisory with a known fix,
329    /// minus — when phase B ([`UpgradeStatus::CandidateVulnerable`]) reports
330    /// that some ids still apply to the checked candidate — those ids,
331    /// since claiming a fix for them would be false. `version` is then the
332    /// highest [`Advisory::fixed_versions`] entry across only the
333    /// *remaining* claimed advisories, not every advisory: computing it over
334    /// the full set first would let an advisory this method just excluded
335    /// (because its own fix is known not to hold) drag the recommendation
336    /// past a lower version that already clears everything actually being
337    /// claimed. Returns `None` when no advisory has a claimable fix.
338    ///
339    /// The subtraction's premise is that the checked candidate is at least
340    /// as new as `version`; when phase B checked an older candidate the
341    /// subtraction is merely over-conservative (it only ever removes
342    /// claims), so this is documented rather than guarded against.
343    ///
344    /// # Limitations
345    ///
346    /// `advisories` is capped at fetch time
347    /// ([`crate::osv::ADVISORY_DISPLAY_CAP`]), so `version` is the max over a
348    /// possibly incomplete subset — the "+N more advisories" hint already
349    /// signals that incompleteness, so this is an accepted under-report, not
350    /// a bug. `Advisory` also retains only `fixed_versions`, never
351    /// `introduced` events, so a version reintroduced above its own last
352    /// known fix (and not yet re-fixed) can still be claimed as a fix
353    /// whenever phase B has not run for this dependency
354    /// ([`UpgradeStatus::NotChecked`]) — the post-edit rescan is what
355    /// surfaces that case.
356    ///
357    /// # Examples
358    ///
359    /// ```
360    /// use deps_core::osv::{Advisory, Capped, DependencyVulnerabilities, UpgradeStatus, VulnSeverity};
361    /// use std::sync::Arc;
362    ///
363    /// fn advisory(id: &str, fixed: &str) -> Arc<Advisory> {
364    ///     Arc::new(Advisory {
365    ///         id: id.to_string(),
366    ///         modified: "2023-01-01T00:00:00Z".to_string(),
367    ///         summary: None,
368    ///         aliases: vec![],
369    ///         severity: VulnSeverity::High,
370    ///         cvss_vector: None,
371    ///         fixed_versions: vec![fixed.to_string()],
372    ///         url: String::new(),
373    ///     })
374    /// }
375    ///
376    /// let dv = DependencyVulnerabilities {
377    ///     advisories: Capped::new(vec![advisory("RUSTSEC-1", "1.2.0")], 1),
378    ///     upgrade_status: UpgradeStatus::NotChecked,
379    ///     fix_target_status: UpgradeStatus::NotChecked,
380    /// };
381    ///
382    /// let fix = dv.recommended_fix().unwrap();
383    /// assert_eq!(fix.version, "1.2.0");
384    /// assert_eq!(fix.advisory_ids, vec!["RUSTSEC-1".to_string()]);
385    /// ```
386    #[must_use]
387    pub fn recommended_fix(&self) -> Option<FixRecommendation> {
388        let still_applying: &[String] = match &self.upgrade_status {
389            UpgradeStatus::CandidateVulnerable { advisory_ids, .. } => advisory_ids.items(),
390            UpgradeStatus::NotChecked | UpgradeStatus::CandidateClean { .. } => &[],
391        };
392
393        let mut claimed: Vec<&Advisory> = self
394            .advisories
395            .items()
396            .iter()
397            .map(Arc::as_ref)
398            .filter(|a| !a.fixed_versions.is_empty())
399            .filter(|a| !still_applying.contains(&a.id))
400            .collect();
401
402        if claimed.is_empty() {
403            return None;
404        }
405
406        // The minimum version that clears every *claimed* advisory — not the
407        // max over every advisory (including ones just excluded above),
408        // which could push the recommendation past a version that resolves
409        // nothing beyond what a lower, still-claimed fix already covers.
410        let version = claimed
411            .iter()
412            .filter_map(|a| a.fixed_versions.last())
413            .max_by(|a, b| super::compare_version_strings(a, b))?
414            .clone();
415
416        claimed.sort_by(|a, b| {
417            severity_rank(b.severity)
418                .cmp(&severity_rank(a.severity))
419                .then_with(|| a.id.cmp(&b.id))
420        });
421
422        Some(FixRecommendation {
423            version,
424            advisory_ids: claimed.into_iter().map(|a| a.id.clone()).collect(),
425        })
426    }
427}

429/// Why a dependency produced no advisories.
430///
431/// Absence from [`VulnerabilityMap`] is never a synonym for "clean" — every
432/// input to [`crate::osv::OsvClient::scan`] gets an entry, and every filtered-out or
433/// failed path must declare itself as one of these reasons rather than
434/// silently vanishing from the map (`architecture.md` §6, §8 invariant 0).
435#[derive(Debug, Clone, Copy, PartialEq, Eq)]
436pub enum SkipReason {
437    /// `dep.source()` was not [`crate::parser::DependencySource::Registry`] (§3 step 0).
438    NonRegistrySource,
439    /// No lockfile-resolved or concrete version was available (§3 steps 1-3).
440    NoConcreteVersion,
441    /// `EcosystemFormatter::osv_package_name` returned `None`.
442    UnmappableName,
443    /// `EcosystemId::osv_ecosystem` returned `None`.
444    UnmappableEcosystem,
445    /// The batch or single-package query failed (network error, non-2xx, malformed JSON,
446    /// or a chunk whose result count did not match its query count).
447    QueryFailed,
448    /// The batch result was truncated (`next_page_token` present) and the
449    /// bounded individual-requery budget was exhausted before this entry
450    /// could be recovered (§8 invariant 2).
451    Truncated,
452}
453
454impl SkipReason {
455    /// Short tag used in the `info`-level scan summary log line.
456    pub(crate) const fn as_str(self) -> &'static str {
457        match self {
458            Self::NonRegistrySource => "non-registry-source",
459            Self::NoConcreteVersion => "no-concrete-version",
460            Self::UnmappableName => "unmappable-name",
461            Self::UnmappableEcosystem => "unmappable-ecosystem",
462            Self::QueryFailed => "query-failed",
463            Self::Truncated => "truncated",
464        }
465    }
466}

468/// Outcome of scanning one dependency.
469///
470/// The three variants are mutually exclusive and collectively exhaustive for
471/// every dependency passed to [`crate::osv::OsvClient::scan`] — see `architecture.md` §6 for
472/// why this must never collapse back to `Option<DependencyVulnerabilities>`.
473#[derive(Debug, Clone)]
474pub enum ScanOutcome {
475    /// Never queried, or the query could not be resolved — say nothing about it.
476    Skipped(SkipReason),
477    /// Queried; OSV reported no advisories.
478    Clean,
479    /// Queried; OSV reported one or more advisories.
480    Vulnerable(DependencyVulnerabilities),
481}

483/// Per-scan result map.
484///
485/// Normally keyed by the normalized dependency name
486/// (`EcosystemFormatter::normalize_package_name` — the same key
487/// [`crate::lsp_helpers::generate_diagnostics_from_cache`] and
488/// [`crate::lsp_helpers::generate_hover`] already use to look up
489/// `cached`/`resolved` versions), but see [`vulnerability_keys`] for the
490/// version-qualified form a duplicated dependency name's occurrences use.
491pub type VulnerabilityMap = HashMap<String, ScanOutcome>;
492
493/// Computes the [`VulnerabilityMap`] key each occurrence in `parse_result`
494/// should be scanned/looked-up under.
495///
496/// Keyed by [`Dependency::name_range`](crate::Dependency::name_range) —
497/// unique per occurrence within one document, so callers holding a specific
498/// `dep` (not just its name) can look their own key up directly.
499///
500/// Normally an occurrence's key is just its normalized name (the common
501/// case, and the only form most `VulnerabilityMap` test fixtures use). When
502/// two or more occurrences of the *same* name resolve to different signatures
503/// — e.g. the same crate under `[dependencies]` and `[dev-dependencies]`, or
504/// multiple `[target.'cfg(...)'.dependencies]` blocks (#394), pinned to
505/// different versions, or mixing a registry source with a git/path fork —
506/// each such occurrence's key is instead qualified with a signature specific
507/// to it, so their OSV results can never collide in the shared map.
508/// Occurrences that share both a name and an identical signature
509/// (registry-source, same in-use version) intentionally keep the plain,
510/// shared key and get scanned once: the OSV result would be identical
511/// either way, so collapsing them is a dedup, not a gap.
512///
513/// Every caller that builds or looks up a `VulnerabilityMap` entry for a
514/// *specific* dependency occurrence — `deps-lsp`'s `build_scan_targets`,
515/// and the vulnerability lookups in `generate_diagnostics_from_cache`,
516/// `generate_hover`, and `generate_code_actions` — must go through this
517/// function so key construction never drifts out of sync between producer
518/// and consumer. A caller with no [`EcosystemId`](crate::EcosystemId) to
519/// give (most test fixtures) skips this and falls back to the plain
520/// normalized name, which still finds any entry a test inserted under that
521/// name directly.
522///
523/// # Examples
524///
525/// ```
526/// use deps_core::lsp_helpers::{
527///     DiagnosticMessages, DiagnosticPolicy, OsvNaming, PackageNaming, PackageRendering,
528///     RequirementResolution, SourcePolicy,
529/// };
530/// use deps_core::osv::vulnerability_keys;
531/// use deps_core::{ConcreteVersion, Dependency, EcosystemId, PackageName, ParseResult, VersionReq};
532/// use std::any::Any;
533/// use std::collections::HashMap;
534/// use tower_lsp_server::ls_types::{Position, Range, Uri};
535///
536/// struct SimpleDep {
537///     name: PackageName,
538///     version_req: Option<VersionReq>,
539///     name_range: Range,
540/// }
541///
542/// impl Dependency for SimpleDep {
543///     fn name(&self) -> &PackageName {
544///         &self.name
545///     }
546///     fn name_range(&self) -> Range {
547///         self.name_range
548///     }
549///     fn version_requirement(&self) -> Option<&VersionReq> {
550///         self.version_req.as_ref()
551///     }
552///     fn version_range(&self) -> Option<Range> {
553///         None
554///     }
555///     fn source(&self) -> deps_core::parser::DependencySource {
556///         deps_core::parser::DependencySource::Registry
557///     }
558///     fn as_any(&self) -> &dyn Any {
559///         self
560///     }
561/// }
562///
563/// struct SimpleParseResult {
564///     deps: Vec<SimpleDep>,
565///     uri: Uri,
566/// }
567///
568/// impl ParseResult for SimpleParseResult {
569///     fn dependencies(&self) -> Vec<&dyn Dependency> {
570///         self.deps.iter().map(|d| d as &dyn Dependency).collect()
571///     }
572///     fn workspace_root(&self) -> Option<&std::path::Path> {
573///         None
574///     }
575///     fn uri(&self) -> &Uri {
576///         &self.uri
577///     }
578///     fn as_any(&self) -> &dyn Any {
579///         self
580///     }
581/// }
582///
583/// struct SimpleFormatter;
584/// impl PackageNaming for SimpleFormatter {}
585/// impl PackageRendering for SimpleFormatter {
586///     fn format_version_for_text_edit(&self, version: &ConcreteVersion) -> String {
587///         version.to_string()
588///     }
589///     fn package_url(&self, name: &PackageName) -> String {
590///         name.to_string()
591///     }
592/// }
593/// impl RequirementResolution for SimpleFormatter {}
594/// impl DiagnosticMessages for SimpleFormatter {}
595/// impl DiagnosticPolicy for SimpleFormatter {}
596/// impl SourcePolicy for SimpleFormatter {}
597/// impl OsvNaming for SimpleFormatter {}
598///
599/// // `time` declared twice, pinned to two different versions.
600/// let parse_result = SimpleParseResult {
601///     deps: vec![
602///         SimpleDep {
603///             name: PackageName::new("time"),
604///             version_req: Some(VersionReq::new("=0.1.43")),
605///             name_range: Range::new(Position::new(0, 0), Position::new(0, 4)),
606///         },
607///         SimpleDep {
608///             name: PackageName::new("time"),
609///             version_req: Some(VersionReq::new("=0.1.44")),
610///             name_range: Range::new(Position::new(3, 0), Position::new(3, 4)),
611///         },
612///     ],
613///     uri: deps_core::test_util::test_uri("/test/Cargo.toml"),
614/// };
615/// let resolved: HashMap<PackageName, ConcreteVersion> = HashMap::new();
616///
617/// let keys = vulnerability_keys(&parse_result, &resolved, &SimpleFormatter, EcosystemId::Cargo);
618/// let deps = parse_result.dependencies();
619/// let key0 = keys.get(&deps[0].name_range()).unwrap();
620/// let key1 = keys.get(&deps[1].name_range()).unwrap();
621/// assert_ne!(key0, key1, "differently-pinned occurrences of one name get distinct keys");
622/// ```
623pub fn vulnerability_keys(
624    parse_result: &dyn crate::ParseResult,
625    resolved: &HashMap<crate::PackageName, crate::ConcreteVersion>,
626    formatter: &dyn crate::lsp_helpers::EcosystemFormatter,
627    ecosystem: crate::EcosystemId,
628) -> HashMap<tower_lsp_server::ls_types::Range, String> {
629    use crate::lsp_helpers::in_use_version;
630
631    let deps = parse_result.dependencies();
632
633    // One signature per occurrence: public-registry-content deps (crates.io itself, or a
634    // verified crates.io mirror per `source_is_public_registry_content` — F1b) carry their
635    // own in-use version (or "u" when none is determinable — a range requirement with no
636    // lock file); every other source (git/path forks, a genuinely different private
637    // registry) always carries "n", since their `ScanOutcome` is always
638    // `Skipped(NonRegistrySource)` regardless of any declared version.
639    let signatures: Vec<(String, String)> = deps
640        .iter()
641        .map(|dep| {
642            let name = formatter.normalize_package_name(dep.name());
643            let signature = if formatter.source_is_public_registry_content(&dep.source()) {
644                match in_use_version(*dep, &name, resolved, formatter, ecosystem) {
645                    Some(v) => format!("v:{v}"),
646                    None => "u".to_string(),
647                }
648            } else {
649                "n".to_string()
650            };
651            (name, signature)
652        })
653        .collect();
654
655    let mut distinct_signatures_by_name: HashMap<&str, std::collections::HashSet<&str>> =
656        HashMap::new();
657    for (name, signature) in &signatures {
658        distinct_signatures_by_name
659            .entry(name.as_str())
660            .or_default()
661            .insert(signature.as_str());
662    }
663
664    deps.iter()
665        .zip(&signatures)
666        .map(|(dep, (name, signature))| {
667            let ambiguous = distinct_signatures_by_name
668                .get(name.as_str())
669                .is_some_and(|s| s.len() > 1);
670            let key = if ambiguous {
671                format!("{name}\u{0}{signature}")
672            } else {
673                name.clone()
674            };
675            (dep.name_range(), key)
676        })
677        .collect()
678}

791/// Returns `true` if `id` matches OSV's advisory id grammar
792/// (`[A-Za-z0-9._-]+`, non-empty, capped at 128 bytes) — the same alphabet
793/// every real id scheme in this space uses (`RUSTSEC-2020-0071`,
794/// `GHSA-xxxx-yyyy-zzzz`, `CVE-2020-26235`); real ids are a few dozen
795/// characters, so the cap exists only to bound how much of a record-supplied
796/// string can ride along into `Diagnostic.code`, hover markdown, and a
797/// `CodeAction` title. `id` is echoed verbatim into a markdown link
798/// destination (`push_vulnerability_hover_section`) and a `Diagnostic.code`,
799/// so this is the parse-boundary chokepoint that keeps a malformed id from
800/// ever reaching either — rejecting it here means every downstream consumer
801/// can treat `Advisory.id` as inherently safe, rather than needing to
802/// sanitize it again at each render site.
803fn is_valid_osv_id(id: &str) -> bool {
804    !id.is_empty()
805        && id.len() <= 128
806        && id
807            .chars()
808            .all(|c| c.is_ascii_alphanumeric() || matches!(c, '.' | '_' | '-'))
809}
810
811impl OsvVulnRecord {
812    /// Converts a raw wire record into the `deps-lsp`-facing [`Advisory`],
813    /// or `None` if the record's id fails [`is_valid_osv_id`] (dropped, same
814    /// as a 404 on `/v1/vulns/{id}` — the dependency renders with whichever
815    /// advisories did resolve, never a half-trusted one). Individual `fixed`
816    /// events failing [`is_safe_version_string`] are dropped the same way,
817    /// but only that entry — the record as a whole still renders with its
818    /// remaining, valid `fixed_versions`.
819    ///
820    /// `osv_name`/`osv_eco` are the package actually queried: a record can
821    /// legitimately cover several unrelated packages sharing one advisory id
822    /// (critique S3), so `affected[]` is filtered to entries whose `package`
823    /// matches (or omits) before `fixed_versions`/severity are extracted —
824    /// otherwise a stranger package's fix version or severity could leak
825    /// into this one's rendering.
826    pub(super) fn into_advisory(self, osv_name: &str, osv_eco: &str) -> Option<Advisory> {
827        if !is_valid_osv_id(&self.id) {
828            tracing::warn!(id = %self.id, "OSV record has a malformed id, dropping");
829            return None;
830        }
831
832        let relevant: Vec<&OsvAffected> = self
833            .affected
834            .iter()
835            .filter(|a| {
836                a.package
837                    .as_ref()
838                    .is_none_or(|p| p.name == osv_name && p.ecosystem == osv_eco)
839            })
840            .collect();
841        // Every `affected[]` entry named a different package: OSV returned
842        // this record in response to our exact query, so that should not
843        // happen in practice. Fall back to using every entry rather than
844        // rendering fixed_versions/severity as empty/Unknown outright.
845        let relevant: Vec<&OsvAffected> = if relevant.is_empty() && !self.affected.is_empty() {
846            tracing::warn!(
847                id = %self.id, osv_name, osv_eco,
848                "no affected[] entry matched the queried package; using all entries"
849            );
850            self.affected.iter().collect()
851        } else {
852            relevant
853        };
854
855        let severity = super::severity::classify(self.database_specific.as_ref(), &relevant);
856        let cvss_vector = self
857            .severity
858            .iter()
859            .find(|s| s.kind == "CVSS_V3")
860            .or_else(|| self.severity.first())
861            .map(|s| s.score.clone());
862
863        let mut fixed_versions: Vec<String> = relevant
864            .iter()
865            .flat_map(|a| a.ranges.iter())
866            .flat_map(|r| r.events.iter())
867            .filter_map(|e| e.fixed.clone())
868            .filter(|v| {
869                let valid = is_safe_version_string(v);
870                if !valid {
871                    tracing::warn!(
872                        id = %self.id, version = %v,
873                        "OSV record has a malformed fixed version, dropping"
874                    );
875                }
876                valid
877            })
878            .collect();
879        fixed_versions.sort_by(|a, b| super::compare_version_strings(a, b));
880        fixed_versions.dedup();
881
882        let url = format!("https://osv.dev/vulnerability/{}", self.id);
883
884        Some(Advisory {
885            id: self.id,
886            modified: self.modified,
887            summary: self.summary,
888            aliases: self.aliases,
889            severity,
890            cvss_vector,
891            fixed_versions,
892            url,
893        })
894    }
895}
