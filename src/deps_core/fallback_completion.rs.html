1//! Shared raw-text scanners for fallback (parse-failure) completion.
2//!
3//! When a manifest fails to parse (typically because the user is mid-edit), ecosystems
4//! fall back to a raw-text heuristic instead of [`crate::ecosystem::Ecosystem::generate_completions`]'s
5//! parsed-AST path: is the cursor inside a dependencies-like section, and if so, what
6//! prefix has the user typed? This module holds the syntax-shape scanners shared by two
7//! or more ecosystems (issue #722) — a manifest-format-specific ecosystem (e.g. Maven's
8//! XML coordinate segments) keeps its own composition in its own crate, calling into
9//! these primitives rather than reimplementing them.
10//!
11//! Each ecosystem composes these into its own
12//! [`Ecosystem::fallback_completion_prefix`](crate::ecosystem::Ecosystem::fallback_completion_prefix)
13//! override: a section-boundary check (this module, or an ecosystem-local equivalent)
14//! followed by [`raw_prefix`] and, where the manifest syntax wraps the completable value
15//! (a JSON string, an XML tag/attribute), a syntax-specific strip.
16

19/// Returns the line of `content` at `position.line`, or `None` if the document has
20/// fewer lines than that (e.g. the cursor is on a not-yet-existing trailing line).
21#[must_use]
22pub fn line_at(content: &str, position: Position) -> Option<&str> {
23    content.lines().nth(position.line as usize)
24}

26/// Extracts what the user has typed on `line` up to the cursor (`character`, a UTF-16
27/// code unit count), trimmed of whitespace.
28///
29/// `character` is converted via [`crate::completion::utf16_to_byte_offset`]; when the
30/// cursor sits beyond the line's UTF-16 length (`None`), this clamps to the full line
31/// rather than panicking on an out-of-bounds slice.
32#[must_use]
33pub fn raw_prefix(line: &str, character: u32) -> &str {
34    let prefix_end = crate::completion::utf16_to_byte_offset(line, character).unwrap_or(line.len());
35    debug_assert!(
36        line.is_char_boundary(prefix_end),
37        "prefix_end must be a char boundary"
38    );
39    line.get(..prefix_end).unwrap_or(line).trim()
40}

42/// Checks if a line is inside a TOML dependencies section.
43///
44/// Looks for `[dependencies]`, `[dev-dependencies]`, `[build-dependencies]` sections
45/// in Cargo.toml or `[project.dependencies]` in pyproject.toml. Shared by Cargo and
46/// PyPI — PyPI's PEP 621 primary dependency list needs its own array-based scan on top
47/// of this (see `deps_pypi`'s `is_in_pypi_project_dependencies_array`), since it is not
48/// a section header at all.
49#[must_use]
50pub fn is_in_toml_dependencies(content: &str, line_number: usize) -> bool {
51    // Walk backwards from current line to find the most recent section header
52    // Collect lines up to target, then iterate backwards
53    let lines: Vec<_> = content.lines().enumerate().take(line_number + 1).collect();
54
55    for (_, line) in lines.iter().rev() {
56        let line = line.trim();
57
58        // Check if this is a section header
59        if line.starts_with('[') && line.ends_with(']') {
60            // Check if it's a dependencies section
61            return line == "[dependencies]"
62                || line == "[dev-dependencies]"
63                || line == "[build-dependencies]"
64                || line == "[workspace.dependencies]"
65                || line == "[project.dependencies]"
66                || line == "[project.optional-dependencies]"
67                || line.starts_with("[target.")
68                    && (line.contains(".dependencies]")
69                        || line.contains(".dev-dependencies]")
70                        || line.contains(".build-dependencies]"));
71        }
72    }
73
74    false
75}

77/// Checks if a line is inside a JSON dependencies-like section.
78///
79/// Looks for `"{key}": {` for any of the given `keys`, e.g. `dependencies` /
80/// `devDependencies` in package.json, or `require` / `require-dev` in composer.json.
81#[must_use]
82pub fn is_in_json_dependencies(content: &str, line_number: usize, keys: &[&str]) -> bool {
83    let mut in_dependencies = false;
84    let mut brace_depth = 0;
85    // Build each `"{key}":` needle once per call rather than once per line.
86    let needles: Vec<String> = keys.iter().map(|key| format!("\"{key}\":")).collect();
87
88    for (i, line) in content.lines().enumerate() {
89        // Early exit: stop if we've passed the target line
90        if i > line_number {
91            break;
92        }
93
94        let trimmed = line.trim();
95
96        // Check if we're entering a dependencies-like section
97        if trimmed.starts_with('"')
98            && needles
99                .iter()
100                .any(|needle| trimmed.contains(needle.as_str()))
101        {
102            in_dependencies = true;
103            brace_depth = 0;
104        }
105
106        // Track brace depth when in dependencies section
107        if in_dependencies {
108            for ch in trimmed.chars() {
109                match ch {
110                    '{' => brace_depth += 1,
111                    '}' => {
112                        brace_depth -= 1;
113                        // If we've closed the dependencies section
114                        if brace_depth <= 0 {
115                            in_dependencies = false;
116                        }
117                    }
118                    _ => {}
119                }
120            }
121
122            // If we're at the target line and inside dependencies section with depth > 0
123            if i == line_number && in_dependencies && brace_depth > 0 {
124                return true;
125            }
126        }
127    }
128
129    false
130}

132/// Checks if a line is inside an XML `<tag>...</tag>` element.
133///
134/// Tracks nested open/close tag counts (ignoring attributes and self-closing tags) to
135/// find whether the target line falls within any occurrence of the element, e.g.
136/// `<dependencies>` in pom.xml (including nested inside `<dependencyManagement>`), or
137/// `<ItemGroup>`/`<packages>` in NuGet's three manifest schemas.
138#[must_use]
139pub fn is_in_xml_tag_section(content: &str, line_number: usize, tag: &str) -> bool {
140    let open_prefix = format!("<{tag}");
141    let close = format!("</{tag}>");
142    let mut depth: usize = 0;
143
144    for (i, line) in content.lines().enumerate() {
145        if i > line_number {
146            break;
147        }
148
149        let opens_here = count_open_tags(line, &open_prefix);
150        depth += opens_here;
151        // A line with an opening tag counts as "inside" even if the same line also
152        // closes it (`<dependencies></dependencies>`), consistent with the target
153        // line being the header itself in `is_in_toml_dependencies`.
154        if i == line_number && opens_here > 0 {
155            return true;
156        }
157
158        depth = depth.saturating_sub(line.matches(close.as_str()).count());
159        if i == line_number && depth > 0 {
160            return true;
161        }
162    }
163
164    false
165}

167/// Counts real `<{open_prefix}...>` tag occurrences on `line`, i.e. `open_prefix`
168/// followed by `>` or whitespace (an attribute) rather than more tag-name characters
169/// (so `<dependencies` doesn't also match a longer, unrelated tag name).
170// `search_from` starts at 0 and is only ever advanced to `idx + open_prefix.len()`,
171// where `idx` is a `str::find` match start (always a char boundary) and the offset
172// lands exactly at the end of that matched substring (also always a char boundary).
173#[allow(clippy::string_slice)]
174fn count_open_tags(line: &str, open_prefix: &str) -> usize {
175    let mut count = 0;
176    let mut search_from = 0;
177
178    while let Some(rel_idx) = line[search_from..].find(open_prefix) {
179        let idx = search_from + rel_idx;
180        let after = &line[idx + open_prefix.len()..];
181        if after.starts_with('>') || after.starts_with(char::is_whitespace) {
182            count += 1;
183        }
184        search_from = idx + open_prefix.len();
185    }
186
187    count
188}

190/// Strips everything up to and including the *last* `>` in `prefix`, also returning the
191/// name of the tag whose content the cursor now sits inside, when there is one.
192///
193/// `<artifactId>gua` -> `("gua", Some("artifactId"))`; returns `(prefix, None)` unchanged
194/// if it contains no `>` at all (e.g. the tag is not yet closed, as when the user is
195/// still typing the tag name itself).
196///
197/// Scans for the last `>`, not the first, to mirror the closest opening tag *before
198/// the cursor*, which is the last one on the line, not the first — a first-`>` version
199/// diverges whenever more than one tag precedes the cursor on a line
200/// (`<groupId>com.google.guava</groupId><artifactId>gua` — the first `>` sits inside
201/// `<groupId>`, well short of the real value).
202///
203/// The returned tag name is `None` in three cases: no `>` at all (see above); the
204/// `<...>` ending at that `>` is a *closing* tag (`</artifactId>` — its name slice
205/// starts with `/`), e.g. cursor right after `<artifactId>guava</artifactId>` — the
206/// last `>` is the line's very last character, correctly yielding an empty stripped
207/// string and no open tag; or loose text after a closed tag (`<artifactId>guava</artifactId>
208/// comm` — same closing-tag case, `Some(" comm")` would otherwise wrongly read as
209/// "inside `artifactId`"). Otherwise the name is the element name immediately after
210/// that tag's `<` (attributes and a trailing `/` stripped, mirroring
211/// [`strip_open_xml_attribute_value`]'s own element extraction).
212#[must_use]
213pub fn strip_leading_xml_tag(prefix: &str) -> (&str, Option<&str>) {
214    // `>`/`<` are single-byte ASCII chars, so `gt + 1`/`lt + 1` are always valid char
215    // boundaries, and slicing at `gt`/`lt` (found via `rfind` on the byte string) is too.
216    #[allow(clippy::string_slice)]
217    {
218        let Some(gt) = prefix.rfind('>') else {
219            return (prefix, None);
220        };
221        let stripped = &prefix[gt + 1..];
222        let Some(lt) = prefix[..gt].rfind('<') else {
223            return (stripped, None);
224        };
225        // A closing tag's name segment (`/artifactId` for `</artifactId>`) splits to an
226        // empty first token, since `/` is itself a delimiter — that's what makes
227        // `name.is_empty()` double as the "not a closing tag" check.
228        let name = prefix[lt + 1..gt]
229            .split(|c: char| c == '/' || c.is_whitespace())
230            .next()
231            .unwrap_or("");
232        if name.is_empty() {
233            (stripped, None)
234        } else {
235            (stripped, Some(name))
236        }
237    }
238}

240/// Extracts the value being typed inside an open attribute (named in `attrs`) at the
241/// end of `prefix`, on one of the elements named in `elements`
242/// (`<PackageReference Include="Newt` -> `Newt`).
243///
244/// Returns an *empty* string whenever the cursor is not inside one of `attrs`'
245/// *unclosed* values on one of `elements` — still inside the element/attribute name
246/// itself (no candidate package-name text yet), inside a different attribute's value,
247/// past an already-closed value, inside a target attribute name on an unrelated
248/// element, or plain non-markup text (comments, a bare word). Deliberately empty
249/// rather than the raw `prefix`, unlike [`strip_leading_xml_tag`]'s "return unchanged"
250/// fallback: an unmodified prefix here would usually contain no `=` either, so a
251/// caller's `contains('=')` guard alone cannot reject it, and a search on that raw
252/// text would query the registry for markup or arbitrary noise.
253///
254/// Scans the whole `prefix` tracking the currently open quote, rather than `rfind` as
255/// [`strip_leading_xml_tag`] does for tag content: an attribute value's opening quote
256/// is the same character as a closed value's closing quote, so telling them apart
257/// requires tracking quote parity from the start of the element instead of searching
258/// backwards from the end. Only sees the element opening tag when it is on the same
259/// line as the cursor.
260#[must_use]
261pub fn strip_open_xml_attribute_value<'a>(
262    prefix: &'a str,
263    elements: &[&str],
264    attrs: &[&str],
265) -> &'a str {
266    let mut quote: Option<char> = None;
267    let mut value_start = 0usize;
268    let mut in_target_attr = false;
269    let mut segment_start = 0usize;
270    let mut element = "";
271
272    for (idx, ch) in prefix.char_indices() {
273        if let Some(open) = quote {
274            if ch == open {
275                quote = None;
276                segment_start = idx + ch.len_utf8();
277            }
278            continue;
279        }
280        if ch == '<' {
281            // `idx + 1` is a char boundary: `<` is a single-byte ASCII char. Reset
282            // here too (not just on a closing quote) so `before_quote` below never
283            // spans back across a *previous* element's tail.
284            segment_start = idx + ch.len_utf8();
285            // char-boundary safe: `segment_start` is `idx + ch.len_utf8()` set
286            // immediately above, always a char boundary.
287            #[allow(clippy::string_slice)]
288            let rest = &prefix[segment_start..];
289            element = rest
290                .split(|c: char| c == '>' || c == '/' || c.is_whitespace())
291                .next()
292                .unwrap_or("");
293            continue;
294        }
295        if ch == '"' || ch == '\'' {
296            // `idx` is a char boundary (from `char_indices`), so this slice is valid.
297            #[allow(clippy::string_slice)]
298            let before_quote = prefix[segment_start..idx].trim_end();
299            let name = before_quote
300                .strip_suffix('=')
301                .unwrap_or(before_quote)
302                .split_whitespace()
303                .next_back()
304                .unwrap_or("");
305            in_target_attr = elements.contains(&element) && attrs.contains(&name);
306            quote = Some(ch);
307            value_start = idx + ch.len_utf8();
308        }
309    }
310
311    if quote.is_some() && in_target_attr {
312        // `value_start` is `idx + ch.len_utf8()` for the opening quote char, always a
313        // char boundary.
314        #[allow(clippy::string_slice)]
315        let value = &prefix[value_start..];
316        // A package id never contains a quote character. The scan above only closes
317        // `quote` on the *matching* delimiter, so an opposite-type quote character
318        // would otherwise survive into the extracted value; stop at the first quote
319        // of either kind instead.
320        return value.split(['"', '\'']).next().unwrap_or(value);
321    }
322    ""
323}

325/// Counts the `"` characters in `segment` that actually open or close a string —
326/// skipping any `\"` escape sequence — along with the byte index of the last such
327/// real quote, if any.
328///
329/// A `"` is escaped, and does not toggle string state, when it is preceded by an *odd*
330/// number of consecutive `\` characters immediately before it (an even count, zero
331/// included, means those backslashes are themselves pairwise-escaped, so the quote is
332/// real) — e.g. in `"a\"b"` the middle `"` is escaped (one preceding `\`), so the
333/// string is `a"b`, not two separate strings. A naive per-`"` count desyncs from real
334/// open/close string state on any segment containing `\"` (#729 code-review, #733: raw
335/// counting both over-counts, flipping a closed key+value pair to look "open", and
336/// under-counts the inverse, flipping a still-open key to look "closed").
337///
338/// Thin wrapper around [`count_real_quotes_with`] for `"`, the quote character every
339/// existing caller of this function needs.
340///
341/// # Examples
342///
343/// ```
344/// use deps_core::fallback_completion::count_real_quotes;
345///
346/// // Two plain quotes, both real: last one is at byte index 7.
347/// assert_eq!(count_real_quotes("\"pytest\""), (2, Some(7)));
348/// // The middle quote is escaped (one preceding `\`), so it isn't counted: only the
349/// // opening quote (index 0) is real.
350/// assert_eq!(count_real_quotes("\"a\\\"b"), (1, Some(0)));
351/// ```
352#[must_use]
353pub fn count_real_quotes(segment: &str) -> (usize, Option<usize>) {
354    count_real_quotes_with(segment, '"')
355}
356
357/// Same escape-aware counting as [`count_real_quotes`], parameterized over which
358/// character counts as a quote.
359///
360/// Gradle's Kotlin/Groovy DSL accepts either `"..."` or `'...'` string literals (e.g.
361/// `version.ref = "..."` or `version.ref = '...'`), so a single hard-coded `"` check
362/// can't serve it — this lets a caller reuse the same escape-aware parity logic for
363/// whichever quote character its manifest syntax actually uses (#738), instead of
364/// keeping a second, naive (non-escape-aware) counter for `'`.
365///
366/// `quote` must not be `\`: the backslash-run tracking that recognizes an escape
367/// always consumes a `\` character first, so passing `\` itself as `quote` can never
368/// match and this always returns `(0, None)`.
369///
370/// # Examples
371///
372/// ```
373/// use deps_core::fallback_completion::count_real_quotes_with;
374///
375/// // Two plain single quotes, both real: last one is at byte index 7.
376/// assert_eq!(count_real_quotes_with("'pytest'", '\''), (2, Some(7)));
377/// // The middle quote is escaped (one preceding `\`), so it isn't counted.
378/// assert_eq!(count_real_quotes_with("'a\\'b", '\''), (1, Some(0)));
379/// ```
380#[must_use]
381pub fn count_real_quotes_with(segment: &str, quote: char) -> (usize, Option<usize>) {
382    let mut count = 0usize;
383    let mut last_real_quote = None;
384    for idx in real_quote_indices(segment, quote) {
385        count += 1;
386        last_real_quote = Some(idx);
387    }
388    (count, last_real_quote)
389}

391/// Finds the byte offset of the first real (escape-aware, see [`count_real_quotes`])
392/// `quote` character in `segment`, scanning forward from the beginning.
393///
394/// The forward-scanning, first-match counterpart to [`count_real_quotes_with`]'s
395/// backward-looking last-match — same escape rule (shared via a private
396/// `real_quote_indices` scan), opposite question. Useful when a string's content is
397/// known to start at byte `0` of `segment` (e.g. the text immediately after an opening
398/// quote), so the first real quote encountered is that string's closing quote.
399///
400/// # Examples
401///
402/// ```
403/// use deps_core::fallback_completion::find_closing_quote;
404///
405/// assert_eq!(find_closing_quote("pytest\" extra", '"'), Some(6));
406/// // The escaped quote is skipped; the real closing quote is found after it.
407/// assert_eq!(find_closing_quote("a\\\"b\"", '"'), Some(4));
408/// assert_eq!(find_closing_quote("no quote here", '"'), None);
409/// ```
410#[must_use]
411pub fn find_closing_quote(segment: &str, quote: char) -> Option<usize> {
412    real_quote_indices(segment, quote).next()
413}
414
415/// Shared escape-aware scan backing both [`count_real_quotes_with`] and
416/// [`find_closing_quote`]: yields the byte offset of every real (non-escaped) `quote`
417/// character in `segment`, in order. See [`count_real_quotes`]'s doc comment for the
418/// escape rule (an odd run of consecutive `\` immediately before `quote` escapes it).
419fn real_quote_indices(segment: &str, quote: char) -> impl Iterator<Item = usize> + '_ {
420    let mut backslash_run = 0usize;
421    segment
422        .char_indices()
423        .filter_map(move |(idx, ch)| match ch {
424            '\\' => {
425                backslash_run += 1;
426                None
427            }
428            ch if ch == quote => {
429                let is_real = backslash_run.is_multiple_of(2);
430                backslash_run = 0;
431                is_real.then_some(idx)
432            }
433            _ => {
434                backslash_run = 0;
435                None
436            }
437        })
438}

440/// Classifies whether `prefix` (the trimmed line text up to the cursor) sits inside an
441/// open JSON string that is a dependency object's *key*.
442///
443/// Covers `package.json`'s `"expr` while the key is still being typed, and
444/// `composer.json`'s `require`/`require-dev` entries — distinguishing that from a
445/// closed key, an open value string, or plain unquoted text.
446///
447/// Uses quote parity — an odd count of real (see [`count_real_quotes`]) `"`
448/// characters means the cursor sits inside an open string literal — adapted for a JSON
449/// object key instead of an XML tag/attribute: the text immediately before the open
450/// string's opening quote decides key-vs-value: a `:` right before it
451/// (`"express": "^4`) means the open string is the *value*, not the key.
452///
453/// Returns `(text_after_the_open_quote, true)` only when the cursor is genuinely
454/// inside an open key string — the one case where a fallback-completion insert should
455/// be the bare candidate name rather than a full `"{name}": "^{latest}"` snippet, the
456/// same "already open" bare-insert shape as
457/// [`strip_leading_xml_tag`]'s `Some("artifactId")` case and
458/// [`strip_open_xml_attribute_value`]'s non-empty result. Every other shape returns
459/// `("", false)`:
460/// - Even quote count with at least one real `"` present means the cursor sits right
461///   after a fully-closed string (`"express"` with the cursor past both quotes) —
462///   quote parity alone cannot tell whether that position wants a new key, is
463///   mid-value, or something else, so this suppresses rather than guesses (#729 critic
464///   S1, matching Maven's "no safe bare text to offer here" discipline for a
465///   non-`artifactId` open tag).
466/// - Odd quote count whose open string's text is preceded by `:` means the cursor is
467///   inside an open *value* string, not the key — a bare package-name insert there
468///   would corrupt the version instead of completing the key (#729 critic S2).
469///
470/// Even quote count with *zero* real `"` present (plain unquoted text, or nothing
471/// typed yet) is the one exception: it returns `(prefix, false)` unchanged, since no
472/// quote has been opened at all and the normal full-pair insert is still correct
473/// there.
474#[must_use]
475pub fn strip_open_json_key(prefix: &str) -> (&str, bool) {
476    let (quote_count, last_quote) = count_real_quotes(prefix);
477    if quote_count == 0 {
478        return (prefix, false);
479    }
480    if quote_count.is_multiple_of(2) {
481        return ("", false);
482    }
483    let Some(last_quote) = last_quote else {
484        return ("", false);
485    };
486    // `quote_count` odd (so >= 1) guarantees `count_real_quotes` found one; `"`
487    // is a single-byte ASCII char, so `last_quote + 1` is always a char boundary.
488    #[allow(clippy::string_slice)]
489    let (before_quote, after_quote) = (&prefix[..last_quote], &prefix[last_quote + 1..]);
490    if before_quote.trim_end().ends_with(':') {
491        ("", false)
492    } else {
493        (after_quote, true)
494    }
495}

497/// Returns the tail of a genuinely still-open quoted string ending at the cursor, or
498/// `None` when there isn't one.
499///
500/// The tail is the text after the last real, escape-aware (see [`count_real_quotes`])
501/// opening `"` in `segment`, returned only when the cursor sits inside a genuinely
502/// still-open quoted string — an odd count of real `"` characters. Returns `None` when
503/// the string is already closed (an even, non-zero count) or no quote has been typed
504/// at all (zero count): in both cases there is no open string to extract a value from.
505///
506/// Shared by ecosystems whose raw-text fallback-completion prefix survives inside a
507/// quoted value (a JSON object key/value, a TOML string-array element) — used instead
508/// of an unconditional `trim_matches('"')`, which cannot distinguish a genuinely open
509/// string from an already-closed one and so cannot tell a caller when a bare insert at
510/// the cursor would corrupt the manifest rather than complete an open value (#734).
511///
512/// # Examples
513///
514/// ```
515/// use deps_core::fallback_completion::open_quoted_tail;
516///
517/// // Still-open string: only the opening quote exists.
518/// assert_eq!(open_quoted_tail("\"flas"), Some("flas"));
519/// // Already-closed string: both quotes exist, nothing left to complete.
520/// assert_eq!(open_quoted_tail("\"pytest\""), None);
521/// // No quote typed yet.
522/// assert_eq!(open_quoted_tail("dependencies = ["), None);
523/// ```
524#[must_use]
525pub fn open_quoted_tail(segment: &str) -> Option<&str> {
526    let (count, last_quote) = count_real_quotes(segment);
527    if count.is_multiple_of(2) {
528        return None;
529    }
530    // `last_quote` is the byte index of a `"` char (from `char_indices`), and `"` is a
531    // single-byte ASCII char, so `last_quote + 1` is always a char boundary.
532    #[allow(clippy::string_slice)]
533    last_quote.map(|pos| &segment[pos + 1..])
534}

547    /// Sums a line's UTF-16 code unit length. `len_utf16()` is always 1 or 2, so a
548    /// short test line can never overflow `u32`.
549    #[allow(clippy::cast_possible_truncation)]
550    fn utf16_len(line: &str) -> u32 {
551        line.chars().map(char::len_utf16).sum::<usize>() as u32
552    }

554    /// `character` beyond the line's UTF-16 length hits `utf16_to_byte_offset`'s
555    /// `None` branch; `unwrap_or(line.len())` must clamp to the full line rather
556    /// than panic, even when the line contains multi-byte characters.

559        let line = "café";
