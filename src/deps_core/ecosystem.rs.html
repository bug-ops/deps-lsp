19/// Canonical, exhaustive identifier for every package ecosystem the workspace supports.
20///
21/// [`Ecosystem::id`] returns a `&'static str` for registry lookups and document
22/// storage, but any code that needs to *branch* on ecosystem identity should match on
23/// this enum instead of re-deriving its own partial match over that string: an
24/// unhandled variant here is a compile error, while an unhandled string is a silent
25/// runtime bug (see the fix for issue #118, where two call sites silently mishandled
26/// ecosystems missing from an incomplete string match).
27///
28/// Deliberately **not** `#[non_exhaustive]`: adding a new ecosystem must force every
29/// exhaustive `match` on this type across the workspace to be updated at compile time.
30///
31/// # Examples
32///
33/// ```
34/// use deps_core::EcosystemId;
35///
36/// let id: EcosystemId = "npm".parse().unwrap();
37/// assert_eq!(id, EcosystemId::Npm);
38/// assert_eq!(id.id(), "npm");
39/// assert_eq!(id.to_string(), "npm");
40///
41/// assert!("unknown".parse::<EcosystemId>().is_err());
42/// ```
43#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
44pub enum EcosystemId {
45    /// Rust Cargo ecosystem (`Cargo.toml`).
46    Cargo,
47    /// JavaScript/TypeScript npm ecosystem (`package.json`).
48    Npm,
49    /// Python PyPI ecosystem (`pyproject.toml`).
50    Pypi,
51    /// Go modules ecosystem (`go.mod`).
52    Go,
53    /// Ruby Bundler ecosystem (`Gemfile`).
54    Bundler,
55    /// Dart/Flutter pub ecosystem (`pubspec.yaml`).
56    Dart,
57    /// Java/Kotlin Maven ecosystem (`pom.xml`).
58    Maven,
59    /// PHP Composer ecosystem (`composer.json`).
60    Composer,
61    /// Java/Kotlin Gradle ecosystem (`build.gradle`, `build.gradle.kts`, version catalogs).
62    Gradle,
63    /// Swift Package Manager ecosystem (`Package.swift`).
64    Swift,
65    /// .NET NuGet ecosystem (`.csproj`/`.fsproj`/`.vbproj`, `Directory.Packages.props`, `packages.config`).
66    NuGet,
67}
68
69impl EcosystemId {
70    /// Returns the canonical string identifier, matching [`Ecosystem::id`] for the
71    /// corresponding ecosystem implementation.
72    #[must_use]
73    pub const fn id(self) -> &'static str {
74        match self {
75            Self::Cargo => "cargo",
76            Self::Npm => "npm",
77            Self::Pypi => "pypi",
78            Self::Go => "go",
79            Self::Bundler => "bundler",
80            Self::Dart => "dart",
81            Self::Maven => "maven",
82            Self::Composer => "composer",
83            Self::Gradle => "gradle",
84            Self::Swift => "swift",
85            Self::NuGet => "nuget",
86        }
87    }
88}

96impl std::str::FromStr for EcosystemId {
97    type Err = crate::error::DepsError;
98
99    fn from_str(s: &str) -> Result<Self, Self::Err> {
100        match s {
101            "cargo" => Ok(Self::Cargo),
102            "npm" => Ok(Self::Npm),
103            "pypi" => Ok(Self::Pypi),
104            "go" => Ok(Self::Go),
105            "bundler" => Ok(Self::Bundler),
106            "dart" => Ok(Self::Dart),
107            "maven" => Ok(Self::Maven),
108            "composer" => Ok(Self::Composer),
109            "gradle" => Ok(Self::Gradle),
110            "swift" => Ok(Self::Swift),
111            "nuget" => Ok(Self::NuGet),
112            _ => Err(crate::error::DepsError::UnsupportedEcosystem(s.to_string())),
113        }
114    }
115}

117/// Parse result trait containing dependencies and metadata.
118///
119/// Implementations hold ecosystem-specific dependency types
120/// but expose them through trait object interfaces.
121pub trait ParseResult: Send + Sync {
122    /// All dependencies found in the manifest
123    fn dependencies(&self) -> Vec<&dyn Dependency>;
124
125    /// Workspace root path (for monorepo support)
126    fn workspace_root(&self) -> Option<&std::path::Path>;
127
128    /// Document URI
129    fn uri(&self) -> &Uri;
130
131    /// Downcast to concrete type for ecosystem-specific operations
132    fn as_any(&self) -> &dyn Any;
133}

135/// Generic dependency trait.
136///
137/// All parsed dependencies must implement this for generic handler access.
138pub trait Dependency: Send + Sync {
139    /// Package name
140    fn name(&self) -> &str;
141
142    /// LSP range of the dependency name
143    fn name_range(&self) -> tower_lsp_server::ls_types::Range;
144
145    /// Version requirement string (e.g., "^1.0", ">=2.0")
146    fn version_requirement(&self) -> Option<&str>;
147
148    /// LSP range of the version string
149    fn version_range(&self) -> Option<tower_lsp_server::ls_types::Range>;
150
151    /// Dependency source (registry, git, path)
152    fn source(&self) -> crate::parser::DependencySource;
153
154    /// Feature flags (ecosystem-specific, empty if not supported)
155    fn features(&self) -> &[String] {
156        &[]
157    }
158
159    /// LSP range of the features array (ecosystem-specific, None if not supported)
160    fn features_range(&self) -> Option<tower_lsp_server::ls_types::Range> {
161        None
162    }
163
164    /// Environment marker expression gating this dependency (e.g. PEP 508's
165    /// `python_version >= '3.8'`). Ecosystem-specific, `None` if not supported
166    /// or not present on this dependency.
167    fn markers(&self) -> Option<&str> {
168        None
169    }
170
171    /// LSP range of the environment marker expression (ecosystem-specific,
172    /// `None` if not supported or not present).
173    fn markers_range(&self) -> Option<tower_lsp_server::ls_types::Range> {
174        None
175    }
176
177    /// Downcast to concrete type
178    fn as_any(&self) -> &dyn Any;
179}

208/// Main trait that all ecosystem implementations must implement.
209///
210/// Each ecosystem (Cargo, npm, PyPI, etc.) provides its own implementation.
211/// This trait defines the contract for parsing manifests, fetching registry data,
212/// and generating LSP responses.
213///
214/// # Type Erasure
215///
216/// This trait uses `Box<dyn Trait>` instead of associated types to allow
217/// runtime polymorphism and dynamic ecosystem registration.
218///
219/// # Examples
220///
221/// ```no_run
222/// use deps_core::{Ecosystem, ParseResult, Registry, EcosystemConfig};
223/// use deps_core::lsp_helpers::EcosystemFormatter;
224/// use std::sync::Arc;
225/// use std::any::Any;
226/// use tower_lsp_server::ls_types::{Uri, CompletionItem, Position};
227///
228/// struct MyFormatter;
229/// impl EcosystemFormatter for MyFormatter {
230///     fn format_version_for_text_edit(&self, version: &str) -> String { version.to_string() }
231///     fn package_url(&self, name: &str) -> String { format!("https://example.com/{name}") }
232/// }
233///
234/// struct MyEcosystem {
235///     registry: Arc<dyn Registry>,
236///     formatter: MyFormatter,
237/// }
238///
239/// impl deps_core::ecosystem::private::Sealed for MyEcosystem {}
240///
241/// impl Ecosystem for MyEcosystem {
242///     fn id(&self) -> &'static str { "my-ecosystem" }
243///     fn display_name(&self) -> &'static str { "My Ecosystem" }
244///     fn manifest_filenames(&self) -> &[&'static str] { &["my-manifest.toml"] }
245///
246///     fn parse_manifest<'a>(
247///         &'a self,
248///         _content: &'a str,
249///         _uri: &'a Uri,
250///     ) -> deps_core::ecosystem::BoxFuture<'a, deps_core::error::Result<Box<dyn ParseResult>>> {
251///         Box::pin(async move { todo!() })
252///     }
253///
254///     fn registry(&self) -> Arc<dyn Registry> { self.registry.clone() }
255///
256///     fn formatter(&self) -> &dyn EcosystemFormatter { &self.formatter }
257///
258///     fn generate_completions<'a>(
259///         &'a self,
260///         _parse_result: &'a dyn ParseResult,
261///         _position: Position,
262///         _content: &'a str,
263///     ) -> deps_core::ecosystem::BoxFuture<'a, Vec<CompletionItem>> {
264///         Box::pin(async move { vec![] })
265///     }
266///
267///     fn as_any(&self) -> &dyn Any { self }
268/// }
269/// ```
270pub trait Ecosystem: Send + Sync + private::Sealed {
271    /// Unique identifier (e.g., "cargo", "npm", "pypi")
272    ///
273    /// This identifier is used for ecosystem registration and routing.
274    fn id(&self) -> &'static str;
275
276    /// Human-readable name (e.g., "Cargo (Rust)", "npm (JavaScript)")
277    ///
278    /// This name is displayed in diagnostic messages and logs.
279    fn display_name(&self) -> &'static str;
280
281    /// Manifest filenames this ecosystem handles (e.g., ["Cargo.toml"])
282    ///
283    /// The ecosystem registry uses these filenames to route file URIs
284    /// to the appropriate ecosystem implementation.
285    fn manifest_filenames(&self) -> &[&'static str];
286
287    /// File extensions this ecosystem handles when the manifest basename is
288    /// not fixed (e.g. `[".csproj", ".fsproj"]` for NuGet project files).
289    ///
290    /// Consulted by [`crate::EcosystemRegistry::get_for_filename`] only after
291    /// an exact [`manifest_filenames`](Ecosystem::manifest_filenames) match
292    /// fails. Empty by default, indicating this ecosystem is routed solely by
293    /// exact filename.
294    fn manifest_extensions(&self) -> &[&'static str] {
295        &[]
296    }
297
298    /// Lock file filenames this ecosystem uses (e.g., ["Cargo.lock"])
299    ///
300    /// Used for file watching - LSP will monitor changes to these files
301    /// and refresh UI when they change. Returns empty slice if ecosystem
302    /// doesn't use lock files.
303    ///
304    /// # Default Implementation
305    ///
306    /// Returns empty slice by default, indicating no lock files are used.
307    fn lockfile_filenames(&self) -> &[&'static str] {
308        &[]
309    }
310
311    /// Parse a manifest file and return parsed result
312    ///
313    /// # Arguments
314    ///
315    /// * `content` - Raw file content
316    /// * `uri` - Document URI for position tracking
317    ///
318    /// # Errors
319    ///
320    /// Returns error if manifest cannot be parsed
321    fn parse_manifest<'a>(
322        &'a self,
323        content: &'a str,
324        uri: &'a Uri,
325    ) -> BoxFuture<'a, crate::error::Result<Box<dyn ParseResult>>>;
326
327    /// Get the registry client for this ecosystem
328    ///
329    /// The registry provides version lookup and package search capabilities.
330    fn registry(&self) -> Arc<dyn Registry>;
331
332    /// Get the lock file provider for this ecosystem.
333    ///
334    /// Returns `None` if the ecosystem doesn't support lock files.
335    /// Lock files provide resolved dependency versions without network requests.
336    fn lockfile_provider(&self) -> Option<Arc<dyn crate::lockfile::LockFileProvider>> {
337        None
338    }
339
340    /// Get the ecosystem-specific formatter for LSP response generation.
341    ///
342    /// The formatter handles version comparison, package URLs, and text formatting.
343    /// Override this to customize LSP response generation.
344    fn formatter(&self) -> &dyn EcosystemFormatter;
345
346    /// Generate inlay hints for the document.
347    ///
348    /// Default implementation delegates to `lsp_helpers::generate_inlay_hints`
349    /// using `self.formatter()`. Override only if custom behavior is needed.
350    fn generate_inlay_hints<'a>(
351        &'a self,
352        parse_result: &'a dyn ParseResult,
353        versions: VersionData<'a>,
354        loading_state: crate::LoadingState,
355        config: &'a EcosystemConfig,
356    ) -> BoxFuture<'a, Vec<InlayHint>> {
357        Box::pin(async move {
358            crate::lsp_helpers::generate_inlay_hints(
359                parse_result,
360                versions,
361                loading_state,
362                config,
363                self.formatter(),
364            )
365        })
366    }
367
368    /// Generate hover information for a position.
369    ///
370    /// Default implementation delegates to `lsp_helpers::generate_hover`
371    /// using `self.formatter()` and `self.registry()`.
372    fn generate_hover<'a>(
373        &'a self,
374        parse_result: &'a dyn ParseResult,
375        position: Position,
376        versions: VersionData<'a>,
377    ) -> BoxFuture<'a, Option<Hover>> {
378        Box::pin(async move {
379            let registry = self.registry();
380            crate::lsp_helpers::generate_hover(
381                parse_result,
382                position,
383                versions,
384                registry.as_ref(),
385                self.formatter(),
386            )
387            .await
388        })
389    }
390
391    /// Generate code actions for a position.
392    ///
393    /// Default implementation delegates to `lsp_helpers::generate_code_actions`
394    /// using `self.formatter()` and `self.registry()`.
395    fn generate_code_actions<'a>(
396        &'a self,
397        parse_result: &'a dyn ParseResult,
398        position: Position,
399        uri: &'a Uri,
400    ) -> BoxFuture<'a, Vec<CodeAction>> {
401        Box::pin(async move {
402            let registry = self.registry();
403            crate::lsp_helpers::generate_code_actions(
404                parse_result,
405                position,
406                uri,
407                registry.as_ref(),
408                self.formatter(),
409            )
410            .await
411        })
412    }
413
414    /// Generate diagnostics for the document.
415    ///
416    /// Default implementation delegates to `lsp_helpers::generate_diagnostics_from_cache`
417    /// using `self.formatter()`.
418    fn generate_diagnostics<'a>(
419        &'a self,
420        parse_result: &'a dyn ParseResult,
421        versions: VersionData<'a>,
422        _uri: &'a Uri,
423    ) -> BoxFuture<'a, Vec<Diagnostic>> {
424        Box::pin(async move {
425            crate::lsp_helpers::generate_diagnostics_from_cache(
426                parse_result,
427                versions,
428                self.formatter(),
429            )
430        })
431    }
432
433    /// Generate completions for a position.
434    ///
435    /// Provides autocomplete suggestions for package names and versions.
436    fn generate_completions<'a>(
437        &'a self,
438        parse_result: &'a dyn ParseResult,
439        position: Position,
440        content: &'a str,
441    ) -> BoxFuture<'a, Vec<CompletionItem>>;
442
443    /// Support for downcasting to concrete ecosystem type
444    ///
445    /// This allows ecosystem-specific operations when needed.
446    fn as_any(&self) -> &dyn Any;
447}
