15/// Soft-sealing mechanism for [`Ecosystem`], shared by every ecosystem crate in this
16/// workspace (`deps-cargo`, `deps-npm`, ...) — each implements [`private::Sealed`] for
17/// its own ecosystem type.
18///
19/// Rust's privacy system has no "visible to this workspace, not beyond" level: `pub(crate)`
20/// would restrict `Sealed` to `deps-core` alone, breaking every sibling ecosystem crate's
21/// `impl Sealed for ...`, since each of those is a separate compilation unit. Making this
22/// module `pub` is therefore required, not a mistake — but it means [`private::Sealed`] is
23/// technically nameable, and implementable, from any crate that depends on `deps-core`, not
24/// only from within this workspace. `#[doc(hidden)]` keeps it out of generated public docs to
25/// avoid inviting that. There is no compiler-enforced wall against it: this is a documented
26/// contract enforced by code review, not a hard guarantee, and [`Ecosystem`]'s default
27/// methods may gain new required behavior without that counting as a breaking change for an
28/// external implementor who ignored this notice.
29#[doc(hidden)]
30pub mod private {
31    /// Marker trait every ecosystem crate in this workspace implements for its own
32    /// ecosystem type, per [`super::private`]'s module doc.
33    ///
34    /// [`Ecosystem`](super::Ecosystem) requires `Self: Sealed`, which is how the
35    /// trait stays extensible (new default methods can be added without
36    /// breaking in-workspace implementors).
37    pub trait Sealed {}
38}

43/// Runs `ecosystem.parse_manifest(content, uri)` on the blocking-thread pool instead of the
44/// calling tokio worker.
45///
46/// Every [`Ecosystem::parse_manifest`] implementation is synchronous work wrapped in an
47/// immediately-ready `async move` block (verified: no implementation contains a real
48/// `.await`). Driving that future to completion via [`tokio::runtime::Handle::block_on`]
49/// from inside [`tokio::task::spawn_blocking`] moves the CPU-bound parse off the tokio
50/// worker without changing the trait's async signature — mirrors
51/// [`crate::lockfile::read_and_parse_lockfile`]'s `spawn_blocking` pattern for the
52/// equivalent lock-file path (#723/#730). This closes #743: a large minified manifest
53/// parsed synchronously on an LSP request's tokio worker would otherwise stall every other
54/// request sharing that worker for the parse's full duration.
55///
56/// `content`/`uri` are cloned internally rather than moved+returned: the caller's own copy
57/// must remain valid (to store into `DocumentState`) even if the blocking task panics, and a
58/// panic here is already an anomalous condition, not a path worth optimizing a clone away
59/// for. Manifest sizes are capped (10MB) and typically far smaller, so the clone's added
60/// *time* cost is negligible next to the parse itself and the thread-pool hop. It does
61/// briefly double transient *peak memory* (both the caller's and the cloned copy live at
62/// once) on exactly the large-manifest path this function targets; a future `Arc<str>`
63/// threaded through the caller would remove that duplication if it ever proves significant.
64///
65/// `Handle::block_on` called from inside a `spawn_blocking` closure is a documented,
66/// supported tokio pattern (distinct from `Runtime::block_on`, which panics if called from
67/// within a runtime). Since the future never actually yields (`Poll::Pending`), `block_on`
68/// resolves on the first poll — negligible overhead beyond the `spawn_blocking` thread-hop
69/// itself.
70///
71/// # Errors
72///
73/// Returns whatever [`Ecosystem::parse_manifest`] returns for a malformed manifest, or a
74/// [`crate::error::DepsError::ParseError`] if the blocking task panics or is cancelled.
75///
76/// # Examples
77///
78/// ```no_run
79/// use deps_core::Ecosystem;
80/// use std::sync::Arc;
81/// use tower_lsp_server::ls_types::Uri;
82///
83/// # async fn example(ecosystem: Arc<dyn Ecosystem>, uri: Uri) -> deps_core::error::Result<()> {
84/// let parsed = deps_core::ecosystem::parse_manifest_blocking(&ecosystem, "content", &uri).await?;
85/// println!("{} dependencies", parsed.dependencies().len());
86/// # Ok(())
87/// # }
88/// ```
89pub async fn parse_manifest_blocking(
90    ecosystem: &Arc<dyn Ecosystem>,
91    content: &str,
92    uri: &Uri,
93) -> crate::error::Result<Box<dyn ParseResult>> {
94    let ecosystem = Arc::clone(ecosystem);
95    let owned_content = content.to_owned();
96    let owned_uri = uri.clone();
97    let handle = tokio::runtime::Handle::current();
98    let parsed = tokio::task::spawn_blocking(move || {
99        handle.block_on(ecosystem.parse_manifest(&owned_content, &owned_uri))
100    })
101    .await
102    .map_err(|e| crate::error::DepsError::ParseError {
103        file_type: format!("manifest at {uri:?}"),
104        source: Box::new(std::io::Error::other(e)),
105    })??;
106
107    // Single chokepoint for #796: every ecosystem's parse result is capped here, once,
108    // rather than each ecosystem crate's own parser bounding its dependency count
109    // independently.
110    Ok(crate::dependency_cap::cap_dependencies(
111        parsed,
112        crate::dependency_cap::MAX_DEPENDENCIES_PER_DOCUMENT,
113    ))
114}

116/// Defines [`EcosystemId`] together with [`EcosystemId::ALL`], [`EcosystemId::id`] and its
117/// [`std::str::FromStr`] impl from one variant list, so the four can never drift apart
118/// (#758) — a 15th ecosystem is added in exactly this one place or the workspace does not
119/// compile. `macro_rules!` is textual-order scoped, so this definition must precede its
120/// invocation below.
121macro_rules! ecosystem_ids {
122    ( $( $(#[$vmeta:meta])* $variant:ident => $id:literal ),+ $(,)? ) => {
123        /// Canonical, exhaustive identifier for every package ecosystem the workspace supports.
124        ///
125        /// [`Ecosystem::id`] returns a `&'static str` for registry lookups and document
126        /// storage, but any code that needs to *branch* on ecosystem identity should match on
127        /// this enum instead of re-deriving its own partial match over that string: an
128        /// unhandled variant here is a compile error, while an unhandled string is a silent
129        /// runtime bug (see the fix for issue #118, where two call sites silently mishandled
130        /// ecosystems missing from an incomplete string match).
131        ///
132        /// Deliberately **not** `#[non_exhaustive]`: adding a new ecosystem must force every
133        /// exhaustive `match` on this type across the workspace to be updated at compile time.
134        ///
135        /// # Examples
136        ///
137        /// ```
138        /// use deps_core::EcosystemId;
139        ///
140        /// let id: EcosystemId = "npm".parse().unwrap();
141        /// assert_eq!(id, EcosystemId::Npm);
142        /// assert_eq!(id.id(), "npm");
143        /// assert_eq!(id.to_string(), "npm");
144        ///
145        /// assert!("unknown".parse::<EcosystemId>().is_err());
146        /// ```
147        #[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
148        pub enum EcosystemId {
149            $( $(#[$vmeta])* $variant, )+
150        }
151
152        impl EcosystemId {
153            /// Every [`EcosystemId`] variant, in declaration order.
154            ///
155            /// Generated from the same list as [`Self::id`] and this type's
156            /// [`std::str::FromStr`] impl, so none of the three can silently drift out of
157            /// sync with the enum's variants (#758) — a forgotten 15th variant here is a
158            /// missing-match compile error, not a silently incomplete set.
159            pub const ALL: &'static [Self] = &[ $( Self::$variant ),+ ];
160
161            /// Returns the canonical string identifier, matching [`Ecosystem::id`] for the
162            /// corresponding ecosystem implementation.
163            #[must_use]
164            pub const fn id(self) -> &'static str {
165                match self {
166                    $( Self::$variant => $id, )+
167                }
168            }
169        }
170
171        impl std::str::FromStr for EcosystemId {
172            type Err = crate::error::DepsError;
173
174            fn from_str(s: &str) -> Result<Self, Self::Err> {
175                match s {
176                    $( $id => Ok(Self::$variant), )+
177                    _ => Err(crate::error::DepsError::UnsupportedEcosystem(s.to_string())),
178                }
179            }
180        }
181    };
182}

351        None
352    }

355    /// `None` if not supported or not present).

357        None

360    /// The raw manifest text spanned by [`version_range`](Dependency::version_range),
361    /// when it differs from [`version_requirement`](Dependency::version_requirement).
362    ///
363    /// Most ecosystems' `version_requirement()` is (up to whitespace) exactly the text at
364    /// `version_range()`, so the default `None` — telling callers to fall back to
365    /// `version_requirement()` — is correct for them. An ecosystem whose parser synthesizes
366    /// a comparator string from a bare literal (e.g. `deps-swift`'s `.exact("4.50.0")`
367    /// becoming requirement `"=4.50.0"` while `version_range()` still spans only `4.50.0`)
368    /// overrides this to return that literal, so `lsp_helpers`' literal-span guard
369    /// (`literal_span_matches`, used by both `generate_code_actions` and
370    /// `collect_update_all_edits`) compares `version_range`'s slice against the text it was
371    /// actually derived from instead of the synthesized comparator, which would otherwise
372    /// never match and silently suppress every fix action for that dependency.
373    ///
374    /// A sibling mechanism already exists for the same underlying problem: `deps-nuget`
375    /// wraps a bare source version as requirement `[1.0.0]`, and `literal_span_matches`
376    /// special-cases that bracket wrapping inline rather than going through this hook. This
377    /// method exists for the general case — an ecosystem whose synthesized requirement is
378    /// not a simple wrap (`deps-swift`'s comparator range is not recoverable from
379    /// `version_requirement()` by stripping fixed characters) needs its own literal, not a
380    /// transform `literal_span_matches` could hard-code.
381    ///
382    /// **Must not** be set when `version_range()` spans only part of a multi-literal
383    /// requirement whose other part(s) are not being rewritten — e.g. a `"lower"..<"upper"`
384    /// range, where `version_range()` covers only `lower`. Reporting `lower` as the literal
385    /// would let the guard pass and an edit rewrite `lower` alone, corrupting the
386    /// requirement (`deps-swift` leaves this `None` for both its range-literal forms for
387    /// exactly this reason — see `crates/deps-swift/src/parser.rs`'s range-form comments).
388    fn version_literal(&self) -> Option<&str> {
389        None
390    }
391
392    /// Downcast to concrete type
393    fn as_any(&self) -> &dyn Any;
394}

498/// How this ecosystem's license strings are sourced.
499///
500/// Drives three ecosystem-specific license behaviors that used to be re-derived
501/// independently at each consumer via a non-exhaustive `matches!`/`==` on
502/// [`EcosystemId`] instead of the sealed [`Ecosystem`] trait every other per-ecosystem
503/// capability goes through (issue #688): hover's "(detected)" qualifier, which only
504/// applies to [`Self::DetectedSpdx`]; whether
505/// [`crate::licenses::resolve_license_entries`] must normalize free text to SPDX
506/// identifiers before hover displays it or a [`crate::licenses::LicensePolicy`]
507/// evaluates it, which only applies to [`Self::PomFreeText`]; and, via
508/// [`Self::requires_dedicated_fetch`] (issue #697), whether
509/// [`Ecosystem::fetch_license`] is a dedicated async fetch or a no-op because the
510/// license already arrived in the hot-path registry response.
511///
512/// # Examples
513///
514/// ```
515/// use deps_core::LicenseSource;
516///
517/// assert_eq!(LicenseSource::default(), LicenseSource::RegistryDeclaredSpdx);
518/// assert_ne!(LicenseSource::PomFreeText, LicenseSource::DetectedSpdx);
519/// ```
520#[non_exhaustive]
521#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
522pub enum LicenseSource {
523    /// Author-declared SPDX identifier(s) already present in the hot-path registry
524    /// response — the default for every ecosystem that doesn't override
525    /// [`Ecosystem::license_source`], and the only variant for which
526    /// [`Self::requires_dedicated_fetch`] is `false`.
527    #[default]
528    RegistryDeclaredSpdx,
529    /// Author-declared SPDX identifier(s) that require a dedicated fetch separate
530    /// from the hot-path registry response: JSR's package-metadata endpoint (Deno).
531    FetchedDeclaredSpdx,
532    /// A detector's best-effort guess, not an author declaration, that requires a
533    /// dedicated fetch: pub.dev's `/score` endpoint (Dart) and GitHub's
534    /// `license.spdx_id` (Swift) are both driven by a license-detection heuristic run
535    /// against repository content, rather than metadata the package author explicitly
536    /// declared to a registry.
537    DetectedSpdx,
538    /// Maven POM `<licenses><license><name>` free text (Gradle), fetched via a
539    /// dedicated request — never an SPDX identifier, so it must be normalized via
540    /// [`crate::licenses::resolve_license_entries`] before display or policy evaluation.
541    PomFreeText,
542}

544impl LicenseSource {
545    /// Whether this source requires [`Ecosystem::fetch_license`]'s dedicated async
546    /// fetch, as opposed to arriving for free in the hot-path registry response.
547    ///
548    /// This is the single answer to "is this a tier-3 license ecosystem" (issue #697):
549    /// capability and shape can no longer disagree, because both come from the same
550    /// [`Ecosystem::license_source`] call.
551    ///
552    /// # Examples
553    ///
554    /// ```
555    /// use deps_core::LicenseSource;
556    ///
557    /// assert!(!LicenseSource::RegistryDeclaredSpdx.requires_dedicated_fetch());
558    /// assert!(LicenseSource::FetchedDeclaredSpdx.requires_dedicated_fetch());
559    /// assert!(LicenseSource::DetectedSpdx.requires_dedicated_fetch());
560    /// assert!(LicenseSource::PomFreeText.requires_dedicated_fetch());
561    /// ```
562    #[must_use]
563    pub const fn requires_dedicated_fetch(self) -> bool {
564        match self {
565            Self::RegistryDeclaredSpdx => false,
566            Self::FetchedDeclaredSpdx | Self::DetectedSpdx | Self::PomFreeText => true,
567        }
568    }

571/// Main trait that all ecosystem implementations must implement.
572///
573/// Each ecosystem (Cargo, npm, PyPI, etc.) provides its own implementation.
574/// This trait defines the contract for parsing manifests, fetching registry data,
575/// and generating LSP responses.
576///
577/// # Type Erasure
578///
579/// This trait uses `Box<dyn Trait>` instead of associated types to allow
580/// runtime polymorphism and dynamic ecosystem registration.
581///
582/// # Sealing
583///
584/// This trait requires `Self: private::Sealed`, making it sealed in the sense described
585/// on that module's doc: a documented contract enforced by code review, not a
586/// compiler-enforced wall. Every sibling ecosystem crate in this workspace (`deps-cargo`,
587/// `deps-npm`, ...) implements [`private::Sealed`] for its own ecosystem type, which requires
588/// `private` to be `pub`; Rust has no visibility level that admits sibling crates while
589/// excluding a truly external one, so this guarantee cannot be enforced any harder than that
590/// without inverting the crate's whole multi-crate extension-point architecture. See
591/// [`private::Sealed`]'s own doc for the full reasoning, and the `impl private::Sealed`
592/// line in the example below for what implementing it in practice looks like.
593///
594/// # Examples
595///
596/// ```no_run
597/// use deps_core::{Ecosystem, ParseResult, Registry, EcosystemConfig, PackageName, ConcreteVersion, Metadata};
598/// use deps_core::completion::{Completions, build_feature_completion};
599/// use deps_core::lsp_helpers::{
600///     DiagnosticMessages, DiagnosticPolicy, EcosystemFormatter, OsvNaming, PackageNaming,
601///     PackageRendering, RequirementResolution, SourcePolicy,
602/// };
603/// use std::sync::Arc;
604/// use std::any::Any;
605/// use tower_lsp_server::ls_types::{
606///     Uri, CompletionItem, CompletionTextEdit, Position, Range, TextEdit,
607/// };
608///
609/// struct MyFormatter;
610/// impl PackageNaming for MyFormatter {}
611/// impl PackageRendering for MyFormatter {
612///     fn format_version_for_text_edit(&self, version: &ConcreteVersion) -> String { version.to_string() }
613///     fn package_url(&self, name: &PackageName) -> String { format!("https://example.com/{name}") }
614/// }
615/// impl RequirementResolution for MyFormatter {}
616/// impl DiagnosticMessages for MyFormatter {}
617/// impl DiagnosticPolicy for MyFormatter {}
618/// impl SourcePolicy for MyFormatter {}
619/// impl OsvNaming for MyFormatter {}
620///
621/// struct MyEcosystem {
622///     registry: Arc<dyn Registry>,
623///     formatter: MyFormatter,
624/// }
625///
626/// // Real in-workspace ecosystem crates implement `Sealed` exactly like this. This line
627/// // compiling here, out-of-crate, is not a bug: as the `# Sealing` section above explains,
628/// // `private::Sealed` is a documented contract, not a compiler-enforced wall — Rust has no
629/// // visibility level that admits sibling workspace crates while excluding a truly external
630/// // one, so any crate that names this path can technically do the same.
631/// impl deps_core::ecosystem::private::Sealed for MyEcosystem {}
632///
633/// impl Ecosystem for MyEcosystem {
634///     fn ecosystem_id(&self) -> deps_core::EcosystemId { deps_core::EcosystemId::Cargo }
635///     fn display_name(&self) -> &'static str { "My Ecosystem" }
636///     fn manifest_filenames(&self) -> &[&'static str] { &["my-manifest.toml"] }
637///
638///     fn parse_manifest<'a>(
639///         &'a self,
640///         _content: &'a str,
641///         _uri: &'a Uri,
642///     ) -> deps_core::ecosystem::BoxFuture<'a, deps_core::error::Result<Box<dyn ParseResult>>> {
643///         Box::pin(async move { todo!() })
644///     }
645///
646///     fn registry(&self) -> Arc<dyn Registry> { self.registry.clone() }
647///
648///     fn formatter(&self) -> &dyn EcosystemFormatter { &self.formatter }
649///
650///     // Hook for `CompletionContext::PackageName` — called by `generate_completions`'s
651///     // default dispatch (below) whenever the cursor sits in a package-name token. A
652///     // real ecosystem would search `self.registry` for `prefix` and build one item per
653///     // match, e.g. via `deps_core::completion::complete_package_names_generic`.
654///     fn complete_package_name<'a>(
655///         &'a self,
656///         _request: deps_core::completion::CompletionRequest<'a>,
657///         prefix: String,
658///         range: Range,
659///     ) -> deps_core::ecosystem::BoxFuture<'a, Completions> {
660///         Box::pin(async move {
661///             let item = CompletionItem {
662///                 label: prefix.clone(),
663///                 text_edit: Some(CompletionTextEdit::Edit(TextEdit { range, new_text: prefix })),
664///                 ..Default::default()
665///             };
666///             Completions::new(vec![item])
667///         })
668///     }
669///
670///     // Hook for `CompletionContext::Version` — called by `generate_completions`'s
671///     // default dispatch whenever the cursor sits in a version-requirement token.
672///     fn complete_version<'a>(
673///         &'a self,
674///         _request: deps_core::completion::CompletionRequest<'a>,
675///         _package_name: PackageName,
676///         _prefix: String,
677///     ) -> deps_core::ecosystem::BoxFuture<'a, Completions> {
678///         Box::pin(async move { Completions::default() })
679///     }
680///
681///     // Hook for `CompletionContext::Feature` — called by `generate_completions`'s
682///     // default dispatch whenever the cursor sits in a feature-flag array entry. Reuses
683///     // the shared `build_feature_completion` helper rather than hand-building the item.
684///     fn complete_feature<'a>(
685///         &'a self,
686///         _request: deps_core::completion::CompletionRequest<'a>,
687///         package_name: PackageName,
688///         prefix: String,

690///         Box::pin(async move {
691///             let item = build_feature_completion(&prefix, &package_name, None);
692///             Completions::new(vec![item])
693///         })
694///     }
695///
696///     fn completion_insert_text(&self, metadata: &dyn Metadata) -> Option<String> {
697///         Some(format!("\"{}\" = \"{}\"", metadata.name(), metadata.latest_version()))
698///     }
699///
700///     fn as_any(&self) -> &dyn Any { self }
701/// }

1044    ///
1045    /// Provides autocomplete suggestions for package names, versions, and features by
1046    /// detecting the [`crate::completion::CompletionContext`] at `position` and dispatching
1047    /// to the matching hook — [`Self::complete_package_name`], [`Self::complete_version`], or
1048    /// [`Self::complete_feature`].
1049    ///
1050    /// The exhaustive `match` over [`crate::completion::CompletionContext`] lives here, in
1051    /// the crate that defines the enum, so `#[non_exhaustive]` does not apply and a new
1052    /// variant is a compile error in this one place instead of a silent no-op downstream
1053    /// (issue #793). An ecosystem that overrides this default opts out of that guarantee and
1054    /// takes the obligation on itself — `deps-maven` and `deps-gradle` do, because they route
1055    /// on their own XML/Groovy context type instead of
1056    /// [`crate::completion::CompletionContext`].
1057    ///
1058    /// `freshness.enabled` gates whether version completion items carry a
1059    /// relative-age `label_details` suffix (issue #145); implementations that
1060    /// delegate to [`crate::completion::complete_versions_generic`] get this for
1061    /// free by threading `freshness` through.
1062    ///
1063    /// The returned [`Completions::is_incomplete`] must reflect *this specific call*
1064    /// (the completion context actually served), not a static worst case for the
1065    /// ecosystem as a whole (#427): a package-name search over an unranked,
1066    /// truncated index should report `true`, while a version completion or any
1067    /// other exhaustive context in the same manifest must report `false`, even for
1068    /// an ecosystem where some contexts are incomplete and others are not.
1069    fn generate_completions<'a>(
1070        &'a self,
1071        parse_result: &'a dyn ParseResult,
1072        position: Position,
1073        content: &'a str,
1074        freshness: crate::FreshnessSettings,
1075    ) -> BoxFuture<'a, Completions> {
1076        Box::pin(async move {
1077            self.prepare_completions();
1078            let request =
1079                crate::completion::CompletionRequest::new(parse_result, position, freshness);
1080            // Exhaustive on purpose (#793): no wildcard arm, so a new
1081            // `CompletionContext` variant is a compile error right here.
1082            match crate::completion::detect_completion_context(parse_result, position, content) {
1083                crate::completion::CompletionContext::PackageName { prefix, range } => {
1084                    self.complete_package_name(request, prefix, range).await
1085                }
1086                crate::completion::CompletionContext::Version {
1087                    package_name,
1088                    prefix,
1089                } => self.complete_version(request, package_name, prefix).await,
1090                crate::completion::CompletionContext::Feature {
1091                    package_name,
1092                    prefix,
1093                } => self.complete_feature(request, package_name, prefix).await,
1094                crate::completion::CompletionContext::None => Completions::default(),
1095            }
1096        })
1097    }
