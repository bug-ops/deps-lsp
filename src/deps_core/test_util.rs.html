7//! Test fixtures and helpers shared across ecosystem crates.
8//!
9//! Test fixtures throughout the workspace write absolute paths in Unix
10//! style (e.g. `/project/Cargo.toml`) for readability. `Uri::from_file_path`
11//! requires a platform-absolute path, and a Unix-style path is not
12//! recognized as absolute on Windows (no drive letter), so calling it
13//! directly with such a literal panics on Windows only. [`test_uri`]
14//! normalizes the path per host platform before constructing the [`Uri`].
15//!
16//! [`assert_dot_segment_gated_or_contained`]/[`assert_dot_segment_gated_or_contained_transformed`]
17//! guard the recurring dot-segment / unvalidated-URL-sink defect class (#337, #341, #349,
18//! #357, #361, #365, #371) against further recurrence.
19//!
20//! [`capture_tracing_output`]/[`capture_tracing_output_async`] let a test assert a
21//! `tracing` call actually fired (originally added standalone in `deps-swift` for #357,
22//! then duplicated per-crate for #380/#378's `warn_rejected_value` coverage before being
23//! consolidated here).
24

27/// Builds a [`Uri`] from a Unix-style absolute test path.
28///
29/// On Windows, a synthetic `C:` drive is prefixed so the path is
30/// recognized as absolute; on other platforms the path is used as-is.
31///
32/// # Panics
33///
34/// Panics if the resulting path is not a valid file URI. This is a test
35/// helper: fixture paths are expected to always be well-formed.
36///
37/// # Examples
38///
39/// ```
40/// use deps_core::test_util::test_uri;
41///
42/// let uri = test_uri("/project/Cargo.toml");
43/// assert!(uri.path().as_str().ends_with("Cargo.toml"));
44/// ```
45#[must_use]
46pub fn test_uri(unix_path: &str) -> Uri {
47    #[cfg(windows)]
48    let owned;
49    #[cfg(windows)]
50    let path: &str = {
51        owned = format!("C:{unix_path}");
52        &owned
53    };
54    #[cfg(not(windows))]
55    let path: &str = unix_path;
56
57    Uri::from_file_path(path).expect("test_uri: fixture path must be a valid file URI")
58}

125/// Canonical adversarial identifier values for the recurring dot-segment /
126/// unvalidated-URL-sink defect class (#337, #341, #349, #357, #361).
127///
128/// A manifest-declared package name, scope, or coordinate segment spliced into a
129/// registry/API URL via `format!`/string interpolation without validation. A bare `.`/`..`
130/// survives naive percent-encoding unchanged (`.` is an RFC 3986 unreserved character) and
131/// is silently removed by a URL parser's dot-segment normalization once the string is
132/// assembled and parsed, letting the request escape the intended host or path prefix; the
133/// remaining entries cover a would-be traversal attempt, whitespace, and query/fragment
134/// injection.
135pub const ADVERSARIAL_URL_SEGMENTS: &[&str] = &[
136    ".",
137    "..",
138    "../../etc/passwd",
139    "a b",
140    "a?b=1",
141    "a#frag",
142    "%2e%2e",
143];
144
145/// Exercises one ecosystem's identifier-to-URL sink against every
146/// [`ADVERSARIAL_URL_SEGMENTS`] entry.
147///
148/// `resolve` should mirror the real request path: apply whatever validation gate
149/// (`is_dot_segment`, `is_safe_package_name`, `is_safe_maven_coordinate_segment`, ...) the
150/// production code runs before building the request, returning `None` when the gate would
151/// reject the identifier (the request is never built, so there is nothing to check), or
152/// `Some(url)` with the URL the identifier resolves to when it reaches the real
153/// fetch-URL-builder function directly.
154///
155/// For every input that reaches `Some(url)`, asserts `url` parses, stays under
156/// `expected_host`/`expected_path_prefix`, and that `segment` itself survives the round
157/// trip. That last check is the one that actually catches a missing/deleted gate: for an
158/// ecosystem whose identifier is the *first* path component (no fixed sub-path to nest
159/// under, e.g. `deps-cargo`'s sparse-index path or `deps-npm`'s bare
160/// `registry.npmjs.org/{name}`), `expected_path_prefix` can only ever be `"/"` — trivially
161/// satisfied by any path — so the prefix check alone is a tautology there. Deleting the
162/// real gate collapses `..`/`.` via dot-segment normalization, which removes it from the
163/// path entirely; the survival check catches that regardless of how trivial
164/// `expected_path_prefix` is, so passing `"/"` is fine as long as this check is also in
165/// effect.
166///
167/// The survival check itself takes one of two forms depending on `segment`:
168/// - For a bare `.`/`..` (the only values `url`'s dot-segment normalization treats
169///   specially): some path segment, once percent-decoded, must *start with* that value.
170///   A whole-path substring search would be too weak here — a coincidental `.` baked into
171///   a static suffix the sink always appends (e.g. `.json`) would satisfy `contains` even
172///   if the real `.`/`..` segment was silently removed, leaving no trace of it anywhere.
173///   `starts_with` (not exact equality) still accommodates a sink that glues the
174///   identifier directly onto a static suffix with no separator (e.g. `deps-bundler`'s
175///   `versions_url`, which decodes a `..` identifier to the segment `"..json"`).
176/// - For every other adversarial entry: the percent-decoded *whole path* must contain
177///   `segment` (or `transform(segment)`) as a substring — safe here since none of those
178///   entries collides with a static suffix the way a bare `.` does.
179///
180/// # Panics
181///
182/// Panics if a returned URL fails to parse, escapes `expected_host` or
183/// `expected_path_prefix`, if the (possibly transformed) segment is empty despite
184/// `resolve` returning `Some`, or if the segment does not survive per the rules above.
185///
186/// # Examples
187///
188/// ```
189/// use deps_core::test_util::assert_dot_segment_gated_or_contained;
190///
191/// fn build(name: &str) -> String {
192///     format!("https://example.com/api/{}", urlencoding::encode(name))
193/// }
194///
195/// fn resolve(name: &str) -> Option<String> {
196///     (name != "." && name != "..").then(|| build(name))
197/// }
198///
199/// assert_dot_segment_gated_or_contained(resolve, "example.com", "/api/");
200/// ```
201pub fn assert_dot_segment_gated_or_contained(
202    resolve: impl Fn(&str) -> Option<String>,
203    expected_host: &str,
204    expected_path_prefix: &str,
205) {
206    assert_dot_segment_gated_or_contained_transformed(
207        resolve,
208        str::to_string,
209        expected_host,
210        expected_path_prefix,
211    );
212}

214/// As [`assert_dot_segment_gated_or_contained`], but for a `resolve` whose production gate
215/// legitimately transforms the identifier before it reaches the URL builder.
216///
217/// E.g. PyPI's PEP 503 `name::normalize`, which collapses `.`/`_`/`-` runs, rather than
218/// passing the identifier through unchanged. `transform` computes what the identifier looks
219/// like once it reaches the sink, so the survival check compares the decoded path against
220/// that instead of the raw adversarial `segment` (which would otherwise never appear
221/// literally, producing a false-positive failure with no real bug behind it).
222///
223/// # Panics
224///
225/// Same conditions as [`assert_dot_segment_gated_or_contained`], with `transform(segment)`
226/// in place of `segment` for the survival check.
227pub fn assert_dot_segment_gated_or_contained_transformed(
228    resolve: impl Fn(&str) -> Option<String>,
229    transform: impl Fn(&str) -> String,
230    expected_host: &str,
231    expected_path_prefix: &str,
232) {
233    for segment in ADVERSARIAL_URL_SEGMENTS {
234        let Some(built) = resolve(segment) else {
235            continue;
236        };
237        let parsed = url::Url::parse(&built).unwrap_or_else(|e| {
238            panic!("adversarial segment {segment:?} produced an unparsable URL {built:?}: {e}")
239        });
240        assert_eq!(
241            parsed.host_str(),
242            Some(expected_host),
243            "adversarial segment {segment:?} escaped host: {built}"
244        );
245        assert!(
246            parsed.path().starts_with(expected_path_prefix),
247            "adversarial segment {segment:?} escaped path prefix {expected_path_prefix:?}: {built}"
248        );
249        let expected_fragment = transform(segment);
250        assert!(
251            !expected_fragment.is_empty(),
252            "adversarial segment {segment:?} transformed to an empty fragment but `resolve` \
253             still returned Some(url) — an empty identifier must be rejected (return `None`) \
254             before reaching the sink, since an empty survival check would trivially pass \
255             (\"\".contains(\"\") is always true) and hide a real gate deletion"
256        );
257        if expected_fragment == "." || expected_fragment == ".." {
258            // A whole-path substring search is too weak here: a coincidental `.` baked
259            // into a static suffix the sink always appends (e.g. `.json`/`.xml`) would
260            // satisfy `contains` even if the real gate is deleted and the actual `.`/`..`
261            // segment was silently removed by dot-segment normalization, leaving no trace
262            // of it anywhere. Require a per-segment match instead: some path segment, once
263            // decoded, must itself start with the dot-segment value — `starts_with` (not
264            // exact equality) accommodates a sink that glues the identifier directly onto a
265            // static suffix with no separator (e.g. deps-bundler's `versions_url`, which
266            // decodes a `..` identifier to the segment `"..json"`).
267            let matched = parsed.path_segments().is_some_and(|mut segs| {
268                segs.any(|seg| {
269                    urlencoding::decode(seg)
270                        .is_ok_and(|decoded| decoded.starts_with(expected_fragment.as_str()))
271                })
272            });
273            assert!(
274                matched,
275                "adversarial segment {segment:?} (expected to survive as {expected_fragment:?}) \
276                 did not survive as its own path segment (silently dropped/collapsed by \
277                 dot-segment normalization?): built url {built}"
278            );
279        } else {
280            let decoded_path = urlencoding::decode(parsed.path()).unwrap_or_else(|e| {
281                panic!(
282                    "adversarial segment {segment:?}'s URL path failed to percent-decode: {built:?}: {e}"
283                )
284            });
285            assert!(
286                decoded_path.contains(&expected_fragment),
287                "adversarial segment {segment:?} (expected to survive as {expected_fragment:?}) \
288                 did not survive intact in the decoded path (silently dropped/collapsed by \
289                 dot-segment normalization?): decoded path {decoded_path:?}, built url {built}"
290            );
291        }
292    }
293}

295// Gated separately from the rest of this module (which also compiles under plain
296// `cfg(test)`, i.e. `cargo test -p deps-core` with no explicit features): `tracing-subscriber`
297// is an *optional* dependency enabled only by the `test-util` feature, so a build that hits
298// this module via bare `cfg(test)` alone would fail to resolve it without this narrower gate.
299#[cfg(feature = "test-util")]
300#[derive(Clone, Default)]
301struct CapturingWriter(std::sync::Arc<std::sync::Mutex<Vec<u8>>>);
302
303#[cfg(feature = "test-util")]
304impl std::io::Write for CapturingWriter {
305    fn write(&mut self, buf: &[u8]) -> std::io::Result<usize> {
306        self.0.lock().unwrap().extend_from_slice(buf);
307        Ok(buf.len())
308    }
309
310    fn flush(&mut self) -> std::io::Result<()> {
311        Ok(())
312    }
313}
314
315#[cfg(feature = "test-util")]
316impl<'a> tracing_subscriber::fmt::MakeWriter<'a> for CapturingWriter {
317    type Writer = Self;
318
319    fn make_writer(&'a self) -> Self::Writer {
320        self.clone()
321    }
322}
323
324#[cfg(feature = "test-util")]
325fn capturing_subscriber_at(
326    max_level: tracing::Level,
327) -> (CapturingWriter, impl tracing::Subscriber) {
328    let writer = CapturingWriter::default();
329    let subscriber = tracing_subscriber::fmt()
330        .with_writer(writer.clone())
331        .with_max_level(max_level)
332        .without_time()
333        .with_target(false)
334        .finish();
335    (writer, subscriber)
336}
337
338#[cfg(feature = "test-util")]
339fn capturing_subscriber() -> (CapturingWriter, impl tracing::Subscriber) {
340    // INFO (not WARN): some call sites emit `tracing::info!` (e.g. deps-swift's
341    // release-dates token-gate skip) that a WARN-only filter would silently drop,
342    // alongside every `warn_rejected_value`/other WARN-level emission this helper
343    // exists to assert on.
344    capturing_subscriber_at(tracing::Level::INFO)
345}

347/// Captures `tracing` output emitted synchronously during `f` into a `String`.
348///
349/// Lets a test assert a `tracing::warn!`/`info!` call actually fired — e.g.
350/// [`crate::lsp_helpers::warn_rejected_value`] — without a real logging sink or a
351/// network-dependent end-to-end path. Filters below `INFO`, so a `tracing::debug!`/`trace!`
352/// emission never appears here — use [`capture_tracing_output_at`] for those.
353///
354/// # Examples
355///
356/// ```
357/// use deps_core::test_util::capture_tracing_output;
358///
359/// let output = capture_tracing_output(|| tracing::warn!("something rejected"));
360/// assert!(output.contains("something rejected"));
361/// ```
362#[cfg(feature = "test-util")]
363#[must_use]
364pub fn capture_tracing_output(f: impl FnOnce()) -> String {
365    let (writer, subscriber) = capturing_subscriber();
366    tracing::subscriber::with_default(subscriber, f);
367    String::from_utf8(writer.0.lock().unwrap().clone()).expect("tracing output is valid utf8")
368}

370/// Like [`capture_tracing_output`], but capturing every level up to and including `max_level`
371/// (e.g. `tracing::Level::DEBUG`).
372///
373/// Needed to positively assert a `tracing::debug!` line fired, or that a `tracing::warn!`
374/// specifically (as opposed to any level) did not — [`capture_tracing_output`]'s fixed `INFO`
375/// filter makes both assertions vacuous, since a `debug!` call is invisible there regardless of
376/// whether the code under test emits it correctly.
377///
378/// # Examples
379///
380/// ```
381/// use deps_core::test_util::capture_tracing_output_at;
382///
383/// let output =
384///     capture_tracing_output_at(tracing::Level::DEBUG, || tracing::debug!("quiet detail"));
385/// assert!(output.contains("quiet detail"));
386/// ```
387#[cfg(feature = "test-util")]
388#[must_use]
389pub fn capture_tracing_output_at(max_level: tracing::Level, f: impl FnOnce()) -> String {
390    let (writer, subscriber) = capturing_subscriber_at(max_level);
391    tracing::subscriber::with_default(subscriber, f);
392    String::from_utf8(writer.0.lock().unwrap().clone()).expect("tracing output is valid utf8")
393}

395/// Async counterpart of [`capture_tracing_output`], for a `tracing` emission inside an
396/// `async fn`/`.await`ed future.
397///
398/// Relies on a `#[tokio::test]` current-thread runtime polling `fut` on the same thread
399/// that installed the subscriber as the thread-local default.
400///
401/// # Examples
402///
403/// ```
404/// use deps_core::test_util::capture_tracing_output_async;
405///
406/// # #[tokio::main]
407/// # async fn main() {
408/// let output = capture_tracing_output_async(async {
409///     tracing::warn!("something rejected");
410/// })
411/// .await;
412/// assert!(output.contains("something rejected"));
413/// # }
414/// ```
415#[cfg(feature = "test-util")]
416pub async fn capture_tracing_output_async(fut: impl std::future::Future<Output = ()>) -> String {
417    let (writer, subscriber) = capturing_subscriber();
418    let guard = tracing::subscriber::set_default(subscriber);
419    fut.await;
420    drop(guard);
421    String::from_utf8(writer.0.lock().unwrap().clone()).expect("tracing output is valid utf8")
422}

454}
