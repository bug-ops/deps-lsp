7/// Registry for all available ecosystems.
8///
9/// This registry manages ecosystem implementations and provides fast lookup
10/// by ecosystem ID or manifest filename. It's designed for thread-safe
11/// concurrent access using DashMap.
12///
13/// # Examples
14///
15/// ```no_run
16/// use deps_core::EcosystemRegistry;
17/// use std::sync::Arc;
18///
19/// let registry = EcosystemRegistry::new();
20///
21/// // Register ecosystems (would be actual implementations)
22/// // registry.register(Arc::new(CargoEcosystem::new(cache.clone())));
23/// // registry.register(Arc::new(NpmEcosystem::new(cache.clone())));
24///
25/// // Look up by filename
26/// if let Some(ecosystem) = registry.get_for_filename("Cargo.toml") {
27///     println!("Found ecosystem: {}", ecosystem.display_name());
28/// }
29///
30/// // List all registered ecosystems
31/// for id in registry.ecosystem_ids() {
32///     println!("Registered: {}", id);
33/// }
34/// ```
35pub struct EcosystemRegistry {
36    /// Map from ecosystem ID to implementation
37    ecosystems: DashMap<&'static str, Arc<dyn Ecosystem>>,
38    /// Map from filename to ecosystem ID (for fast lookup)
39    filename_map: DashMap<&'static str, &'static str>,
40    /// Map from lowercased file extension (e.g. ".csproj") to ecosystem ID.
41    ///
42    /// Consulted only when an exact `filename_map` lookup misses. Kept as a
43    /// separate `DashMap` rather than folded into `filename_map` so extension
44    /// routing cannot silently shadow an exact filename. Two ecosystems
45    /// claiming the same extension is a configuration error caught by a
46    /// `debug_assert!` in [`register`](EcosystemRegistry::register) — in a
47    /// release build (or if the assertion is otherwise skipped) `DashMap::insert`
48    /// silently last-write-wins, so the outcome is registration-order-dependent,
49    /// not deterministic.
50    extension_map: DashMap<&'static str, &'static str>,
51}
52
53impl EcosystemRegistry {
54    /// Create a new empty registry
55    ///
56    /// # Examples
57    ///
58    /// ```
59    /// use deps_core::EcosystemRegistry;
60    ///
61    /// let registry = EcosystemRegistry::new();
62    /// assert_eq!(registry.ecosystem_ids().len(), 0);
63    /// ```
64    pub fn new() -> Self {
65        Self {
66            ecosystems: DashMap::new(),
67            filename_map: DashMap::new(),
68            extension_map: DashMap::new(),
69        }
70    }
71
72    /// Register an ecosystem implementation
73    ///
74    /// This method registers the ecosystem and creates filename mappings
75    /// for all manifest filenames declared by the ecosystem.
76    ///
77    /// # Arguments
78    ///
79    /// * `ecosystem` - Arc-wrapped ecosystem implementation
80    ///
81    /// # Examples
82    ///
83    /// ```no_run
84    /// use deps_core::EcosystemRegistry;
85    /// use std::sync::Arc;
86    ///
87    /// let registry = EcosystemRegistry::new();
88    /// // registry.register(Arc::new(CargoEcosystem::new(cache)));
89    /// ```
90    pub fn register(&self, ecosystem: Arc<dyn Ecosystem>) {
91        let id = ecosystem.id();
92
93        // Register filename mappings
94        for filename in ecosystem.manifest_filenames() {
95            self.filename_map.insert(*filename, id);
96        }
97
98        // Register extension mappings (fallback for unbounded basenames, e.g. *.csproj)
99        for extension in ecosystem.manifest_extensions() {
100            debug_assert!(
101                self.extension_map
102                    .get(extension)
103                    .is_none_or(|owner| *owner == id),
104                "extension {extension:?} already claimed by ecosystem {:?}, cannot also assign it to {id:?}",
105                self.extension_map.get(extension).map(|o| *o),
106            );
107            self.extension_map.insert(*extension, id);
108        }
109
110        // Register ecosystem
111        self.ecosystems.insert(id, ecosystem);
112    }
113
114    /// Get ecosystem by ID
115    ///
116    /// # Arguments
117    ///
118    /// * `id` - Ecosystem identifier (e.g., "cargo", "npm", "pypi")
119    ///
120    /// # Returns
121    ///
122    /// * `Some(Arc<dyn Ecosystem>)` - Registered ecosystem
123    /// * `None` - No ecosystem registered with this ID
124    ///
125    /// # Examples
126    ///
127    /// ```no_run
128    /// use deps_core::EcosystemRegistry;
129    ///
130    /// let registry = EcosystemRegistry::new();
131    /// if let Some(ecosystem) = registry.get("cargo") {
132    ///     println!("Found: {}", ecosystem.display_name());
133    /// }
134    /// ```
135    pub fn get(&self, id: &str) -> Option<Arc<dyn Ecosystem>> {
136        self.ecosystems.get(id).map(|e| Arc::clone(&e))
137    }
138
139    /// Get ecosystem for a filename
140    ///
141    /// Lookup is two-stage: an exact, case-sensitive match against
142    /// registered manifest filenames (e.g. `"Cargo.toml"`) is tried first;
143    /// if that misses, the filename's extension is matched case-insensitively
144    /// against registered [`Ecosystem::manifest_extensions`]. This asymmetry
145    /// is deliberate — exact filenames are canonical and case-sensitive on
146    /// Unix filesystems, while extensions on unbounded basenames (e.g.
147    /// `*.csproj`) come from MSBuild/Windows projects that commonly vary
148    /// case (`MyApp.CSPROJ`). So `MyApp.CSPROJ` routes via the extension
149    /// fallback, but `packages.Config` does **not** match the exact-name
150    /// entry for `packages.config`.
151    ///
152    /// # Arguments
153    ///
154    /// * `filename` - Manifest filename (e.g., "Cargo.toml", "package.json")
155    ///
156    /// # Returns
157    ///
158    /// * `Some(Arc<dyn Ecosystem>)` - Ecosystem handling this filename
159    /// * `None` - No ecosystem handles this filename
160    ///
161    /// # Examples
162    ///
163    /// ```no_run
164    /// use deps_core::EcosystemRegistry;
165    ///
166    /// let registry = EcosystemRegistry::new();
167    /// if let Some(ecosystem) = registry.get_for_filename("Cargo.toml") {
168    ///     println!("Cargo.toml handled by: {}", ecosystem.display_name());
169    /// }
170    /// ```
171    pub fn get_for_filename(&self, filename: &str) -> Option<Arc<dyn Ecosystem>> {
172        if let Some(id) = self.filename_map.get(filename) {
173            return self.get(*id);
174        }
175
176        // Avoid the rsplit_once/format! allocation below when no ecosystem has registered
177        // an extension (extension routing is only used by nuget today) — this runs on
178        // every exact-match miss, including every did_open/did_change via get_for_uri.
179        if self.extension_map.is_empty() {
180            return None;
181        }
182
183        let (_, extension) = filename.rsplit_once('.')?;
184        let lowercased = format!(".{}", extension.to_lowercase());
185        let id = self.extension_map.get(lowercased.as_str())?;
186        self.get(*id)
187    }
188
189    /// Get ecosystem from URI
190    ///
191    /// Extracts the filename from the URI path and looks up the ecosystem.
192    ///
193    /// # Arguments
194    ///
195    /// * `uri` - Document URI (file:///path/to/Cargo.toml)
196    ///
197    /// # Returns
198    ///
199    /// * `Some(Arc<dyn Ecosystem>)` - Ecosystem handling this file
200    /// * `None` - No ecosystem handles this file type or URI parsing failed
201    ///
202    /// # Examples
203    ///
204    /// ```no_run
205    /// use deps_core::EcosystemRegistry;
206    /// use tower_lsp_server::ls_types::Uri;
207    ///
208    /// let registry = EcosystemRegistry::new();
209    /// let uri = Uri::from_file_path("/home/user/project/Cargo.toml").unwrap();
210    ///
211    /// if let Some(ecosystem) = registry.get_for_uri(&uri) {
212    ///     println!("File handled by: {}", ecosystem.display_name());
213    /// }
214    /// ```
215    pub fn get_for_uri(&self, uri: &Uri) -> Option<Arc<dyn Ecosystem>> {
216        let path = uri.path().as_str();
217        let filename = path.rsplit('/').next()?;
218        self.get_for_filename(filename)
219    }

221    /// Get all registered ecosystem IDs
222    ///
223    /// Returns a vector of all ecosystem IDs currently registered.
224    /// This is useful for debugging and listing available ecosystems.
225    ///
226    /// # Returns
227    ///
228    /// Vector of ecosystem ID strings
229    ///
230    /// # Examples
231    ///
232    /// ```no_run
233    /// use deps_core::EcosystemRegistry;
234    ///
235    /// let registry = EcosystemRegistry::new();
236    /// // registry.register(cargo_ecosystem);
237    /// // registry.register(npm_ecosystem);
238    ///
239    /// for id in registry.ecosystem_ids() {
240    ///     println!("Registered ecosystem: {}", id);
241    /// }
242    /// ```
243    pub fn ecosystem_ids(&self) -> Vec<&'static str> {
244        self.ecosystems.iter().map(|e| *e.key()).collect()
245    }

247    /// Get ecosystem for a lock file name
248    ///
249    /// # Arguments
250    ///
251    /// * `filename` - Lock file name (e.g., "Cargo.lock", "package-lock.json")
252    ///
253    /// # Returns
254    ///
255    /// * `Some(Arc<dyn Ecosystem>)` - Ecosystem using this lock file
256    /// * `None` - No ecosystem uses this lock file
257    ///
258    /// # Examples
259    ///
260    /// ```no_run
261    /// use deps_core::EcosystemRegistry;
262    ///
263    /// let registry = EcosystemRegistry::new();
264    /// // registry.register(cargo_ecosystem);
265    ///
266    /// if let Some(ecosystem) = registry.get_for_lockfile("Cargo.lock") {
267    ///     println!("Cargo.lock handled by: {}", ecosystem.display_name());
268    /// }
269    /// ```
270    pub fn get_for_lockfile(&self, filename: &str) -> Option<Arc<dyn Ecosystem>> {
271        for entry in self.ecosystems.iter() {
272            let ecosystem = entry.value();
273            if ecosystem.lockfile_filenames().contains(&filename) {
274                return Some(Arc::clone(ecosystem));
275            }
276        }
277        None
278    }
279
280    /// Get all lock file patterns for file watching
281    ///
282    /// Returns glob patterns (e.g., "**/Cargo.lock") for all registered ecosystems.
283    ///
284    /// # Examples
285    ///
286    /// ```no_run
287    /// use deps_core::EcosystemRegistry;
288    ///
289    /// let registry = EcosystemRegistry::new();
290    /// // registry.register(cargo_ecosystem);
291    /// // registry.register(npm_ecosystem);
292    ///
293    /// let patterns = registry.all_lockfile_patterns();
294    /// for pattern in patterns {
295    ///     println!("Watching pattern: {}", pattern);
296    /// }
297    /// ```
298    pub fn all_lockfile_patterns(&self) -> Vec<String> {
299        let mut patterns = Vec::new();
300        for entry in self.ecosystems.iter() {
301            let ecosystem = entry.value();
302            for filename in ecosystem.lockfile_filenames() {
303                patterns.push(format!("**/{}", filename));
304            }
305        }
306        patterns
307    }
308}
