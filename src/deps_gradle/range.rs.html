1//! Gradle bracket-interval version range parsing and containment.
2//!
3//! Gradle's [rich version][spec] grammar shares its bracket-interval syntax
4//! (`[1.0,2.0)`, `[1.0]`, `[1.5,)`, `(,2.0]`) with Maven, but unlike Maven has no top-level
5//! comma union — a single interval is the whole requirement. Both grammars are handled by
6//! the shared single-interval parser in [`deps_maven::interval`]; the only divergence is
7//! Gradle's reversed-bracket exclusive notation (`]1.2,1.5]` for an exclusive lower bound,
8//! `[1.1,2.0[` for an exclusive upper bound), selected here via
9//! `deps_maven::interval::BracketStyle::AllowReversed`. Dynamic versions (`1.0+`) and
10//! `latest.*` selectors are handled directly in `formatter.rs`, not here.
11//!
12//! [spec]: https://docs.gradle.org/current/userguide/rich_versions.html
13
14use deps_maven::interval::{BracketStyle, VersionRange, contains, parse_interval};
15
16/// Parses a Gradle bracket-interval `requirement`, once.
17///
18/// Supports `[1.0,2.0)`, `[1.0]`, `[1.5,)`, `(,2.0]`, and Gradle's reversed-bracket exclusive
19/// notation (`]1.2,1.5]`, `[1.1,2.0[`). Returns `None` for an unparseable requirement
20/// (unbalanced/unrecognized delimiters, empty bounds, a nested bracket, an extra
21/// comma-separated component) rather than panicking.
22///
23/// Used by `GradleFormatter::compile_requirement` to parse the requirement once per
24/// dependency; the resulting `VersionRange` is then tested against each candidate version
25/// via `contains` with no re-parsing.
26pub(crate) fn parse_range(requirement: &str) -> Option<VersionRange> {
27    parse_interval(requirement.trim(), BracketStyle::AllowReversed)
28}
29
30/// Checks whether `version` satisfies a Gradle bracket-interval `requirement`.
31///
32/// Convenience wrapper around `parse_range` + [`contains`] for callers that don't need to
33/// test more than one candidate against the same requirement.
34pub fn satisfies(version: &str, requirement: &str) -> bool {
35    match parse_range(requirement) {
36        Some(range) => contains(version, &range),
37        None => false,
38    }
39}

41#[cfg(test)]
42mod tests {
43    use super::*;
44
45    #[test]
46    fn test_satisfies_bound_digit_count_normalization() {
47        // #182 fix propagated through the shared parser: missing trailing
48        // segments normalize as zero.
49        assert!(satisfies("1.5", "]1.2,1.5.0]"));
50        assert!(!satisfies("2.0.0", "[1.1,2.0["));
51    }
52
53    #[test]
54    fn test_satisfies_rejects_union_syntax() {
55        // Gradle's grammar has no top-level comma union — a Maven-style union string must
56        // be rejected outright, not silently parsed as one interval with a corrupted bound.
57        assert!(!satisfies("1.5", "[1.0,2.0),[3.0,4.0)"));
58        assert!(!satisfies("2.5", "[1.0,2.0),[3.0,4.0)"));
59        assert!(!satisfies("3.5", "[1.0,2.0),[3.0,4.0)"));
60        assert!(!satisfies("9.9", "[1.0,2.0),[3.0,4.0)"));
61    }
62
63    #[test]
64    fn test_satisfies_reversed_bracket_exclusive_lower() {
65        // `]1.2,1.5]`: exclusive lower bound, inclusive upper bound.
66        assert!(!satisfies("1.2", "]1.2,1.5]"));
67        assert!(satisfies("1.3", "]1.2,1.5]"));
68        assert!(satisfies("1.5", "]1.2,1.5]"));
69        assert!(!satisfies("1.6", "]1.2,1.5]"));
70    }
71
72    #[test]
73    fn test_satisfies_reversed_bracket_exclusive_upper() {
74        // `[1.1,2.0[`: inclusive lower bound, exclusive upper bound.
75        assert!(satisfies("1.1", "[1.1,2.0["));
76        assert!(satisfies("1.5", "[1.1,2.0["));
77        assert!(!satisfies("2.0", "[1.1,2.0["));
78        assert!(!satisfies("1.0", "[1.1,2.0["));
79    }
80
81    #[test]
82    fn test_satisfies_single_delimiter_does_not_panic() {
83        // #187: a single-character requirement must be rejected, not panic on a
84        // start > end slice (AllowReversed makes `[`/`]` valid on both sides).
85        assert!(!satisfies("1.0", "["));
86        assert!(!satisfies("1.0", "]"));
87    }
