1//! Best-effort license fetch from a Maven Central POM (issue #660, spec 010 plan §1
2//! tier 3).
3//!
4//! `deps_dev_system` excludes the Gradle ecosystem (unlike the Maven ecosystem crate,
5//! which shares the exact same `groupId:artifactId` coordinate format but is
6//! tier-2/deps.dev-covered), and `maven-metadata.xml` — the version-list endpoint
7//! [`deps_maven::MavenCentralRegistry`] already fetches — carries no license field at
8//! all. Only a specific version's POM file's `<licenses>` block does; live-verified
9//! 2026-09-08 against `com.squareup.okhttp3:okhttp:4.12.0`, whose POM carries
10//! `<licenses><license><name>The Apache Software License, Version 2.0</name>...`.
11//!
12//! Deliberately targets Maven Central only (`repo1.maven.org`), not the Google Maven /
13//! Gradle Plugin Portal fallbacks `MavenCentralRegistry`'s version lookup also tries —
14//! those two are edge cases for the small minority of Android/plugin-only coordinates,
15//! and this is a best-effort secondary signal (NFR-003 graceful degradation already
16//! covers a POM that 404s there).
17//!
18//! **Parent POM traversal (issue #692)**: a Maven multi-module project commonly declares
19//! `<licenses>` only on a shared parent POM, leaving each module's own (leaf) POM to
20//! reference it via `<parent>` — Guava's own leaf POM (`guava-32.0.1-jre.pom`) has no
21//! `<licenses>` element at all; the license is declared only on `guava-parent`'s POM.
22//! [`fetch_license_from`] follows a leaf POM's `<parent>` coordinate when its own
23//! `<licenses>` block is empty, bounded by [`MAX_POM_FETCHES`] so a pathological (or
24//! adversarial) parent chain can't turn one hover request into an unbounded fetch chain.
25

28use quick_xml::Reader;
29use quick_xml::events::Event;
30use std::sync::Arc;

32/// Maximum number of `<licenses><license><name>` entries [`parse_pom_licenses`] retains
33/// from a single POM (issue #690; raised from an initial 8 to 64 per impl-critic S2).
34/// Generously above any real-world POM's actual license count (dual-licensing is the
35/// practical ceiling) and, more importantly, above what
36/// `deps_core::licenses::evaluate`'s `license_policy` check needs: that function scans
37/// every entry in the returned `Vec` with no cap of its own, so an 8-entry retention cap
38/// could have silently hidden a policy violation carried by a real 9th+ license entry.
39/// `deps-core::lsp_helpers::hover`'s separate `MAX_LICENSE_ENTRIES_RENDERED` (8) only
40/// truncates hover *rendering* — it says nothing about how many entries diagnostics
41/// evaluate, so this constant must not be tied to it. CPU/memory cost on a hostile
42/// response is bounded independently by [`MAX_POM_LICENSE_BYTES_SCANNED`], not by this
43/// cap. Kept as a Gradle-crate-local constant rather than a shared `deps-core` export
44/// since no other ecosystem crate parses this shape yet (see this module's `pom_url` doc
45/// for the same "not worth a cross-crate helper yet" reasoning).
46const MAX_POM_LICENSE_ENTRIES: usize = 64;
47
48/// Hard cap, in bytes of `reader.buffer_position()`, on how far into a POM
49/// [`parse_pom_licenses`] will read (issue #690, impl-critic S1 — through three rounds of
50/// counterexamples). A cap on *retained* entries alone doesn't bound CPU, and every
51/// attempt to budget by counting a specific event shape instead (a `<name>` text node, a
52/// `<name>` `Event::Start`) turned out to be shape-dependent and bypassable: a truly empty
53/// `<name></name>` emits no `Event::Text`; a self-closing `<name/>` emits `Event::Empty`,
54/// never `Event::Start`; and a `<license>` with no `<name>` child at all — or a flood of
55/// unrelated junk elements — emits neither. Each of those consumed **zero** budget under a
56/// shape-keyed counter while quick-xml still walked the reader to EOF (measured: a 32 MiB
57/// flood of bare `<a/>` elements took ~117 ms with zero counted nodes). A byte-position
58/// bound sidesteps this entirely — it advances on *every* event regardless of what it is,
59/// so it cannot be starved by omitting one element shape. A real POM's `<licenses>` block
60/// (and everything preceding it) is realistically well under this bound; the reader simply
61/// stops handing back events once it's exhausted, degrading gracefully to whatever was
62/// found so far — acceptable for this best-effort secondary signal (see module docs).
63const MAX_POM_LICENSE_BYTES_SCANNED: usize = 1024 * 1024;

65/// Maven Central's repository root. Mirrors `deps-maven`'s own (private)
66/// `MAVEN_REPO_BASE` — kept as an independent constant rather than a shared one so this
67/// crate does not need a new public export from `deps-maven` for a single string.
68const MAVEN_REPO_BASE: &str = "https://repo1.maven.org/maven2";

70/// Builds the Maven Central POM URL for `coordinate` (`"group:artifact"`) at
71/// `version`, or `None` if `coordinate` isn't in that shape or any path segment (each
72/// dot-separated `group` component, `artifact`, or `version`) fails
73/// [`is_safe_maven_coordinate_segment`].
74///
75/// Security P3 (impl-critic): a bare exact-match `.`/`..` reject is not enough here —
76/// Gradle's own coordinate/version parsers allow characters (notably `/`) this
77/// function used to interpolate raw, so a `version` (or `group`/`artifact` component)
78/// of e.g. `"../../../etc/passwd"` was never exactly `".."` and passed the old check
79/// unfiltered, then split the URL path into extra traversal segments once
80/// interpolated. Reuses [`is_safe_maven_coordinate_segment`] — the same allowlist
81/// `deps-maven`/`deps-gradle`'s own coordinate formatters already validate this exact
82/// `groupId`/`artifactId` shape against (`deps-maven/src/registry.rs:589-614`,
83/// `deps-gradle/src/formatter.rs:171-174`) — rather than switching to percent-encoding
84/// alone: this codebase has already hit the case where percent-encoding does *not*
85/// neutralize a traversal segment (`deps-dart::registry::package_metadata_url`'s doc,
86/// #349) — a segment of exactly `.`/`..` survives unreserved-character percent-encoding
87/// unchanged, since `.` is an RFC 3986 unreserved character no encoder in this
88/// codebase's `urlencoding` dependency escapes. An allowlist that fails closed on `/`
89/// and any other non-`[A-Za-z0-9._-]` character closes both the embedded-slash
90/// injection *and* the exact-dot-segment case in one already-audited helper, so no
91/// separate encoding step is needed.
92///
93/// The `groupId:artifactId` -> URL-path construction itself is shared with
94/// `deps-maven::registry::metadata_urls` via [`deps_core::maven_coordinate_path`] (#702) —
95/// this function only adds the `version`/`.pom` filename suffix on top.
96fn pom_url(base: &str, coordinate: &str, version: &str) -> Option<String> {
97    let (group, artifact) = coordinate.split_once(':')?;
98    if !is_safe_maven_coordinate_segment(version) {
99        return None;
100    }
101    let coordinate_path = deps_core::maven_coordinate_path(group, artifact)?;
102    Some(format!(
103        "{base}/{coordinate_path}/{version}/{artifact}-{version}.pom"
104    ))
105}

107/// Bounds how many POM fetches [`fetch_license_from`] performs for one license lookup —
108/// the requested (leaf) coordinate itself, plus up to `MAX_POM_FETCHES - 1` `<parent>`
109/// hops when each POM in the chain declares no `<licenses>` block of its own (issue
110/// #692). A real Maven parent chain is rarely more than one or two levels deep (a leaf
111/// module -> its immediate parent POM), so this leaves headroom for that common shape
112/// while still capping a pathological or adversarial chain at a small, fixed number of
113/// network round trips.
114const MAX_POM_FETCHES: u8 = 3;
115
116/// Fetches `coordinate`'s (`"group:artifact"`) license at `version` from Maven
117/// Central.
118///
119/// Returns an empty `Vec` (never an error) when `coordinate` isn't in the expected
120/// `"group:artifact"` shape, the fetch fails (network error, 404 — a large fraction of
121/// Gradle plugin/Android coordinates live on Google Maven or the Gradle Plugin Portal
122/// instead, not Maven Central), or neither the POM nor any `<parent>` POM within
123/// [`MAX_POM_FETCHES`] declares a `<licenses>` block — graceful degradation (NFR-003),
124/// since this is a best-effort secondary signal, not core version data.
125pub(crate) async fn fetch_license(
126    cache: &Arc<HttpCache>,
127    coordinate: &str,
128    version: &str,
129) -> Vec<String> {
130    fetch_license_from(cache, MAVEN_REPO_BASE, coordinate, version).await
131}

133/// [`fetch_license`]'s implementation, parameterized over the repository base URL so
134/// tests can point it at a mockito server instead of live Maven Central — mirrors
135/// `deps-dart::PubDevRegistry::with_base`/`deps-swift`'s equivalent test-only base
136/// override (tester gap: unlike Swift/Dart/Deno's fetch layer, Gradle's had no
137/// HTTP-mocked coverage at all, only the malformed-coordinate short-circuit below).
138///
139/// Follows `<parent>` POM coordinates (issue #692) when a fetched POM's own
140/// `<licenses>` is empty, up to [`MAX_POM_FETCHES`] total fetches (the leaf plus its
141/// parent chain). Each hop re-validates its coordinate/version through [`pom_url`] —
142/// the same allowlist the leaf fetch uses — so a malicious `<parent>` value can no more
143/// escape Maven Central's URL space than a malicious leaf coordinate could.
144///
145/// Deliberately the sole instrumented function on this path (issue #823): the actual
146/// bounded fetch loop lives in [`fetch_license_hops`], a plain, uninstrumented helper
147/// that only returns data, so `hops` is recorded exactly once here rather than at each
148/// of that loop's several exit paths — see that function's doc for why.
149#[tracing::instrument(
150    skip_all,
151    fields(package = ?coordinate, version = ?version, hops = tracing::field::Empty),
152    level = "debug"
153)]
154async fn fetch_license_from(
155    cache: &Arc<HttpCache>,
156    base: &str,
157    coordinate: &str,
158    version: &str,
159) -> Vec<String> {
160    let (licenses, hops) = fetch_license_hops(cache, base, coordinate, version).await;
161    tracing::Span::current().record("hops", hops);
162    licenses
163}
164
165/// Runs [`fetch_license_from`]'s bounded POM fetch loop and returns the discovered
166/// licenses together with the number of POM fetches actually performed (issue #823),
167/// i.e. how many times [`HttpCache::get_cached`] was actually called — *not* how many
168/// loop iterations ran, since the [`pom_url`] guard can reject a coordinate/version
169/// before any network call is made (the documented malformed-coordinate short-circuit,
170/// both for the leaf and for a malformed `<parent>` hop) and that iteration must not be
171/// counted as a fetch.
172///
173/// Kept as a separate, non-instrumented function (impl-critic M2 on issue #823) so
174/// [`fetch_license_from`] can record the `hops` span field at its single return point
175/// instead of at every one of this loop's exit paths, where a future added `return`
176/// could silently omit it — the same silent-omission failure mode issue #819 removes
177/// from the completion-context dispatch, reintroduced one level down if each exit had
178/// to remember its own `Span::current().record` call.
179async fn fetch_license_hops(
180    cache: &Arc<HttpCache>,
181    base: &str,
182    coordinate: &str,
183    version: &str,
184) -> (Vec<String>, u64) {
185    let mut current_coordinate = coordinate.to_string();
186    let mut current_version = version.to_string();
187
188    for hop in 0..MAX_POM_FETCHES {
189        let Some(url) = pom_url(base, &current_coordinate, &current_version) else {
190            // No network call happens on this path, so `hop` (not `hop + 1`) is the
191            // count of fetches actually performed so far.
192            return (Vec::new(), u64::from(hop));
193        };
194        let pom = match cache.get_cached(&url).await {
195            Ok(data) => parse_pom(&data),
196            Err(e) => {
197                // Logs both the requested (leaf) coordinate and the current hop's —
198                // code-review nit: a parent-hop failure logged only the reassigned
199                // parent coordinate, making it hard to correlate back to the dependency
200                // the manifest/hover actually shows (e.g. `guava-parent` instead of
201                // `guava` for a failed hop past a successfully-fetched leaf POM).
202                tracing::debug!(
203                    requested_coordinate = coordinate,
204                    requested_version = version,
205                    coordinate = current_coordinate,
206                    version = current_version,
207                    error = %e,
208                    "gradle license pom fetch failed"
209                );
210                return (Vec::new(), u64::from(hop) + 1);
211            }
212        };
213        if !pom.licenses.is_empty() {
214            return (pom.licenses, u64::from(hop) + 1);
215        }
216        match pom.parent {
217            Some((parent_coordinate, parent_version)) => {
218                current_coordinate = parent_coordinate;
219                current_version = parent_version;
220            }
221            None => return (Vec::new(), u64::from(hop) + 1),
222        }
223    }
224
225    (Vec::new(), u64::from(MAX_POM_FETCHES))
226}

228/// One parsed Maven POM XML document's license-relevant content: its own declared
229/// licenses, and — when present — the `<parent>` coordinate/version
230/// [`fetch_license_from`] follows next if [`Self::licenses`] is empty (issue #692).
231struct PomInfo {
232    licenses: Vec<String>,
233    /// `("group:artifact", version)` from `<parent><groupId>`/`<artifactId>`/`<version>`,
234    /// present only when the POM declares all three.
235    parent: Option<(String, String)>,
236}
237
238/// Extracts every `<licenses><license><name>` text value, and the `<parent>` coordinate
239/// if any, from a POM XML document.
240///
241/// A dependency can declare more than one license (dual-licensed artifacts, e.g. EPL
242/// and GPL-with-classpath-exception), so this collects all of them — consistent with
243/// every other ecosystem's `license: Vec<String>` shape (spec 010 plan §1 "License
244/// shape" decision). Non-UTF-8 or malformed XML degrades to an empty `Vec` and no
245/// parent rather than an error, matching this module's overall graceful-degradation
246/// contract.
247fn parse_pom(data: &[u8]) -> PomInfo {
248    let Ok(content) = std::str::from_utf8(data) else {
249        return PomInfo {
250            licenses: Vec::new(),
251            parent: None,
252        };
253    };
254
255    let mut reader = Reader::from_str(content);
256    reader.config_mut().trim_text(true);
257
258    let mut licenses = Vec::new();
259    let mut in_licenses = false;
260    let mut in_license = false;
261    let mut in_name = false;
262
263    // Scoped to `<parent>...</parent>` (via `in_parent`) so the project's own top-level
264    // `<groupId>`/`<artifactId>`/`<version>` elements — which every POM also has,
265    // outside `<parent>` — are never mistaken for the parent coordinate.
266    let mut in_parent = false;
267    let mut in_parent_group_id = false;
268    let mut in_parent_artifact_id = false;
269    let mut in_parent_version = false;
270    let mut parent_group_id = String::new();
271    let mut parent_artifact_id = String::new();
272    let mut parent_version = String::new();
273
274    loop {
275        // Shape-independent budget (impl-critic S1, through three rounds of
276        // counterexamples that each defeated a shape-keyed counter): advances on every
277        // event regardless of what it is, so it cannot be starved by a document that
278        // simply omits the element type a narrower counter was watching for.
279        if exhausted_with(
280            licenses.len(),
281            reader.buffer_position(),
282            MAX_POM_LICENSE_ENTRIES,
283            MAX_POM_LICENSE_BYTES_SCANNED,
284        ) {
285            break;
286        }
287        match reader.read_event() {
288            Ok(Event::Start(ref e)) => match e.local_name().as_ref() {
289                "licenses" => in_licenses = true,
290                "license" if in_licenses => in_license = true,
291                "name" if in_license => in_name = true,
292                "parent" => in_parent = true,
293                "groupId" if in_parent => in_parent_group_id = true,
294                "artifactId" if in_parent => in_parent_artifact_id = true,
295                "version" if in_parent => in_parent_version = true,
296                _ => {}
297            },
298            Ok(Event::Text(ref e)) if in_name => {
299                let raw = e.trim();
300                if raw.len() <= MAX_POM_LICENSE_NAME_RAW_CHARS {
301                    let raw = raw.to_string();
302                    let text = quick_xml::escape::unescape(&raw)
303                        .map(|c| c.into_owned())
304                        .unwrap_or(raw);
305                    if !text.is_empty() {
306                        licenses.push(text);
307                    }
308                }
309            }
310            Ok(Event::Text(ref e)) if in_parent_group_id => {
311                parent_group_id = e.trim().to_string();
312            }
313            Ok(Event::Text(ref e)) if in_parent_artifact_id => {
314                parent_artifact_id = e.trim().to_string();
315            }
316            Ok(Event::Text(ref e)) if in_parent_version => {
317                parent_version = e.trim().to_string();
318            }
319            Ok(Event::End(ref e)) => match e.local_name().as_ref() {
320                "licenses" => in_licenses = false,
321                "license" => in_license = false,
322                "name" => in_name = false,
323                "parent" => in_parent = false,
324                "groupId" => in_parent_group_id = false,
325                "artifactId" => in_parent_artifact_id = false,
326                "version" => in_parent_version = false,
327                _ => {}
328            },
329            Ok(Event::Eof) | Err(_) => break,
330            _ => {}
331        }
332    }
333
334    let parent =
335        if parent_group_id.is_empty() || parent_artifact_id.is_empty() || parent_version.is_empty()
336        {
337            None
338        } else {
339            Some((
340                format!("{parent_group_id}:{parent_artifact_id}"),
341                parent_version,
342            ))
343        };
344
345    PomInfo { licenses, parent }
346}
