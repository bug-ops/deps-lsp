14/// Unresolved Gradle variable reference (`$var`, `${var}`), or an explicit empty
15/// version-catalog entry (`[versions] foo = ""`) that a `version.ref` could point at.
16fn is_unresolved(requirement: &str) -> bool {
17    requirement.is_empty() || requirement.contains('$')
18}
19
20/// Strips Gradle's rich-version strict/preferred shorthand
21/// (`{strictlyVersion}!!{preferredVersion}`, e.g. the degenerate suffix form `1.2.3!!`
22/// or the full infix form `[1.7,1.8[!!1.7.25`), returning only the `strictlyVersion`
23/// half. Per Gradle's rich-version semantics, `strictly` is the hard constraint and
24/// `preferred` is only a soft conflict-resolution tiebreak among versions that already
25/// satisfy it — a version inside the strict range/pin satisfies the requirement
26/// regardless of whether it matches the preferred pointer, so every "does this version
27/// satisfy the requirement" comparison in [`gradle_version_matches`] and
28/// [`GradleFormatter::compile_requirement`] must operate on the `strictlyVersion`
29/// constraint alone, never the preference. `requirement` is `trim_end`ed before the
30/// marker check: `libs.versions.toml` catalog values reach here un-trimmed
31/// (`catalog::extract_version` returns the raw string verbatim), unlike the DSL
32/// parsers, which already trim their capture.
33fn strip_strict_marker(requirement: &str) -> &str {
34    requirement
35        .trim_end()
36        .split_once("!!")
37        .map_or(requirement, |(strictly, _)| strictly.trim_end())
38}
39
40/// A `-SNAPSHOT` pin (e.g. `7.0.0-SNAPSHOT`) resolves against Maven Central's snapshot
41/// repository, which `deps-maven`'s `MavenCentralRegistry` (reused for Gradle resolution)
42/// never queries — release-repo `maven-metadata.xml` never lists snapshot versions, so
43/// `available` can never contain one. Treated as always satisfied, like an unresolved
44/// variable or `latest.*`.
45fn is_snapshot(requirement: &str) -> bool {
46    requirement.ends_with("-SNAPSHOT")
47}

49/// Decides whether `version` satisfies a Gradle `requirement` — shared by
50/// `version_satisfies_requirement` and [`GradleFormatter::compile_requirement`]'s matcher,
51/// since Gradle has no separate "loose" vs. "precise" comparator to distinguish (mirrors
52/// `deps-maven`'s formatter, which shares the same shape for the same reason).
53///
54/// #249 review (M4, root cause of S1): this function's branch order is a separate copy from
55/// `compile_requirement`'s below — the malformed-range guard that function adds ahead of its
56/// own copy of this order has no equivalent here (this function has none; a malformed range
57/// simply falls through to `crate::range::satisfies`'s fail-closed `false`, which is correct
58/// for the "loose satisfies" question this function answers). Reordering the branches here
59/// must be checked against `compile_requirement`'s branch order and guard placement too.
60fn gradle_version_matches(version: &str, requirement: &str) -> bool {
61    // Checked on the raw string *before* stripping the `!!` marker: an unresolved
62    // Gradle variable reference can appear in the `strictlyVersion` half (e.g.
63    // `${r}!!1.7.25`), and stripping first would discard the `$` along with it,
64    // silently treating an unresolved requirement as a concrete one to compare.
65    //
66    // Deliberate, harmless asymmetry with `compile_requirement` below, which has no
67    // equivalent raw-string check: `strip_strict_marker` already returns the
68    // `strictlyVersion` half with any `$` intact, so the post-strip `is_unresolved`
69    // check a few lines down covers the same case on its own — this raw-string check
70    // is pure defense-in-depth for a caller of this loose matcher standalone. Its
71    // only observable effect is over-permissive, never under-permissive: a
72    // `preferredVersion` half containing an unresolved variable (e.g.
73    // `[1.7,1.8[!!${r}`) makes this function report "satisfied" for every version,
74    // including ones outside the strict range, whereas `compile_requirement`'s
75    // matcher (no raw-string check, and never reached with such a requirement in
76    // production since `requirement_is_unsatisfiable` gates on the raw-string
77    // `requirement_is_unresolved` first) would correctly reject an out-of-range
78    // version. Never produces a false "outdated" badge or a spurious edit, so this
79    // is not a bug — just don't "fix" the two functions back into lockstep by
80    // deleting this check without checking C3's post-strip coverage still holds.
81    if is_unresolved(requirement) {
82        return true;
83    }
84    let requirement = strip_strict_marker(requirement);
85    // Unresolved Gradle variable reference (`$var`/`${var}`), or an empty version-catalog
86    // entry (`[versions] foo = ""`) — skip comparison. Re-checked post-strip for the
87    // degenerate case where the `strictlyVersion` half itself is empty (a malformed
88    // bare `"!!"` requirement), which the raw check above does not catch since the raw
89    // string is `"!!"`, not empty.
90    if is_unresolved(requirement) {
91        return true;
92    }
93    if requirement == "latest" || requirement.starts_with("latest.") {
94        return true;
95    }
96    if is_snapshot(requirement) {
97        return true;
98    }
99    if let Some(prefix) = requirement.strip_suffix('+') {
100        return version == prefix.trim_end_matches('.') || version.starts_with(prefix);
101    }
102    // `]` is included alongside `[`/`(` because Gradle's reversed-bracket exclusive
103    // notation (`]1.2,1.5]`) is a leading delimiter in its own right, not just a
104    // trailing one.
105    if requirement.starts_with(['[', '(', ']']) {
106        return crate::range::satisfies(version, requirement);
107    }
108    version == requirement
109}

111/// Precise Gradle version/range matcher, compiled once per dependency by
112/// [`GradleFormatter::compile_requirement`] — a bracket-interval range is parsed once into a
113/// [`deps_maven::interval::VersionRange`] here rather than being re-parsed for every
114/// candidate version scanned. `requirement_is_unsatisfiable` already gates on
115/// `requirement_is_unresolved` before calling `compile_requirement`, so the unresolved and
116/// `latest.*` short-circuits are unreachable from that caller in practice; they stay so this
117/// matcher is correct if used standalone.
118enum GradleMatcher {
119    /// Unresolved `$var`/`${var}`, `latest`/`latest.*`, or a `-SNAPSHOT` pin.
120    AlwaysSatisfied,
121    /// A dynamic `1.0.+` prefix — the text before the trailing `+`.
122    DynamicPrefix(String),
123    /// A bracket-interval range, pre-parsed by [`crate::range::parse_range`].
124    Range(deps_maven::interval::VersionRange),
125    /// A bare exact version.
126    Exact(String),
127}
128
129impl RequirementMatcher for GradleMatcher {
130    fn matches(&self, version: &ConcreteVersion) -> Option<bool> {
131        let version = version.as_str();
132        Some(match self {
133            Self::AlwaysSatisfied => true,
134            Self::DynamicPrefix(prefix) => {
135                version == prefix.trim_end_matches('.') || version.starts_with(prefix.as_str())
136            }
137            Self::Range(range) => deps_maven::interval::contains(version, range),
138            Self::Exact(target) => version == target,
139        })
140    }
141}

143impl PackageNaming for GradleFormatter {
144    /// Validates a Gradle coordinate's `group:artifact` shape and character set.
145    ///
146    /// Gradle resolves through `deps_maven::MavenCentralRegistry` and shares Maven's
147    /// `groupId:artifactId` coordinate shape (see `deps-gradle/src/ecosystem.rs`), so this
148    /// mirrors [`deps_maven`]'s `MavenFormatter::validate_package_name` exactly, reusing
149    /// [`is_safe_maven_coordinate_segment`] rather than duplicating it — letting the
150    /// "Invalid package name" diagnostic surface the accurate reason instead of the
151    /// generic "Unknown package" a registry-side rejection produces (#375).
152    ///
153    /// Unlike Maven's `${property}`-specific `is_unresolved`, this uses Gradle's own
154    /// `is_unresolved`, which also short-circuits on an unresolved `$var`/`${var}`
155    /// reference or Gradle-catalog-alias placeholder — valid Gradle syntax, not a
156    /// malformed coordinate.
157    ///
158    /// # Errors
159    ///
160    /// Returns [`InvalidPackageName`] if `name` has no `:` separator, or if either the
161    /// `group` or `artifact` segment fails [`is_safe_maven_coordinate_segment`] — but
162    /// never when `name` is unresolved per `is_unresolved`, which is accepted instead.
163    fn validate_package_name(&self, name: &str) -> Result<(), InvalidPackageName> {
164        if is_unresolved(name) {
165            return Ok(());
166        }
167        let Some((group_id, artifact_id)) = name.split_once(':') else {
168            return Err(InvalidPackageName::new(
169                "coordinate must be in 'group:artifact' form",
170            ));
171        };
172        if !is_safe_maven_coordinate_segment(group_id) {
173            return Err(InvalidPackageName::new("group contains invalid characters"));
174        }
175        if !is_safe_maven_coordinate_segment(artifact_id) {
176            return Err(InvalidPackageName::new(
177                "artifact contains invalid characters",
178            ));
179        }
180        Ok(())
181    }
182}

184impl PackageRendering for GradleFormatter {
185    fn format_version_for_text_edit(&self, version: &ConcreteVersion) -> String {
186        let version = version.as_str();
187        version.to_string()
188    }
189
190    /// Preserves Gradle's rich-version strict/preferred shorthand
191    /// (`{strictlyVersion}!!{preferredVersion}`) when `current` carries it — a bare
192    /// `format_version_for_text_edit` replacement would otherwise silently downgrade
193    /// a strict constraint to a normal one.
194    ///
195    /// Only the degenerate suffix form (`1.2.3!!`, no `preferredVersion`) is
196    /// rewritten — to `{version}!!` — since the strict pin itself is what "update
197    /// version" means to bump there, and there is nothing else in the requirement to
198    /// preserve. The full infix form (`[1.7,1.8[!!1.7.25`) is left unchanged rather
199    /// than rewriting the `preferredVersion` half: since Gradle's strict constraint
200    /// always wins conflict resolution, bumping the preference to a version outside
201    /// the hand-written strict range (a likely outcome for "update to latest") would
202    /// silently write a no-op edit that *looks* like an update but changes nothing —
203    /// worse than the original silently-dropped-marker bug, since the manifest now
204    /// reads as though it were updated. There is no single rewrite that is safe in
205    /// general without inspecting the strict range's bounds, which is out of scope
206    /// here. `current` is `trim`med before the marker check for the same
207    /// un-trimmed-catalog-value reason as `strip_strict_marker` above. A no-op
208    /// return here is safely excluded from `deps-core`'s `collect_update_all_edits`
209    /// ("Update N outdated dependencies" lens) by its own no-op guard.
210    fn format_version_replacing(&self, version: &ConcreteVersion, current: &str) -> String {
211        let version = version.as_str();
212        let trimmed = current.trim();
213        match trimmed.split_once("!!") {
214            Some((_, "")) => format!("{version}!!"),
215            Some(_) => trimmed.to_string(),
216            None => self.format_version_for_text_edit(&ConcreteVersion::new(version)),
217        }
218    }
219
220    fn package_url(&self, name: &PackageName) -> String {
221        deps_maven::registry::package_url(name.as_str())
222    }
223}

225impl RequirementResolution for GradleFormatter {
226    fn version_satisfies_requirement(&self, version: &ConcreteVersion, requirement: &str) -> bool {
227        let version = version.as_str();
228        gradle_version_matches(version, requirement)
229    }
230
231    fn requirement_is_unresolved(&self, requirement: &VersionReq) -> bool {
232        is_unresolved(requirement.as_str())
233    }
234
235    /// Uses [`compile_requirement_unless`] (see that function and
236    /// [`deps_core::lsp_helpers::RequirementResolution::compile_requirement`] for the shared "undecidable" contract).
237    ///
238    /// The undecidable predicate rejects a malformed range (leading `[`/`(`/`]` but
239    /// `crate::range::parse_range` fails) — checked unconditionally, first, before any
240    /// other branch: without this guard ahead of the `AlwaysSatisfied`/dynamic-prefix
241    /// short-circuits below, a malformed bracket range that also happens to end in `+`
242    /// (e.g. `"[1.0,2.0]+"`) would be misclassified as a dynamic prefix — which decides
243    /// `Some(false)` for every real candidate — instead of correctly suppressing the check.
244    ///
245    /// #249 review (M4): this is a separate branch-order copy from `gradle_version_matches`
246    /// above — see the note on that function before reordering either one.
247    fn compile_requirement(&self, requirement: &VersionReq) -> Option<Box<dyn RequirementMatcher>> {
248        // `!!` is Gradle's rich-version strict/preferred shorthand (see
249        // `gradle_version_matches`/`strip_strict_marker`) — stripped once here, first, so
250        // every branch below (the malformed-range guard, dynamic-prefix, range, exact)
251        // operates on the `strictlyVersion` spelling underneath without needing its own
252        // separate strip. Unlike `gradle_version_matches` (re-derives everything from the raw
253        // string on every call), this matcher is pre-parsed once, so the stripped spelling must
254        // be what actually gets stored in the `GradleMatcher` variant — storing the unstripped
255        // string would make e.g. `Exact` compare against a target that includes `"!!"`.
256        let requirement = strip_strict_marker(requirement.as_str());
257        compile_requirement_unless(
258            requirement,
259            |r| r.starts_with(['[', '(', ']']) && crate::range::parse_range(r).is_none(),
260            |r| {
261                if is_unresolved(&r) || r == "latest" || r.starts_with("latest.") {
262                    return GradleMatcher::AlwaysSatisfied;
263                }
264                if is_snapshot(&r) {
265                    return GradleMatcher::AlwaysSatisfied;
266                }
267                if let Some(prefix) = r.strip_suffix('+') {
268                    return GradleMatcher::DynamicPrefix(prefix.to_string());
269                }
270                // `]` is included alongside `[`/`(` because Gradle's reversed-bracket
271                // exclusive notation (`]1.2,1.5]`) is a leading delimiter in its own
272                // right, not just a trailing one. The undecidable guard above already
273                // ensures `parse_range` succeeds here.
274                if r.starts_with(['[', '(', ']'])
275                    && let Some(range) = crate::range::parse_range(&r)
276                {
277                    return GradleMatcher::Range(range);
278                }
279                GradleMatcher::Exact(r)
280            },
281        )
282    }
283}
