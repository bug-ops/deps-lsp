20/// Configuration words that follow neither the suffix convention
21/// ([`CONFIGURATION_SUFFIXES`]) nor the prefix convention
22/// ([`CONFIGURATION_PREFIXES`]) and must be matched literally.
23///
24/// Includes the pre-Gradle-7 legacy configurations `compile`, `testCompile`,
25/// and `provided`: Gradle still parses them regardless of which DSL a build
26/// script uses, so a `build.gradle.kts` migrated from an old `build.gradle`
27/// (or targeting a project not yet updated) can legitimately contain them.
28/// Restricting recognition to one DSL would be silent drift, not an
29/// intentional scoping decision. `classpath` has no variant form at all.
30const CONFIGURATION_LITERALS: &[&str] = &["classpath", "compile", "testCompile", "provided"];
31
32/// `(bare, suffix)` pairs for configuration base words that Gradle core
33/// plugins (`java`, `java-library`) and common first-party plugins (Android
34/// Gradle Plugin, `java-test-fixtures`) combine with an arbitrary
35/// variant/source-set prefix, e.g. `debugImplementation`,
36/// `androidTestImplementation`, `testFixturesImplementation` — and
37/// `compileOnlyApi`, matched by the `Api` suffix.
38///
39/// `bare` is the word used with no prefix (`implementation`); `suffix` is the
40/// same word capitalized, as it appears after a prefix. A literal whitelist
41/// can't keep up with plugin-registered configurations following this
42/// convention (arbitrary build types/flavors in Android, custom source
43/// sets), so membership is decided by suffix instead — safe because Gradle
44/// itself treats any name ending in one of these words as that kind of
45/// configuration.
46const CONFIGURATION_SUFFIXES: &[(&str, &str)] = &[
47    ("implementation", "Implementation"),
48    ("api", "Api"),
49    ("compileOnly", "CompileOnly"),
50    ("runtimeOnly", "RuntimeOnly"),
51    ("annotationProcessor", "AnnotationProcessor"),
52];
53
54/// Kotlin annotation-processing plugin configurations (`kapt`, KSP's `ksp`)
55/// follow a *prefix* convention instead: bare for the main source set, or
56/// `<word><Variant>` for others — e.g. `kaptTest`, `kaptAndroidTest`,
57/// `kspDebug`, `kspCommonMainMetadata`. [`CONFIGURATION_SUFFIXES`] can't
58/// express this since the variant comes after the word, not before it.
59const CONFIGURATION_PREFIXES: &[&str] = &["kapt", "ksp"];
60
61/// Returns whether `config` is a recognized Gradle dependency configuration
62/// word: a legacy literal ([`CONFIGURATION_LITERALS`]), a name ending in one
63/// of [`CONFIGURATION_SUFFIXES`]'s base words, or a name starting with one of
64/// [`CONFIGURATION_PREFIXES`]'s words followed by a capitalized variant.
65pub(crate) fn is_dependency_configuration(config: &str) -> bool {
66    CONFIGURATION_LITERALS.contains(&config)
67        || CONFIGURATION_SUFFIXES
68            .iter()
69            .any(|(bare, suffix)| config == *bare || config.ends_with(suffix))
70        || CONFIGURATION_PREFIXES.iter().any(|prefix| {
71            config == *prefix
72                || config
73                    .strip_prefix(prefix)
74                    .is_some_and(|rest| rest.starts_with(|c: char| c.is_ascii_uppercase()))
75        })
76}

78/// Returns whether `trimmed` (a line with surrounding whitespace stripped)
79/// opens a Gradle `dependencies { }` block.
80///
81/// Requires the brace to follow the `dependencies` keyword (any amount of
82/// whitespace, including none, in between) so that unrelated blocks whose
83/// name merely starts with the word — e.g. Android's `dependenciesInfo { }`
84/// — aren't mistaken for it.
85pub(crate) fn opens_dependencies_block(trimmed: &str) -> bool {
86    trimmed
87        .strip_prefix("dependencies")
88        .is_some_and(|rest| rest.trim_start().starts_with('{'))
89}

91/// Builds a [`GradleDependency`] from a regex match's captures.
92///
93/// All Groovy and Kotlin DSL regex variants share identical capture group
94/// indices — 1: configuration, 2: group id, 3: artifact id, 4: version (when
95/// present) — so `has_version` alone distinguishes the two capture shapes.
96pub(crate) fn build_dependency(
97    caps: &Captures<'_>,
98    line: &str,
99    line_idx: u32,
100    has_version: bool,
101    config: &str,
102) -> GradleDependency {
103    debug_assert_eq!(
104        caps.len(),
105        if has_version { 5 } else { 4 },
106        "regex capture count must match has_version so group 4 (version) is only read when present"
107    );
108
109    let match_start = caps.get(0).map_or(0, |m| m.start());
110    let group_id = caps.get(2).map_or("", |m| m.as_str()).to_string();
111    let artifact_id = caps.get(3).map_or("", |m| m.as_str()).to_string();
112    let name = format!("{group_id}:{artifact_id}");
113    let name_range = find_name_range(line, line_idx, match_start, &group_id, &artifact_id);
114
115    let (version_req, version_range) = if has_version {
116        let version = caps.get(4).map_or("", |m| m.as_str()).trim().to_string();
117        let version_range = find_version_range(line, line_idx, match_start, &version);
118        (Some(version.into()), Some(version_range))
119    } else {
120        (None, None)
121    };
122
123    GradleDependency {
124        group_id,
125        artifact_id,
126        name: name.into(),
127        name_range,
128        version_req,
129        version_range,
130        configuration: config.to_string(),
131    }
132}

140/// Resolves `$var` and `${var}` references in dependency versions using the given properties map.
141///
142/// If a version is a variable reference and the variable is found in `properties`,
143/// the version is replaced with the resolved value. The version_range is kept as-is
144/// (pointing to the variable reference in source).
145pub fn resolve_variables(deps: &mut [GradleDependency], properties: &HashMap<String, String>) {
146    for dep in deps.iter_mut() {
147        if let Some(ref ver) = dep.version_req
148            && let Some(resolved) = resolve_variable_ref(ver.as_str(), properties)
149        {
150            dep.version_req = Some(resolved.into());
151        }
152    }
153}
154
155/// Returns the resolved value if `value` is a `$name` or `${name}` reference. Returns `None` otherwise.
156fn resolve_variable_ref(value: &str, properties: &HashMap<String, String>) -> Option<String> {
157    let trimmed = value.trim();
158    if let Some(name) = trimmed.strip_circumfix("${", '}') {
159        properties.get(name).cloned()
160    } else if let Some(name) = trimmed.strip_prefix('$') {
161        properties.get(name).cloned()
162    } else {
163        None
164    }
165}

167pub fn parse_gradle(content: &str, uri: &Uri) -> Result<GradleParseResult> {
168    let path = uri.path().to_string();
169    let mut result = if path.ends_with("libs.versions.toml") {
170        catalog::parse_version_catalog(content, uri)?
171    } else if path.ends_with("settings.gradle.kts") || path.ends_with("settings.gradle") {
172        settings::parse_settings(content, uri)?
173    } else if path.ends_with(".gradle.kts") {
174        kotlin::parse_kotlin_dsl(content, uri)?
175    } else if path.ends_with(".gradle") {
176        groovy::parse_groovy_dsl(content, uri)?
177    } else {
178        return Ok(GradleParseResult {
179            dependencies: vec![],
180            uri: uri.clone(),
181        });
182    };
183
184    // Resolve variable references for build files (not catalogs or settings)
185    if (path.ends_with("build.gradle.kts") || path.ends_with("build.gradle"))
186        && let Some(dir) = std::path::Path::new(&path).parent()
187    {
188        let props = properties::load_gradle_properties(dir);
189        if !props.is_empty() {
190            resolve_variables(&mut result.dependencies, &props);
191        }
192    }
193
194    Ok(result)
195}

218/// Returns the number of UTF-16 code units in `s`.
219pub(crate) fn utf16_len(s: &str) -> usize {
220    s.chars().map(|c| c.len_utf16()).sum()
221}
222
223/// Finds the LSP range of `"group_id:artifact_id"` within the dependency
224/// declaration's own match span (`line[match_start..]`), not the whole line.
225///
226/// Scoping to `match_start` matters when two dependencies share an identical
227/// coordinate on one line — e.g.
228/// `implementation("a:b:1.0.0"); testImplementation("a:b:1.0.0")` — so the
229/// second dependency's range isn't mis-attributed to the first's position.
230pub(crate) fn find_name_range(
231    line: &str,
232    line_idx: u32,
233    match_start: usize,
234    group_id: &str,
235    artifact_id: &str,
236) -> Range {
237    let scoped = &line[match_start..];
238    let search = format!("{group_id}:{artifact_id}");
239    if let Some(rel) = scoped.find(&search) {
240        let abs_start = match_start + rel;
241        let col_u32 = utf16_len(&line[..abs_start]) as u32;
242        let end_u32 = col_u32 + utf16_len(&search) as u32;
243        Range::new(
244            Position::new(line_idx, col_u32),
245            Position::new(line_idx, end_u32),
246        )
247    } else {
248        Range::default()
249    }
250}
251
252/// Finds the LSP range of `version` after the second `:` within the
253/// dependency declaration's own match span (`line[match_start..]`), not the
254/// whole line.
255///
256/// Scoping to `match_start` matters when two dependencies share the same
257/// version string on one line — e.g.
258/// `implementation("a:b:1.0.0"); implementation("c:d:1.0.0")` — so the
259/// second dependency's range isn't mis-attributed to the first's position.
260pub(crate) fn find_version_range(
261    line: &str,
262    line_idx: u32,
263    match_start: usize,
264    version: &str,
265) -> Range {
266    let scoped = &line[match_start..];
267    let second_colon = scoped
268        .char_indices()
269        .filter(|(_, c)| *c == ':')
270        .nth(1)
271        .map(|(i, _)| i);
272
273    if let Some(colon_pos) = second_colon {
274        let after_colon = &scoped[colon_pos + 1..];
275        if let Some(rel) = after_colon.find(version) {
276            let abs_start = match_start + colon_pos + 1 + rel;
277            let col_start = utf16_len(&line[..abs_start]) as u32;
278            let col_end = col_start + utf16_len(version) as u32;
279            return Range::new(
280                Position::new(line_idx, col_start),
281                Position::new(line_idx, col_end),
282            );
283        }
284    }
285    Range::default()
286}
