9//! PyPI/Python support for deps-lsp.
10//!
11//! This crate provides parsing, validation, and registry client functionality
12//! for Python dependency management in `pyproject.toml` files (PEP 621 and
13//! Poetry formats) and in `requirements.txt`/`constraints.txt` files (pip's
14//! requirements file format).
15//!
16//! # Features
17//!
18//! - **PEP 621 Support**: Parse `[project.dependencies]` and `[project.optional-dependencies]`
19//! - **Poetry Support**: Parse `[tool.poetry.dependencies]` and `[tool.poetry.group.*.dependencies]`
20//! - **requirements.txt / constraints.txt Support**: Parse pip's line-oriented requirements
21//!   file format, including comments, continuations, per-requirement options, and includes
22//! - **PEP 508 Parsing**: Handle complex dependency specifications with extras and markers
23//! - **PEP 440 Versions**: Validate and compare Python version specifiers
24//! - **PEP 503 Name Normalization**: One canonical normalizer ([`name::normalize`]) shared by
25//!   registry lookups, the formatter, and lock file parsing
26//! - **PyPI API Client**: Fetch package metadata from PyPI JSON API with HTTP caching
27//!
28//! # Architecture
29//!
30//! deps-pypi follows the same architecture as deps-cargo and deps-npm:
31//! - **Types**: `PypiDependency`, `PypiVersion`, `PypiPackage` with LSP range tracking
32//! - **Parser**: Parse both PEP 621 and Poetry formats using `toml-span`
33//! - **Registry**: PyPI JSON API client with HTTP caching
34//! - **Error Handling**: Typed errors with `thiserror`
35//!
36//! # Examples
37//!
38//! ## Parsing pyproject.toml
39//!
40//! ```no_run
41//! use deps_pypi::PypiParser;
42//! use tower_lsp_server::ls_types::Uri;
43//!
44//! let content = r#"
45//! [project]
46//! dependencies = [
47//!     "requests>=2.28.0,<3.0",
48//!     "flask[async]>=3.0",
49//! ]
50//! "#;
51//!
52//! let parser = PypiParser::new();
53//! let uri = Uri::from_file_path("/project/pyproject.toml").unwrap();
54//! let result = parser.parse_content(content, &uri).unwrap();
55//!
56//! assert_eq!(result.dependencies.len(), 2);
57//! assert_eq!(result.dependencies[0].name, "requests");
58//! assert_eq!(result.dependencies[1].extras, vec!["async"]);
59//! ```
60//!
61//! ## Fetching versions from PyPI
62//!
63//! ```no_run
64//! use deps_pypi::PypiRegistry;
65//! use deps_core::HttpCache;
66//! use std::sync::Arc;
67//!
68//! # #[tokio::main]
69//! # async fn main() {
70//! let cache = Arc::new(HttpCache::new());
71//! let registry = PypiRegistry::new(cache);
72//!
73//! let versions = registry.get_versions("requests").await.unwrap();
74//! assert!(!versions.is_empty());
75//!
76//! let latest = registry
77//!     .get_latest_matching("requests", ">=2.28.0,<3.0")
78//!     .await
79//!     .unwrap();
80//! assert!(latest.is_some());
81//! # }
82//! ```
83//!
84//! ## Supported Formats
85//!
86//! ### PEP 621 (Standard)
87//!
88//! ```toml
89//! [project]
90//! dependencies = [
91//!     "requests>=2.28.0,<3.0",
92//!     "flask[async]>=3.0",
93//!     "numpy>=1.24; python_version>='3.9'",
94//! ]
95//!
96//! [project.optional-dependencies]
97//! dev = ["pytest>=7.0", "mypy>=1.0"]
98//! ```
99//!
100//! ### Poetry
101//!
102//! ```toml
103//! [tool.poetry.dependencies]
104//! python = "^3.9"
105//! requests = "^2.28.0"
106//! flask = {version = "^3.0", extras = ["async"]}
107//!
108//! [tool.poetry.group.dev.dependencies]
109//! pytest = "^7.0"
110//! mypy = "^1.0"
111//! ```
