1//! In-memory PyPI package-name search index, built once from the live PEP 691 Simple
2//! API project list.
3//!
4//! PyPI removed its XML-RPC search API and offers no first-party ranked search, so
5//! this mimics how PyCharm and similar tools fill the gap: download the full list of
6//! ~882k project names, normalize and sort it locally, and serve prefix matches with
7//! no popularity ranking. See `.local/plans/419-pypi-search.md` for the full design
8//! rationale (issue #419).
9//!
10//! # Lifecycle
11//!
12//! The index is built at most once per process (see [`IndexCell`]): there is no TTL
13//! and it is never refreshed once [`IndexState::Ready`]. Measured against the live
14//! endpoint, the index's `X-PyPI-Last-Serial` advances roughly 1,820 times per hour,
15//! so any TTL short enough to matter would mean re-downloading the ~9.6 MB gzipped
16//! body on almost every expiry — worse than simply accepting that a long-lived
17//! process serves a name list that is at most a few hours stale.
18//!
19//! A failed build (network error, malformed body, or an implausibly small parse —
20//! see [`MIN_PLAUSIBLE_INDEX_ENTRIES`]) is retried with exponential backoff rather
21//! than on every call, so a persistently failing link costs one attempt per backoff
22//! window, not one per keystroke.
23

33/// URL for PyPI's full Simple API project index (PEP 691).
34pub(crate) const SIMPLE_INDEX_URL: &str = "https://pypi.org/simple/";
35
36/// Origin the index fetch's redirect policy is pinned to (L1, security review):
37/// this is the one fetch in the workspace carrying a [`SIMPLE_INDEX_MAX_BYTES`]
38/// (96 MiB) budget rather than [`deps_core::BodyLimit::DEFAULT`]'s 32 MiB, so a
39/// cross-host redirect is worth stopping explicitly rather than following
40/// `reqwest`'s default up-to-10-hop policy. Only constrains *redirects*, not the
41/// initial request — `index_url` itself is always [`SIMPLE_INDEX_URL`] in
42/// production (or a test's own mock server URL), never user-controlled.
43const PYPI_TRUSTED_ORIGIN: &str = "https://pypi.org/";
44
45/// Upper bound on the index response body. Roughly 2x the ~43 MB decompressed size
46/// observed live (2026-08-31), bounded well under [`deps_core::cache::BodyLimit`]'s
47/// own 128 MiB ceiling so this doesn't silently grow the cache's largest-response
48/// footprint if PyPI's project count keeps climbing.
49const SIMPLE_INDEX_MAX_BYTES: BodyLimit = BodyLimit::new(96 * 1024 * 1024);
50
51/// Upper bound on the number of parsed project entries, independent of
52/// [`SIMPLE_INDEX_MAX_BYTES`]'s byte cap — defense in depth against a body that is
53/// small but contains an implausible number of tiny entries.
54///
55/// Enforced *during* deserialization by `deserialize_projects`'s streaming
56/// `Visitor`, not by truncating an already-fully-materialized `Vec` afterward: a
57/// 96 MiB body of minimal `{"name":"a"}` entries deserializes to ~7.7M elements,
58/// and holding all of them live at once (rather than normalizing/filtering each
59/// one and discarding it immediately) previously peaked at ~370 MB RSS for a
60/// 1-entry result — the cap did nothing to prevent that. See `deserialize_projects`.
61const MAX_INDEX_ENTRIES: usize = 2_000_000;
62
63/// Upper bound on a single raw (pre-normalization) project name, checked before
64/// [`crate::name::normalize`] runs.
65///
66/// PyPI's own package-name validation caps a real name at 214 characters, so 1024
67/// bytes loses no legitimate name. Checked first because `normalize` itself
68/// allocates three full copies of its input (`to_lowercase` → `replace` → `join`):
69/// a single ~95 MiB name in an otherwise well-formed body peaked at ~400 MB RSS
70/// before the (still-necessary) [`deps_core::is_safe_package_name`] gate discarded
71/// it anyway — this check makes that discard immediate instead.
72const MAX_RAW_NAME_LEN: usize = 1024;
73
74/// A successful parse with fewer than this many entries is treated as a failed
75/// build rather than cached as [`IndexState::Ready`].
76///
77/// Dropping the TTL (see the module doc) means a successful build is never
78/// invalidated later, so a body that arrives truncated-but-still-valid-JSON would
79/// otherwise pin the process to a degraded index for its entire lifetime with no
80/// path back. The live index carries ~882k entries; 100,000 is comfortably below
81/// any plausible legitimate response while still well above what a truncated
82/// transfer would produce.
83const MIN_PLAUSIBLE_INDEX_ENTRIES: usize = 100_000;
84
85/// Base backoff after the first failed build attempt.
86const RETRY_BASE_BACKOFF: Duration = Duration::from_secs(30);
87
88/// Ceiling on the computed backoff duration, regardless of how many attempts have
89/// failed.
90const RETRY_MAX_BACKOFF: Duration = Duration::from_mins(15);
91
92/// Ceiling on the *exponent* used to compute backoff, independent of
93/// [`RETRY_MAX_BACKOFF`]'s ceiling on the resulting duration.
94///
95/// `attempts` (see [`IndexState::Failed`]) grows without bound on a permanently
96/// failing link — a long-lived editor session can accumulate far more than 64 of
97/// them. Capping only the backoff *duration* would still compute `1u64 <<
98/// (attempts - 1)` first, which panics on overflow in debug builds once `attempts`
99/// exceeds 64. Clamping the exponent itself means the shift is always well-formed,
100/// independent of how large `attempts` gets.
101const MAX_BACKOFF_EXPONENT: u32 = 10;
102
103/// Computes the backoff duration for the `attempts`-th failed build (`attempts >=
104/// 1`): `min(RETRY_BASE_BACKOFF * 2^(attempts - 1), RETRY_MAX_BACKOFF)`, with the
105/// exponent itself clamped to [`MAX_BACKOFF_EXPONENT`] so the shift can never
106/// overflow regardless of how large `attempts` grows.
107///
108/// # Examples
109///
110/// ```ignore
111/// assert_eq!(backoff_for(1), Duration::from_secs(30));
112/// assert_eq!(backoff_for(2), Duration::from_secs(60));
113/// ```
114fn backoff_for(attempts: u32) -> Duration {
115    let exponent = attempts.saturating_sub(1).min(MAX_BACKOFF_EXPONENT);
116    let secs = RETRY_BASE_BACKOFF
117        .as_secs()
118        .saturating_mul(1u64 << exponent);
119    Duration::from_secs(secs.min(RETRY_MAX_BACKOFF.as_secs()))
120}

122/// A sorted, PEP 503-normalized set of PyPI project names, laid out as one
123/// concatenated `String` blob plus an offset table rather than a `Vec<String>`.
124///
125/// At ~882k entries a `Vec<String>` would cost roughly 12 MB of name bytes plus
126/// ~28 MB of per-`String` allocator/pointer overhead; the blob layout keeps the
127/// total near 15 MB in the common case. That figure is the *typical* case, not the
128/// bound: [`MAX_INDEX_ENTRIES`] bounds entry count and [`SIMPLE_INDEX_MAX_BYTES`]
129/// bounds the wire body, but neither bounds the blob directly, so a maximally
130/// adversarial (but within-cap) body — many short names packed into the 96 MiB
131/// limit — could parse to roughly 70 MB of blob plus 8 MB of offsets, ~78 MB total.
132/// This is this struct's own *steady-state* size once parsing has finished and the
133/// input body has been dropped; see [`deserialize_projects`] for the (separately
134/// bounded) transient peak *during* parsing.
135pub(crate) struct PackageIndex {
136    /// Every entry's normalized name, concatenated in sorted order.
137    blob: String,
138    /// `n + 1` byte offsets into `blob`; entry `i` spans `blob[offsets[i]..offsets[i
139    /// + 1]]`.
140    offsets: Vec<u32>,
141}
142
143impl PackageIndex {
144    /// Builds an index from already deduplicated, sorted, normalized names.
145    fn from_sorted_names(names: Vec<String>) -> Self {
146        let mut blob = String::with_capacity(names.iter().map(String::len).sum());
147        let mut offsets = Vec::with_capacity(names.len() + 1);
148        offsets.push(0u32);
149        for name in &names {
150            blob.push_str(name);
151            // `SIMPLE_INDEX_MAX_BYTES` (96 MiB) bounds the source body far below
152            // u32::MAX bytes, so this never actually saturates; the clamp is
153            // defense in depth rather than a reachable path.
154            offsets.push(u32::try_from(blob.len()).unwrap_or(u32::MAX));
155        }
156        Self { blob, offsets }
157    }
158
159    /// Number of entries in the index.
160    fn len(&self) -> usize {
161        self.offsets.len() - 1
162    }
163
164    /// The normalized name at `index`.
165    fn name(&self, index: usize) -> &str {
166        &self.blob[self.offsets[index] as usize..self.offsets[index + 1] as usize]
167    }
168
169    /// Index of the first entry that is `>= prefix`, via binary search
170    /// (`slice::partition_point`'s underlying algorithm, adapted to the blob
171    /// layout rather than a slice of `&str`).
172    fn lower_bound(&self, prefix: &str) -> usize {
173        let mut lo = 0usize;
174        let mut hi = self.len();
175        while lo < hi {
176            let mid = lo + (hi - lo) / 2;
177            if self.name(mid) < prefix {
178                lo = mid + 1;
179            } else {
180                hi = mid;
181            }
182        }
183        lo
184    }
185
186    /// Returns up to `limit` normalized names starting with `normalized_prefix`, in
187    /// sorted (alphabetical, unranked) order.
188    ///
189    /// O(log n + limit): [`Self::lower_bound`] locates the first candidate, then a
190    /// forward scan collects matches until either a non-matching entry or `limit`
191    /// is reached.
192    pub(crate) fn prefix_matches(&self, normalized_prefix: &str, limit: usize) -> Vec<&str> {
193        if limit == 0 || normalized_prefix.is_empty() {
194            return Vec::new();
195        }
196        let mut out = Vec::with_capacity(limit.min(self.len()));
197        for index in self.lower_bound(normalized_prefix)..self.len() {
198            let name = self.name(index);
199            if !name.starts_with(normalized_prefix) {
200                break;
201            }
202            out.push(name);
203            if out.len() >= limit {
204                break;
205            }
206        }
207        out
208    }
209}

211/// State of the lazily-built, build-once package index.
212enum IndexState {
213    /// No build has ever been attempted.
214    NotBuilt,
215    /// A build succeeded; served forever (see the module doc — there is no TTL).
216    Ready(Arc<PackageIndex>),
217    /// The most recent build attempt failed. Retried only once
218    /// `last_attempt.elapsed() >= backoff_for(attempts)`.
219    Failed {
220        last_attempt: Instant,
221        attempts: u32,
222    },
223}
224
225/// Holds the build-once [`PackageIndex`] state behind a [`std::sync::RwLock`] (never
226/// held across an `.await`, matching the discipline `HttpCache` itself uses), plus a
227/// [`tokio::sync::Mutex`] that serializes the build itself so N concurrently-queued
228/// completion requests trigger exactly one download rather than one each.
229pub(crate) struct IndexCell {
230    state: RwLock<IndexState>,
231    build_lock: tokio::sync::Mutex<()>,
232}
233
234impl IndexCell {
235    pub(crate) fn new() -> Self {
236        Self {
237            state: RwLock::new(IndexState::NotBuilt),
238            build_lock: tokio::sync::Mutex::new(()),
239        }
240    }
241
242    /// Reads the current state. Lock poisoning (only reachable if a prior writer
243    /// panicked mid-update, which none of this module's critical sections can do
244    /// mid-guard) is recovered from rather than propagated, since a poisoned search
245    /// index is still safe to treat as "not built yet".
246    fn read_state(&self) -> RwLockReadGuard<'_, IndexState> {
247        self.state.read().unwrap_or_else(PoisonError::into_inner)
248    }
249
250    fn write_state(&self) -> RwLockWriteGuard<'_, IndexState> {
251        self.state.write().unwrap_or_else(PoisonError::into_inner)
252    }
253
254    /// Returns the ready index, if any, without touching the network.
255    pub(crate) fn ready(&self) -> Option<Arc<PackageIndex>> {
256        match &*self.read_state() {
257            IndexState::Ready(index) => Some(Arc::clone(index)),
258            IndexState::NotBuilt | IndexState::Failed { .. } => None,
259        }
260    }
261
262    /// Whether a build attempt should be made right now: never built yet, or
263    /// previously failed and its backoff window has elapsed.
264    fn needs_attempt(&self) -> bool {
265        match &*self.read_state() {
266            IndexState::Ready(_) => false,
267            IndexState::NotBuilt => true,
268            IndexState::Failed {
269                last_attempt,
270                attempts,
271            } => last_attempt.elapsed() >= backoff_for(*attempts),
272        }
273    }
274
275    fn failed_attempts(&self) -> u32 {
276        match &*self.read_state() {
277            IndexState::Failed { attempts, .. } => *attempts,
278            IndexState::NotBuilt | IndexState::Ready(_) => 0,
279        }
280    }
281
282    fn set_ready(&self, index: PackageIndex) {
283        *self.write_state() = IndexState::Ready(Arc::new(index));
284    }
285
286    fn set_failed(&self, prior_attempts: u32) {
287        *self.write_state() = IndexState::Failed {
288            last_attempt: Instant::now(),
289            attempts: prior_attempts.saturating_add(1),
290        };
291    }
292}

294/// Starts building `cell`'s index in the background if [`IndexCell::needs_attempt`]
295/// says one is due, otherwise returns immediately without spawning anything.
296///
297/// Safe to call unconditionally and often (every completion request in a Python
298/// manifest calls this via [`crate::registry::PypiRegistry::warm_search_index`]):
299/// the cheap synchronous [`IndexCell::needs_attempt`] check short-circuits the
300/// common case (already [`IndexState::Ready`]) without spawning a task at all, and
301/// the single-flight [`IndexCell::build_lock`] — re-checked *after* it is acquired —
302/// ensures that even many callers racing on a cold cell produce exactly one fetch.
303///
304/// The spawned task's `JoinHandle` is retained and awaited by a supervisor task
305/// that logs a panic instead of discarding it silently (mirroring the fix in
306/// commit b8cd6210 for the cold-start-limiter cleanup task).
307pub(crate) fn trigger_index_build(cache: Arc<HttpCache>, index_url: String, cell: Arc<IndexCell>) {
308    if !cell.needs_attempt() {
309        return;
310    }
311
312    let build_task = tokio::spawn(async move {
313        ensure_index_built(&cache, &index_url, &cell).await;
314    });
315    tokio::spawn(async move {
316        if let Err(e) = build_task.await {
317            tracing::error!("PyPI simple index build task exited unexpectedly: {e}");
318        }
319    });
320}
321
322/// Builds the index if [`IndexCell::needs_attempt`] still holds after acquiring the
323/// single-flight lock (double-checked locking, N3: the first check in
324/// [`trigger_index_build`] happens before the lock, so a racing caller that loses
325/// the race must re-check once it actually holds the lock rather than repeat the
326/// download).
327async fn ensure_index_built(cache: &HttpCache, index_url: &str, cell: &IndexCell) {
328    let _guard = cell.build_lock.lock().await;
329
330    if !cell.needs_attempt() {
331        return;
332    }
333
334    let prior_attempts = cell.failed_attempts();
335
336    match fetch_and_parse_index(cache, index_url).await {
337        Some(index) if index.len() >= MIN_PLAUSIBLE_INDEX_ENTRIES => {
338            tracing::info!("PyPI simple index built: {} entries", index.len());
339            cell.set_ready(index);
340        }
341        Some(too_small) => {
342            tracing::warn!(
343                "PyPI simple index parsed but implausibly small ({} entries, expected at least \
344                 {MIN_PLAUSIBLE_INDEX_ENTRIES}); treating as a failed build so backoff retries",
345                too_small.len()
346            );
347            cell.set_failed(prior_attempts);
348        }
349        None => cell.set_failed(prior_attempts),
350    }
351}

353/// Fetches the Simple API index and parses it on a blocking thread. Never returns
354/// `Err`: every failure mode (network, HTTP status, malformed JSON, or a panicked
355/// parse task) is logged and folded into `None`, since a failed background index
356/// build must degrade to "search still returns empty", not propagate an error.
357async fn fetch_and_parse_index(cache: &HttpCache, index_url: &str) -> Option<PackageIndex> {
358    let body = match cache
359        .get_transport_only_with_headers_limited_trusted_origin(
360            index_url,
361            &[(reqwest::header::ACCEPT, SIMPLE_API_ACCEPT)],
362            SIMPLE_INDEX_MAX_BYTES,
363            PYPI_TRUSTED_ORIGIN,
364        )
365        .await
366    {
367        Ok(body) => body,
368        Err(e) => {
369            tracing::warn!("failed to fetch PyPI simple index: {e}");
370            return None;
371        }
372    };
373
374    match tokio::task::spawn_blocking(move || parse_index(&body)).await {
375        Ok(Ok(index)) => Some(index),
376        Ok(Err(e)) => {
377            tracing::warn!("failed to parse PyPI simple index: {e}");
378            None
379        }
380        Err(e) => {
381            tracing::warn!("PyPI simple index parse task panicked: {e}");
382            None
383        }
384    }
385}

387/// PEP 691 Simple API response shape, narrowed to the one field this module needs.
388///
389/// `projects` deserializes straight into normalized, filtered, capped `String`
390/// names (via [`deserialize_projects`]) rather than an intermediate
391/// `Vec<Project<'a>>` — see that function's doc for why the intermediate shape was
392/// a real memory-amplification bug (#419 S3/M1), not just an inefficiency.
393#[derive(Deserialize)]
394struct SimpleIndex {
395    #[serde(deserialize_with = "deserialize_projects")]
396    projects: Vec<String>,
397}
398
399/// A single project entry as it appears on the wire. `Cow<'a, str>`, not `&'a
400/// str`: `serde_json` can only borrow a string that needs no unescaping, so a
401/// plain `&str` here would make a *single* escaped name anywhere in the
402/// ~882k-entry response fail the whole parse and silently drop every name. PEP
403/// 508 package-name rules should preclude characters that need JSON escaping, but
404/// the parser must not rest on that.
405#[derive(Deserialize)]
406struct Project<'a> {
407    #[serde(borrow)]
408    name: Cow<'a, str>,
409}
410
411/// Deserializes the `projects` JSON array directly into a bounded, normalized,
412/// filtered `Vec<String>` — never materializing a `Vec<Project>` of the full
413/// input size.
414///
415/// This is the fix for #419 S3/M1: the original shape deserialized into
416/// `Vec<Project<'a>>` first and applied [`MAX_INDEX_ENTRIES`] via `.take(...)`
417/// *afterward*, so the cap bounded nothing about peak memory — a 96 MiB body of
418/// minimal `{"name":"a"}` entries decodes to ~7.7M elements, and holding all of
419/// them live simultaneously (rather than processing and discarding each one)
420/// measured at ~370 MB peak RSS for a final 1-entry index. Streaming via this
421/// [`serde::de::Visitor`] means each element exists only for the duration of one
422/// loop iteration: normalized and pushed if it survives every filter, dropped
423/// immediately otherwise, with `names` never growing past [`MAX_INDEX_ENTRIES`].
424///
425/// The raw-length gate ([`MAX_RAW_NAME_LEN`], M2) runs first, before
426/// [`crate::name::normalize`] (three allocations per call) or
427/// [`deps_core::is_safe_package_name`] even look at the name — both would reject
428/// an oversized name anyway, but only after paying its allocation cost.
429///
430/// The sequence is drained to its end even after `names` reaches
431/// [`MAX_INDEX_ENTRIES`] (each further element is still parsed and immediately
432/// dropped, not accumulated): a `SeqAccess` must be fully consumed for the
433/// deserializer's cursor to land correctly past the closing `]`, so stopping
434/// early would corrupt parsing of whatever JSON follows.
435fn deserialize_projects<'de, D>(deserializer: D) -> std::result::Result<Vec<String>, D::Error>
436where
437    D: serde::Deserializer<'de>,
438{
439    struct ProjectsVisitor;
440
441    impl<'de> serde::de::Visitor<'de> for ProjectsVisitor {
442        type Value = Vec<String>;
443
444        fn expecting(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
445            formatter.write_str("a sequence of PyPI Simple API project objects")
446        }
447
448        fn visit_seq<A>(self, mut seq: A) -> std::result::Result<Self::Value, A::Error>
449        where
450            A: serde::de::SeqAccess<'de>,
451        {
452            let mut names = Vec::new();
453            while let Some(project) = seq.next_element::<Project<'de>>()? {
454                if names.len() >= MAX_INDEX_ENTRIES || project.name.len() > MAX_RAW_NAME_LEN {
455                    continue;
456                }
457                let normalized = crate::name::normalize(&project.name);
458                if normalized.is_empty() || !deps_core::is_safe_package_name(&normalized) {
459                    continue;
460                }
461                names.push(normalized);
462            }
463            Ok(names)
464        }
465    }
466
467    deserializer.deserialize_seq(ProjectsVisitor)
468}

470/// Parses a Simple API index response body into a sorted [`PackageIndex`].
471///
472/// Each name is normalized via [`crate::name::normalize`] (the same normalizer used
473/// for registry lookups, so a query and the index it searches always agree on
474/// spelling), filtered through [`deps_core::is_safe_package_name`], and deduplicated
475/// — normalization collapses some distinct raw spellings onto the same key. PyPI's
476/// own wire order is a punctuation-insensitive collation, not byte order, so the
477/// names are always re-sorted here regardless of input order.
478///
479/// Intended to run inside [`tokio::task::spawn_blocking`]: normalizing and sorting
480/// up to [`MAX_INDEX_ENTRIES`] names is CPU-bound work that must not sit on the
481/// async runtime.
482fn parse_index(body: &[u8]) -> Result<PackageIndex> {
483    let index: SimpleIndex =
484        deps_core::parse_json_checked(body).map_err(|e| DepsError::ApiResponse {
485            package: "<pypi-simple-index>".to_string(),
486            registry: REGISTRY,
487            source: e,
488        })?;
489
490    let mut names = index.projects;
491    names.sort_unstable();
492    names.dedup();
493
494    Ok(PackageIndex::from_sorted_names(names))
495}

497/// Builds a Simple API response body plausible enough to clear
498/// [`MIN_PLAUSIBLE_INDEX_ENTRIES`] (padded with synthetic entries), plus
499/// `extra_names`, for tests elsewhere in the crate that need a real successful
500/// index build against a mock server (`crate::registry`, `crate::ecosystem`).
501#[cfg(test)]
502pub(crate) fn sample_index_body(extra_names: &[&str]) -> String {
503    let mut projects: Vec<String> = (0..MIN_PLAUSIBLE_INDEX_ENTRIES)
504        .map(|i| format!(r#"{{"name": "synthetic-package-{i}"}}"#))
505        .collect();
506    projects.extend(
507        extra_names
508            .iter()
509            .map(|name| format!(r#"{{"name": "{name}"}}"#)),
510    );
511    format!(
512        r#"{{"meta": {{"api-version": "1.4"}}, "projects": [{}]}}"#,
513        projects.join(",")
514    )
515}

768    async fn test_trigger_index_build_single_flight_against_mock() {
769        let mut server = mockito::Server::new_async().await;
770        let mock = server
771            .mock("GET", "/simple/")
772            .with_status(200)
773            .with_body(sample_index_body(&["requests"]))
774            .expect(1)
775            .create_async()
776            .await;
777
778        let cache = Arc::new(HttpCache::new());
779        let cell = Arc::new(IndexCell::new());
780        let index_url = format!("{}/simple/", server.url());
781
782        // N concurrent triggers must still produce exactly one fetch.
783        let mut handles = Vec::new();
784        for _ in 0..5 {
785            handles.push(tokio::spawn({
786                let cache = Arc::clone(&cache);
787                let index_url = index_url.clone();
788                let cell = Arc::clone(&cell);
789                async move { trigger_index_build(cache, index_url, cell) }
790            }));
791        }
792        for h in handles {
793            h.await.unwrap();
794        }
795
796        // Poll (rather than a single fixed sleep) so the test isn't flaky under
797        // slower CI machines: parsing ~100k entries plus the HTTP round trip can
798        // occasionally exceed a short fixed delay.
799        for _ in 0..100 {
800            if cell.ready().is_some() {
801                break;
802            }
803            tokio::time::sleep(Duration::from_millis(50)).await;
804        }
805
806        assert!(
807            cell.ready().is_some(),
808            "index should have built successfully"
809        );
810        mock.assert_async().await;
811    }

836    #[tokio::test]
837    async fn test_ensure_index_built_failure_then_backoff_then_retry() {
838        let mut server = mockito::Server::new_async().await;
839        let mock = server
840            .mock("GET", "/simple/")
841            .with_status(500)
842            .expect(2)
843            .create_async()
844            .await;
845
846        let cache = HttpCache::new();
847        let cell = IndexCell::new();
848        let index_url = format!("{}/simple/", server.url());
849
850        ensure_index_built(&cache, &index_url, &cell).await;
851        assert_eq!(cell.failed_attempts(), 1);
852
853        // Still within backoff: a second call must not touch the network again.
854        ensure_index_built(&cache, &index_url, &cell).await;
855        assert_eq!(
856            cell.failed_attempts(),
857            1,
858            "still within backoff, must not have retried"
859        );
860
861        // #419 M4 regression (§6.7's "after the window elapses, exactly one more
862        // attempt" half, previously untested): once `backoff_for(1)` (30s) has
863        // elapsed, the *next* call must retry exactly once more, not stay stuck at
864        // `attempts == 1` forever. `set_failed` (and so `last_attempt`) is only
865        // ever mutated under `build_lock` inside `ensure_index_built` itself, so
866        // directly rewriting the cell's state to simulate elapsed wall-clock time
867        // is the only way to test this without pausing the tokio clock crate-wide.
868        *cell.write_state() = IndexState::Failed {
869            last_attempt: Instant::now().checked_sub(Duration::from_secs(31)).unwrap(),
870            attempts: 1,
871        };
872        ensure_index_built(&cache, &index_url, &cell).await;
873        assert_eq!(
874            cell.failed_attempts(),
875            2,
876            "backoff window elapsed: must have retried exactly once more"
877        );
878
879        mock.assert_async().await;
880    }
