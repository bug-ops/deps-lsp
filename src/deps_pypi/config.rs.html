1//! Private/custom PyPI index resolution — `--index-url`/`--extra-index-url`
2//! (`requirements.txt`), Poetry `[[tool.poetry.source]]`, and uv
3//! `[tool.uv.index]`/`[tool.uv.sources]`.
4//!
5//! # Security model (read before touching this module)
6//!
7//! A `requirements.txt`/`pyproject.toml` index declaration is attacker-controlled the moment a
8//! hostile repository is cloned and opened — this LSP parses on file open, before any build
9//! ever runs. Phase 1 carries no authentication at all (spec Out of Scope: no URL userinfo, no
10//! `keyring`/`.netrc`), which closes the credential half of the threat model this module would
11//! otherwise have to solve, but two things still apply:
12//!
13//! - **No credential-shaped value is ever parsed.** [`PypiIndexUrl::new`] rejects any URL
14//!   carrying `username()`/`password()` outright (FR-006/FR-011) — there is no expansion step
15//!   for PyPI config (unlike npm's `${VAR}`), so [`InvalidEntry::raw`] and every
16//!   `tracing::warn!` here name the as-written value with any embedded userinfo and query
17//!   string/fragment stripped first (see [`deps_core::net_policy::RedactedUrl`]) — the raw
18//!   value is otherwise preserved so a warning or a
19//!   [`DependencySource::CustomRegistry`] naming an unresolved primary/named source still shows
20//!   the user what they actually typed, minus the credential.
21//! - **FR-005's resolution order is the load-bearing security invariant of this whole
22//!   feature.** Case (a) (an explicit `--index-url`/Poetry primary/uv `default`): the
23//!   explicit primary is checked first, then extras — a deliberate user choice, no
24//!   disclosure risk. Case (b) (no explicit primary, extras only): declared extras are
25//!   checked **before** the implicit public `pypi.org` fallback, never the reverse — this is
26//!   what stops a private package's name from being sent to `pypi.org` before the user's own
27//!   declared index has had a chance, and what stops a same-named public package from
28//!   silently shadowing a private one. See [`PypiIndexConfig::resolve_source_for`] and
29//!   [`ResolvedChain`]'s docs.
30//!
31//! See `specs/033-pypi-private-index-support/spec.md` FR-001–FR-014 and
32//! `specs/033-pypi-private-index-support/plan.md` §1/§3 for the design review this module
33//! implements.

46/// A validated, normalized, https-only PyPI-protocol index URL with no embedded userinfo.
47///
48/// Mirrors `deps_npm::config::NpmRegistryIndex` (see FR-006/FR-011); kept `deps-pypi`-local
49/// rather than promoted to `deps-core` per this spec's Open Questions (consolidate only once a
50/// third near-identical type makes the duplication concrete).
51#[derive(Debug, Clone, PartialEq, Eq, Hash)]
52pub struct PypiIndexUrl {
53    /// The validated URL, normalized by stripping a trailing `/` — matches
54    /// `simple_api_url`'s existing `{base}/{name}/` join convention (PEP 503).
55    normalized: String,
56}
57
58impl PypiIndexUrl {
59    /// Validates and normalizes `raw` against `policy`.
60    ///
61    /// # Errors
62    ///
63    /// Returns [`PypiIndexUrlError`] if `raw` does not parse as a URL, is not `https` (outside
64    /// the `cfg(test)`/`test-util` loopback carve-out), carries a userinfo component, or
65    /// resolves to a host class the current `policy` blocks.
66    ///
67    /// # Examples
68    ///
69    /// ```
70    /// use deps_core::net_policy::RegistryAccessPolicy;
71    /// use deps_pypi::config::PypiIndexUrl;
72    ///
73    /// let policy = RegistryAccessPolicy::default();
74    /// assert!(PypiIndexUrl::new("https://pypi.mycorp.example/simple", &policy).is_ok());
75    /// assert!(PypiIndexUrl::new("http://pypi.mycorp.example/simple", &policy).is_err());
76    /// assert!(PypiIndexUrl::new("https://user:pass@pypi.mycorp.example", &policy).is_err());
77    /// ```
78    pub fn new(raw: &str, policy: &RegistryAccessPolicy) -> Result<Self, PypiIndexUrlError> {
79        let url = validate_index_url(raw, raw, "pypi", PolicyGate::Enforce(policy))?;
80        let normalized = url.as_str().trim_end_matches('/').to_string();
81        Ok(Self { normalized })
82    }
83
84    /// The normalized index URL. Never carries a trailing `/`.
85    #[must_use]
86    pub fn as_str(&self) -> &str {
87        &self.normalized
88    }
89}

91impl std::fmt::Display for PypiIndexUrl {
92    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
93        f.write_str(self.as_str())
94    }
95}

97/// A present-but-unusable index entry — an invalid URL, a policy-blocked host, or a
98/// well-formed-but-non-https/userinfo-bearing value.
99///
100/// Carries the raw value as written, **with any embedded userinfo and query string/fragment
101/// redacted** (M1 fix, extended by #767 S2a, structurally enforced by #789's
102/// [`deps_core::net_policy::RedactedUrl`]), so [`PypiIndexConfig::resolve_source_for`] can
103/// build [`DependencySource::CustomRegistry`] for an explicit primary/named source, or log a
104/// warning naming what the user wrote for a
105/// dropped extra, without ever holding or surfacing the credential itself: a
106/// `CustomRegistry.url` can reach hover/diagnostics text, and a `UserInfoPresent` rejection is
107/// exactly the case where `raw` would otherwise still contain `user:pass@` (a query-string
108/// credential can reach the same field via any other rejection reason).
109///
110/// Output-only: constructed internally by this module's own `resolve_entry`, never by
111/// external code — no constructor is provided.
112#[non_exhaustive]
113#[derive(Debug, Clone)]
114pub struct InvalidEntry {
115    /// The raw index value, as written in the source file, with any `user:pass@`/`user@`
116    /// userinfo component and any query string/fragment stripped.
117    pub raw: RedactedUrl,
118    /// Why it was rejected.
119    pub reason: PypiIndexUrlError,
120}
121
122/// Validates and normalizes one raw index value, logging a `tracing::warn!` naming the raw
123/// value (redacted — see [`deps_core::net_policy::RedactedUrl`]) on failure. `pub(crate)`:
124/// every parser
125/// surface (`requirements.rs`, `pyproject.rs`) that discovers a candidate index value calls
126/// this before handing the result to a [`PypiIndexConfig`] setter.
127pub(crate) fn resolve_entry(
128    raw: &str,
129    policy: &RegistryAccessPolicy,
130) -> Result<PypiIndexUrl, InvalidEntry> {
131    PypiIndexUrl::new(raw, policy).map_err(|reason| {
132        // #767 S2a: `RedactedUrl`, not `redact_userinfo` alone — this value is also
133        // stored in `InvalidEntry::raw`, which can surface as `DependencySource::CustomRegistry`'s
134        // hover/diagnostics text, so a query-string credential must be stripped too, not just
135        // userinfo. Host and path still survive, so hover stays identifiable.
136        let redacted = RedactedUrl::new(raw);
137        tracing::warn!(raw = %redacted, %reason, "PyPI index URL failed validation");
138        InvalidEntry {
139            raw: redacted,
140            reason,
141        }
142    })
143}

145/// One fully-resolved, ready-to-register routing chain — produced by
146/// [`PypiIndexConfig::resolved_chains`], consumed by
147/// `PypiRegistry::register_alternate`.
148///
149/// Output-only: constructed internally by [`PypiIndexConfig::resolved_chains`], never by
150/// external code — no constructor is provided.
151#[non_exhaustive]
152#[derive(Debug, Clone)]
153pub struct ResolvedChain {
154    /// Composite identity — becomes both the router's `alternates` map key and the
155    /// `DependencySource::AlternateRegistry.index` value for a plain (non-named-source)
156    /// dependency.
157    ///
158    /// For a primary/extras chain (case a/b): an opaque, single-line hashed token produced by
159    /// [`deps_core::hash_routing_key`] (`"pypi-chain"`) over the ordered hop strings plus the
160    /// [`Self::implicit_public_fallback`] flag — **never** a newline-joined or otherwise
161    /// URL-shaped value: `deps-core`'s `AlternateRegistry.index` doc describes this field as
162    /// "the resolved index URL" for Cargo/npm, and a value that merely *looks* like a URL
163    /// (but isn't one) would violate that contract for any future reader — see plan.md's N6
164    /// fix. An explicit chain `[A, B]` and an implicit chain that happens to resolve to the
165    /// same hops `[A, B]` plus the fallback flag produce **different** keys (this is what
166    /// closes the C2 aliasing defect: two files sharing a primary but differing extras never
167    /// collide, and editing a file's extras changes its key on the next reparse).
168    ///
169    /// For a named-source chain (Poetry `source =`/uv `index =`): the source's own literal
170    /// URL, matching Cargo/npm's convention for a single resolved index.
171    pub key: String,
172    /// Ordered, already-validated hops. Hop 0 becomes the registered client's own
173    /// `simple_base`; the rest become its `fallback_chain`. Never empty — see
174    /// [`PypiIndexConfig::resolved_chains`]'s zero-hop handling.
175    pub hops: Vec<PypiIndexUrl>,
176    /// `true` only for a case-(b) chain (spec FR-005(b)) whose final hop is the implicit
177    /// public `pypi.org` root, appended at registration time rather than present in
178    /// [`Self::hops`] — `PypiRegistry::register_alternate` builds that hop itself.
179    pub implicit_public_fallback: bool,
180}
181
182impl ResolvedChain {
183    fn chain(hops: Vec<PypiIndexUrl>, implicit_public_fallback: bool) -> Self {
184        let flag = if implicit_public_fallback {
185            "true"
186        } else {
187            "false"
188        };
189        let key = deps_core::hash_routing_key(
190            "pypi-chain",
191            hops.iter()
192                .map(PypiIndexUrl::as_str)
193                .chain(std::iter::once(flag)),
194        );
195        Self {
196            key,
197            hops,
198            implicit_public_fallback,
199        }
200    }
201
202    fn named_source(url: PypiIndexUrl) -> Self {
203        Self {
204            key: url.as_str().to_string(),
205            hops: vec![url],
206            implicit_public_fallback: false,
207        }
208    }
209}

211/// The effective case-(b) hop list (spec FR-005(b)): declared extras, in order, plus a final
212/// hop that is either a concrete uv `default = true` index or the implicit public fallback.
213struct CaseBChain {
214    hops: Vec<PypiIndexUrl>,
215    implicit_public_fallback: bool,
216}

218/// Resolved index configuration for one `requirements.txt`/`pyproject.toml` file.
219///
220/// Built once per parse (two-pass for `requirements.txt` — see `parser::requirements`'s
221/// module doc; single-pass for `pyproject.toml`, whose TOML tree is fully available before any
222/// dependency is resolved), consulted per-dependency via [`Self::resolve_source_for`].
223#[derive(Debug, Default)]
224pub struct PypiIndexConfig {
225    /// Explicit `--index-url`, a Poetry `primary`/`default`-priority source (including one
226    /// with no `priority` key at all — FR-007), or a Poetry `explicit`/unrecognized-priority
227    /// source contributes nothing here. uv **never** populates this field (FR-013's r3
228    /// correction) — a pure-uv config always routes through [`Self::case_b_chain`] instead.
229    /// `None` when no explicit primary is declared (spec FR-005(b) applies).
230    primary: Option<Result<PypiIndexUrl, InvalidEntry>>,
231    /// `--extra-index-url` values (declaration order preserved) plus Poetry
232    /// `supplemental`/`secondary`-priority sources plus every non-`default`/non-`explicit` uv
233    /// `[tool.uv.index]` entry — FR-005's fallback chain. An `Err(InvalidEntry)` here is
234    /// dropped (with a warning already logged by [`resolve_entry`]) rather than escalated,
235    /// per FR-006's extra-specific rule.
236    extras: Vec<Result<PypiIndexUrl, InvalidEntry>>,
237    /// uv's `default = true` index, if any (uv permits at most one) — uv's lowest-priority,
238    /// last-resort hop, replacing the implicit public fallback in that final slot. `None` for
239    /// every non-uv config, and for a uv config that declares no `default` entry (the
240    /// implicit public fallback is used instead).
241    tail_hop: Option<Result<PypiIndexUrl, InvalidEntry>>,
242    /// Poetry `[[tool.poetry.source]]` entries (all priorities, including `explicit`) keyed
243    /// by `name`, plus every uv `[tool.uv.index]` entry keyed by its own `name` — consulted
244    /// only when a dependency declares `source = "<name>"` (Poetry) or an `index = "<name>"`
245    /// uv-sources binding (FR-013), never auto-included in the primary/extras/tail chain.
246    named_sources: HashMap<String, Result<PypiIndexUrl, InvalidEntry>>,
247}
248
249impl PypiIndexConfig {
250    /// An empty config — every dependency resolves to plain [`DependencySource::Registry`],
251    /// byte-identical to today (US-004).
252    #[must_use]
253    pub fn new() -> Self {
254        Self::default()
255    }
256
257    /// FR-002: sets (overwriting any prior value) the explicit `--index-url`/Poetry
258    /// `primary`-equivalent primary. Matches pip's own `argparse` `store` semantics for
259    /// `--index-url` (not `append`, unlike `--extra-index-url`) — the last occurrence in a
260    /// file wins.
261    pub fn set_primary(&mut self, raw: &str, policy: &RegistryAccessPolicy) {
262        self.primary = Some(resolve_entry(raw, policy));
263    }
264
265    /// Like [`Self::set_primary`], but for a caller that already resolved (or is re-using an
266    /// already-resolved) candidate — avoids re-validating and double-logging a value that
267    /// must also be reachable as a named source (Poetry/uv).
268    ///
269    /// **First registration wins** (validator finding S3, fixes a silent-overwrite bug): a
270    /// second call is ignored, with a `tracing::warn!` naming the raw value that got dropped
271    /// — unlike [`Self::set_primary`] (`requirements.txt`'s `--index-url`, where pip's own
272    /// `argparse` `store` semantics make *last*-wins correct), Poetry has no documented
273    /// ordering for multiple `primary`/`default`-priority (or unlabeled) `[[tool.poetry.source]]`
274    /// entries, so silently picking whichever happened to parse last is an arbitrary,
275    /// non-deterministic-looking outcome — keeping the first and logging every later one is
276    /// deterministic and discoverable instead.
277    pub(crate) fn set_primary_resolved(&mut self, result: Result<PypiIndexUrl, InvalidEntry>) {
278        if self.primary.is_some() {
279            // `Ok(url)` is a *validated* index, but `PypiIndexUrl::new` only rejects
280            // userinfo, not a query string/fragment — a legitimate token embedded as
281            // `?token=...` survives validation and must still be redacted here, the same
282            // way `deps-nuget`'s `fail_closed` redacts its own already-valid `Ok` arm.
283            let raw = match &result {
284                Ok(url) => RedactedUrl::new(url.as_str()),
285                Err(invalid) => invalid.raw.clone(),
286            };
287            tracing::warn!(
288                raw = %raw,
289                "multiple primary-priority index sources declared; keeping the first, \
290                 ignoring this one"
291            );
292            return;
293        }
294        self.primary = Some(result);
295    }
296
297    /// FR-003: appends an `--extra-index-url`/Poetry supplemental-priority/uv non-default
298    /// entry to the fallback chain, in declaration order.
299    pub fn add_extra(&mut self, raw: &str, policy: &RegistryAccessPolicy) {
300        self.extras.push(resolve_entry(raw, policy));
301    }
302
303    /// Like [`Self::add_extra`], but for an already-resolved candidate — see
304    /// [`Self::set_primary_resolved`]'s rationale.
305    pub(crate) fn add_extra_resolved(&mut self, result: Result<PypiIndexUrl, InvalidEntry>) {
306        self.extras.push(result);
307    }
308
309    /// FR-013: sets (overwriting any prior value — uv permits at most one) uv's
310    /// `default = true` index, an already-resolved candidate.
311    pub(crate) fn set_tail_hop_resolved(&mut self, result: Result<PypiIndexUrl, InvalidEntry>) {
312        self.tail_hop = Some(result);
313    }
314
315    /// FR-007/FR-013: registers a Poetry/uv named source, reachable via
316    /// [`Self::resolve_source_for`] with `Some(name)`. An already-resolved candidate — see
317    /// [`Self::set_primary_resolved`]'s rationale.
318    pub(crate) fn add_named_source_resolved(
319        &mut self,
320        name: String,
321        result: Result<PypiIndexUrl, InvalidEntry>,
322    ) {
323        self.named_sources.insert(name, result);
324    }
325
326    /// The valid (`Ok`) subset of [`Self::extras`], in declaration order.
327    fn valid_extras(&self) -> Vec<PypiIndexUrl> {
328        self.extras
329            .iter()
330            .filter_map(|e| e.as_ref().ok())
331            .cloned()
332            .collect()
333    }
334
335    /// The effective case-(b) chain (spec FR-005(b)), or `None` when this config has no
336    /// case-(b) declaration at all (no extras, no uv tail hop) — the "nothing declared"
337    /// state, distinct from "declared but every hop turned out invalid" (the zero-hop case,
338    /// which returns `Some` with an empty `hops`).
339    ///
340    /// An invalid uv `default` entry (`tail_hop` is `Some(Err(_))`) degrades to the implicit
341    /// public fallback rather than being dropped with no replacement — the same
342    /// fail-toward-availability rule FR-006 applies to every other extra.
343    fn case_b_chain(&self) -> Option<CaseBChain> {
344        if self.extras.is_empty() && self.tail_hop.is_none() {
345            return None;
346        }
347        let mut hops = self.valid_extras();
348        let implicit_public_fallback = match &self.tail_hop {
349            Some(Ok(tail)) => {
350                hops.push(tail.clone());
351                false
352            }
353            Some(Err(_)) | None => true,
354        };
355        Some(CaseBChain {
356            hops,
357            implicit_public_fallback,
358        })
359    }
360
361    /// FR-002/FR-003/FR-005/FR-006/FR-007/FR-013: resolves one dependency's
362    /// [`DependencySource`].
363    ///
364    /// `named_source` is `Some("internal")` for a dependency declaring `source = "internal"`
365    /// (Poetry) or an `index = "internal"` uv-sources binding; `None` for every other
366    /// dependency (routes through `primary`/`extras`/`tail_hop` per FR-005 instead).
367    ///
368    /// - A named-source reference resolves to that source's own URL, or
369    ///   [`DependencySource::CustomRegistry`] if the name is unresolved or the source is
370    ///   invalid (fail-closed, never a silent `pypi.org` fallback — FR-006).
371    /// - No override anywhere (no primary, no case-(b) declaration) -> plain
372    ///   [`DependencySource::Registry`] (US-004).
373    /// - An explicit primary present but invalid -> [`DependencySource::CustomRegistry`]
374    ///   (fail-closed — never falls through to the extras chain, matching an explicit
375    ///   `--index-url`'s *replace*, not *add*, semantics).
376    /// - Otherwise -> [`DependencySource::AlternateRegistry`] pointing at the chain
377    ///   [`ResolvedChain::key`] this same config's [`Self::resolved_chains`] registers.
378    #[must_use]
379    pub fn resolve_source_for(&self, named_source: Option<&str>) -> DependencySource {
380        if let Some(name) = named_source {
381            return match self.named_sources.get(name) {
382                Some(Ok(url)) => DependencySource::AlternateRegistry {
383                    index: url.as_str().to_string(),
384                    mirrors_crates_io: false,
385                },
386                Some(Err(invalid)) => DependencySource::CustomRegistry {
387                    url: invalid.raw.to_string(),
388                },
389                None => DependencySource::CustomRegistry {
390                    url: name.to_string(),
391                },
392            };
393        }
394
395        match &self.primary {
396            Some(Ok(primary)) => {
397                let mut hops = vec![primary.clone()];
398                hops.extend(self.valid_extras());
399                DependencySource::AlternateRegistry {
400                    index: ResolvedChain::chain(hops, false).key,
401                    mirrors_crates_io: false,
402                }
403            }
404            Some(Err(invalid)) => DependencySource::CustomRegistry {
405                url: invalid.raw.to_string(),
406            },
407            None => match self.case_b_chain() {
408                Some(chain) if !chain.hops.is_empty() => DependencySource::AlternateRegistry {
409                    index: ResolvedChain::chain(chain.hops, chain.implicit_public_fallback).key,
410                    mirrors_crates_io: false,
411                },
412                _ => DependencySource::Registry,
413            },
414        }
415    }
416
417    /// Every chain this config implies, ready for registration — FR-005(a)/(b) resolved to
418    /// concrete hop lists, plus one single-hop chain per valid named source. Empty when the
419    /// file declares nothing (US-004) or when every case-(b) hop turned out invalid with no
420    /// explicit primary (N5's zero-hop case — [`Self::resolve_source_for`] returns plain
421    /// `Registry` in that case, and there is nothing to register).
422    #[must_use]
423    pub fn resolved_chains(&self) -> Vec<ResolvedChain> {
424        let mut chains = Vec::new();
425
426        match &self.primary {
427            Some(Ok(primary)) => {
428                let mut hops = vec![primary.clone()];
429                hops.extend(self.valid_extras());
430                chains.push(ResolvedChain::chain(hops, false));
431            }
432            Some(Err(_)) => {}
433            None => {
434                if let Some(chain) = self.case_b_chain()
435                    && !chain.hops.is_empty()
436                {
437                    chains.push(ResolvedChain::chain(
438                        chain.hops,
439                        chain.implicit_public_fallback,
440                    ));
441                }
442            }
443        }
444
445        for url in self.named_sources.values().flatten() {
446            chains.push(ResolvedChain::named_source(url.clone()));
447        }
448
449        chains
450    }
451}
