1//! PyPI registry client.
2//!
3//! Provides access to the PyPI registry via:
4//! - Simple API (<https://pypi.org/simple/{package}/>), PEP 691 JSON variant,
5//!   for version lookups (`get_versions`) — smaller than the full JSON API
6//! - Package metadata API (<https://pypi.org/pypi/{package}/json>) for hover
7//!   metadata (`get_package_metadata`), which needs `summary`/`project_urls`
8//!   that the Simple API doesn't carry
9//!
10//! All HTTP requests are cached aggressively using ETag/Last-Modified headers.
11
12use crate::types::{PypiPackage, PypiVersion};
13use deps_core::{DepsError, HttpCache, Result};
14use pep440_rs::{Version, VersionSpecifiers};
15use serde::Deserialize;
16use std::any::Any;
17use std::str::FromStr;
18use std::sync::Arc;

20const PYPI_BASE: &str = "https://pypi.org/pypi";
21
22/// Base URL for the PEP 691 Simple API, used by `get_versions`.
23const PYPI_SIMPLE_BASE: &str = "https://pypi.org/simple";
24
25/// `Accept` header requesting the PEP 691 Simple API JSON representation.
26/// Roughly a third smaller than the full JSON API (verified against
27/// `django`: 619,755 bytes full vs 411,376 bytes Simple API).
28const SIMPLE_API_ACCEPT: &str = "application/vnd.pypi.simple.v1+json";
29
30/// Display name for PyPI used in not-found and API-response error messages.
31pub const REGISTRY: &str = "PyPI";
32
33/// Base URL for package pages on pypi.org
34pub const PYPI_URL: &str = "https://pypi.org/project";
35
36/// Normalize package name according to PEP 503.
37///
38/// Converts package name to lowercase and replaces underscores/dots with hyphens,
39/// then filters out consecutive hyphens. This ensures consistent package lookups
40/// regardless of how the package name is written.
41///
42/// # Examples
43///
44/// ```
45/// # use deps_pypi::registry::normalize_package_name;
46/// assert_eq!(normalize_package_name("Flask"), "flask");
47/// assert_eq!(normalize_package_name("django_rest_framework"), "django-rest-framework");
48/// assert_eq!(normalize_package_name("Pillow.Image"), "pillow-image");
49/// assert_eq!(normalize_package_name("my__package"), "my-package");
50/// ```
51pub fn normalize_package_name(name: &str) -> String {
52    name.to_lowercase()
53        .replace(&['_', '.'][..], "-")
54        .split('-')
55        .filter(|s| !s.is_empty())
56        .collect::<Vec<_>>()
57        .join("-")
58}
59
60/// Returns the URL for a package's page on pypi.org.
61///
62/// Package names are normalized and URL-encoded to prevent path traversal attacks.
63pub fn package_url(name: &str) -> String {
64    let normalized = normalize_package_name(name);
65    format!("{}/{}", PYPI_URL, urlencoding::encode(&normalized))
66}
67
68/// Builds the Simple API request URL for `normalized`'s version listing.
69///
70/// The name segment is URL-encoded (matching `package_url`) since
71/// `normalize_package_name` only collapses `-`/`_`/`.` separators and leaves
72/// characters like `/`, `?`, `#` untouched.
73fn simple_api_url(normalized: &str) -> String {
74    format!("{PYPI_SIMPLE_BASE}/{}/", urlencoding::encode(normalized))
75}
76
77/// Builds the JSON API request URL for `normalized`'s package metadata.
78fn metadata_url(normalized: &str) -> String {
79    format!("{PYPI_BASE}/{}/json", urlencoding::encode(normalized))
80}
81
82/// Converts a 404 response into `DepsError::PackageNotFound`, passing through
83/// any other error unchanged.
84fn not_found_or(err: DepsError, name: &str) -> DepsError {
85    if matches!(err, DepsError::HttpStatus { status: 404, .. }) {
86        DepsError::PackageNotFound {
87            package: name.to_string(),
88            registry: REGISTRY,
89        }
90    } else {
91        err
92    }
93}

95/// Client for interacting with the PyPI registry.
96///
97/// Uses the PyPI JSON API for package metadata.
98/// All requests are cached via the provided HttpCache.
99///
100/// # Examples
101///
102/// ```no_run
103/// # use deps_pypi::PypiRegistry;
104/// # use deps_core::HttpCache;
105/// # use std::sync::Arc;
106/// # #[tokio::main]
107/// # async fn main() {
108/// let cache = Arc::new(HttpCache::new());
109/// let registry = PypiRegistry::new(cache);
110///
111/// let versions = registry.get_versions("requests").await.unwrap();
112/// assert!(!versions.is_empty());
113/// # }
114/// ```
115#[derive(Clone)]
116pub struct PypiRegistry {
117    cache: Arc<HttpCache>,
118}
119
120impl PypiRegistry {
121    /// Creates a new PyPI registry client with the given HTTP cache.
122    pub const fn new(cache: Arc<HttpCache>) -> Self {
123        Self { cache }
124    }
125
126    /// Fetches all versions for a package from PyPI's Simple API (PEP 691).
127    ///
128    /// Requests the JSON representation (`Accept:
129    /// application/vnd.pypi.simple.v1+json`), which is smaller than the full
130    /// JSON API and provides the version list directly, without needing to
131    /// derive versions from release-file names.
132    ///
133    /// Returns versions sorted newest-first. Filters out yanked versions by default.
134    ///
135    /// # Errors
136    ///
137    /// Returns an error if:
138    /// - HTTP request fails
139    /// - Response body is invalid UTF-8
140    /// - JSON parsing fails
141    /// - Package does not exist
142    ///
143    /// # Examples
144    ///
145    /// ```no_run
146    /// # use deps_pypi::PypiRegistry;
147    /// # use deps_core::HttpCache;
148    /// # use std::sync::Arc;
149    /// # #[tokio::main]
150    /// # async fn main() {
151    /// let cache = Arc::new(HttpCache::new());
152    /// let registry = PypiRegistry::new(cache);
153    ///
154    /// let versions = registry.get_versions("flask").await.unwrap();
155    /// assert!(!versions.is_empty());
156    /// # }
157    /// ```
158    pub async fn get_versions(&self, name: &str) -> Result<Vec<PypiVersion>> {
159        let normalized = normalize_package_name(name);
160        let url = simple_api_url(&normalized);
161        let data = self
162            .cache
163            .get_cached_with_headers(&url, &[(reqwest::header::ACCEPT, SIMPLE_API_ACCEPT)])
164            .await
165            .map_err(|e| not_found_or(e, name))?;
166
167        parse_simple_api_response(name, &data)
168    }
169
170    /// Finds the latest version matching the given PEP 440 version specifier.
171    ///
172    /// Only returns non-yanked, non-prerelease versions by default.
173    ///
174    /// # Errors
175    ///
176    /// Returns an error if:
177    /// - HTTP request fails
178    /// - Package does not exist
179    /// - Version specifier is invalid
180    ///
181    /// # Examples
182    ///
183    /// ```no_run
184    /// # use deps_pypi::PypiRegistry;
185    /// # use deps_core::HttpCache;
186    /// # use std::sync::Arc;
187    /// # #[tokio::main]
188    /// # async fn main() {
189    /// let cache = Arc::new(HttpCache::new());
190    /// let registry = PypiRegistry::new(cache);
191    ///
192    /// let latest = registry.get_latest_matching("flask", ">=3.0,<4.0").await.unwrap();
193    /// assert!(latest.is_some());
194    /// # }
195    /// ```
196    pub async fn get_latest_matching(
197        &self,
198        name: &str,
199        req_str: &str,
200    ) -> Result<Option<PypiVersion>> {
201        let versions = self.get_versions(name).await?;
202
203        // PEP 440 uses empty string for "any version"
204        let normalized_req = if req_str == "*" { "" } else { req_str };
205
206        let specs = VersionSpecifiers::from_str(normalized_req)
207            .map_err(|e| DepsError::InvalidVersionReq(format!("{req_str}: {e}")))?;
208
209        Ok(versions.into_iter().find(|v| {
210            if let Ok(version) = Version::from_str(&v.version) {
211                specs.contains(&version) && !v.yanked && !v.is_prerelease()
212            } else {
213                false
214            }
215        }))
216    }
217
218    /// Searches for packages by name/keywords.
219    ///
220    /// Note: PyPI does not provide an official search API, so this returns
221    /// an empty result for now. Future implementation could use third-party
222    /// search services or scraping.
223    ///
224    /// # Errors
225    ///
226    /// Currently always returns Ok with empty vector.
227    ///
228    /// # Examples
229    ///
230    /// ```no_run
231    /// # use deps_pypi::PypiRegistry;
232    /// # use deps_core::HttpCache;
233    /// # use std::sync::Arc;
234    /// # #[tokio::main]
235    /// # async fn main() {
236    /// let cache = Arc::new(HttpCache::new());
237    /// let registry = PypiRegistry::new(cache);
238    ///
239    /// let results = registry.search("flask", 10).await.unwrap();
240    /// // Currently returns empty, to be implemented
241    /// # }
242    /// ```
243    pub async fn search(&self, _query: &str, _limit: usize) -> Result<Vec<PypiPackage>> {
244        // TODO: Implement search using third-party API or scraping
245        // PyPI deprecated their XML-RPC search API
246        Ok(Vec::new())
247    }
248
249    /// Fetches package metadata including description and project URLs.
250    ///
251    /// # Errors
252    ///
253    /// Returns an error if:
254    /// - HTTP request fails
255    /// - Package does not exist
256    /// - JSON parsing fails
257    pub async fn get_package_metadata(&self, name: &str) -> Result<PypiPackage> {
258        let normalized = normalize_package_name(name);
259        let url = metadata_url(&normalized);
260        let data = self
261            .cache
262            .get_cached(&url)
263            .await
264            .map_err(|e| not_found_or(e, name))?;
265
266        parse_package_info(name, &data)
267    }
268}

270// Implement Registry trait for PypiRegistry
271impl deps_core::Registry for PypiRegistry {
272    fn get_versions<'a>(
273        &'a self,
274        name: &'a str,
275    ) -> deps_core::ecosystem::BoxFuture<
276        'a,
277        deps_core::error::Result<Vec<Box<dyn deps_core::Version>>>,
278    > {
279        Box::pin(async move {
280            let versions = Self::get_versions(self, name).await?;
281            Ok(versions
282                .into_iter()
283                .map(|v| Box::new(v) as Box<dyn deps_core::Version>)
284                .collect())
285        })
286    }
287
288    fn get_latest_matching<'a>(
289        &'a self,
290        name: &'a str,
291        req: &'a str,
292    ) -> deps_core::ecosystem::BoxFuture<
293        'a,
294        deps_core::error::Result<Option<Box<dyn deps_core::Version>>>,
295    > {
296        Box::pin(async move {
297            let version = Self::get_latest_matching(self, name, req).await?;
298            Ok(version.map(|v| Box::new(v) as Box<dyn deps_core::Version>))
299        })
300    }
301
302    fn search<'a>(
303        &'a self,
304        query: &'a str,
305        limit: usize,
306    ) -> deps_core::ecosystem::BoxFuture<
307        'a,
308        deps_core::error::Result<Vec<Box<dyn deps_core::Metadata>>>,
309    > {
310        Box::pin(async move {
311            let packages = Self::search(self, query, limit).await?;
312            Ok(packages
313                .into_iter()
314                .map(|p| Box::new(p) as Box<dyn deps_core::Metadata>)
315                .collect())
316        })
317    }
318
319    fn package_url(&self, name: &str) -> String {
320        package_url(name)
321    }
322
323    fn as_any(&self) -> &dyn Any {
324        self
325    }

358/// A file's yanked status per PEP 592: either `false` (not yanked) or a
359/// string giving the yank reason. `get_versions` only needs the yanked/not
360/// distinction, so the reason text itself is discarded on parse.
361#[derive(Debug, Deserialize)]
362#[serde(untagged)]
363enum Yanked {
364    Flag(bool),
365    Reason(#[expect(dead_code, reason = "reason text not surfaced by get_versions")] String),
366}
367
368impl Default for Yanked {
369    fn default() -> Self {
370        Self::Flag(false)
371    }
372}
373
374impl Yanked {
375    const fn is_yanked(&self) -> bool {
376        match self {
377            Self::Flag(b) => *b,
378            Self::Reason(_) => true,
379        }
380    }
381}

383/// Archive/wheel file extensions recognized on PyPI's Simple API index, used
384/// to strip the trailing extension off an sdist-style filename (one with no
385/// further `-`-delimited tags after the version) when deriving its version.
386const KNOWN_ARCHIVE_EXTENSIONS: &[&str] = &[
387    ".tar.gz", ".tar.bz2", ".tar.xz", ".tar.lz", ".tar.Z", ".zip", ".whl", ".egg", ".tar",
388];
389
390/// Derives the release version directly from `filename`'s structure, given
391/// the package's PEP 503 normalized name, in O(filename length) with no
392/// dependency on the number of known versions.
393///
394/// PyPI release filenames are `{name}-{version}[-...].{ext}`, but the
395/// `{name}` segment on disk may use the project's original casing and
396/// original `-`/`_`/`.` separators (e.g. `zope.interface-3.3.0b1.tar.gz` for
397/// normalized name `zope-interface`) rather than the normalized spelling.
398/// This walks `normalized_name` against `filename` byte-for-byte,
399/// case-insensitively, treating any run of `-`/`_`/`.` in `filename` as
400/// equivalent to a single `-` in `normalized_name`, then cuts the remainder
401/// at the first `-` (wheel tags) or a recognized archive extension (sdists).
402///
403/// Returns `None` if `filename` doesn't conform closely enough to derive a
404/// version unambiguously; callers skip attributing that file rather than
405/// guess (see [`build_yanked_map`]).
406fn parse_version_from_filename<'a>(filename: &'a str, normalized_name: &str) -> Option<&'a str> {
407    let bytes = filename.as_bytes();
408    let mut fi = 0usize;
409    for nb in normalized_name.bytes() {
410        if nb == b'-' {
411            let start = fi;
412            while matches!(bytes.get(fi), Some(b'-' | b'_' | b'.')) {
413                fi += 1;
414            }
415            if fi == start {
416                return None;
417            }
418        } else {
419            match bytes.get(fi) {
420                Some(&fb) if fb.to_ascii_lowercase() == nb => fi += 1,
421                _ => return None,
422            }
423        }
424    }
425    if bytes.get(fi) != Some(&b'-') {
426        return None;
427    }
428    let rest = &filename[fi + 1..];
429    match rest.find('-') {
430        Some(end) => Some(&rest[..end]),
431        None => KNOWN_ARCHIVE_EXTENSIONS
432            .iter()
433            .find_map(|ext| rest.strip_suffix(ext)),
434    }
435}

437/// Builds a per-version yanked map from a Simple API `files` list.
438///
439/// A version is yanked if any of its release files are yanked (PyPI itself
440/// treats a release as yanked once any file under it is, since new uploads
441/// to an already-yanked version are rejected).
442///
443/// Derives each file's version in O(1) via [`parse_version_from_filename`]
444/// and resolves it against `versions` in two tiers, the second tried only
445/// if the first misses:
446/// 1. Exact string match — the common case, and the only tier that can tell
447///    apart distinct-but-PEP-440-equal strings like `1.0` and `1.0.0`
448///    (legacy packages can list both as separate releases).
449/// 2. Match as a parsed [`Version`] — PyPI filenames sometimes spell a
450///    version differently than its canonical form (e.g. `4.21.0_rc_1` in a
451///    wheel filename for the canonical `4.21.0rc1`), which `Version`'s
452///    `Eq`/`Hash` normalize away.
453///
454/// A file whose filename doesn't structurally parse, or whose derived
455/// version matches neither tier, is skipped rather than guessed at via
456/// substring search — an earlier substring-based fallback could misattribute
457/// a file to an unrelated version whose digits happened to appear elsewhere
458/// in the filename (e.g. a platform tag), which is worse than the version's
459/// yanked status resting on its other, better-formed release files.
460fn build_yanked_map(
461    files: &[SimpleFile],
462    versions: &[String],
463    normalized_name: &str,
464) -> std::collections::HashMap<String, bool> {
465    let version_set: std::collections::HashSet<&str> =
466        versions.iter().map(String::as_str).collect();
467
468    // Built lazily: most real-world responses resolve every file via the
469    // exact-string tier and never need this.
470    let mut parsed_versions: Option<std::collections::HashMap<Version, &str>> = None;
471
472    let mut yanked: std::collections::HashMap<String, bool> = std::collections::HashMap::new();
473    for file in files {
474        let matched =
475            parse_version_from_filename(&file.filename, normalized_name).and_then(|candidate| {
476                version_set.get(candidate).copied().or_else(|| {
477                    let parsed = Version::from_str(candidate).ok()?;
478                    let map = parsed_versions.get_or_insert_with(|| {
479                        versions
480                            .iter()
481                            .filter_map(|v| Some((Version::from_str(v).ok()?, v.as_str())))
482                            .collect()
483                    });
484                    map.get(&parsed).copied()
485                })
486            });
487
488        let Some(version) = matched else {
489            continue;
490        };
491        let entry = yanked.entry(version.to_string()).or_insert(false);
492        *entry |= file.yanked.is_yanked();
493    }
494    yanked
495}

497/// Parse the version list from a PyPI Simple API (PEP 691) JSON response.
498fn parse_simple_api_response(package_name: &str, data: &[u8]) -> Result<Vec<PypiVersion>> {
499    let response: SimpleApiResponse =
500        serde_json::from_slice(data).map_err(|e| DepsError::ApiResponse {
501            package: package_name.to_string(),
502            registry: REGISTRY,
503            source: e,
504        })?;
505
506    let normalized_name = normalize_package_name(package_name);
507    let yanked_map = build_yanked_map(&response.files, &response.versions, &normalized_name);
508
509    let mut versions_with_parsed: Vec<(PypiVersion, Version)> = response
510        .versions
511        .into_iter()
512        .filter_map(|version_str| {
513            let parsed = Version::from_str(&version_str).ok()?;
514            let yanked = yanked_map.get(&version_str).copied().unwrap_or(false);
515            Some((
516                PypiVersion {
517                    version: version_str,
518                    yanked,
519                },
520                parsed,
521            ))
522        })
523        .collect();
524
525    // Sort by version (newest first) using pre-parsed versions
526    versions_with_parsed.sort_by(|a, b| b.1.cmp(&a.1));
527
528    Ok(versions_with_parsed.into_iter().map(|(v, _)| v).collect())
529}

531/// Parse package info from PyPI JSON response.
532fn parse_package_info(package_name: &str, data: &[u8]) -> Result<PypiPackage> {
533    let response: PypiResponse =
534        serde_json::from_slice(data).map_err(|e| DepsError::ApiResponse {
535            package: package_name.to_string(),
536            registry: REGISTRY,
537            source: e,
538        })?;
539
540    let project_urls = response
541        .info
542        .project_urls
543        .unwrap_or_default()
544        .into_iter()
545        .collect();
546
547    Ok(PypiPackage {
548        name: response.info.name,
549        summary: response.info.summary,
550        project_urls,
551        latest_version: response.info.version,
552    })
553}
