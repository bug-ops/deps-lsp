21/// Builds a completion item that inserts the full GitHub URL for `.package(url: "...")`.
22///
23/// The completion fires with the cursor inside the `url:` string literal (see
24/// [`SwiftEcosystem::generate_completions`]), so the insertable text must always be a
25/// full URL — never the bare `owner/repo` identity — regardless of how much of the
26/// scheme the user has typed so far. `replace_range` should be the dependency's
27/// `name_range()` (the byte span of the whole URL literal) whenever the caller can resolve
28/// it — the base builder's own range is a placeholder `(0,0)-(0,0)` that does not contain
29/// the real cursor position and would corrupt the document if used as-is. When `None` (the
30/// dependency containing the cursor could not be found), falls back to `insert_text`-only
31/// — the same safe pattern used by `create_package_completion_item` in `deps-lsp` — rather
32/// than guessing a range.
33///
34/// Returns `None` when `url` doesn't pass [`is_safe_registry_url`], or when the base
35/// builder itself rejects `package.name` — a malicious/compromised search result must
36/// not reach the manifest as an unsanitized `TextEdit`, so the item is dropped rather
37/// than built with unsafe text.
38fn build_url_completion(
39    package: &SwiftPackage,
40    replace_range: Option<LspRange>,
41) -> Option<CompletionItem> {
42    let url = package
43        .repository
44        .clone()
45        .unwrap_or_else(|| format!("https://github.com/{}", package.name));
46
47    if !is_safe_registry_url(&url) {
48        warn_rejected_value("is_safe_registry_url", "swift url completion", &url);
49        return None;
50    }
51
52    let mut item = deps_core::completion::build_package_completion(package, LspRange::default())?;
53
54    item.insert_text = Some(url.clone());
55    item.filter_text = Some(url.clone());
56    item.sort_text = Some(url.clone());
57    item.text_edit = replace_range.map(|range| {
58        CompletionTextEdit::Edit(TextEdit {
59            range,
60            new_text: url,
61        })
62    });
63
64    Some(item)
65}
66
67/// Strips a leading `https://github.com/` (or `https://github.com`) scheme from a
68/// completion prefix, leaving the search query GitHub's repository search expects.
69fn strip_github_prefix(prefix: &str) -> &str {
70    prefix
71        .strip_prefix("https://github.com/")
72        .or_else(|| prefix.strip_prefix("https://github.com"))
73        .unwrap_or(prefix)
74}

76/// Swift/SPM ecosystem implementation.
77///
78/// Provides LSP functionality for Package.swift files, including:
79/// - Dependency parsing with position tracking
80/// - Version information from GitHub tags
81/// - Inlay hints for latest versions
82/// - Hover tooltips with package metadata
83/// - Code actions for version updates
84/// - Diagnostics for unknown packages
85pub struct SwiftEcosystem {
86    registry: Arc<SwiftRegistry>,
87    formatter: SwiftFormatter,
88    lockfile_provider: Arc<SwiftLockParser>,
89}

91impl SwiftEcosystem {
92    /// Creates a new Swift ecosystem with the given HTTP cache.
93    pub fn new(cache: Arc<deps_core::HttpCache>) -> Self {
94        Self {
95            registry: Arc::new(SwiftRegistry::new(cache)),
96            formatter: SwiftFormatter,
97            lockfile_provider: Arc::new(SwiftLockParser),
98        }
99    }
100
101    async fn complete_package_urls(
102        &self,
103        query: &str,
104        replace_range: Option<LspRange>,
105    ) -> Vec<CompletionItem> {
106        if !deps_core::completion::is_valid_completion_prefix_len(query) {
107            return vec![];
108        }
109
110        let results = match self.registry.search(query, 20).await {
111            Ok(r) => r,
112            Err(e) => {
113                tracing::warn!("Swift registry search failed for '{}': {}", query, e);
114                return vec![];
115            }
116        };
117
118        results
119            .iter()
120            .filter_map(|package| build_url_completion(package, replace_range))
121            .collect()
122    }
123
124    async fn complete_versions(
125        &self,
126        package_name: &deps_core::PackageName,
127        prefix: &str,
128        freshness: deps_core::FreshnessSettings,
129    ) -> Vec<CompletionItem> {
130        deps_core::completion::complete_versions_generic(
131            self.registry.as_ref(),
132            package_name,
133            prefix,
134            &[],
135            freshness,
136        )
137        .await
138    }
139}

143impl Ecosystem for SwiftEcosystem {
144    fn id(&self) -> &'static str {
145        "swift"
146    }
147
148    fn display_name(&self) -> &'static str {
149        "Swift (SPM)"
150    }
151
152    fn manifest_filenames(&self) -> &[&'static str] {
153        &["Package.swift"]
154    }
155
156    fn lockfile_filenames(&self) -> &[&'static str] {
157        &["Package.resolved"]
158    }
159
160    fn parse_manifest<'a>(
161        &'a self,
162        content: &'a str,
163        uri: &'a Uri,
164    ) -> deps_core::ecosystem::BoxFuture<'a, Result<Box<dyn ParseResultTrait>>> {
165        Box::pin(async move {
166            let result = crate::parser::parse_package_swift(content, uri)?;
167            Ok(Box::new(result) as Box<dyn ParseResultTrait>)
168        })
169    }
170
171    fn registry(&self) -> Arc<dyn Registry> {
172        self.registry.clone() as Arc<dyn Registry>
173    }
174
175    fn lockfile_provider(&self) -> Option<Arc<dyn deps_core::lockfile::LockFileProvider>> {
176        Some(self.lockfile_provider.clone() as Arc<dyn deps_core::lockfile::LockFileProvider>)
177    }
178
179    fn formatter(&self) -> &dyn EcosystemFormatter {
180        &self.formatter
181    }
182
183    fn generate_completions<'a>(
184        &'a self,
185        parse_result: &'a dyn ParseResultTrait,
186        position: Position,
187        content: &'a str,
188        freshness: deps_core::FreshnessSettings,
189    ) -> deps_core::ecosystem::BoxFuture<'a, Completions> {
190        Box::pin(async move {
191            use deps_core::completion::{CompletionContext, detect_completion_context};
192
193            let context = detect_completion_context(parse_result, position, content);
194
195            match context {
196                CompletionContext::PackageName { prefix, range } => {
197                    // The completion context only fires with the cursor inside an existing
198                    // dependency's url: "..." literal (see module docs), so `range` (the
199                    // dependency's `name_range()`, computed by `detect_completion_context`)
200                    // is already the exact span the completion must replace.
201                    self.complete_package_urls(strip_github_prefix(&prefix), Some(range))
202                        .await
203                }
204                CompletionContext::Version {
205                    package_name,
206                    prefix,
207                } => {
208                    self.complete_versions(&package_name, &prefix, freshness)
209                        .await
210                }
211                CompletionContext::Feature { .. } => vec![],
212                CompletionContext::None => vec![],
213            }
214            .into()
215        })
216    }
217
218    fn as_any(&self) -> &dyn Any {
219        self
220    }
221}
