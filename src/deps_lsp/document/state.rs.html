47/// State for a single open document.
48///
49/// Stores the document content, parsed dependency information, and cached
50/// version data for a single file. The state is updated when the document
51/// changes or when version information is fetched from the registry.
52///
53/// Supports multiple package ecosystems via the trait-based `ParseResult`.
54///
55/// # Examples
56///
57/// ```no_run
58/// use deps_core::EcosystemId;
59/// use deps_lsp::document::DocumentState;
60///
61/// let state = DocumentState::new_without_parse_result(
62///     EcosystemId::Cargo,
63///     "[dependencies]\nserde = \"1.0\"".into(),
64/// );
65///
66/// assert!(state.cached_versions.is_empty());
67/// ```
68pub struct DocumentState {
69    /// Package ecosystem identifier, exhaustively typed.
70    pub ecosystem: EcosystemId,
71    /// Original document content
72    pub content: String,
73    /// Parsed result as trait object, wrapped in `Arc` (rather than `Box`) so
74    /// [`Self::parse_result_arc`] can hand a caller a cheap owned clone — letting it
75    /// release the DashMap shard `Ref` before an `.await` on a registry-bound
76    /// `generate_*` call without deep-cloning ecosystem-specific parse data (#319).
77    parse_result: Option<Arc<dyn ParseResult>>,
78    /// Latest known version and full version list per package, fetched together in a
79    /// single registry round trip (see [`PackageVersions`]).
80    pub cached_versions: HashMap<PackageName, PackageVersions>,
81    /// Resolved versions from lock file
82    pub resolved_versions: HashMap<PackageName, ConcreteVersion>,
83    /// OSV.dev scan results, keyed by normalized package name. Empty until
84    /// the first background scan completes; carried across document edits
85    /// by `preserve_cache` so it is not wiped on every keystroke.
86    pub vulnerabilities: VulnerabilityMap,
87    /// Yanked, deprecation, and fetch-failure findings from the lifecycle's registry
88    /// fetch, keyed by **normalized** package name. This is deliberately a different
89    /// type from `FetchResult`'s raw-keyed triple: the split makes a forgotten
90    /// normalization at a store/merge site a compile error rather than a silent bug for
91    /// ecosystems where normalization changes the name (e.g. PyPI). See
92    /// [`DependencyOutcome`](deps_core::DependencyOutcome) for what each of the three
93    /// channels means. Empty until the first fetch completes; carried across document
94    /// edits by `preserve_cache` so it doesn't flicker off on every keystroke.
95    pub outcomes: DependencyOutcomes,
96    /// Last successful parse time
97    pub parsed_at: Instant,
98    /// Current loading state for registry data
99    pub loading_state: LoadingState,
100    /// When the current loading operation started (for timeout/metrics)
101    pub loading_started_at: Option<Instant>,
102    /// LSP document version from the client's `didOpen`/`didChange`, `None` if this
103    /// state was populated from disk (cold start) rather than an LSP notification.
104    ///
105    /// Threaded into `WorkspaceEdit.document_changes` so the client can reject a batch
106    /// edit whose ranges were computed against a buffer state it has since moved past
107    /// (see `handlers::code_lens`).
108    pub version: Option<i32>,
109}

131/// Tracks recent cold start attempts per URI to prevent DOS.
132///
133/// Uses rate limiting with a configurable minimum interval between
134/// cold start attempts for the same URI. This prevents malicious or
135/// buggy clients from overwhelming the server with rapid file loading
136/// requests.
137///
138/// # Examples
139///
140/// ```
141/// use deps_lsp::document::ColdStartLimiter;
142/// use std::time::Duration;
143///
144/// let limiter = ColdStartLimiter::new(Duration::from_secs(10));
145/// let uri = deps_core::test_util::test_uri("/test.toml");
146///
147/// assert!(limiter.allow_cold_start(&uri));
148/// assert!(!limiter.allow_cold_start(&uri)); // Rate limited
149/// ```
150#[derive(Debug)]
151pub struct ColdStartLimiter {
152    /// Maps URI to last cold start attempt time.
153    last_attempts: DashMap<Uri, Instant>,
154    /// Minimum interval between cold start attempts for the same URI, in
155    /// milliseconds. Atomic so `set_min_interval` can live-update it from
156    /// `did_change_configuration` (issue #499) without disturbing in-flight
157    /// `allow_cold_start` callers.
158    min_interval_ms: AtomicU64,
159}
160
161impl ColdStartLimiter {
162    /// Creates a new cold start limiter with the specified minimum interval.
163    pub fn new(min_interval: Duration) -> Self {
164        Self {
165            last_attempts: DashMap::new(),
166            min_interval_ms: AtomicU64::new(min_interval.as_millis() as u64),
167        }
168    }
169
170    /// Updates the minimum interval between cold start attempts.
171    ///
172    /// Takes effect on the next `allow_cold_start` call. Used to apply a
173    /// live-reloaded `cold_start.rate_limit_ms` (issue #499).
174    pub fn set_min_interval(&self, min_interval: Duration) {
175        self.min_interval_ms
176            .store(min_interval.as_millis() as u64, Ordering::Relaxed);
177    }
178
179    /// Returns true if cold start is allowed, false if rate limited.
180    ///
181    /// Updates the last attempt time if the cold start is allowed.
182    pub fn allow_cold_start(&self, uri: &Uri) -> bool {
183        let min_interval = Duration::from_millis(self.min_interval_ms.load(Ordering::Relaxed));
184        let now = Instant::now();
185
186        // Check last attempt time
187        if let Some(mut entry) = self.last_attempts.get_mut(uri) {
188            let elapsed = now.duration_since(*entry);
189            if elapsed < min_interval {
190                let retry_after = min_interval.checked_sub(elapsed).unwrap();
191                tracing::warn!(
192                    "Cold start rate limited for {:?} (retry after {:?})",
193                    uri,
194                    retry_after
195                );
196                return false;
197            }
198            *entry = now;
199        } else {
200            self.last_attempts.insert(uri.clone(), now);
201        }
202
203        true
204    }
205
206    /// Cleans up old entries periodically.
207    ///
208    /// Removes entries older than `max_age` to prevent unbounded memory growth.
209    /// Should be called from a background task.
210    pub fn cleanup_old_entries(&self, max_age: Duration) {
211        let now = Instant::now();
212        self.last_attempts
213            .retain(|_, instant| now.duration_since(*instant) < max_age);
214    }
215
216    /// Returns the number of tracked URIs.
217    #[cfg(test)]
218    pub fn tracked_count(&self) -> usize {
219        self.last_attempts.len()
220    }
221}

566/// security M3): `first_queued_at` is set once, by the change that starts a new pending
567/// entry, and survives every later union — a continuous burst of config changes arriving
568/// faster than the debounce window must not starve the reparse forever, so a debounce
569/// worker forces a drain once this timestamp is old enough, regardless of whether it was
570/// itself superseded by a newer generation.
571struct PendingReparse {
572    scope: crate::config::ReparseScope,
573    first_queued_at: Instant,
574}

689    /// Unions `scope` into the pending coalesced reparse and bumps the generation counter
690    /// (issue #592), returning the new generation.
691    ///
692    /// Union happens first, and both operations happen while the same lock is held —
693    /// bumping the generation first would let a debounce worker that wakes between the two
694    /// steps drain a union still missing this call's own scope. `first_queued_at` is set
695    /// only when this call starts a fresh pending entry (`None` -> `Some`); a union onto an
696    /// already-pending entry keeps the original timestamp, so [`Self::pending_reparse_overdue`]
697    /// measures from the *first* unhandled change in a burst, not the latest (security M3:
698    /// otherwise a burst arriving faster than the debounce window could starve the reparse
699    /// indefinitely).
700    pub(crate) fn queue_reparse(&self, scope: crate::config::ReparseScope) -> u64 {
701        {
702            let mut pending = self
703                .pending_reparse
704                .lock()
705                .unwrap_or_else(std::sync::PoisonError::into_inner);
706            *pending = Some(match pending.take() {
707                Some(existing) => PendingReparse {
708                    scope: existing.scope.union(scope),
709                    first_queued_at: existing.first_queued_at,
710                },
711                None => PendingReparse {
712                    scope,
713                    first_queued_at: Instant::now(),
714                },
715            });
716        }
717        self.config_generation.fetch_add(1, Ordering::SeqCst) + 1
718    }

727    /// Whether the pending coalesced reparse has been waiting at least `max_wait` since it
728    /// was first queued (issue #592 security M3). A debounce worker that finds itself
729    /// superseded by a newer config generation normally defers to that newer worker — but
730    /// under a continuous burst arriving faster than the debounce window, every worker would
731    /// see itself superseded forever. Once this returns `true`, the worker that observes it
732    /// must drain and reparse regardless of its own generation being stale, capping the
733    /// worst-case staleness a live-reloaded, security-relevant setting can be left
734    /// unapplied to open documents' rendered data.
735    pub(crate) fn pending_reparse_overdue(&self, max_wait: Duration) -> bool {
736        self.pending_reparse
737            .lock()
738            .unwrap_or_else(std::sync::PoisonError::into_inner)
739            .as_ref()
740            .is_some_and(|pending| pending.first_queued_at.elapsed() >= max_wait)
741    }

743    /// Drains the pending coalesced reparse scope, if any (issue #592).
744    ///
745    /// Returns `None` if no config change is pending — reachable even for a worker that
746    /// just passed its own generation check: one that wakes between a newer change's union
747    /// and its generation bump can lose the race to drain first (see
748    /// [`Self::queue_reparse`]'s ordering). Callers must treat `None` as "nothing to do",
749    /// never `unwrap`/`expect` it.
750    pub(crate) fn take_pending_reparse(&self) -> Option<crate::config::ReparseScope> {
751        self.pending_reparse
752            .lock()
753            .unwrap_or_else(std::sync::PoisonError::into_inner)
754            .take()
755            .map(|pending| pending.scope)
756    }

758    /// Fires `workspace/inlayHint/refresh` and `workspace/codeLens/refresh` as
759    /// detached, capability-gated, timeout-bounded background requests (issue #493).
760    ///
761    /// Neither refresh feeds anything downstream (hover/inlay-hint/code-lens
762    /// handlers recompute on demand from already-committed document state), so a
763    /// failure or timeout is only logged and never blocks the caller's critical
764    /// path — the OSV vulnerability commit and diagnostics publish this is called
765    /// alongside. The capability gate skips clients that never declared support (and
766    /// so may never reply); the timeout additionally bounds a client that declares
767    /// support but stops replying, so detached tasks can't accumulate without limit.
768    pub fn spawn_refresh_requests(&self, client: &Client) {
769        if self.inlay_hint_refresh_supported() {
770            let client = client.clone();
771            tokio::spawn(async move {
772                match tokio::time::timeout(CLIENT_REFRESH_TIMEOUT, client.inlay_hint_refresh())
773                    .await
774                {
775                    Ok(Ok(())) => {}
776                    Ok(Err(e)) => tracing::debug!("inlay_hint_refresh failed: {:?}", e),
777                    Err(_) => tracing::debug!(
778                        "inlay_hint_refresh timed out after {CLIENT_REFRESH_TIMEOUT:?}"
779                    ),
780                }
781            });
782        }
783        if self.code_lens_refresh_supported() {
784            let client = client.clone();
785            tokio::spawn(async move {
786                match tokio::time::timeout(CLIENT_REFRESH_TIMEOUT, client.code_lens_refresh()).await
787                {
788                    Ok(Ok(())) => {}
789                    Ok(Err(e)) => tracing::debug!("code_lens_refresh failed: {:?}", e),
790                    Err(_) => tracing::debug!(
791                        "code_lens_refresh timed out after {CLIENT_REFRESH_TIMEOUT:?}"
792                    ),
793                }
794            });
795        }
796    }

812    /// Extracts owned data from a document without exposing the DashMap shard `Ref`
813    /// to the caller.
814    ///
815    /// `extract` runs synchronously while the shard lock is held and must return only
816    /// owned or `Arc`-cloned data (e.g. via [`DocumentState::parse_result_arc`]); the
817    /// `Ref` this method acquires is dropped before the call returns, so *that*
818    /// particular guard can never leak across an `.await` through `T`. This does not by
819    /// itself prevent `extract` from independently capturing and returning some other,
820    /// unrelated `Ref` (e.g. from a second `get_document` call on `state`) — `extract`'s
821    /// closure environment is not restricted to this method's own guard. The
822    /// project-wide backstop against the DashMap Ref-across-await hazard (#333) in
823    /// general is the `await-holding-invalid-types` lint configured in the workspace
824    /// `clippy.toml` (#334), not this method's type signature alone.
825    ///
826    /// # Examples
827    ///
828    /// ```no_run
829    /// # use deps_lsp::document::ServerState;
830    /// # use tower_lsp_server::ls_types::Uri;
831    /// # async fn example(state: &ServerState, uri: &Uri) {
832    /// let content_len = state.with_document(uri, |doc| doc.content.len());
833    /// # }
834    /// ```

836        &self,
837        uri: &Uri,
838        extract: impl FnOnce(&DocumentState) -> T,
839    ) -> Option<T> {
840        self.documents.get(uri).map(|doc| extract(&doc))

891    /// Spawns a background task for a document, supervising it for panics.
892    ///
893    /// If a task already exists for the given URI, it is aborted before the new task is
894    /// registered. This ensures only one background task runs per document.
895    ///
896    /// Typical use case: fetching version data asynchronously after document open or
897    /// change.
898    ///
899    /// Takes `self: &Arc<Self>` (rather than `&self`) because this also spawns a
900    /// supervisor task that awaits `task` for its whole lifetime (issue #632): if the
901    /// background fetch task panics instead of reaching `DocumentState::set_loaded`/
902    /// `set_failed`, the document's `loading_state` would otherwise stay `Loading`
903    /// forever, permanently suppressing diagnostics for it. The supervisor forces
904    /// `loading_state` to `Failed` only on a genuine panic — never when `task` is
905    /// cancelled (a newer call to this method, or [`Self::cancel_background_task`]
906    /// superseding it), since an intentionally aborted task's document state is already
907    /// owned by whatever superseded it. It additionally re-checks whether it is still the
908    /// current task for its URI right before writing (issue #632 critic S3): a task that
909    /// already panicked (finished, not cancelled) before a newer task was registered for
910    /// the same URI must not have its belated supervisor clobber that newer task's
911    /// in-flight load.
912    pub async fn spawn_background_task(self: &Arc<Self>, uri: Uri, task: JoinHandle<()>) {
913        let task_id = task.id();
914        let abort_handle = task.abort_handle();
915
916        {
917            let mut tasks = self.tasks.write().await;
918            // Cancel existing task if any
919            if let Some((_, old_task)) = tasks.insert(uri.clone(), (task_id, abort_handle)) {
920                old_task.abort();
921            }
922        }
923
924        let state = Arc::clone(self);
925        tokio::spawn(async move {
926            if let Err(join_error) = task.await
927                && !join_error.is_cancelled()
928            {
929                if !state.is_current_background_task(&uri, task_id).await {
930                    tracing::debug!(
931                        "background task for {:?} panicked ({}) after being superseded; \
932                         ignoring stale panic",
933                        uri,
934                        join_error
935                    );
936                    return;
937                }
938                tracing::error!(
939                    "background task for {:?} panicked ({}); forcing loading_state to Failed",
940                    uri,
941                    join_error
942                );
943                state.force_document_failed_with_not_attempted(&uri);
944            }
945        });
946    }

948    /// Whether `task_id` is still the background task currently registered for `uri`
949    /// (issue #632 critic S3).
950    ///
951    /// A background task's own supervisor (see [`Self::spawn_background_task`]) checks
952    /// this immediately before forcing a panicked task's document to `Failed`: by the
953    /// time a supervisor observes a panic, a newer [`Self::spawn_background_task`] call
954    /// may already have superseded it (aborting an already-finished — and thus
955    /// unabortable — handle is a documented no-op), and that newer task now owns the
956    /// document's `loading_state`.
957    async fn is_current_background_task(&self, uri: &Uri, task_id: tokio::task::Id) -> bool {
958        self.tasks
959            .read()
960            .await
961            .get(uri)
962            .is_some_and(|(current_id, _)| *current_id == task_id)
963    }

965    /// Forces `uri`'s document out of `Loading` into `Failed`, seeding a
966    /// [`deps_core::FetchFailure::NotAttempted`] finding for every dependency that has
967    /// neither a cached registry version nor a lockfile-resolved one (issue #632 critic
968    /// S2).
969    ///
970    /// Without the seeding, `generate_diagnostics_from_cache`'s unknown-package rule would
971    /// render every such dependency as "Unknown package" — a registry lookup that was
972    /// never actually attempted looks identical to one that came back empty — instead of
973    /// "lookup could not be determined", turning a silently suppressed diagnostic into a
974    /// misleading one. `set_fetch_failure_if_absent` (already used for a similar
975    /// never-queried case, source collisions — see `document::lifecycle`) never clobbers a
976    /// genuine `Actionable`/`Transient` finding a partial fetch may have already recorded
977    /// for the same dependency.
978    ///
979    /// A no-op if the document is missing, or (issue #632 critic M3) if it isn't currently
980    /// `Loading` — this must never clobber a document that a newer background task has
981    /// already taken past `Loading`, or one that panicked before ever reaching
982    /// `DocumentState::set_loading`.
983    pub(crate) fn force_document_failed_with_not_attempted(&self, uri: &Uri) {
984        let Some(mut doc) = self.documents.get_mut(uri) else {
985            return;
986        };
987        if doc.loading_state != LoadingState::Loading {
988            return;
989        }
990
991        let not_attempted: Vec<String> = self
992            .ecosystem_registry
993            .get(doc.ecosystem_id())
994            .zip(doc.parse_result())
995            .map(|(ecosystem, parse_result)| {
996                let formatter = ecosystem.formatter();
997                parse_result
998                    .dependencies()
999                    .into_iter()
1000                    .filter(|dep| formatter.can_resolve_source(&dep.source()))
1001                    .map(|dep| dep.name().clone())
1002                    .filter(|name| {
1003                        !doc.cached_versions.contains_key(name)
1004                            && !doc.resolved_versions.contains_key(name)
1005                    })
1006                    .map(|name| formatter.normalize_package_name(&name))
1007                    .collect()
1008            })
1009            .unwrap_or_default();
1010
1011        if !not_attempted.is_empty() {
1012            let mut outcomes = doc.outcomes.clone();
1013            for name in not_attempted {
1014                outcomes.set_fetch_failure_if_absent(name, deps_core::FetchFailure::NotAttempted);
1015            }
1016            doc.replace_outcomes(outcomes);
1017        }
1018
1019        doc.set_failed();
1020    }
