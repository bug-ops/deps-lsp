31/// Pairs each distinct dependency name in `parse_result` with the source its occurrence(s)
32/// resolve to, for the background registry fetch to route through
33/// `Registry::get_versions_from`/`get_latest_matching_from` (spec FR-001).
34///
35/// Two gates, applied in order:
36///
37/// 1. **Resolvability** (closes a review-flagged leak): a dependency whose source is not
38///    resolvable at all (`!formatter.can_resolve_source(source)` — Git, Path, an unresolved
39///    `CustomRegistry` alias, ...) is dropped from the result entirely, never reaching the
40///    fetch. Without this gate, `CargoRegistry`'s (and every other source-aware registry's)
41///    `_ =>` default-to-crates.io arm would silently look up a private/unresolvable name
42///    against the ecosystem's *public* registry — exactly the leak this feature's own
43///    hover/code-actions/diagnostics gating was built to close, just reached through the
44///    highest-traffic path (the background fetch feeding inlay hints and cached
45///    diagnostics) instead.
46/// 2. **Collision** (spec FR-011): when two occurrences of the same name both resolve
47///    (gate 1 passed for both) to two *different* sources — e.g. a genuine resolution bug
48///    producing two distinct index URLs for what should be one registry — both are dropped
49///    from the map and the name is added to the returned collision set instead of being
50///    fetched. The fetch result is shared across every occurrence of a name
51///    (`FetchResult::versions` is name-keyed), so silently picking a source here would
52///    silently apply it to occurrences whose author may have intended a different
53///    registry. A `tracing::warn!` names both resolved sources, using message text
54///    distinguishable from `deps-cargo`'s own FR-003 unresolved-alias warning.
55///
56/// Returns `(sources, collided)`: `sources` is ready to fetch as-is; `collided` must be
57/// merged into `DocumentState::outcomes`' fetch-failure channel by the caller so
58/// `generate_diagnostics_from_cache` reports "lookup could not be determined" rather than
59/// a false "Unknown package" for a dependency that was never actually queried.
60pub(crate) fn dedup_dependencies_by_source(
61    parse_result: &dyn deps_core::ParseResult,
62    formatter: &dyn deps_core::lsp_helpers::EcosystemFormatter,
63) -> (
64    HashMap<PackageName, deps_core::parser::DependencySource>,
65    HashSet<PackageName>,
66) {
67    use std::collections::hash_map::Entry;
68
69    let mut by_name: HashMap<PackageName, deps_core::parser::DependencySource> = HashMap::new();
70    let mut collided: HashSet<PackageName> = HashSet::new();
71
72    for dep in parse_result
73        .dependencies()
74        .into_iter()
75        .filter(|dep| formatter.can_resolve_source(&dep.source()))
76    {
77        let name = dep.name().clone();
78        let source = dep.source();
79        match by_name.entry(name.clone()) {
80            Entry::Vacant(entry) => {
81                entry.insert(source);
82            }
83            Entry::Occupied(entry) => {
84                if *entry.get() != source && collided.insert(name.clone()) {
85                    tracing::warn!(
86                        package = %name,
87                        source_a = ?entry.get(),
88                        source_b = ?source,
89                        "dependency declared against two different resolved registries; \
90                         skipping version resolution for all occurrences"
91                    );
92                }
93            }
94        }
95    }
96
97    for name in &collided {
98        by_name.remove(name);
99    }
100    (by_name, collided)
101}

130/// Result of parallel version fetching.
131pub(crate) struct FetchResult {
132    /// Successfully fetched versions (package -> latest + full version list)
133    pub(crate) versions: HashMap<PackageName, PackageVersions>,
134    /// Yanked-version findings, keyed by **raw** package name (unlike
135    /// `DocumentState::outcomes`, which is normalized-keyed — see
136    /// §3.1 of the design), to (the version string found yanked, its
137    /// `RemovalStatus`). The status rides alongside so #205's package-level
138    /// deprecation diagnostic can gate its yanked-check suppression on
139    /// `AdvisoryDeprecated` specifically, never a genuine `Yanked` finding.
140    /// Callers must re-key through `EcosystemFormatter::normalize_package_name`
141    /// before merging into document state.
142    pub(crate) yanked_versions: HashMap<PackageName, (ConcreteVersion, RemovalStatus)>,
143    /// Package-level deprecation findings (issue #205), keyed by **raw** package name
144    /// (same raw/normalized split as `yanked_versions` above). Derived from the
145    /// `resolved`/"latest" pick in the fetch loop below, not by scanning the full
146    /// `versions` list — see that loop's comments for why.
147    pub(crate) deprecations: HashMap<PackageName, Deprecation>,
148    /// Packages whose registry fetch errored or timed out, keyed by **raw**
149    /// package name (same raw/normalized split as `yanked_versions` above).
150    /// Lets diagnostic generation (#267) distinguish "the registry said this
151    /// package doesn't exist" from "the registry couldn't be asked" instead
152    /// of conflating both into a misleading "Unknown package" diagnostic.
153    pub(crate) fetch_failed: HashMap<PackageName, FetchFailure>,
154    /// Packages whose registry fetch succeeded but produced zero comparable versions
155    /// (#550), keyed by **raw** package name (same raw/normalized split as
156    /// `yanked_versions` above). Distinct from `fetch_failed`: the registry was
157    /// successfully asked and the package demonstrably exists — it just has nothing a
158    /// version-comparison rule can use — so `generate_diagnostics_from_cache` must
159    /// report neither "Registry lookup failed" nor "Unknown package" for it.
160    pub(crate) no_comparable_versions: HashSet<PackageName>,
161    /// Number of packages whose registry fetch did not succeed, counting both a genuine
162    /// fetch failure (timeout, error — recorded in `fetch_failed` above) and a not-found
163    /// lookup (the registry answered "no such package", never recorded in `fetch_failed`,
164    /// see #267 C1). Only the `fetch_failed` subset produces an inline "Registry lookup
165    /// failed" diagnostic, so this count can exceed `fetch_failed.len()` (#276 S2, #490).
166    pub(crate) failed_count: usize,
167    /// First actionable error message (shown to user via `window/showMessage`)
168    pub(crate) first_error: Option<String>,
169    /// SPDX license identifier(s) for the resolved/"latest" pick, for every package
170    /// whose `Version::license` on the already-fetched version-list entry is
171    /// non-empty (issue #660/#661 tier-1 backfill) — today, only the native-list
172    /// ecosystems (PyPI, Composer) ever populate this; every other ecosystem's
173    /// `Version::license` default is empty, so this map stays empty for them.
174    /// Deliberately *not* threaded into [`PackageVersions`] itself (that type is
175    /// constructed identically across ~40 call sites throughout the workspace,
176    /// including files outside this crate's ownership for this change) —
177    /// `merge_registry_fetch_result` merges this map directly into
178    /// [`crate::document::DocumentState::licenses`] instead, the same map the tier-3
179    /// background pre-fetch (`run_license_prefetch`) already populates for
180    /// Dart/Swift/Gradle/Deno. A merge (not replace), since the two sources are
181    /// always disjoint per document (one ecosystem per document) but run as
182    /// independent, non-ordered background tasks.
183    pub(crate) licenses: HashMap<PackageName, Vec<String>>,
184}

186/// Fetches latest versions for multiple packages in parallel with progress reporting.
187///
188/// Returns a [`FetchResult`] containing successfully fetched versions and failure count.
189/// Packages that fail to fetch are omitted from the versions map.
190///
191/// This function executes all registry requests concurrently with per-dependency
192/// timeout isolation, preventing slow packages from blocking others.
193///
194/// Alongside the primary fetch, checks whether the in-use version of a
195/// dependency has been yanked (#233), for registries that [report yank
196/// data](Registry::reports_yanked). Unlike the original design, this is not
197/// a second registry round trip: `registry.get_versions` below already
198/// fetches the full, unfiltered version list once per package (see
199/// [`PackageVersions`]), so the in-use-version check is a zero-cost
200/// in-memory search over a list already in hand, run for every dependency
201/// with a known in-use version rather than only when it differs from
202/// `latest`.
203///
204/// # Arguments
205///
206/// * `registry` - Package registry to fetch from
207/// * `package_names` - List of package names to fetch
208/// * `in_use` - Raw dependency name -> the version(s) this project actually
209///   has (lockfile-resolved or a concrete pin) for every occurrence of that
210///   name in the manifest, checked against the fetched version list for
211///   yank status
212/// * `progress` - Optional progress tracker (will be updated after each fetch)
213/// * `timeout_secs` - Timeout for each individual package fetch (default: 10s)
214/// * `max_concurrent` - Maximum concurrent fetches (default: 20); clamped to `>= 1`
215///   internally, since `buffer_unordered(0)` would hang forever (issue #833)
216///
217/// # Timeout Behavior
218///
219/// Each package fetch is wrapped in an individual timeout. If a package
220/// takes longer than `timeout_secs` to fetch, it fails fast with a warning
221/// and does NOT block other packages.
222///
223/// # Performance
224///
225/// With 50 dependencies and 100ms per request:
226/// - Sequential: 50 × 100ms = 5000ms
227/// - Parallel (no timeout): max(100ms) ≈ 150ms
228/// - Parallel (10s timeout, 1 slow package at 30s): max(10s) ≈ 10s
229#[allow(
230    clippy::too_many_arguments,
231    reason = "internal (non-pub) call-site-controlled fetch tuning + ecosystem-context \
232              parameters; grouping into a config struct would only move, not reduce, the \
233              per-call-site churn across this module's ~15 production and test call sites"
234)]
235pub(crate) async fn fetch_latest_versions_parallel(
236    registry: Arc<dyn Registry>,
237    package_sources: DepSources,
238    in_use: &HashMap<PackageName, Vec<String>>,
239    progress_sender: Option<ProgressSender>,
240    freshness: deps_core::freshness::FreshnessSettings,
241    timeout_secs: u64,
242    max_concurrent: usize,
243    minimum_stability: Option<&str>,
244) -> FetchResult {
245    use futures::stream::{self, StreamExt};
246    use std::time::Duration;
247
248    let fetched = Arc::new(std::sync::atomic::AtomicUsize::new(0));
249    let failed = Arc::new(std::sync::atomic::AtomicUsize::new(0));
250    let first_error: Arc<std::sync::Mutex<Option<String>>> = Arc::new(std::sync::Mutex::new(None));
251    // Separate from `first_error` (#480): a not-found error is deliberately excluded
252    // from `fetch_failed` below (it isn't evidence of a registry-side problem), but
253    // without this, whichever concurrent fetch happened to finish first could still win
254    // the `first_error` race and put a misleading "not found" message in the one-time
255    // toast even when the batch's real, actionable failure is e.g. a rate limit hit by
256    // 20 other dependencies. Any error that *does* count toward `fetch_failed`
257    // (including a timeout) always wins the toast over a not-found, regardless of
258    // finishing order; only a not-found-only batch falls back to `first_error`.
259    //
260    // Unlike `first_error`, this is not a shared `Arc<Mutex>` written from inside the
261    // four match arms below — each task instead returns its own `(name, message)` via
262    // `failed_name`, and the priority error is derived by folding those in completion
263    // order once every task has finished (see the loop below). `fetch_failed` and the
264    // priority error are thereby always in sync by construction: both come from the
265    // same `failed_name` value, so a future edit to one can no longer silently drift
266    // from the other, which two independently hand-maintained writes could (#480).
267    let timeout = Duration::from_secs(timeout_secs);
268    let wildcard_req = deps_core::VersionReq::new("*");
269    let check_yanked = registry.reports_yanked();
270
271    let results: Vec<_> = stream::iter(package_sources)
272        .map(|(name, source)| {
273            let registry = Arc::clone(&registry);
274            let fetched = Arc::clone(&fetched);
275            let failed = Arc::clone(&failed);
276            let first_error = Arc::clone(&first_error);
277            let progress_sender = progress_sender.clone();
278            let wildcard_req = &wildcard_req;
279            let in_use_versions = in_use.get(&name).cloned().unwrap_or_default();
280            async move {
281                fetch_and_classify_package(
282                    registry.as_ref(),
283                    name,
284                    source,
285                    in_use_versions,
286                    wildcard_req,
287                    freshness,
288                    timeout,
289                    minimum_stability,
290                    check_yanked,
291                    &fetched,
292                    &failed,
293                    &first_error,
294                    progress_sender.as_ref(),
295                )
296                .await
297            }
298        })
299        // `.max(1)`: `CacheConfig.max_concurrent_fetches` is a `pub` field, so an in-crate
300        // direct field assignment (see the test setup at `server.rs:1680`) bypasses
301        // `with_max_concurrent_fetches`'s own clamp; this is defence-in-depth against a
302        // future in-crate caller doing the same with `0`. `buffer_unordered(0)` never
303        // polls its source stream — it returns `Pending` forever instead of erroring,
304        // hanging every fetch through this document indefinitely (issue #833).
305        .buffer_unordered(max_concurrent.max(1))
306        .collect()
307        .await;
308
309    let mut versions = HashMap::with_capacity(results.len());
310    let mut yanked_versions = HashMap::new();
311    let mut fetch_failed = HashMap::new();
312    let mut deprecations = HashMap::new();
313    let mut no_comparable_versions = HashSet::new();
314    let mut licenses = HashMap::new();
315    // First actionable failure in completion order — `results` is collected from
316    // `buffer_unordered`, so its order already reflects real finishing order, the same
317    // order a shared `Arc<Mutex>` written from inside each task would have observed.
318    let mut priority_error: Option<String> = None;
319    for (version, yanked, failed_name, deprecation, no_comparable_versions_name, license) in results
320    {
321        if let Some((name, v)) = version {
322            versions.insert(name, v);
323        }
324        if let Some((name, v, status)) = yanked {
325            yanked_versions.insert(name, (v, status));
326        }
327        if let Some((name, failure, message)) = failed_name {
328            fetch_failed.insert(name, failure);
329            if priority_error.is_none() {
330                priority_error = Some(message);
331            }
332        }
333        if let Some((name, d)) = deprecation {
334            deprecations.insert(name, d);
335        }
336        if let Some(name) = no_comparable_versions_name {
337            no_comparable_versions.insert(name);
338        }
339        if let Some((name, license)) = license {
340            licenses.insert(name, license);
341        }
342    }
343
344    // `priority_error` (an actual fetch failure — rate limit, timeout, outage, ...)
345    // always wins the toast over `first_error` (which may be a not-found race winner);
346    // `first_error` is the fallback only for a batch whose only failures were
347    // not-found (#480).
348    let error_message =
349        priority_error.or_else(|| first_error.lock().unwrap_or_else(|p| p.into_inner()).take());
350
351    FetchResult {
352        versions,
353        yanked_versions,
354        fetch_failed,
355        deprecations,
356        no_comparable_versions,
357        failed_count: failed.load(std::sync::atomic::Ordering::Relaxed),
358        first_error: error_message,
359        licenses,
360    }
361}

363/// Per-package outcome returned by [`fetch_and_classify_package`]: the resolved
364/// `(name, PackageVersions)` entry, a yanked finding, a fetch failure, a package-level
365/// deprecation finding, a name whose fetch succeeded with no comparable versions
366/// (#550), and the resolved/"latest" pick's license when the ecosystem's already-fetched
367/// version-list entries carry it (issue #660/#661 tier-1 backfill — see
368/// [`FetchResult::licenses`]) — folded into [`fetch_latest_versions_parallel`]'s
369/// aggregate `FetchResult` once every package in the stream has finished.
370///
371/// The license entry specifically comes from `select_latest_matching_with_context`'s
372/// pick below (critic S1: previously documented here as "the resolved version's
373/// license", which is wrong — this function never reads `resolved_versions` at all, it
374/// picks the latest version matching the requirement/stability floor, same as
375/// `PackageVersions.latest`).
376type PackageFetchOutcome = (
377    Option<(PackageName, PackageVersions)>,
378    Option<(PackageName, ConcreteVersion, RemovalStatus)>,
379    Option<(PackageName, FetchFailure, String)>,
380    Option<(PackageName, Deprecation)>,
381    Option<PackageName>,
382    Option<(PackageName, Vec<String>)>,
383);

385/// Fetches, classifies, and version-selects a single package within
386/// [`fetch_latest_versions_parallel`]'s concurrent stream: one round trip for the full
387/// version list, an in-memory "latest" pick with a `get_latest_matching_from` fallback
388/// when the list-based pick fails on a non-empty list, yanked/deprecation extraction,
389/// and updates to the shared `fetched`/`failed`/`first_error` counters the stream
390/// aggregates across every package.
391#[allow(
392    clippy::too_many_arguments,
393    reason = "mirrors the per-package async closure this was extracted from — every \
394              parameter is either call-site fetch tuning already threaded through \
395              fetch_latest_versions_parallel or a counter/sender shared across the \
396              whole stream; grouping into a struct would only move, not reduce, churn"
397)]
398async fn fetch_and_classify_package(
399    registry: &dyn Registry,
400    name: PackageName,
401    source: deps_core::parser::DependencySource,
402    in_use_versions: Vec<String>,
403    wildcard_req: &VersionReq,
404    freshness: deps_core::freshness::FreshnessSettings,
405    timeout: Duration,
406    minimum_stability: Option<&str>,
407    check_yanked: bool,
408    fetched: &std::sync::atomic::AtomicUsize,
409    failed: &std::sync::atomic::AtomicUsize,
410    first_error: &std::sync::Mutex<Option<String>>,
411    progress_sender: Option<&ProgressSender>,
412) -> PackageFetchOutcome {
413    // Single round trip: the full version list is fetched once, and "latest"
414    // is a pure in-memory pick over it (`Registry::select_latest_matching`) —
415    // no second registry call, so the retained full list costs nothing extra
416    // over the network (see `PackageVersions`). `get_versions_from` (source-
417    // aware, spec FR-001) rather than `get_versions`: this populates
418    // `published_at` for registries that support it (#339), matching hover's
419    // existing freshness-aware call, AND routes a resolved
420    // `DependencySource::AlternateRegistry` to its own index instead of the
421    // ecosystem's default registry — registries with no override forward
422    // straight to `get_versions` at zero extra cost either way.
423    let result = tokio::time::timeout(
424        timeout,
425        registry.get_versions_from(&name, &source, freshness),
426    )
427    .await;
428
429    let mut yanked: Option<(PackageName, ConcreteVersion, RemovalStatus)> = None;
430    let mut failed_name: Option<(PackageName, FetchFailure, String)> = None;
431    let mut deprecation: Option<(PackageName, Deprecation)> = None;
432    let mut license: Option<(PackageName, Vec<String>)> = None;
433    // Set only when the fetch (and its `get_latest_matching` fallback) both
434    // genuinely succeeded yet resolved to no version at all (#550) — see the
435    // `Ok(Ok(None))` fallback arm below.
436    let mut no_comparable_versions = false;
437    let version = match result {
438        Ok(Ok(versions)) => {
439            let available: Arc<[ConcreteVersion]> = versions
440                .iter()
441                .map(|v| v.version_string().clone())
442                .collect();
443            // Retained alongside `available` so `generate_diagnostics_from_cache`
444            // can flag a requirement satisfiable only by a yanked version — see
445            // `PackageVersions::yanked`. Gated on `check_yanked`: a registry that
446            // cannot answer `removal_status()` (§#298) must not populate this list
447            // with an untrustworthy always-`Available` signal. Carries each
448            // entry's own `RemovalStatus` (#437) so the #247 diagnostic path can
449            // gate its package-level-deprecation suppression on `AdvisoryDeprecated`
450            // specifically, never on a genuine `Yanked` finding.
451            let yanked_list: Arc<[(ConcreteVersion, RemovalStatus)]> = if check_yanked {
452                versions
453                    .iter()
454                    .filter_map(|v| {
455                        let status = v.removal_status();
456                        status
457                            .is_flagged()
458                            .then(|| (v.version_string().clone(), status))
459                    })
460                    .collect()
461            } else {
462                Arc::from([])
463            };
464            // `.get(idx)` rather than `versions[idx]`: `select_latest_matching`
465            // is a public `Registry` trait method, so an out-of-tree
466            // implementation returning a stale index must not panic this task.
467            // `_with_context` (not the plain method) so a registry with
468            // manifest-level stability state (Composer's `minimum-stability`,
469            // #424 S1) can apply it — every other registry's default
470            // implementation just forwards to the plain method unchanged.
471            let resolved = if let Some(v) = registry
472                .select_latest_matching_with_context(&versions, wildcard_req, minimum_stability)
473                .and_then(|idx| versions.get(idx))
474            {
475                let latest = v.version_string().clone();
476                tracing::debug!(package = %name, version = %latest, "fetched");
477                Some((
478                    latest,
479                    v.removal_status(),
480                    v.published_at(),
481                    v.deprecation().cloned(),
482                    v.license().to_vec(),
483                ))
484            } else {
485                // The pure list-based pick found nothing — for most
486                // ecosystems this genuinely means "no version found", but
487                // for a registry whose list endpoint can be incomplete
488                // (e.g. Go's `/@v/list`, which never enumerates
489                // pseudo-versions and can be entirely empty for an
490                // untagged module) it may just mean the list alone isn't
491                // enough. Fall back to the registry's own
492                // `get_latest_matching`, which some registries answer from
493                // a different, more complete source (Go's `/@latest`). This
494                // costs a second network call, but only in this already-rare
495                // "list-based pick failed" case, not the common path.
496                let fallback = tokio::time::timeout(
497                    timeout,
498                    registry.get_latest_matching_from(
499                        &name,
500                        &source,
501                        wildcard_req,
502                        minimum_stability,
503                    ),
504                )
505                .await;
506                match fallback {
507                    Ok(Ok(Some(v))) => {
508                        let latest = v.version_string().clone();
509                        tracing::debug!(
510                            package = %name,
511                            version = %latest,
512                            "fetched via get_latest_matching fallback"
513                        );
514                        Some((
515                            latest,
516                            v.removal_status(),
517                            v.published_at(),
518                            v.deprecation().cloned(),
519                            v.license().to_vec(),
520                        ))
521                    }
522                    Ok(Ok(None)) => {
523                        tracing::debug!(package = %name, "no version found");
524                        // Both the list-based pick and this fallback
525                        // genuinely succeeded and found nothing — the
526                        // package demonstrably exists (the fetch itself
527                        // never errored), it just has zero versions this
528                        // registry can compare against (#550), e.g. a
529                        // repository whose only tags don't parse as full
530                        // semver. Distinct from every branch below that
531                        // sets `failed_name`.
532                        no_comparable_versions = true;
533                        None
534                    }
535                    Ok(Err(e)) => {
536                        tracing::warn!(
537                            package = %name,
538                            error = %e,
539                            "fetch fallback failed"
540                        );
541                        failed.fetch_add(1, std::sync::atomic::Ordering::Relaxed);
542                        let mut fe = first_error.lock().unwrap_or_else(|p| p.into_inner());
543                        if fe.is_none() {
544                            *fe = Some(e.to_string());
545                        }
546                        drop(fe);
547                        // A genuine not-found (the registry was
548                        // successfully asked and said "no such
549                        // package") is not a fetch failure — only
550                        // an unanswerable request is (#267 C1).
551                        if !e.is_not_found() {
552                            failed_name = Some((name.clone(), e.fetch_failure(), e.to_string()));
553                        }
554                        None
555                    }
556                    Err(_) => {
557                        tracing::warn!(
558                            package = %name,
559                            "fetch fallback timed out ({}s)",
560                            timeout.as_secs()
561                        );
562                        failed.fetch_add(1, std::sync::atomic::Ordering::Relaxed);
563                        failed_name = Some((
564                            name.clone(),
565                            FetchFailure::Transient,
566                            format!(
567                                "{name}: registry request timed out after {}s",
568                                timeout.as_secs()
569                            ),
570                        ));
571                        None
572                    }
573                }
574            };
575
576            if check_yanked {
577                // Row 1 (§4.7): the picked "latest" itself yanked —
578                // zero extra cost, since it's already in hand.
579                // Unreachable in production for an *enabled*
580                // registry under today's hardcoded wildcard (one
581                // never returns a yanked version for `*`), but
582                // stays correct as a defense-in-depth check.
583                if let Some((latest, status, _, _, _)) = &resolved
584                    && status.is_flagged()
585                {
586                    yanked = Some((name.clone(), latest.clone(), *status));
587                }
588
589                // Row 2/3 (§4.7, revised under #206): `versions`
590                // is the full, already-fetched, unfiltered list —
591                // no second registry round trip is needed to
592                // check whether the in-use version was yanked,
593                // unlike the pre-#206 probe design. Checked for
594                // every dependency with a known in-use version,
595                // not just when it differs from `latest`, since
596                // it's now a free in-memory lookup either way. A
597                // yanked in-use version wins over an already
598                // -recorded yanked `latest` — it's the version
599                // the user actually has.
600                //
601                // Multiple occurrences of the same name (#394,
602                // e.g. under both `[dependencies]` and
603                // `[target.*.dependencies]`) can carry different
604                // in-use versions — every one is checked so a
605                // yanked pin on any occurrence is never missed
606                // just because another occurrence happens to
607                // share the registry lookup.
608                // Filters on `is_flagged()` inside the `find` predicate itself
609                // (not via a separate `.filter()` on the first version-string
610                // match) so a registry response with more than one entry sharing
611                // `iv`'s version string still finds a flagged one if any exists —
612                // mirroring the pre-#205 `.any(matches && flagged)` scan rather
613                // than narrowing to "is the *first* same-string entry flagged".
614                if let Some((iv, status)) = in_use_versions.iter().find_map(|iv| {
615                    versions
616                        .iter()
617                        .find(|v| {
618                            v.version_string() == iv.as_str() && v.removal_status().is_flagged()
619                        })
620                        .map(|v| (iv, v.removal_status()))
621                }) {
622                    yanked = Some((name.clone(), iv.as_str().into(), status));
623                }
624            }
625
626            // #205: the package-level deprecation finding is derived from the
627            // same `Version` `resolved` already picked as "latest" — covering
628            // the `get_latest_matching_with_context` fallback branch above too,
629            // whose returned `Version` is not a member of `versions` at all. See
630            // `FetchResult::deprecations`'s docs for why this must not instead
631            // scan `versions`.
632            if let Some((_, _, _, dep_info, _)) = &resolved
633                && let Some(dep_info) = dep_info
634            {
635                deprecation = Some((name.clone(), dep_info.clone()));
636            }
637
638            // Issue #660/#661 tier-1 backfill: extracted from the same `resolved` pick
639            // before `.map()` below consumes it — non-empty only for the native-list
640            // ecosystems whose `Version::license` isn't the default empty (PyPI,
641            // Composer today). Filtered here (not left to the aggregation loop) so a
642            // `Some((name, vec![]))` entry — indistinguishable from "no data" once
643            // merged into `DocumentState::licenses` — never gets inserted.
644            license = resolved
645                .as_ref()
646                .map(|(_, _, _, _, lic)| lic)
647                .filter(|lic| !lic.is_empty())
648                .map(|lic| (name.clone(), lic.clone()));
649
650            resolved.map(|(latest, _, published_at, _, _)| {
651                let mut versions = PackageVersions::new(latest, available).with_yanked(yanked_list);
652                if let Some(published_at) = published_at {
653                    versions = versions.with_published_at(published_at);
654                }
655                (name.clone(), versions)
656            })
657        }
658        Ok(Err(e)) => {
659            // Issue #483: while offline, every fetch fails by design — this
660            // would otherwise log a per-dependency WARNING for every open/edit,
661            // contradicting the toast suppression two call sites away in this
662            // same file for being "unusable".
663            if e.is_offline() {
664                tracing::debug!(package = %name, "fetch skipped: offline");
665            } else {
666                tracing::warn!(package = %name, error = %e, "fetch failed");
667            }
668            failed.fetch_add(1, std::sync::atomic::Ordering::Relaxed);
669            let mut fe = first_error.lock().unwrap_or_else(|p| p.into_inner());
670            if fe.is_none() {
671                *fe = Some(e.to_string());
672            }
673            drop(fe);
674            // A genuine not-found (the registry was successfully
675            // asked and said "no such package") is not a fetch
676            // failure — only an unanswerable request is (#267
677            // C1). Marking it `fetch_failed` here would make
678            // `generate_diagnostics_from_cache` report "Registry
679            // lookup failed" for the common typo'd-name case
680            // instead of "Unknown package", inverting the bug
681            // this field exists to fix.
682            if !e.is_not_found() {
683                failed_name = Some((name.clone(), e.fetch_failure(), e.to_string()));
684            }
685            None
686        }
687        Err(_) => {
688            tracing::warn!(package = %name, "fetch timed out ({}s)", timeout.as_secs());
689            failed.fetch_add(1, std::sync::atomic::Ordering::Relaxed);
690            failed_name = Some((
691                name.clone(),
692                FetchFailure::Transient,
693                format!(
694                    "{name}: registry request timed out after {}s",
695                    timeout.as_secs()
696                ),
697            ));
698            None
699        }
700    };
701
702    let count = fetched.fetch_add(1, std::sync::atomic::Ordering::Relaxed) + 1;
703    if let Some(sender) = progress_sender {
704        sender.send(count);
705    }
706
707    let no_comparable_versions_name = no_comparable_versions.then(|| name.clone());
708    (
709        version,
710        yanked,
711        failed_name,
712        deprecation,
713        no_comparable_versions_name,
714        license,
715    )
716}

718/// Decides whether a fetch-failure toast should be shown for this fetch cycle, and what
719/// its message should be — a pure decision, factored out of the two call sites in
720/// `handle_document_open` and `handle_document_change` so both share one policy and the
721/// policy itself is unit-testable without an LSP transport.
722///
723/// `failed_count` counts both genuine fetch failures and not-found lookups (see #276 S2),
724/// so the message deliberately says "could not be resolved" rather than "failed to fetch"
725/// or anything containing "lookup failed" — that phrasing is the exact inline "Registry
726/// lookup failed" diagnostic text, which excludes not-found by design (#267 C1), so reusing
727/// it here would recreate the same overcount confusion in different words. "could not be
728/// resolved" covers both the "Registry lookup failed" and "Unknown package" diagnostic
729/// outcomes, so the count stays checkable against their union (#490).
730///
731/// Returns `None` when there were no failures at all, or when `offline` is set (issue
732/// #483): every fetch fails by design while `network.offline` is set, so toasting on every
733/// document open/change would make offline mode unusable.
734pub(crate) fn fetch_failure_toast(
735    failed_count: usize,
736    first_error: Option<&str>,
737    offline: bool,
738) -> Option<String> {
739    if failed_count == 0 || offline {
740        return None;
741    }
742    Some(format!(
743        "deps-lsp: {failed_count} package(s) could not be resolved: {}",
744        first_error.unwrap_or("timeout or network error")
745    ))
746}

748/// Fans the registry fetch out for the added/version-changed dependencies determined by the
749/// caller's diff: marks the document loading, opens an LSP progress notification when the
750/// client supports it, resolves each dependency occurrence to a fetchable source (deduping
751/// same-name collisions across different resolved sources), and fetches latest versions in
752/// parallel. Returns everything the caller needs to merge the result and end the progress
753/// notification, without exposing the intermediate `in_use`/`dep_sources` bookkeeping.
754#[allow(clippy::too_many_arguments)]
755pub(crate) async fn fetch_registry_versions_for_change(
756    uri: &Uri,
757    state: &ServerState,
758    client: &Client,
759    ecosystem: &dyn Ecosystem,
760    resolved_versions: &HashMap<PackageName, ConcreteVersion>,
761    resolved_version_candidates: &HashMap<PackageName, Vec<ConcreteVersion>>,
762    deps_to_fetch: Vec<PackageName>,
763    freshness_settings: deps_core::FreshnessSettings,
764    fetch_timeout_secs: u64,
765    max_concurrent_fetches: usize,
766    refetch: RefetchPolicy,
767) -> (
768    Option<RegistryProgress>,
769    FetchResult,
770    Vec<PackageName>,
771    HashSet<PackageName>,
772) {
773    tracing::info!(
774        count = deps_to_fetch.len(),
775        "fetching versions for added/version-changed dependencies"
776    );
777
778    // Mark as loading and start progress. Under `RefetchPolicy::AllDependencies` the
779    // routing itself changed (issue #592), not just the manifest, so the cache built under
780    // the *old* routing can no longer be vouched for — drop it under the same lock that
781    // sets `Loading`, so no reader ever observes a half-updated state (critic M2: this
782    // replaces a separate drop-then-set_loading sequence, which would leave a window
783    // between two independent `get_mut` acquisitions).
784    if let Some(mut doc) = state.documents.get_mut(uri) {
785        if refetch == RefetchPolicy::AllDependencies {
786            drop_cache_for_forced_refetch(&mut doc, &deps_to_fetch, ecosystem.formatter());
787        }
788        doc.set_loading();
789    }
790
791    let (progress, progress_sender) = if state.supports_progress() {
792        match tokio::time::timeout(
793            std::time::Duration::from_secs(2),
794            RegistryProgress::start(client.clone(), uri.as_str(), deps_to_fetch.len()),
795        )
796        .await
797        {
798            Ok(Ok((p, s))) => (Some(p), Some(s)),
799            _ => (None, None),
800        }
801    } else {
802        (None, None)
803    };
804
805    // Build the in-use-version map (§4.6) and the added/changed dependencies' resolved
806    // sources (spec FR-001/FR-011) from the freshly-committed parse result and the
807    // resolved versions just loaded above.
808    let (in_use, minimum_stability, dep_sources, collided_names): (
809        HashMap<PackageName, Vec<String>>,
810        Option<String>,
811        DepSources,
812        HashSet<PackageName>,
813    ) = match state.get_document(uri) {
814        Some(doc) => match doc.parse_result() {
815            Some(pr) => {
816                let (sources, collided_names) =
817                    dedup_dependencies_by_source(pr, ecosystem.formatter());
818                let dep_sources = deps_to_fetch
819                    .iter()
820                    .filter_map(|name| sources.get(name).map(|s| (name.clone(), s.clone())))
821                    .collect();
822                (
823                    collect_in_use_versions(
824                        pr,
825                        resolved_versions,
826                        resolved_version_candidates,
827                        ecosystem.formatter(),
828                        ecosystem.ecosystem_id(),
829                    ),
830                    composer_minimum_stability(pr),
831                    dep_sources,
832                    collided_names,
833                )
834            }
835            None => (HashMap::new(), None, Vec::new(), HashSet::new()),
836        },
837        None => (HashMap::new(), None, Vec::new(), HashSet::new()),
838    };
839
840    // Fetch latest versions only for NEW dependencies
841    //
842    // Captured before `dep_sources` is moved into the call below: every raw name a
843    // fetch was actually attempted for this round, used by the #550
844    // no-comparable-versions merge further down to distinguish "attempted and
845    // resolved fine this round" (clear any stale marker) from "not attempted this
846    // round" (leave any existing marker untouched) — unlike `fetched_names` in the
847    // merge step, this can't be derived from `fetch_result.versions`'s keys, since a
848    // no-comparable-versions package is by definition never one of them.
849    let attempted_names: Vec<PackageName> =
850        dep_sources.iter().map(|(name, _)| name.clone()).collect();
851    let registry = ecosystem.registry();
852    let fetch_result = fetch_latest_versions_parallel(
853        registry,
854        dep_sources,
855        &in_use,
856        progress_sender,
857        freshness_settings,
858        fetch_timeout_secs,
859        max_concurrent_fetches,
860        minimum_stability.as_deref(),
861    )
862    .await;
863
864    (progress, fetch_result, attempted_names, collided_names)
865}

867/// Merges a completed registry fetch into the document's cache and outcome maps —
868/// newly fetched versions, yanked/fetch-failure markers (re-keyed raw -> normalized),
869/// collided names recorded as not-attempted, and deprecation / no-comparable-versions
870/// bookkeeping — then marks the document loaded or failed depending on `success`. Returns
871/// `(failed_count, first_error)`, the two `FetchResult` fields this function does not
872/// consume, so the caller can still raise the fetch-failure toast after `fetch_result`
873/// itself has been moved in here.
874pub(crate) fn merge_registry_fetch_result(
875    state: &ServerState,
876    uri: &Uri,
877    formatter: &dyn deps_core::lsp_helpers::EcosystemFormatter,
878    fetch_result: FetchResult,
879    attempted_names: &[PackageName],
880    collided_names: HashSet<PackageName>,
881    success: bool,
882) -> (usize, Option<String>) {
883    if let Some(mut doc) = state.documents.get_mut(uri) {
884        // Captured before `fetch_result.versions` is consumed below: every name
885        // successfully fetched this round, used by the S1 deprecation-clearing
886        // loop further down to distinguish "fetched and clean" from "not fetched
887        // this round" — only the former may clear a stale finding.
888        let fetched_names: Vec<PackageName> = fetch_result.versions.keys().cloned().collect();
889        for (name, version) in fetch_result.versions {
890            doc.cached_versions.insert(name, version);
891        }
892        // Issue #660/#661 tier-1 backfill: merge (never replace — see
893        // `DocumentState::merge_licenses`'s docs), so this coexists safely with the
894        // independent tier-3 background pre-fetch's own write to the same map.
895        doc.merge_licenses(fetch_result.licenses);
896        // Re-key raw -> normalized (§3.1), same as the didOpen path.
897        for (name, version) in fetch_result.yanked_versions {
898            doc.outcomes
899                .set_yanked(formatter.normalize_package_name(&name), version);
900        }
901        for (name, failure) in fetch_result.fetch_failed {
902            doc.outcomes
903                .set_fetch_failure(formatter.normalize_package_name(&name), failure);
904        }
905        // `set_fetch_failure_if_absent` (not `set_fetch_failure`): a collided name
906        // normalizing to the same key as a genuine failure just recorded above must
907        // not clobber it (impl-critic M2).
908        for name in collided_names {
909            doc.outcomes.set_fetch_failure_if_absent(
910                formatter.normalize_package_name(&name),
911                FetchFailure::NotAttempted,
912            );
913        }
914        merge_deprecations_after_fetch(
915            &mut doc,
916            &fetched_names,
917            fetch_result.deprecations,
918            formatter,
919        );
920        merge_no_comparable_versions_after_fetch(
921            &mut doc,
922            attempted_names,
923            fetch_result.no_comparable_versions,
924            formatter,
925        );
926        if success {
927            doc.set_loaded();
928        } else {
929            doc.set_failed();
930        }
931    }
932
933    (fetch_result.failed_count, fetch_result.first_error)
934}
