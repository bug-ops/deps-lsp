22/// Builds the OSV scan targets for one manifest's dependencies, applying the
23/// version-selection policy from `architecture.md` §3 in order:
24///
25/// 0. Skip unless `formatter.source_is_public_registry_content(&dep.source())` — a patched
26///    git/path fork must never be flagged with a CVE for a version it does
27///    not actually contain, and neither must a genuinely different private registry's
28///    dependency (only a verified crates.io mirror counts as public-registry content,
29///    F1/F1b).
30/// 1. Use the lock-file-resolved version if present.
31/// 2. Otherwise use the declared requirement, if it is already concrete.
32/// 3. Otherwise skip — querying a fabricated version is a silent false
33///    negative, which is worse than not scanning at all.
34///
35/// **Go exception** (#228 follow-up, unified with #235's
36/// [`deps_core::lsp_helpers::RequirementResolution::manifest_requirement_is_resolved_version`]):
37/// step 1 is skipped entirely for a dependency whose manifest requirement is
38/// itself the resolved version (a Go `require`-directive dependency), going
39/// straight to step 2. Go's `go.mod` `require` line is already an exact
40/// pinned version, never a range, unlike Cargo/npm where the manifest is a
41/// range and the lockfile holds the pin. go.sum-derived `resolved_versions`
42/// is unreliable here: go.sum is a checksum ledger that `go get`/`go build`
43/// only ever append to (only `go mod tidy` prunes it), so its
44/// last-occurrence-wins parse can surface a version still recorded in the
45/// file but no longer selected by Go's MVS — silently querying OSV against
46/// the wrong version. Routing through the formatter hook (rather than a bare
47/// `ecosystem == EcosystemId::Go` check) also excludes Go's `exclude`/
48/// `replace` directive pseudo-dependencies, whose `version_requirement()` is
49/// not an in-use version.
50///
51/// Every dependency that does **not** become a [`deps_core::osv::ScanTarget`]
52/// gets an explicit [`deps_core::osv::ScanOutcome::Skipped`] entry in the
53/// returned map instead of silently vanishing (critique C1) — absence from
54/// [`deps_core::osv::VulnerabilityMap`] must never happen for an input this
55/// function considered.
56///
57/// Each dependency's map/target key comes from
58/// [`deps_core::osv::vulnerability_keys`] rather than a bare
59/// `formatter.normalize_package_name(dep.name())` (#394 S2): when two
60/// occurrences of one name resolve to different in-use versions (or mix a
61/// registry source with a git/path fork), their keys are disambiguated so
62/// one occurrence's OSV result never overwrites another's in the shared
63/// [`deps_core::osv::VulnerabilityMap`]. Occurrences that share both a name
64/// and an identical in-use version keep the plain key and are scanned once —
65/// a dedup, not a gap, since the OSV result would be identical either way.
66fn build_scan_targets(
67    parse_result: &dyn deps_core::ParseResult,
68    resolved_versions: &HashMap<PackageName, ConcreteVersion>,
69    resolved_version_candidates: &HashMap<PackageName, Vec<ConcreteVersion>>,
70    formatter: &dyn deps_core::lsp_helpers::EcosystemFormatter,
71    ecosystem: EcosystemId,
72) -> (
73    Vec<deps_core::osv::ScanTarget>,
74    deps_core::osv::VulnerabilityMap,
75) {
76    use deps_core::osv::{ScanOutcome, SkipReason};
77
78    let mut targets = Vec::new();
79    let mut skipped = deps_core::osv::VulnerabilityMap::new();
80    let keys = deps_core::osv::vulnerability_keys(
81        parse_result,
82        resolved_versions,
83        Some(resolved_version_candidates),
84        formatter,
85        ecosystem,
86    );
87
88    for dep in parse_result.dependencies() {
89        let normalized_name = formatter.normalize_package_name(dep.name());
90        let key = keys
91            .get(&dep.name_range())
92            .cloned()
93            .unwrap_or_else(|| normalized_name.clone());
94
95        if !formatter.source_is_public_registry_content(&dep.source()) {
96            skipped.insert(key, ScanOutcome::Skipped(SkipReason::NonRegistrySource));
97            continue;
98        }
99
100        // lockfile holds the pin) — so for a Go `require` dependency the
101        // manifest itself is the authoritative version, not go.sum. go.sum is
102        // a checksum ledger that `go get`/`go build` only ever append to
103        // (only `go mod tidy` prunes it), so its last-occurrence-wins parse
104        // can yield a stale version still recorded in the file but no longer
105        // selected by Go's MVS, silently mismatching whatever's actually in
106        // use. Skipping the lockfile lookup avoids feeding that stale version
107        // to OSV (excludes/replaces fall through to the lockfile lookup below
108        // like any other ecosystem, since their `version_requirement()` is
109        // not an in-use version — see `manifest_requirement_is_resolved_version`).
110        let version = resolve_in_use_version(
111            dep,
112            &normalized_name,
113            resolved_versions,
114            Some(resolved_version_candidates),
115            formatter,
116            ecosystem,
117        );
118
119        let Some(version) = version else {
120            skipped.insert(key, ScanOutcome::Skipped(SkipReason::NoConcreteVersion));
121            continue;
122        };
123
124        let Some(osv_name) = formatter.osv_package_name(dep) else {
125            skipped.insert(key, ScanOutcome::Skipped(SkipReason::UnmappableName));
126            continue;
127        };
128
129        targets.push(deps_core::osv::ScanTarget::new(
130            key,
131            osv_name,
132            formatter.osv_version(&version),
133            version,
134        ));
135    }
136
137    (targets, skipped)
138}

140/// Logs the per-document scan summary unconditionally (critique C1) —
141/// including when every dependency was filtered out before ever reaching
142/// [`deps_core::osv::OsvClient::scan`], which previously produced no log line
143/// at all and defeated §8 invariant 0's purpose of making "not scanned"
144/// observable.
145fn log_osv_run_summary(vulnerabilities: &deps_core::osv::VulnerabilityMap) {
146    let mut clean = 0usize;
147    let mut vulnerable = 0usize;
148    let mut skipped = 0usize;
149    for outcome in vulnerabilities.values() {
150        match outcome {
151            deps_core::osv::ScanOutcome::Clean => clean += 1,
152            deps_core::osv::ScanOutcome::Vulnerable(_) => vulnerable += 1,
153            deps_core::osv::ScanOutcome::Skipped(_) => skipped += 1,
154        }
155    }
156    tracing::info!(
157        "OSV: document scan complete, {} dependencies considered, {clean} clean, {vulnerable} vulnerable, {skipped} skipped",
158        vulnerabilities.len(),
159    );
160}

162/// Phase A output, carried from the concurrently-spawned scan task into
163/// phase B (run later, after the registry fetch resolves — critique S1).
164pub(crate) struct OsvScanResult {
165    /// Document content at the moment the scan started, to guard the
166    /// eventual write against a cross-generation stale commit (critique M4).
167    content_snapshot: String,
168    vulnerabilities: deps_core::osv::VulnerabilityMap,
169    /// `key -> osv_name`, needed to build phase B candidates.
170    osv_name_by_key: HashMap<String, String>,
171    /// `key -> dep.name()` (raw, pre-normalization), the fallback
172    /// `cached_versions` lookup needs since that map is keyed by the raw
173    /// name while `key` is normalized (critique S2) — they differ for
174    /// Composer/Swift/NuGet-style ecosystems.
175    raw_name_by_key: HashMap<String, String>,
176}

178/// Phase A: builds scan targets, runs [`deps_core::osv::OsvClient::scan`], and
179/// merges in the pre-filter skips — all before the registry fetch is known
180/// to have completed, so this must be `tokio::spawn`ed by the caller and run
181/// concurrently with it, never awaited inline (critique S2/original design
182/// note: joining here would gate the inlay-hint refresh that must happen
183/// immediately after the registry fetch).
184///
185/// Returns `None` only when there is nothing to report at all (no
186/// dependencies reached any of steps 0-3, including the pre-filter skips —
187/// i.e. an empty manifest).
188#[tracing::instrument(skip_all, fields(uri = ?uri, ecosystem = ecosystem.id()))]
189pub(crate) async fn run_osv_scan_phase_a(
190    uri: Uri,
191    state: Arc<ServerState>,
192    ecosystem: Arc<dyn Ecosystem>,
193    fetch_timeout_secs: u64,
194) -> Option<OsvScanResult> {
195    let ecosystem_id = ecosystem.ecosystem_id();
196
197    let (content_snapshot, targets, mut vulnerabilities, raw_name_by_key) = {
198        let doc = state.get_document(&uri)?;
199        let parse_result = doc.parse_result()?;
200        let (targets, skipped) = build_scan_targets(
201            parse_result,
202            &doc.resolved_versions,
203            &doc.resolved_version_candidates,
204            ecosystem.formatter(),
205            ecosystem_id,
206        );
207        // Keyed the same way `targets`/`skipped` are (#394 S2: possibly
208        // version-qualified, not just the plain normalized name) so phase B's
209        // `raw_name_by_key.get(key)` fallback below still finds this
210        // occurrence's raw name when its key was disambiguated.
211        let vuln_keys = deps_core::osv::vulnerability_keys(
212            parse_result,
213            &doc.resolved_versions,
214            Some(&doc.resolved_version_candidates),
215            ecosystem.formatter(),
216            ecosystem_id,
217        );
218        let raw_name_by_key: HashMap<String, String> = parse_result
219            .dependencies()
220            .into_iter()
221            .map(|d| {
222                let key = vuln_keys
223                    .get(&d.name_range())
224                    .cloned()
225                    .unwrap_or_else(|| ecosystem.formatter().normalize_package_name(d.name()));
226                (key, d.name().to_string())
227            })
228            .collect();
229        (doc.content.clone(), targets, skipped, raw_name_by_key)
230    };
231
232    if targets.is_empty() && vulnerabilities.is_empty() {
233        return None;
234    }
235
236    let osv_name_by_key: HashMap<String, String> = targets
237        .iter()
238        .map(|t| (t.key.clone(), t.osv_name.clone()))
239        .collect();
240
241    if !targets.is_empty() {
242        let timeout_duration =
243            Duration::from_secs(fetch_timeout_secs.min(OSV_SCAN_TIMEOUT_CEILING_SECS));
244        let scanned = state
245            .osv
246            .scan(ecosystem_id, &targets, timeout_duration)
247            .await;
248        vulnerabilities.extend(scanned);
249    }
250
251    log_osv_run_summary(&vulnerabilities);
252
253    Some(OsvScanResult {
254        content_snapshot,
255        vulnerabilities,
256        osv_name_by_key,
257        raw_name_by_key,
258    })
259}

261/// Background pre-fetch of each dependency's license, for whichever ecosystems
262/// override [`deps_core::Ecosystem::fetch_license`] (issue #660/#688, spec 010 plan §1
263/// tier 3) — today, pub.dev's `/score` endpoint (Dart), the GitHub repository API
264/// (Swift), a Maven Central POM fetch (Gradle), and the JSR per-version API (Deno).
265/// Mirrors
266/// [`run_osv_scan_phase_a`]'s spawn-concurrently-with-the-registry-fetch shape, but
267/// commits directly with no phase B. Callers must `.await` the returned
268/// `JoinHandle` (via `tokio::spawn`) before their own diagnostics publish, exactly
269/// like the OSV `osv_task` join — a tier-3 license-policy violation must be able to
270/// appear in the *first* diagnostics publish after the triggering edit, not only
271/// whenever some later, unrelated event happens to regenerate diagnostics (code-review
272/// round 3 finding #3).
273///
274/// The commit at the end is staleness-guarded (`doc.content == content_snapshot`,
275/// mirroring [`run_osv_phase_b_and_commit`]'s identical guard) and merges rather than
276/// replaces (round 3 finding #1/#2): two overlapping edits can spawn two overlapping
277/// pre-fetches, and without the guard the older one finishing last could silently
278/// overwrite the newer one's results with stale data; without a merge, a transient
279/// per-dependency fetch failure this round (already filtered out below, before this
280/// point) would drop that dependency's previously-cached, still-valid license instead
281/// of just failing to refresh it. `DocumentState::merge_licenses`'s own additive
282/// contract already provides exactly this — a genuinely *removed* dependency's stale
283/// entry is reclaimed separately, by the manifest-diff pruning loop in
284/// `commit_parsed_document`, not by this function replacing the whole map.
285///
286/// **What version each source actually reflects is per-ecosystem, not uniform**
287/// (critic S1 — corrects this doc's previous blanket "only ever targets the
288/// resolved/in-use version" claim): Gradle's POM fetch and Deno's JSR API are
289/// genuinely version-specific (fetched at the dependency's resolved/in-use version, the
290/// `version` passed into [`Ecosystem::fetch_license`]). Dart's `get_license` calls
291/// pub.dev's per-*package* `/score` endpoint, which carries no version parameter at
292/// all — it reflects pana's detection on whatever pub.dev last scored, not necessarily
293/// the resolved version. Swift's `get_license` calls GitHub's `GET /repos/{owner}/{repo}`,
294/// which reflects the repository's *default branch*, not the resolved version's tag.
295/// `resolve_in_use_version` below is still required as a *gate* for all four (no version
296/// resolved means nothing to look up), but for Dart/Swift it does not pin which
297/// version's license is actually returned.
298///
299/// Filters on [`deps_core::lsp_helpers::SourcePolicy::source_is_public_registry_content`]
300/// (critic M3/S6), the same stricter filter [`build_scan_targets`]'s OSV path already
301/// uses, not the looser [`deps_core::lsp_helpers::SourcePolicy::can_resolve_source`]: a
302/// patched git/path fork is resolvable but must never have its license misattributed to
303/// the upstream registry package it forked from — the identical "is this really the
304/// same package" problem OSV's stricter filter exists to solve.
305///
306/// No-op (returns immediately) for every ecosystem whose
307/// <code>ecosystem.[license_source](deps_core::Ecosystem::license_source)().[requires_dedicated_fetch](deps_core::LicenseSource::requires_dedicated_fetch)()</code>
308/// is `false` (issue #697) — every ecosystem except the four above.
309pub(crate) async fn run_license_prefetch(
310    uri: Uri,
311    state: Arc<ServerState>,
312    ecosystem: Arc<dyn Ecosystem>,
313    fetch_timeout_secs: u64,
314) {
315    let ecosystem_id = ecosystem.ecosystem_id();
316    if !ecosystem.license_source().requires_dedicated_fetch() {
317        return;
318    }
319
320    let (content_snapshot, targets): (String, Vec<(PackageName, String)>) = {
321        let Some(doc) = state.get_document(&uri) else {
322            return;
323        };
324        let Some(parse_result) = doc.parse_result() else {
325            return;
326        };
327        let formatter = ecosystem.formatter();
328        let targets = parse_result
329            .dependencies()
330            .into_iter()
331            .filter(|d| formatter.source_is_public_registry_content(&d.source()))
332            .filter_map(|d| {
333                let normalized = formatter.normalize_package_name(d.name());
334                let version = resolve_in_use_version(
335                    d,
336                    normalized.as_str(),
337                    &doc.resolved_versions,
338                    Some(&doc.resolved_version_candidates),
339                    formatter,
340                    ecosystem_id,
341                )?;
342                Some((d.name().clone(), version))
343            })
344            .collect();
345        (doc.content.clone(), targets)
346    };
347
348    if targets.is_empty() {
349        return;
350    }
351
352    use futures::stream::{self, StreamExt};
353
354    let timeout_duration = Duration::from_secs(fetch_timeout_secs.clamp(
355        LICENSE_PREFETCH_TIMEOUT_FLOOR_SECS,
356        LICENSE_PREFETCH_TIMEOUT_CEILING_SECS,
357    ));
358    let ecosystem = &ecosystem;
359    let licenses: HashMap<PackageName, Vec<String>> = stream::iter(targets)
360        .map(|(name, version)| async move {
361            let found = tokio::time::timeout(
362                timeout_duration,
363                ecosystem.fetch_license(name.as_str(), &version),
364            )
365            .await
366            .unwrap_or_else(|_| {
367                tracing::debug!(package = %name, "tier-3 license fetch timed out");
368                Vec::new()
369            });
370            (name, found)
371        })
372        .buffer_unordered(LICENSE_PREFETCH_CONCURRENCY)
373        .filter(|(_, found)| std::future::ready(!found.is_empty()))
374        .collect()
375        .await;
376
377    if let Some(mut doc) = state.documents.get_mut(&uri) {
378        if doc.content == content_snapshot {
379            doc.merge_licenses(licenses);
380        } else {
381            tracing::debug!(
382                "dropping stale tier-3 license pre-fetch result: document content changed mid-fetch"
383            );
384        }
385    }
386}

388/// Bounds how many concurrent per-dependency license fetches [`run_license_prefetch`]
389/// issues at once — same rationale as `fetch_latest_versions_parallel`'s
390/// `max_concurrent`, scaled down: tier-3 documents rarely carry more than a handful of
391/// dependencies (Dart/Swift/Gradle/Deno are all comparatively small ecosystems in this
392/// project's usage), and this pre-fetch is a background nice-to-have, not on the hover
393/// critical path, so there is no latency pressure to fan out aggressively.
394const LICENSE_PREFETCH_CONCURRENCY: usize = 8;
395
396/// Ceiling on the per-dependency tier-3 license fetch timeout, independent of the
397/// configured `fetch_timeout_secs` (critic S4/M2: the direct `Ecosystem::fetch_license`
398/// call in [`run_license_prefetch`] previously had no bound at all, unlike every other
399/// registry call in this codebase, e.g. `fetch_and_classify_package`'s
400/// `tokio::time::timeout(timeout, ...)`). Mirrors
401/// [`OSV_SCAN_TIMEOUT_CEILING_SECS`]'s rationale: the shared `reqwest` client behind
402/// `HttpCache` already imposes its own client-wide 30s timeout, so a per-call timeout
403/// longer than that would never actually bind.
404const LICENSE_PREFETCH_TIMEOUT_CEILING_SECS: u64 = 30;

406/// Floor on the per-dependency tier-3 license fetch timeout, independent of the
407/// configured `fetch_timeout_secs` (issue #692 critic M2). `fetch_timeout_secs` is
408/// user-configurable down to a minimum of 1s (see `config.rs`'s validation), but
409/// `deps_gradle::license::fetch_license_from` may now perform up to
410/// `deps_gradle::license::MAX_POM_FETCHES` **sequential** HTTPS round trips inside the
411/// single `tokio::time::timeout` this budget bounds, where it previously performed one
412/// — a low `fetch_timeout_secs` would otherwise silently starve exactly the
413/// parent-chained artifacts (e.g. Guava) issue #692 exists to resolve. This pre-fetch is
414/// a background nice-to-have, never on the hover critical path (see
415/// [`run_license_prefetch`]'s doc), so raising its effective minimum costs nothing but a
416/// few extra seconds before this best-effort signal gives up.
417const LICENSE_PREFETCH_TIMEOUT_FLOOR_SECS: u64 = 10;

419/// Phase B: for every dependency phase A flagged [`deps_core::osv::ScanOutcome::Vulnerable`],
420/// checks whether the version currently recommended (the registry's latest,
421/// now that the registry fetch has resolved — critique S1) is itself
422/// affected (B.1), then independently verifies each dependency's recommended
423/// *fix target* F (B.2, #462 — see [`run_osv_fix_target_verification`]),
424/// before committing the result into `DocumentState.vulnerabilities`.
425///
426/// Must be called only *after* the registry fetch has updated
427/// `doc.cached_versions`: calling it concurrently with that fetch (as the
428/// original implementation did, by folding phase B into the same spawned
429/// task as phase A) reads `cached_versions` before it holds the registry's
430/// actual latest version, so hover could report the *already-installed*
431/// version as "also affected" instead of the true latest.
432///
433/// The write is guarded against a cross-generation stale commit (critique
434/// M4): `spawn_background_task` aborts the *previous* task only after the
435/// new `DocumentState` is already installed, so an in-flight scan from stale
436/// content could otherwise commit advisories computed against content the
437/// document no longer has. B.1 and B.2 share one `phase_b_deadline` (#462
438/// critic S2/M1) rather than each getting a fresh `fetch_timeout_secs`
439/// budget: B.2 is a second sequential network round-trip added *before* this
440/// guard, so giving it its own full budget would both double phase B's
441/// worst-case wall-clock time (contradicting NFR-002's singular "existing...
442/// budget" framing) and widen the window in which a mid-scan edit discards
443/// this whole result, `upgrade_status` included. Sharing one deadline caps
444/// the total at the original ceiling, same as before this fix existed.
445pub(crate) async fn run_osv_phase_b_and_commit(
446    uri: &Uri,
447    state: &Arc<ServerState>,
448    ecosystem_id: EcosystemId,
449    formatter: &dyn deps_core::lsp_helpers::EcosystemFormatter,
450    fetch_timeout_secs: u64,
451    mut result: OsvScanResult,
452) {
453    let vulnerable_keys: Vec<String> = result
454        .vulnerabilities
455        .iter()
456        .filter(|(_, outcome)| matches!(outcome, deps_core::osv::ScanOutcome::Vulnerable(_)))
457        .map(|(key, _)| key.clone())
458        .collect();
459
460    if !vulnerable_keys.is_empty() {
461        let phase_b_deadline = Instant::now()
462            + Duration::from_secs(fetch_timeout_secs.min(OSV_SCAN_TIMEOUT_CEILING_SECS));
463
464        // B.1 (US-002, unchanged by #462): checks the registry's "latest" candidate.
465        // `latest_native_by_key` is kept for B.2 below, which needs the same native
466        // "latest" string to detect a fix target F that coincides with latest (FR-002)
467        // without re-deriving it from `doc.cached_versions` a second time.
468        let latest_native_by_key: HashMap<String, String> = {
469            let Some(doc) = state.get_document(uri) else {
470                return;
471            };
472            vulnerable_keys
473                .iter()
474                .filter_map(|key| {
475                    let latest = doc
476                        .cached_versions
477                        .get(key.as_str())
478                        .or_else(|| {
479                            let raw = result.raw_name_by_key.get(key)?;
480                            doc.cached_versions.get(raw.as_str())
481                        })?
482                        .latest
483                        .clone();
484                    Some((key.clone(), latest.to_string()))
485                })
486                .collect()
487        };
488
489        let candidates: Vec<deps_core::osv::ScanTarget> = vulnerable_keys
490            .iter()
491            .filter_map(|key| {
492                let osv_name = result.osv_name_by_key.get(key)?.clone();
493                let latest_native = latest_native_by_key.get(key)?.clone();
494                Some(deps_core::osv::ScanTarget::new(
495                    key.clone(),
496                    osv_name,
497                    formatter.osv_version(&latest_native),
498                    latest_native,
499                ))
500            })
501            .collect();
502
503        if !candidates.is_empty() {
504            let timeout_duration = phase_b_deadline.saturating_duration_since(Instant::now());
505            let statuses = state
506                .osv
507                .check_candidates(ecosystem_id, &candidates, timeout_duration)
508                .await;
509            for (key, status) in statuses {
510                if let Some(deps_core::osv::ScanOutcome::Vulnerable(dv)) =
511                    result.vulnerabilities.get_mut(&key)
512                {
513                    dv.upgrade_status = status;
514                }
515            }
516        }
517
518        run_osv_fix_target_verification(
519            &mut result.vulnerabilities,
520            &vulnerable_keys,
521            &result.osv_name_by_key,
522            &latest_native_by_key,
523            ecosystem_id,
524            formatter,
525            &state.osv,
526            phase_b_deadline,
527        )
528        .await;
529    }
530
531    if let Some(mut doc) = state.documents.get_mut(uri) {
532        if doc.content == result.content_snapshot {
533            doc.update_vulnerabilities(result.vulnerabilities);
534        } else {
535            tracing::debug!("dropping stale OSV scan result: document content changed mid-scan");
536        }
537    }
538}

540/// Synthetic [`deps_core::osv::ScanTarget::key`] suffix marking a fix-target (F) live-check
541/// candidate as distinct from the same dependency's "latest" candidate (B.1) within the
542/// shared `VulnerabilityMap` key space — see [`run_osv_fix_target_verification`].
543const FIX_TARGET_KEY_SUFFIX: &str = "\u{0}fix";
544
545/// B.2 (#462): independently verifies each vulnerable dependency's recommended fix target F
546/// (`DependencyVulnerabilities::recommended_fix`'s `version`), which B.1 above never scans —
547/// B.1 only ever checks the registry's "latest" candidate, and F is frequently a different,
548/// older version (the minimal version that clears the advisories B.1's "latest" check did
549/// not already exclude). Without this, `generate_code_actions` could offer F as a verified
550/// fix when OSV was never asked about F itself — see
551/// `deps_core::osv::DependencyVulnerabilities::fix_target_status`'s doc, which this function
552/// populates, and the orphaned-TODO history in this function's git blame (formerly tracked
553/// only by a `// TODO(critic): ... see #216 critique D1` comment; now #462).
554///
555/// Resolution order per dependency, cheapest first:
556/// 1. F equals the already-checked "latest" candidate (FR-002) — reuse `upgrade_status`,
557///    no extra call.
558/// 2. Otherwise, queue a live [`deps_core::osv::OsvClient::check_candidates`] check for F,
559///    batched into a single call across every dependency that reaches this branch (NFR-001)
560///    — never one call per dependency. There is no data-derived shortcut here: a proof
561///    checking F against the advisories `recommended_fix()` computed F *from* is a
562///    tautology at its only call site (critic C1) — phase A only ever queries the
563///    dependency's declared version, so an advisory affecting some version strictly
564///    between declared and F, but not the declared version itself, is invisible to any
565///    check built only from `self.advisories`. Only a live query of F itself can surface
566///    that advisory (the actual gap #462 exists to close).
567///
568/// A dependency whose F fails [`deps_core::lsp_helpers::is_safe_version_string`], or whose
569/// `osv_name` is unavailable, is skipped (logged at `debug`), never crashes the scan. A
570/// dependency whose live check above times out or fails simply keeps `fix_target_status`
571/// left at `NotChecked` (case 2's `HashMap` result never carries that key) — the safe,
572/// fail-closed degradation FR-004/NFR-002 call for: `deps_core::lsp_helpers::code_actions::fix_target_is_verified`
573/// treats an unresolved status as unverified and omits the fix action, so a stalled
574/// verification never surfaces a fix action that was never actually checked.
575#[allow(clippy::too_many_arguments)]
576async fn run_osv_fix_target_verification(
577    vulnerabilities: &mut deps_core::osv::VulnerabilityMap,
578    vulnerable_keys: &[String],
579    osv_name_by_key: &HashMap<String, String>,
580    latest_native_by_key: &HashMap<String, String>,
581    ecosystem_id: EcosystemId,
582    formatter: &dyn deps_core::lsp_helpers::EcosystemFormatter,
583    osv: &deps_core::osv::OsvClient,
584    phase_b_deadline: Instant,
585) {
586    use deps_core::osv::ScanOutcome;
587
588    let (resolved, live_check_candidates) = collect_fix_target_resolutions(
589        vulnerabilities,
590        vulnerable_keys,
591        osv_name_by_key,
592        latest_native_by_key,
593        formatter,
594    );
595
596    for (key, status) in resolved {
597        if let Some(ScanOutcome::Vulnerable(dv)) = vulnerabilities.get_mut(key.as_str()) {
598            dv.fix_target_status = status;
599        }
600    }
601
602    if live_check_candidates.is_empty() {
603        return;
604    }
605
606    let timeout_duration = phase_b_deadline.saturating_duration_since(Instant::now());
607    let statuses = osv
608        .check_candidates(ecosystem_id, &live_check_candidates, timeout_duration)
609        .await;
610
611    apply_live_fix_target_statuses(vulnerabilities, statuses);
612}

614/// Outcome of [`resolve_fix_target`] for one vulnerable dependency.
615#[derive(Debug, PartialEq, Eq)]
616enum FixTargetResolution {
617    /// No fix recommended, F failed [`deps_core::lsp_helpers::is_safe_version_string`], or no
618    /// `osv_name` is on record for this key — nothing to verify or record; `fix_target_status`
619    /// stays untouched (left at `NotChecked`).
620    Skip,
621    /// F's status was resolved without a network call by reusing the already-checked
622    /// "latest" candidate's result (FR-002, F == latest).
623    Resolved(deps_core::osv::UpgradeStatus),
624    /// F differs from latest and needs a live [`deps_core::osv::OsvClient::check_candidates`]
625    /// check — carries the [`deps_core::osv::ScanTarget`] to batch into the caller's single
626    /// combined call (NFR-001), keyed with [`FIX_TARGET_KEY_SUFFIX`] so its result cannot
627    /// collide with the same dependency's "latest" candidate in the same `VulnerabilityMap`
628    /// key space.
629    NeedsLiveCheck(deps_core::osv::ScanTarget),
630}
631
632/// Pure (network-free) decision logic for [`run_osv_fix_target_verification`]'s per-dependency
633/// resolution order — see that function's doc for the two cases and their rationale. Split
634/// out so each case is unit-testable without an `OsvClient`/network dependency.
635fn resolve_fix_target(
636    dv: &deps_core::osv::DependencyVulnerabilities,
637    key: &str,
638    latest_native_by_key: &HashMap<String, String>,
639    osv_name_by_key: &HashMap<String, String>,
640    formatter: &dyn deps_core::lsp_helpers::EcosystemFormatter,
641) -> FixTargetResolution {
642    use deps_core::lsp_helpers::is_safe_version_string;
643    use deps_core::osv::ScanTarget;
644
645    let Some(fix) = dv.recommended_fix() else {
646        return FixTargetResolution::Skip;
647    };
648    let version_native = formatter.osv_version_to_native(&fix.version);
649    if !is_safe_version_string(&version_native) {
650        tracing::debug!(
651            key,
652            version = %fix.version,
653            "OSV #462: fix-target version failed validation, skipping verification"
654        );
655        return FixTargetResolution::Skip;
656    }
657
658    if latest_native_by_key.get(key) == Some(&version_native) {
659        return FixTargetResolution::Resolved(dv.upgrade_status.clone());
660    }
661
662    let Some(osv_name) = osv_name_by_key.get(key).cloned() else {
663        tracing::debug!(
664            key,
665            "OSV #462: no osv_name on record for fix-target verification, skipping"
666        );
667        return FixTargetResolution::Skip;
668    };
669    FixTargetResolution::NeedsLiveCheck(ScanTarget::new(
670        format!("{key}{FIX_TARGET_KEY_SUFFIX}"),
671        osv_name,
672        fix.version,
673        version_native,
674    ))
675}

677/// Pure aggregation step of [`run_osv_fix_target_verification`]: resolves every vulnerable
678/// dependency's fix target via [`resolve_fix_target`], splitting immediately-resolvable
679/// results (`resolved`) from the ones that need a live check (`live_check_candidates`) — the
680/// latter collected into one `Vec` across *every* dependency before the caller's single
681/// `check_candidates` call, so multiple dependencies needing a live check always batch into
682/// one network round-trip rather than one per dependency (NFR-001). Split out from the async
683/// orchestrator specifically so this batching/aggregation behavior is unit-testable without
684/// an `OsvClient`.
685fn collect_fix_target_resolutions(
686    vulnerabilities: &deps_core::osv::VulnerabilityMap,
687    vulnerable_keys: &[String],
688    osv_name_by_key: &HashMap<String, String>,
689    latest_native_by_key: &HashMap<String, String>,
690    formatter: &dyn deps_core::lsp_helpers::EcosystemFormatter,
691) -> (
692    Vec<(String, deps_core::osv::UpgradeStatus)>,
693    Vec<deps_core::osv::ScanTarget>,
694) {
695    use deps_core::osv::ScanOutcome;
696
697    let mut resolved = Vec::new();
698    let mut live_check_candidates = Vec::new();
699
700    for key in vulnerable_keys {
701        let Some(ScanOutcome::Vulnerable(dv)) = vulnerabilities.get(key.as_str()) else {
702            continue;
703        };
704        match resolve_fix_target(dv, key, latest_native_by_key, osv_name_by_key, formatter) {
705            FixTargetResolution::Skip => {}
706            FixTargetResolution::Resolved(status) => resolved.push((key.clone(), status)),
707            FixTargetResolution::NeedsLiveCheck(target) => live_check_candidates.push(target),
708        }
709    }
710
711    (resolved, live_check_candidates)
712}
713
714/// Applies a live [`deps_core::osv::OsvClient::check_candidates`] result keyed with
715/// [`FIX_TARGET_KEY_SUFFIX`] back onto the matching dependency's `fix_target_status`.
716///
717/// A key absent from `statuses` (timeout, OSV outage, or a chunk `check_candidates` itself
718/// dropped) simply leaves that dependency's `fix_target_status` untouched — still
719/// `NotChecked` if it was never set, which is exactly the fail-closed degradation
720/// FR-004/NFR-002 call for (never a panic, never a fabricated "verified" status).
721fn apply_live_fix_target_statuses(
722    vulnerabilities: &mut deps_core::osv::VulnerabilityMap,
723    statuses: HashMap<String, deps_core::osv::UpgradeStatus>,
724) {
725    use deps_core::osv::ScanOutcome;
726
727    for (synthetic_key, status) in statuses {
728        let Some(key) = synthetic_key.strip_suffix(FIX_TARGET_KEY_SUFFIX) else {
729            continue;
730        };
731        if let Some(ScanOutcome::Vulnerable(dv)) = vulnerabilities.get_mut(key) {
732            dv.fix_target_status = status;
733        }
734    }
735}
