14/// Whether a reparse should only re-fetch what `DependencyDiff` calls for, or force a
15/// full re-fetch of every dependency regardless of diff (issue #592).
16///
17/// A config change that alters registry *routing* (`registries.workspace_registries`,
18/// `registries.nuget_user_profile_sources`) feeds `DependencyDiff::compute` an unchanged
19/// dependency set — the manifest didn't change, only where its dependencies resolve from —
20/// so the default `Self::Diff` policy's `deps_to_fetch` would stay empty and
21/// `run_document_change_task`'s `deps_to_fetch.is_empty()` early return would leave the
22/// document displaying versions resolved under the *old* routing indefinitely (the same bug
23/// class #424 already documents for `minimum-stability`).
24#[derive(Debug, Clone, Copy, PartialEq, Eq)]
25pub(crate) enum RefetchPolicy {
26    /// Re-fetch only what `DependencyDiff` calls for (added / version-changed dependencies)
27    /// — correct for every real document edit, where the diff already answers "what needs
28    /// re-checking".
29    Diff,
30    /// Force a re-fetch of every dependency in the new parse result, and drop any
31    /// previously cached version/fetch-failure data before doing so (in
32    /// `fetch_registry_versions_for_change`) — the routing itself changed, so data
33    /// obtained under the old routing can no longer be vouched for.
34    AllDependencies,
35}

37/// Builds `dep_name -> [in_use_version, ...]` (§4.5/§4.6) for every
38/// dependency with a known in-use version, for the yanked-check probe in
39/// `fetch_latest_versions_parallel`. Skips non-registry dependencies
40/// (git/path forks, step 0 of `build_scan_targets`'s ladder) so a patched
41/// fork is never flagged for a registry version it does not contain.
42///
43/// One entry per *occurrence* of a name, not a single collapsed value: the
44/// same dependency name can appear more than once in a manifest (the same
45/// crate under `[dependencies]`/`[dev-dependencies]` or multiple
46/// `[target.'cfg(...)'.dependencies]` blocks — #394). A HashMap keyed by
47/// name alone would silently drop all but the last occurrence's in-use
48/// version from the yanked probe below.
49pub(crate) fn collect_in_use_versions(
50    parse_result: &dyn deps_core::ParseResult,
51    resolved_versions: &HashMap<PackageName, ConcreteVersion>,
52    resolved_version_candidates: &HashMap<PackageName, Vec<ConcreteVersion>>,
53    formatter: &dyn deps_core::lsp_helpers::EcosystemFormatter,
54    ecosystem: EcosystemId,
55) -> HashMap<PackageName, Vec<String>> {
56    let mut map: HashMap<PackageName, Vec<String>> = HashMap::new();
57    for dep in parse_result
58        .dependencies()
59        .into_iter()
60        .filter(|dep| formatter.source_is_public_registry_content(&dep.source()))
61    {
62        let normalized_name = formatter.normalize_package_name(dep.name());
63        if let Some(v) = resolve_in_use_version(
64            dep,
65            &normalized_name,
66            resolved_versions,
67            Some(resolved_version_candidates),
68            formatter,
69            ecosystem,
70        ) {
71            map.entry(dep.name().clone()).or_default().push(v);
72        }
73    }
74    map
75}

77/// Builds `name -> [version_requirement, ...]` for every dependency in `pr`,
78/// one entry per occurrence, the shape `DependencyDiff::compute` needs. A
79/// `HashMap<PackageName, Option<VersionReq>>` (single value per name) would
80/// silently collapse a duplicate name to its last occurrence, losing any
81/// edit made to an earlier one (#394).
82///
83/// Occurrence order is whatever `pr.dependencies()` returns, which for
84/// `deps-cargo` is *not* document order for multiple `[target.*]` blocks
85/// (see `DependencyDiff::compute`'s doc). A consequence worth knowing: if
86/// an edit only renames a `[target.'cfg(...)'.dependencies]` expression
87/// (no version change), that occurrence can sort into a different position
88/// in the new `Vec` than the old one, so `old.get(name) != new.get(name)`
89/// trips even though every individual version requirement is unchanged —
90/// a spurious but harmless `version_changed` (one extra registry
91/// refetch/OSV rescan for that name, never a missed or misattributed one).
92pub(crate) fn dependency_version_map(
93    pr: &dyn deps_core::ParseResult,
94) -> HashMap<PackageName, Vec<Option<VersionReq>>> {
95    let mut map: HashMap<PackageName, Vec<Option<VersionReq>>> = HashMap::new();
96    for d in pr.dependencies() {
97        map.entry(d.name().clone())
98            .or_default()
99            .push(d.version_requirement().cloned());
100    }
101    map
102}

104/// Builds a `cached_versions` map from lock-file-resolved versions, ahead of any registry
105/// fetch.
106///
107/// `available` is deliberately left empty (`PackageVersions::latest_without_list`, not a
108/// plausible-looking one-element list) — this runs before any registry fetch, and
109/// `requirement_is_unsatisfiable` treats an empty `available` as "still loading, skip"
110/// (FR-004). Using `latest_only` here instead would populate a bogus single-entry list and
111/// let the unsatisfiable-requirement check compute a false verdict on every document open,
112/// before the fetch that's supposed to suppress it has a chance to run.
113pub(crate) fn cached_versions_from_lockfile(
114    resolved: &HashMap<PackageName, ConcreteVersion>,
115) -> HashMap<PackageName, PackageVersions> {
116    resolved
117        .iter()
118        .map(|(name, version)| {
119            (
120                name.clone(),
121                PackageVersions::latest_without_list(version.clone()),
122            )
123        })
124        .collect()
125}

127/// Splits a parsed [`deps_core::lockfile::ResolvedPackages`] into the collapsed
128/// `dep_name -> version` map (`ResolvedPackages::iter`, unchanged FR-005 fast path) and a
129/// sibling `dep_name -> [version, ...]` map (issue #649) holding every retained lock-file
130/// entry for names with more than one — built from `ResolvedPackages::iter_all`, and
131/// deliberately omitting a single-occurrence name entirely (NFR-003: the common case never
132/// pays for a candidates-map lookup).
133///
134/// Shared by [`load_resolved_versions`] and `DepsLanguageServer`'s watched-lock-file-change
135/// handler (`server.rs`) — both re-parse a lock file and need the identical split.
136pub(crate) fn split_resolved_packages(
137    resolved: &deps_core::lockfile::ResolvedPackages,
138) -> (
139    HashMap<PackageName, ConcreteVersion>,
140    HashMap<PackageName, Vec<ConcreteVersion>>,
141) {
142    let versions = resolved
143        .iter()
144        .map(|(name, pkg)| (PackageName::new(name.as_str()), pkg.version.clone().into()))
145        .collect();
146    let candidates = resolved
147        .iter_all()
148        .filter(|(_, versions)| versions.len() > 1)
149        .map(|(name, versions)| {
150            (
151                PackageName::new(name.as_str()),
152                versions
153                    .iter()
154                    .map(|pkg| ConcreteVersion::from(pkg.version.clone()))
155                    .collect(),
156            )
157        })
158        .collect();
159    (versions, candidates)
160}

162/// Loads resolved versions from lock file for a given manifest URI.
163///
164/// Uses the ecosystem's lockfile provider to parse the lock file, then
165/// [`split_resolved_packages`]. Both returned maps are empty if no lock file is found or
166/// parsing fails.
167pub(crate) async fn load_resolved_versions(
168    uri: &Uri,
169    state: &ServerState,
170    ecosystem: &dyn Ecosystem,
171) -> (
172    HashMap<PackageName, ConcreteVersion>,
173    HashMap<PackageName, Vec<ConcreteVersion>>,
174) {
175    let lock_provider = match ecosystem.lockfile_provider() {
176        Some(p) => p,
177        None => {
178            tracing::debug!("No lock file provider for ecosystem {}", ecosystem.id());
179            return (HashMap::new(), HashMap::new());
180        }
181    };
182
183    let lockfile_path = match lock_provider.locate_lockfile(uri) {
184        Some(path) => path,
185        None => {
186            tracing::debug!("No lock file found for {:?}", uri);
187            return (HashMap::new(), HashMap::new());
188        }
189    };
190
191    match state
192        .lockfile_cache
193        .get_or_parse(lock_provider.as_ref(), &lockfile_path)
194        .await
195    {
196        Ok(resolved) => {
197            tracing::info!(
198                "Loaded {} resolved versions from {}",
199                resolved.len(),
200                lockfile_path.display()
201            );
202            split_resolved_packages(&resolved)
203        }
204        Err(e) => {
205            tracing::warn!("Failed to parse lock file: {}", e);
206            (HashMap::new(), HashMap::new())
207        }
208    }
209}

215    /// N5 regression guard: the lock-file-population path must build every
216    /// `PackageVersions` with an **empty** `available` list, never a populated one — an
217    /// empty `available` is what makes `requirement_is_unsatisfiable`'s FR-004 guard
218    /// suppress the check before any registry fetch has run. This is the exact function
219    /// `handle_document_open`'s background task calls, so a regression here (e.g.
220    /// swapping `latest_without_list` for `latest_only`) is caught directly, without
221    /// racing the background task.
222    #[test]
223    fn test_cached_versions_from_lockfile_has_empty_available() {
224        let mut resolved = HashMap::new();
225        resolved.insert(PackageName::new("serde"), "1.0.195".into());
226        resolved.insert(PackageName::new("tokio"), "1.35.0".into());
227
228        let cached = cached_versions_from_lockfile(&resolved);
229
230        assert_eq!(cached.len(), 2);
231        let serde = cached.get(&PackageName::new("serde")).unwrap();
232        assert_eq!(serde.latest, "1.0.195");
233        assert!(
234            serde.available.is_empty(),
235            "lock-file-populated entries must have an empty available list, got: {:?}",
236            serde.available
237        );
238        // Issue #227 C3: a locked/pinned version's age is not actionable, so this
239        // instant-display path must never attach a stale `published_at` — there is no
240        // second parallel map here that could drift out of sync with `latest`, since
241        // both live on the same `PackageVersions` entry.
242        assert_eq!(serde.published_at, None);
243        let tokio = cached.get(&PackageName::new("tokio")).unwrap();
244        assert_eq!(tokio.latest, "1.35.0");
245        assert!(tokio.available.is_empty());
246        assert_eq!(tokio.published_at, None);
247    }
248
249    #[test]
250    fn test_cached_versions_from_lockfile_empty_input_is_empty_output() {
251        let resolved = HashMap::new();
252        assert!(cached_versions_from_lockfile(&resolved).is_empty());
253    }
