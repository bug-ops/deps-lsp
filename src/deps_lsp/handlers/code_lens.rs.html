1//! Code lens handler: "Update N outdated dependencies".
2//!
3//! Mirrors `handlers::code_actions`' delegation pattern, but produces a single
4//! document-scoped lens (or none) bound to [`COMMAND_ID`] rather than a per-position
5//! action. The command handler that applies the edit lives in `server::execute_command`,
6//! not here — it needs the raw `Vec<TextEdit>`, not the wrapped `CodeLens`.
7
8use crate::config::DepsConfig;
9use crate::document::{ServerState, ensure_document_loaded};
10use deps_core::VersionData;
11use std::sync::Arc;
12use tokio::sync::RwLock;
13use tower_lsp_server::Client;
14use tower_lsp_server::ls_types::{CodeLens, CodeLensParams};
15
16/// The `workspace/executeCommand` id bound to the lens produced here.
17pub const COMMAND_ID: &str = "deps-lsp.updateAllOutdated";
18
19/// Handles `textDocument/codeLens` requests using trait-based delegation.
20///
21/// Returns zero or one lens for the document. Zero when: the code lens feature is
22/// disabled (`enabled` is `false`); the document cannot be loaded; it has no parse
23/// result; it is not [ready for a batch
24/// update](crate::document::DocumentState::is_ready_for_batch_update) (version data is
25/// still loading, or the document has no known LSP version — the same two conditions
26/// `execute_update_all_outdated` requires, so the lens never renders a click target the
27/// command would then refuse); or every dependency is up to date / not safely editable
28/// (see `deps_core::lsp_helpers::collect_update_all_edits`).
29#[tracing::instrument(
30    skip(state, params, client, config),
31    fields(uri = ?params.text_document.uri, ecosystem = tracing::field::Empty)
32)]
33pub async fn handle_code_lens(
34    state: Arc<ServerState>,
35    params: CodeLensParams,
36    enabled: bool,
37    client: Client,
38    config: Arc<RwLock<DepsConfig>>,
39) -> Vec<CodeLens> {
40    if !enabled {
41        return vec![];
42    }
43
44    let uri = &params.text_document.uri;
45
46    // Ensure document is loaded (cold start support)
47    if !ensure_document_loaded(uri, Arc::clone(&state), client, Arc::clone(&config)).await {
48        tracing::warn!("Could not load document for code lens: {:?}", uri);
49        return vec![];
50    }
51
52    let (offline, severities) = {
53        let config = config.read().await;
54        (config.network.offline, config.diagnostics.to_severities())
55    };
56
57    // Own everything `generate_code_lenses` needs and release the DashMap shard `Ref`
58    // before awaiting it (#333): `with_document` only ever hands `extract` a borrowed
59    // `&DocumentState` synchronously, so the guard can't leak across the `.await` below.
60    let Some((ecosystem, parse_result, content, cached_versions, resolved_versions)) = state
61        .with_document(uri, |doc| {
62            let ecosystem = state.ecosystem_registry.get(doc.ecosystem_id())?;
63
64            // Refuse the same conditions `execute_update_all_outdated` requires before
65            // acting, so the lens never renders a click target the command would then
66            // refuse: version data isn't `Loading` (avoids counting against an empty
67            // cache, also mirrors `diagnostics::generate_diagnostics_internal`), and the
68            // document has a known LSP version (`None` means it was loaded from disk
69            // after a missed `didOpen` — see `DocumentState::is_ready_for_batch_update`).
70            if !doc.is_ready_for_batch_update() {
71                return None;
72            }
73
74            let parse_result = doc.parse_result_arc()?;
75            Some((
76                ecosystem,
77                parse_result,
78                doc.content.clone(),
79                doc.cached_versions.clone(),
80                doc.resolved_versions.clone(),
81            ))
82        })
83        .flatten()
84    else {
85        return vec![];
86    };
87
88    tracing::Span::current().record("ecosystem", ecosystem.id());
89
90    let versions = VersionData::new(&cached_versions, &resolved_versions).with_offline(offline);
91    let mut lenses = ecosystem
92        .generate_code_lenses(
93            parse_result.as_ref(),
94            &content,
95            versions,
96            uri,
97            COMMAND_ID,
98            severities,
99        )
100        .await;
101
102    // M4 (#640): the bulk "Pin N {noun} to commit SHA" lens lives here, not as an
103    // ecosystem's own `generate_code_lenses` override, so no override can silently drop
104    // it. Same on/off toggle as the mutable-ref-pin diagnostic (issue #633): a lens is
105    // push-based/permanently rendered, unlike the pull-based per-position quickfix, so
106    // disabling the flag must suppress the lens too, not just the diagnostic.
107    if severities.mutable_ref_pin_enabled {
108        let count = ecosystem
109            .collect_pin_all_to_sha_edits(parse_result.as_ref(), versions)
110            .len();
111        lenses.extend(deps_core::lsp_helpers::build_pin_all_to_sha_lens(
112            count,
113            ecosystem.pin_all_to_sha_noun(),
114            uri,
115        ));
116    }
117
118    lenses
119}

157    /// #333 liveness regression: `handle_code_lens` must release the DashMap shard
158    /// `Ref` on the document *before* awaiting `Ecosystem::generate_code_lenses`, so a
159    /// concurrent `documents.get_mut` on the same URI (e.g. a `didChange`) is never
160    /// blocked behind an in-flight (or stuck) lens generation.
161    ///
162    /// `BlockingEcosystem::generate_code_lenses` waits on a `Barrier` before blocking
163    /// forever (`std::future::pending`), standing in for an override that performs real
164    /// I/O — the worst case for a shard `Ref` held across the call. The test only
165    /// proceeds to race the writer once that future has demonstrably started executing
166    /// (via the barrier); a concurrent write racing here must complete almost
167    /// immediately, proving the `Ref` was already dropped before the call was awaited.
168    #[tokio::test(flavor = "multi_thread", worker_threads = 4)]
169    async fn test_concurrent_document_write_not_blocked_by_in_flight_code_lens() {
170        use crate::document::DocumentState;
171        use crate::test_utils::blocking_ecosystem::{
172            BlockingEcosystem, BlockingHook, MockParseResult,
173        };
174        use deps_core::ParseResult;
175        use tokio::sync::Barrier;
176
177        let state = Arc::new(ServerState::new());
178        let started = Arc::new(Barrier::new(2));
179        state
180            .ecosystem_registry
181            .register(Arc::new(BlockingEcosystem {
182                started: Arc::clone(&started),
183                hook: BlockingHook::CodeLenses,
184            }));
185
186        let uri = deps_core::test_util::test_uri("/test/Cargo.toml");
187        let content = "[dependencies]\nserde = \"1.0\"\n".to_string();
188        let parse_result: Box<dyn ParseResult> = Box::new(MockParseResult { uri: uri.clone() });
189        let mut doc =
190            DocumentState::new_from_parse_result(EcosystemId::Cargo, content, parse_result);
191        doc.set_version(Some(1));
192        state.update_document(uri.clone(), doc);
193
194        let (client, config) = create_test_client_and_config();
195
196        let handler_task = tokio::spawn({
197            let state = Arc::clone(&state);
198            let uri = uri.clone();
199            async move { handle_code_lens(state, params(uri), true, client, config).await }
200        });
201
202        // Block until `generate_code_lenses` has actually started executing — i.e.
203        // `handle_code_lens` has reached (and is now inside) the await — before racing
204        // the writer below. Timeout-wrapped so a regression that makes the handler
205        // never reach the awaited call fails loudly instead of hanging forever.
206        tokio::time::timeout(std::time::Duration::from_secs(5), started.wait())
207            .await
208            .expect("handle_code_lens did not reach generate_code_lenses within 5s");
209
210        // Spawned onto its own task (rather than awaited inline) deliberately: see
211        // `completion.rs`'s equivalent #319 regression test for why `DashMap::get_mut`
212        // needs a real async yield point to race against `tokio::time::timeout`.
213        let write_task = tokio::spawn({
214            let state = Arc::clone(&state);
215            let uri = uri.clone();
216            async move {
217                state.documents.get_mut(&uri).unwrap().set_loading();
218            }
219        });
220        let write_result =
221            tokio::time::timeout(std::time::Duration::from_millis(500), write_task).await;
222
223        handler_task.abort();
224
225        assert!(
226            write_result.is_ok(),
227            "#333 regression: a concurrent documents.get_mut on the same URI must not \
228             block on an in-flight generate_code_lenses call — the DashMap shard Ref \
229             must be dropped before the call is awaited, not after it"
230        );
231    }

350    /// Cross-ecosystem consistency matrix for `collect_update_all_edits` (required by
351    /// `.claude/rules/continuous-improvement.md`'s cross-ecosystem rule and §6 of the
352    /// design plan). Exercises the real per-ecosystem parser/formatter, including the
353    /// fragile ecosystems the §4.4 literal-span guard must skip: every "passes" fixture
354    /// asserts the *resulting document text is a valid, re-parseable declaration* (not
355    /// merely "an edit exists"); every "skipped" fixture asserts no edit is produced at
356    /// all, proving the guard — not the absence of a fixture — is what stops it.
357    // `clippy::string_slice` has no `allow-*-in-tests` clippy.toml knob (unlike
358    // indexing_slicing/unwrap_used/expect_used), so it needs an explicit allow here.
359    #[allow(clippy::string_slice)]
360    mod cross_ecosystem_tests {
361        use super::*;
362        use std::collections::HashMap;
363        use tower_lsp_server::ls_types::TextEdit;
364
365        /// Applies a single `TextEdit` to `content`, using the exact inverse of the
366        /// `Position`-to-byte-offset conversion `collect_update_all_edits` used to build
367        /// the edit's range in the first place.
368        fn apply_single_edit(content: &str, edit: &TextEdit) -> String {
369            let table = deps_core::LineOffsetTable::new(content);
370            let start = table.position_to_byte_offset(content, edit.range.start);
371            let end = table.position_to_byte_offset(content, edit.range.end);
372            format!("{}{}{}", &content[..start], edit.new_text, &content[end..])
373        }
374
375        /// Asserts the ecosystem produces exactly one edit for `content`, and that
376        /// applying it yields text which both contains `expected_fragment` and still
377        /// parses successfully under the same ecosystem parser.
378        async fn assert_single_edit_produces_valid_declaration(
379            ecosystem: &dyn deps_core::Ecosystem,
380            uri: &tower_lsp_server::ls_types::Uri,
381            content: &str,
382            cached: HashMap<deps_core::PackageName, deps_core::PackageVersions>,
383            expected_fragment: &str,
384        ) {
385            // Held per `deps_core::fs_probe::snapshot_guard`'s doc: called with every
386            // ecosystem under test here (including cargo/npm/nuget/gradle, which
387            // transitively touch fs_probe), and this shared helper runs in the same binary
388            // as `document/loader.rs`'s diffing test.
389            let _guard = deps_core::fs_probe::snapshot_guard_async().await;
390            let parse_result = ecosystem
391                .parse_manifest(content, uri)
392                .await
393                .expect("fixture must parse");
394            let resolved = HashMap::new();
395            let edits = deps_core::collect_update_all_edits(
396                parse_result.as_ref(),
397                content,
398                deps_core::VersionData::new(&cached, &resolved),
399                ecosystem.formatter(),
400            );
401            assert_eq!(edits.len(), 1, "expected exactly one edit for this fixture");
402
403            let new_content = apply_single_edit(content, &edits[0]);
404            assert!(
405                new_content.contains(expected_fragment),
406                "resulting text should contain {expected_fragment:?}, got: {new_content}"
407            );
408            ecosystem
409                .parse_manifest(&new_content, uri)
410                .await
411                .unwrap_or_else(|e| panic!("resulting text failed to re-parse: {e}"));
412        }

782        /// Seeds `ecosystem`'s shared `TagIndex` for `name` with one `tag -> sha` entry,
783        /// downcasting through `Ecosystem::registry()`/`Registry::as_any()` — this test
784        /// module never drives a live registry fetch (`collect_update_all_edits` is pure
785        /// over `parse_result`/`content`/cached `VersionData`), so the index has to be
786        /// seeded directly for the SHA-pin `format_version_replacing_for` branch to have
787        /// anything to resolve.
788        #[cfg(feature = "github-actions")]
789        fn seed_gha_tag_index(
790            ecosystem: &dyn deps_core::Ecosystem,
791            name: &str,
792            tag: &str,
793            sha: &str,
794        ) {
795            let registry = ecosystem.registry();
796            let gha_registry = registry
797                .as_any()
798                .downcast_ref::<deps_github_actions::GithubActionsRegistry>()
799                .expect("github-actions ecosystem must back onto a GithubActionsRegistry");
800            let mut index = deps_github_actions::registry::TagIndex::default();
801            index.tag_to_sha.insert(tag.to_string(), sha.to_string());
802            index.sha_to_tag.insert(sha.to_string(), tag.to_string());
803            gha_registry.tag_index().insert(
804                deps_core::PackageName::new(name),
805                std::sync::Arc::new(index),
806            );
807        }
