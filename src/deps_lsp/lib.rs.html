10//! The `deps-lsp` binary crate: wires the 14 ecosystem crates into a running
11//! `tower-lsp-server` [`LanguageServer`](tower_lsp_server::LanguageServer)
12//! implementation.
13//!
14//! [`register_ecosystems`] registers every feature-enabled ecosystem crate
15//! against an [`EcosystemRegistry`], threading live-updatable settings
16//! ([`EcosystemRuntime`]) into the ones that need them. [`server::Backend`]
17//! is the `LanguageServer` implementation itself; `document` holds the
18//! per-document state machine driving hover/completion/diagnostics.

39/// Live-updatable settings [`register_ecosystems`] threads into every ecosystem that needs them.
40///
41/// Bundled into one struct (issue #561, M3) rather than growing that function's arity again
42/// for each new cross-ecosystem live flag.
43#[non_exhaustive]
44#[derive(Debug, Clone)]
45pub struct EcosystemRuntime {
46    /// Gates every workspace-declared registry index host (spec #443,
47    /// `registries.workspace_registries`).
48    pub policy: Arc<deps_core::net_policy::RegistryAccessPolicy>,
49    /// `registries.nuget_user_profile_sources` (issue #561, FR-006) — whether a NuGet
50    /// user-profile-tier `NuGet.Config` source with no repo-declared counterpart becomes a
51    /// routing hop, not just a credential source. Default `false`.
52    pub nuget_user_profile_sources: Arc<AtomicBool>,
53    /// `registries.gitlab_instance_host` (issue #466, spec FR-005a/FR-011a) — the raw
54    /// configured GitLab instance host string, or `None` when unset. A feature-agnostic
55    /// `Arc<RwLock<Option<String>>>` (not a `deps-gitlab-ci` type) since this struct is
56    /// un-`cfg`'d — see `deps_gitlab_ci::host::GitlabInstanceHost`'s docs for why host
57    /// validation lives in that crate instead, applied on read.
58    pub gitlab_instance_host: Arc<std::sync::RwLock<Option<String>>>,
59}
60
61impl EcosystemRuntime {
62    /// Constructs the runtime from its three live-updatable settings handles.
63    ///
64    /// Needed because [`Self`] is `#[non_exhaustive]`: a struct literal only works inside
65    /// this crate, so every other crate — including anything embedding [`register_ecosystems`],
66    /// this lib's advertised entry point — must use this constructor instead. Unlike this
67    /// crate's config-DTO builders (e.g. [`config::InlayHintsConfig::new`]), all three fields
68    /// are required here: none has a sensible default, so there is no accompanying `with_*`
69    /// chain — a future field can still add one without breaking this signature.
70    ///
71    /// # Examples
72    ///
73    /// ```
74    /// use deps_core::net_policy::RegistryAccessPolicy;
75    /// use deps_lsp::EcosystemRuntime;
76    /// use std::sync::atomic::AtomicBool;
77    /// use std::sync::{Arc, RwLock};
78    ///
79    /// let runtime = EcosystemRuntime::new(
80    ///     Arc::new(RegistryAccessPolicy::default()),
81    ///     Arc::new(AtomicBool::new(false)),
82    ///     Arc::new(RwLock::new(None)),
83    /// );
84    /// assert!(!runtime.nuget_user_profile_sources.load(std::sync::atomic::Ordering::Relaxed));
85    /// ```
86    #[must_use]
87    pub fn new(
88        policy: Arc<deps_core::net_policy::RegistryAccessPolicy>,
89        nuget_user_profile_sources: Arc<AtomicBool>,
90        gitlab_instance_host: Arc<std::sync::RwLock<Option<String>>>,
91    ) -> Self {
92        Self {
93            policy,
94            nuget_user_profile_sources,
95            gitlab_instance_host,
96        }
97    }
98}

100/// Declares an ecosystem: re-exports types and registers at runtime.
101///
102/// `$types` re-export rule (#834 §6 last bullet — the list used to be asymmetric with no
103/// stated rule: `*Formatter` re-exported for 8/14 ecosystems, `*LockParser` for 4/14,
104/// `*Registry` for 13/14): **always list every `*Registry`/`*Formatter`/`*LockParser`
105/// (or ecosystem-specific-named lock parser, e.g. `GoSumParser`) type the ecosystem crate
106/// actually defines and re-exports from its own crate root.** Omit only when the crate
107/// genuinely has none — e.g. `deps-gradle` has no `GradleRegistry` (reuses
108/// `deps_maven::MavenCentralRegistry` directly), and several ecosystems omit a
109/// `*LockParser` entry because the crate has no `lockfile` module at all (Maven, Gradle,
110/// GitHub Actions, GitLab CI — no lock file format exists for these; Deno — `deno.lock`
111/// exists as a format, but this crate has no parser for it yet, a real coverage gap rather
112/// than "no format" like the other four).
113macro_rules! ecosystem {
114    ($feature:literal, $crate_name:ident, $ecosystem:ident, [$($types:ident),* $(,)?]) => {
115        #[cfg(feature = $feature)]
116        pub use $crate_name::{$ecosystem, $($types),*};
117    };
118}
119
120/// Registers ecosystem if feature is enabled.
121macro_rules! register {
122    ($feature:literal, $ecosystem:ident, $registry:expr, $cache:expr) => {
123        #[cfg(feature = $feature)]
124        $registry.register(Arc::new($ecosystem::new(Arc::clone($cache))));
125    };
126}

357/// Registers all enabled ecosystems.
358///
359/// `cargo` is special-cased (spec #443/#441, plan-1b §1.6): unlike `register!`'s generic
360/// `Ecosystem::new(cache)` call, `CargoEcosystem` needs `policy` threaded through
361/// `CargoEcosystem::with_context` so `ServerState`'s live-updatable
362/// `Arc<RegistryAccessPolicy>` (see `document::state::ServerState::registry_policy`) is the
363/// exact same handle every Cargo parse reads — `initialize`/`did_change_configuration`
364/// updating it then takes effect immediately, with no need to reconstruct the ecosystem.
365///
366/// `npm` and `deno` are special-cased (#312): when both features are enabled, they share
367/// one `NpmRegistry` instance — built once here and handed to both `NpmEcosystem` and
368/// `DenoEcosystem`'s `npm:`-scheme half via `with_registry`/`with_npm` — instead of each
369/// constructing its own. `NpmRegistry` is cheaply `Clone` (its `HttpCache` and
370/// freshness-path publish-time map are both `Arc`-wrapped internally), so this dedupes the
371/// freshness path's full-packument fetch and its publish-time cache for a package
372/// appearing in both `package.json` and a `deno.json` `npm:`-specifier dependency, on top
373/// of the plain cached GETs the shared `cache` already dedupes.
374/// Returns every ecosystem id this call threaded the live `RegistryAccessPolicy` handle
375/// into (issue #592 security M1) — the single source of truth `config::reparse_scope`'s
376/// caller consults to scope a `registries.workspace_registries` reparse, so that set can
377/// never drift from what this function actually wires up. Adding a 6th policy-consuming
378/// ecosystem means editing this function anyway (to thread `policy` through its parse
379/// context); pushing its id onto the returned list at that same call site keeps the two
380/// facts — "receives the policy" and "is in the reparse scope" — physically inseparable,
381/// rather than duplicated across two independently-editable places.
382pub fn register_ecosystems(
383    registry: &EcosystemRegistry,
384    cache: Arc<HttpCache>,
385    runtime: &EcosystemRuntime,
386) -> Vec<&'static str> {
387    let policy = Arc::clone(&runtime.policy);
388    // Keeps `policy` used even when the `cargo` feature (its only consumer) is compiled out.
389    let _ = &policy;
390    let mut workspace_registry_ecosystems = Vec::new();
391
392    #[cfg(feature = "cargo")]
393    {
394        let context = deps_cargo::parser::CargoParseContext::new(
395            Arc::clone(&policy),
396            Arc::new(deps_cargo::config::ConfigFileCache::new()),
397        );
398        registry.register(Arc::new(CargoEcosystem::with_context(
399            Arc::clone(&cache),
400            context,
401        )));
402        workspace_registry_ecosystems.push("cargo");
403    }
404
405    #[cfg(all(feature = "npm", feature = "deno"))]
406    {
407        let npm_context = deps_npm::config::NpmParseContext::new(
408            Arc::clone(&policy),
409            Arc::new(deps_npm::config::NpmConfigCache::new()),
410            Arc::new(deps_npm::catalog::PnpmWorkspaceCache::new()),
411        );
412        let npm_registry = Arc::new(NpmRegistry::new(Arc::clone(&cache)));
413        registry.register(Arc::new(NpmEcosystem::with_context(
414            Arc::clone(&npm_registry),
415            npm_context,
416        )));
417        workspace_registry_ecosystems.push("npm");
418        // `DenoEcosystem::with_npm` shares the registry above but is never handed `policy`
419        // itself (its own `.npmrc`-style workspace registry concept doesn't exist yet), so
420        // "deno" deliberately never joins this list.
421        registry.register(Arc::new(DenoEcosystem::with_npm(
422            Arc::clone(&cache),
423            npm_registry.as_ref().clone(),
424        )));
425    }
426    // npm is written out explicitly rather than via `register!` (spec 032, S3): that macro's
427    // `NpmEcosystem::new(cache)` would give npm a default, disconnected `NpmParseContext` —
428    // its `.npmrc` reachability policy would never see a live `initialize`/
429    // `didChangeConfiguration` update.
430    #[cfg(all(feature = "npm", not(feature = "deno")))]
431    {
432        let npm_context = deps_npm::config::NpmParseContext::new(
433            Arc::clone(&policy),
434            Arc::new(deps_npm::config::NpmConfigCache::new()),
435            Arc::new(deps_npm::catalog::PnpmWorkspaceCache::new()),
436        );
437        registry.register(Arc::new(NpmEcosystem::with_context(
438            Arc::new(NpmRegistry::new(Arc::clone(&cache))),
439            npm_context,
440        )));
441        workspace_registry_ecosystems.push("npm");
442    }
443    #[cfg(all(feature = "deno", not(feature = "npm")))]
444    register!("deno", DenoEcosystem, registry, &cache);
445
446    // pypi is written out explicitly rather than via `register!` (spec 033, mirroring npm's
447    // spec 032 S3 precedent): that macro's `PypiEcosystem::new(cache)` would give pypi a
448    // default, disconnected `RegistryAccessPolicy` — its private-index reachability policy
449    // would never see a live `initialize`/`didChangeConfiguration` update.
450    #[cfg(feature = "pypi")]
451    {
452        registry.register(Arc::new(PypiEcosystem::with_policy(
453            Arc::new(PypiRegistry::new(Arc::clone(&cache))),
454            Arc::clone(&policy),
455        )));
456        workspace_registry_ecosystems.push("pypi");
457    }
458
459    // go is written out explicitly rather than via `register!` (spec 034, mirroring npm's
460    // spec 032 S3 precedent): that macro's `GoEcosystem::new(cache)` would give Go a
461    // default, disconnected `GoParseContext` — its `$GOENV` reachability policy would never
462    // see a live `initialize`/`didChangeConfiguration` update.
463    #[cfg(feature = "go")]
464    {
465        let go_context = deps_go::config::GoParseContext::new(
466            Arc::clone(&policy),
467            Arc::new(deps_go::config::GoEnvCache::new()),
468            deps_go::config::goenv_path(),
469        );
470        registry.register(Arc::new(GoEcosystem::with_context(
471            Arc::new(GoRegistry::new(Arc::clone(&cache))),
472            go_context,
473        )));
474        workspace_registry_ecosystems.push("go");
475    }
476    register!("bundler", BundlerEcosystem, registry, &cache);
477    register!("dart", DartEcosystem, registry, &cache);
478    register!("maven", MavenEcosystem, registry, &cache);
479    register!("gradle", GradleEcosystem, registry, &cache);
480    register!("swift", SwiftEcosystem, registry, &cache);
481    register!("composer", ComposerEcosystem, registry, &cache);
482
483    // nuget is written out explicitly rather than via `register!` (issue #523, mirroring
484    // npm's/pypi's identical precedent): that macro's `NuGetEcosystem::new(cache)` would give
485    // nuget a default, disconnected `RegistryAccessPolicy` — its private-feed reachability
486    // policy would never see a live `initialize`/`didChangeConfiguration` update.
487    #[cfg(feature = "nuget")]
488    {
489        let nuget_context = deps_nuget::config::NuGetParseContext::new(
490            Arc::clone(&policy),
491            Arc::new(deps_nuget::config::NuGetConfigCache::new()),
492            Arc::clone(&runtime.nuget_user_profile_sources),
493        );
494        registry.register(Arc::new(NuGetEcosystem::with_context(
495            Arc::new(NuGetRegistry::new(Arc::clone(&cache))),
496            nuget_context,
497        )));
498        workspace_registry_ecosystems.push("nuget");
499    }
500
501    register!("github-actions", GithubActionsEcosystem, registry, &cache);
502
503    // gitlab-ci is written out explicitly rather than via `register!` (issue #466, mirroring
504    // github-actions'/nuget's identical precedent): that macro's `GitlabCiEcosystem::new(cache)`
505    // would give it a default, disconnected `registries.gitlab_instance_host` — its
506    // self-hosted-instance resolution and the single token-host rule (spec FR-005a/FR-011a)
507    // would never see a live `initialize`/`didChangeConfiguration` update.
508    #[cfg(feature = "gitlab-ci")]
509    registry.register(Arc::new(GitlabCiEcosystem::with_context(
510        Arc::clone(&cache),
511        Arc::clone(&policy),
512        Arc::clone(&runtime.gitlab_instance_host),
513    )));
514
515    workspace_registry_ecosystems
516}

530    /// Smoke test: `register_ecosystems` must not panic under any feature combination.
531    /// Per-ecosystem "is it actually registered" coverage moved to
532    /// [`test_ecosystem_id_all_registered`] (#758) — driven by
533    /// [`deps_core::EcosystemId::ALL`] instead of this hand-written, drift-prone 14-line list.
534    #[test]
535    fn test_register_ecosystems() {
536        let registry = Arc::new(EcosystemRegistry::new());
537        let cache = Arc::new(HttpCache::new());
538        register_ecosystems(&registry, Arc::clone(&cache), &test_runtime());
539    }

541    /// Issue #592 security M1: every id `register_ecosystems` returns must actually be a
542    /// registered ecosystem (catches a typo'd `push` literal) and, for this feature set,
543    /// must exactly match the five ecosystems known to thread `RegistryAccessPolicy` through
544    /// their parse context — a regression here means either a policy-consuming ecosystem
545    /// was added without pushing its id (fails closed for `config::reparse_scope`), or an id
546    /// was pushed for an ecosystem that no longer receives the policy (harmless over-scoping,
547    /// but signals the two facts drifted anyway).
548    #[test]
549    #[allow(
550        clippy::vec_init_then_push,
551        reason = "each push is independently feature-gated, so a `vec![]` literal can't \
552                  express the feature-conditional membership"
553    )]
554    fn test_register_ecosystems_workspace_registry_list_matches_registered_ecosystems() {
555        let registry = Arc::new(EcosystemRegistry::new());
556        let cache = Arc::new(HttpCache::new());
557        let workspace_registry_ecosystems =
558            register_ecosystems(&registry, Arc::clone(&cache), &test_runtime());
559
560        for id in &workspace_registry_ecosystems {
561            assert!(
562                registry.get(id).is_some(),
563                "{id:?} was returned as policy-consuming but is not a registered ecosystem"
564            );
565        }
566
567        let mut expected = Vec::new();
568        #[cfg(feature = "cargo")]
569        expected.push("cargo");
570        #[cfg(feature = "npm")]
571        expected.push("npm");
572        #[cfg(feature = "pypi")]
573        expected.push("pypi");
574        #[cfg(feature = "go")]
575        expected.push("go");
576        #[cfg(feature = "nuget")]
577        expected.push("nuget");
578        expected.sort_unstable();
579        let mut actual = workspace_registry_ecosystems.clone();
580        actual.sort_unstable();
581        assert_eq!(
582            actual, expected,
583            "workspace-registry-policy ecosystem set changed — update this test's `expected` \
584             list alongside whatever registration change caused it"
585        );
586    }

588    /// Layer 1a (#758): completeness — every [`deps_core::EcosystemId::ALL`] variant must
589    /// actually be registered by [`register_ecosystems`]. Driven by `ALL` itself, not
590    /// `registry.ecosystem_ids()`, so an ecosystem declared in the enum but never wired in
591    /// fails this test instead of silently vanishing from coverage.
592    ///
593    /// Gated on every ecosystem feature at once: `ALL` always lists 14 variants regardless of
594    /// which features are enabled for this build, so this specific claim — "all 14 are
595    /// present" — is only meaningful, and only makes sense to check, in an all-features build.
596    /// The per-ecosystem invariants that don't depend on all 14 being present live in the
597    /// ungated [`test_registered_ecosystems_universal_invariants`] instead (#758 impl-critic
598    /// S2): unlike this completeness check, those must keep working under any feature subset,
599    /// the way the two hand-written per-feature lists this pair replaces used to.
600    #[cfg(all(
601        feature = "cargo",
602        feature = "npm",
603        feature = "pypi",
604        feature = "go",
605        feature = "bundler",
606        feature = "dart",
607        feature = "maven",
608        feature = "gradle",
609        feature = "swift",
610        feature = "composer",
611        feature = "nuget",
612        feature = "deno",
613        feature = "github-actions",
614        feature = "gitlab-ci"
615    ))]
616    #[test]
617    fn test_ecosystem_id_all_registered() {
618        let registry = Arc::new(EcosystemRegistry::new());
619        let cache = Arc::new(HttpCache::new());
620        register_ecosystems(&registry, Arc::clone(&cache), &test_runtime());
621
622        for id in deps_core::EcosystemId::ALL {
623            assert!(
624                registry.get(id.id()).is_some(),
625                "{id:?} is in EcosystemId::ALL but was not registered by register_ecosystems"
626            );
627        }
628    }

630    /// Layer 1b (#758): universal, offline invariants every ecosystem this build actually
631    /// registers must satisfy. Deliberately **ungated** — unlike
632    /// [`test_ecosystem_id_all_registered`]'s completeness claim, none of these invariants
633    /// depend on all 14 ecosystems being present, so this iterates whatever

740    /// Whether `formatter`'s own comparator (preferring
741    /// [`deps_core::lsp_helpers::RequirementResolution::compile_requirement`], falling back to
742    /// [`deps_core::lsp_helpers::RequirementResolution::version_satisfies_requirement`] when it
743    /// declines to compile) treats a bare `requirement` as an exact pin: it must match
744    /// `requirement` itself but reject both a higher patch (`"1.2.9"`) and a higher
745    /// minor/major (`"9.9.9"`) — the same "matches only this one version" shape
746    /// [`deps_core::lsp_helpers::concrete_pin_version`] asserts. Also correctly says `false`
747    /// for a genuine partial-version range (e.g. `"1.2"`), since that legitimately matches
748    /// more than one candidate.
749    #[cfg(any(
750        feature = "cargo",
751        feature = "npm",
752        feature = "go",
753        feature = "bundler",
754        feature = "dart",
755        feature = "maven",
756        feature = "composer",
757        feature = "gradle",
758        feature = "nuget",
759        feature = "deno",
760        feature = "github-actions",
761        feature = "gitlab-ci",
762        feature = "swift",
763        feature = "pypi"
764    ))]
765    fn formatter_treats_bare_version_as_exact_pin(
766        formatter: &dyn deps_core::lsp_helpers::EcosystemFormatter,
767        bare: &str,
768    ) -> bool {
769        use deps_core::{ConcreteVersion, VersionReq};
770
771        let requirement = VersionReq::new(bare);
772        let matches = |candidate: &str| -> bool {
773            let version = ConcreteVersion::from(candidate);
774            if let Some(matcher) = formatter.compile_requirement(&requirement) {
775                matcher.matches(&version) == Some(true)
776            } else {
777                formatter.version_satisfies_requirement(&version, bare)
778            }
779        };
780
781        matches(bare) && !matches("9.9.9") && !matches("1.2.9")
782    }

784    /// How a given ecosystem's bare-version-is-a-pin verdict relates to its own formatter's
785    /// comparator — see [`bare_version_agreement_expectation`].
786    #[cfg(any(
787        feature = "cargo",
788        feature = "npm",
789        feature = "go",
790        feature = "bundler",
791        feature = "dart",
792        feature = "maven",
793        feature = "composer",
794        feature = "gradle",
795        feature = "nuget",
796        feature = "deno",
797        feature = "github-actions",
798        feature = "gitlab-ci",
799        feature = "swift",
800        feature = "pypi"
801    ))]
802    #[derive(Debug, Clone, Copy, PartialEq, Eq)]
803    enum BareVersionAgreementExpectation {
804        /// `deps-core`'s `concrete_pin_version` and the ecosystem's own comparator must
805        /// agree on whether a bare full version (`"1.2.3"`) is an exact pin. When `true`,
806        /// also checked against a bare *partial* version (`"1.2"`, impl-critic M1) —
807        /// gated per-ecosystem because a partial-version *requirement* is not a concept
808        /// every `Concrete`-policy ecosystem actually has: Go's own comparator, for
809        /// instance, treats a bare string as a version *prefix* to support pseudo-version
810        /// and `+incompatible`-suffix matching (`go_version_matches`), which makes it
811        /// (correctly, for its real purpose) accept `"1.2"` against a candidate `"1.2.9"`
812        /// — a false "divergence" against `deps-core` if compared as though `"1.2"` were
813        /// a genuine partial-version requirement, which go.mod's `require` directive
814        /// never actually contains (it is always a complete version). Only the
815        /// ecosystems with a real bare-partial-is-a-range grammar (`AlwaysRange`'s Cargo
816        /// caret, `ConcreteIfFullVersion`'s X-range/moving-tag ecosystems) get `true`.
817        Checked { test_partial: bool },
818        /// The ecosystem's own comparator would disagree with `deps-core`'s verdict, but
819        /// the parser can never actually emit a bare requirement in the first place, so the
820        /// divergence never reaches `concrete_pin_version` in practice (Swift, PyPI).
821        LatentOnly,
822        /// `deps-core` deliberately reports a bare requirement as a pin even though the
823        /// ecosystem's own comparator disagrees — a documented approximation, not an
824        /// oversight (NuGet).
825        DeliberateApproximation,
826    }
827
828    /// #669 regression guard: `deps-core`'s `bare_requirement_policy` hand-maintains a
829    /// per-ecosystem model of "is a bare version requirement a pin or a range", but every
830    /// ecosystem's own formatter already has the authoritative answer via
831    /// `compile_requirement`/`version_satisfies_requirement`, and nothing kept the two in
832    /// sync — this already caused two shipped bugs (#664 npm/Composer, #667 Deno). For every
833    /// ecosystem this crate can register, classifies it via [`BareVersionAgreementExpectation`]
834    /// and checks the matching invariant: `Checked` ecosystems must agree on both a bare full
835    /// version (`"1.2.3"`) and a bare partial version (`"1.2"`, impl-critic M1); `LatentOnly`
836    /// and `DeliberateApproximation` ecosystems must instead still exhibit the disagreement
837    /// their exemption relies on (impl-critic M2) — so an alignment on either side (a parser
838    /// change, a comparator change) fails this test instead of silently going stale.
839    ///
840    /// The `match` in [`bare_version_agreement_expectation`] is deliberately exhaustive
841    /// (`.claude/CLAUDE.md`'s bug-class-#118 rule): a future 15th ecosystem must get an
842    /// explicit arm — added to `Checked` or listed as a commented, reviewed exemption —
843    /// rather than silently falling through a wildcard.
844    #[cfg(any(
845        feature = "cargo",
846        feature = "npm",
847        feature = "go",
848        feature = "bundler",
849        feature = "dart",
850        feature = "maven",
851        feature = "composer",
852        feature = "gradle",
853        feature = "nuget",
854        feature = "deno",
855        feature = "github-actions",
856        feature = "gitlab-ci",
857        feature = "swift",
858        feature = "pypi"
859    ))]
860    fn bare_version_agreement_expectation(
861        id: deps_core::EcosystemId,
862    ) -> BareVersionAgreementExpectation {
863        use BareVersionAgreementExpectation::{Checked, DeliberateApproximation, LatentOnly};
864
865        match id {
866            // Cargo (caret), and the `ConcreteIfFullVersion` ecosystems (npm/Composer/Deno's
867            // X-ranges, GitHub Actions'/GitLab CI's moving-major tags): a bare *partial*
868            // version is a real, distinct requirement shape from a bare full version under
869            // these ecosystems' own grammar, so both are worth checking.
870            deps_core::EcosystemId::Cargo
871            | deps_core::EcosystemId::Npm
872            | deps_core::EcosystemId::Composer
873            | deps_core::EcosystemId::Deno
874            | deps_core::EcosystemId::GithubActions
875            | deps_core::EcosystemId::GitlabCi => Checked { test_partial: true },
876            // Go/Bundler/Dart/Maven/Gradle: no partial-version requirement concept exists in
877            // these ecosystems' own manifests (a bare version is always a complete one), so a
878            // synthetic partial input like `"1.2"` isn't a meaningful requirement to compare —
879            // only the full-version case is checked. (Go's own comparator in particular
880            // treats a bare string as a version *prefix*, for pseudo-version/`+incompatible`
881            // matching, not as a partial-version range — comparing it against `"1.2"` as if it
882            // were a partial-range requirement produces a false divergence.)
883            deps_core::EcosystemId::Go
884            | deps_core::EcosystemId::Bundler
885            | deps_core::EcosystemId::Dart
886            | deps_core::EcosystemId::Maven
887            | deps_core::EcosystemId::Gradle => Checked {
888                test_partial: false,
889            },
890            // Swift: `SwiftFormatter::compile_requirement` parses a requirement via
891            // `semver::VersionReq`, whose bare-string default is a caret range — the same
892            // divergence NuGet has — but `deps-swift`'s parser always emits an explicit
893            // range spelling (`">=X, <Y"`) or an exact `"=X"` pin, never a bare `"X.Y.Z"`
894            // string (see `deps-swift/src/parser.rs`'s `upToNextMajor`/`.exact(...)`
895            // handling), so this can't fire today. Re-review if the parser ever changes to
896            // emit a bare form.
897            deps_core::EcosystemId::Swift => LatentOnly,
898            // PyPI: the parser retains the pep440 comparator on every requirement (e.g. an
899            // exact pin parses to `"==1.2.3"`, never bare `"1.2.3"` — see `deps-core`'s
900            // `concrete_pin_version_strips_pep440_double_equals_comparator`), so a bare
901            // requirement never reaches this check either. Re-review if the parser ever
902            // changes to emit a bare form.
903            deps_core::EcosystemId::Pypi => LatentOnly,
904            // NuGet (#669): a bare `Version="X"` is really an unbounded minimum floor under
905            // `NuGetFormatter`'s own comparator, but `deps-core` deliberately still reports
906            // it as a pin — restore resolves a direct `PackageReference` to its floor
907            // version in practice, mirrored by `NuGetFormatter::is_requirement_up_to_date`
908            // treating the same bare floor as a pin for outdated-checking. See
909            // `deps-core`'s `bare_requirement_policy` doc for the full rationale, including
910            // why the alternative (an always-range policy) was tried and reverted.
911            deps_core::EcosystemId::NuGet => DeliberateApproximation,
912        }
913    }

1038            }
1039

1041                self.status

1045                self
1046            }

1064        // above — a prerelease-only flag is a ranking preference for "latest", not a hard
1065        // removal from existence. `is_prerelease()` is overridden directly rather than
1066        // relying on a hyphenated version string, so this fixture is unambiguous regardless
1067        // of which ecosystem-specific parser (if any) `select_latest_matching` re-parses
1068        // `version_string()` with.

1074            fn version_string(&self) -> &deps_core::ConcreteVersion {
1075                &self.version
1076            }

1079                true

1081

1083                self
1084            }
1085        }

1088            vec![
1089                Box::new(PrereleaseOnlyVersion {
1090                    version: "2.0.0-beta2".into(),
1091                }),
1092                Box::new(PrereleaseOnlyVersion {
1093                    version: "2.0.0-beta1".into(),
1094                }),

1099        let cache = Arc::new(HttpCache::new());
1100        register_ecosystems(&registry, Arc::clone(&cache), &test_runtime());
1101
1102        let req = VersionReq::new("*");
