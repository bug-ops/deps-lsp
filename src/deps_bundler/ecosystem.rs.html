17/// Bundler ecosystem implementation.
18///
19/// Provides LSP functionality for Gemfile files, including:
20/// - Dependency parsing with position tracking
21/// - Version information from rubygems.org
22/// - Inlay hints for latest versions
23/// - Hover tooltips with gem metadata
24/// - Code actions for version updates
25/// - Diagnostics for unknown/yanked gems
26pub struct BundlerEcosystem {
27    registry: Arc<RubyGemsRegistry>,
28    formatter: BundlerFormatter,
29}
30
31impl BundlerEcosystem {
32    /// Creates a new Bundler ecosystem with the given HTTP cache.
33    pub fn new(cache: Arc<deps_core::HttpCache>) -> Self {
34        Self {
35            registry: Arc::new(RubyGemsRegistry::new(cache)),
36            formatter: BundlerFormatter,
37        }
38    }
39
40    async fn complete_package_names(&self, prefix: &str, range: Range) -> Vec<CompletionItem> {
41        deps_core::completion::complete_package_names_generic(
42            self.registry.as_ref(),
43            prefix,
44            20,
45            range,
46        )
47        .await
48    }
49
50    async fn complete_versions(
51        &self,
52        package_name: &deps_core::PackageName,
53        prefix: &str,
54        freshness: deps_core::FreshnessSettings,
55    ) -> Vec<CompletionItem> {
56        deps_core::completion::complete_versions_generic(
57            self.registry.as_ref(),
58            package_name,
59            prefix,
60            &['~', '>', '<', '=', '!'],
61            freshness,
62        )
63        .await
64    }
65}

69impl Ecosystem for BundlerEcosystem {
70    fn ecosystem_id(&self) -> deps_core::EcosystemId {
71        deps_core::EcosystemId::Bundler
72    }
73
74    fn display_name(&self) -> &'static str {
75        "Bundler (Ruby)"
76    }
77
78    fn manifest_filenames(&self) -> &[&'static str] {
79        &["Gemfile"]
80    }
81
82    fn lockfile_filenames(&self) -> &[&'static str] {
83        &["Gemfile.lock"]
84    }
85
86    fn parse_manifest<'a>(
87        &'a self,
88        content: &'a str,
89        uri: &'a Uri,
90    ) -> deps_core::ecosystem::BoxFuture<'a, Result<Box<dyn ParseResultTrait>>> {
91        Box::pin(async move {
92            let result = crate::parser::parse_gemfile(content, uri)?;
93            Ok(Box::new(result) as Box<dyn ParseResultTrait>)
94        })
95    }
96
97    fn registry(&self) -> Arc<dyn Registry> {
98        self.registry.clone() as Arc<dyn Registry>
99    }
100
101    fn lockfile_provider(&self) -> Option<Arc<dyn deps_core::lockfile::LockFileProvider>> {
102        Some(Arc::new(crate::lockfile::GemfileLockParser))
103    }
104
105    fn formatter(&self) -> &dyn EcosystemFormatter {
106        &self.formatter
107    }
108
109    fn complete_package_name<'a>(
110        &'a self,
111        _request: deps_core::completion::CompletionRequest<'a>,
112        prefix: String,
113        range: Range,
114    ) -> deps_core::ecosystem::BoxFuture<'a, Completions> {
115        Box::pin(async move { self.complete_package_names(&prefix, range).await.into() })
116    }
117
118    fn complete_version<'a>(
119        &'a self,
120        request: deps_core::completion::CompletionRequest<'a>,
121        package_name: deps_core::PackageName,
122        prefix: String,
123    ) -> deps_core::ecosystem::BoxFuture<'a, Completions> {
124        Box::pin(async move {
125            self.complete_versions(&package_name, &prefix, request.freshness)
126                .await
127                .into()
128        })
129    }
130
131    fn completion_insert_text(&self, metadata: &dyn deps_core::Metadata) -> Option<String> {
132        let name = metadata.name();
133        let latest = metadata.latest_version().as_str();
134        Some(format!("gem \"{name}\", \"~> {latest}\""))
135    }
136
137    fn as_any(&self) -> &dyn Any {
138        self
139    }
140}

183        let cache = Arc::new(deps_core::HttpCache::new());
184        let ecosystem = BundlerEcosystem::new(cache);

187

189        let position = Position::new(1, 7); // cursor after "ra" in "rails"

191        let context = deps_core::completion::detect_completion_context(

194            content,
195        );
196
197        match context {
198            deps_core::completion::CompletionContext::PackageName { prefix, range } => {
199                assert_eq!(prefix, "ra");
