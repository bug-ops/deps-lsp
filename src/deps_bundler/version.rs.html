1//! Version comparison utilities for Ruby gems.
2//!
3//! Provides version comparison and requirement matching for Bundler ecosystem,
4//! ported directly from RubyGems' own `Gem::Version#<=>`,
5//! `Gem::Version#canonical_segments`, `#release`, and `#bump` so that
6//! ordering and `~>` requirement matching match `Gem::Version`/
7//! `Gem::Requirement` exactly, including prerelease-vs-prerelease
8//! tie-breaking and pessimistic-operator ceiling rules (#323, #327).
9//!
10//! `canonical_segments` itself is a literal string-level port of RubyGems' two
11//! `canonical_segments` regexes (see `strip_trailing_padding`/`strip_padding_before_tag`)
12//! rather than a token-level re-derivation of the same reduction, since a token-level
13//! re-derivation had twice previously diverged from Ruby's actual algorithm on narrow
14//! shapes it didn't happen to consider (#331).
15

21/// A single alternating digit/letter run extracted from a version string.
22///
23/// Mirrors the segments produced by `Gem::Version#segments`
24/// (`version.scan(/[0-9]+|[a-z]+/i)`): a numeric run is kept as its raw digit
25/// string rather than parsed into a fixed-width integer, so an arbitrarily
26/// long digit run (e.g. 20+ digits) is never silently dropped on overflow
27/// (#327 M8).
28#[derive(Clone, Debug)]
29enum Token {
30    /// A run of ASCII digits, compared as an arbitrary-precision
31    /// non-negative integer via [`cmp_digits`].
32    Numeric(String),
33    /// A run of ASCII letters, compared case-sensitively like Ruby's default
34    /// string `<=>`.
35    Alpha(String),
36}
37
38impl PartialEq for Token {
39    fn eq(&self, other: &Self) -> bool {
40        match (self, other) {
41            (Self::Numeric(a), Self::Numeric(b)) => cmp_digits(a, b) == Ordering::Equal,
42            (Self::Alpha(a), Self::Alpha(b)) => a == b,
43            _ => false,
44        }
45    }
46}

50/// Whether every character of a digit run is `0`.
51fn is_zero_digits(s: &str) -> bool {
52    s.bytes().all(|b| b == b'0')
53}
54
55/// Compares two non-negative digit strings as arbitrary-precision integers.
56///
57/// Leading zeros are stripped first, so the comparison never overflows a
58/// fixed-width type regardless of how many digits either run has (#327 M8).
59fn cmp_digits(a: &str, b: &str) -> Ordering {
60    let a = a.trim_start_matches('0');
61    let b = b.trim_start_matches('0');
62    a.len().cmp(&b.len()).then_with(|| a.cmp(b))
63}

65/// Tokenizes `version` into alternating digit/letter runs, skipping
66/// separators (`.`, `+`, ...).
67///
68/// A literal `-` is first rewritten to `.pre.`, mirroring
69/// `Gem::Version#initialize`'s `@version.gsub!("-", ".pre.")` — RubyGems
70/// treats a hyphenated suffix as an implicit `pre` segment, so `1.0.0-1`
71/// tokenizes identically to `1.0.0.pre.1` and sorts as a prerelease of
72/// `1.0.0` rather than above it (#327 M7).
73///
74/// RubyGems' own prerelease tags can be dot-separated (`3.7.0.pre1`) or
75/// glued directly onto the preceding numeric segment (`0.2.19b1`) —
76/// scanning the whole string instead of splitting on `.` first handles both
77/// shapes uniformly (#323).
78fn tokenize(version: &str) -> Vec<Token> {
79    let normalized = version.replace('-', ".pre.");
80    let bytes = normalized.as_bytes();
81    let mut tokens = Vec::new();
82    let mut i = 0;
83    while i < bytes.len() {
84        let start = i;
85        if bytes[i].is_ascii_digit() {
86            while i < bytes.len() && bytes[i].is_ascii_digit() {
87                i += 1;
88            }
89            tokens.push(Token::Numeric(normalized[start..i].to_string()));
90        } else if bytes[i].is_ascii_alphabetic() {
91            while i < bytes.len() && bytes[i].is_ascii_alphabetic() {
92                i += 1;
93            }
94            tokens.push(Token::Alpha(normalized[start..i].to_string()));
95        } else {
96            i += 1;
97        }
98    }
99    tokens
100}

102/// Strips a trailing run of `.`/`0` bytes that is itself preceded by a letter or a literal
103/// `.`, mirroring RubyGems' `Gem::Version#canonical_segments` regex
104/// `/(?<=[a-zA-Z.])[.0]+\z/` — e.g. `"1.0.0"` -> `"1."` (dropping the redundant trailing
105/// zero segments; the leftover `.` is inert, `tokenize` ignores it).
106///
107/// Rust's `regex` crate has no lookbehind, so this walks byte positions directly instead of
108/// porting the pattern as a literal `Regex`: it finds the smallest index `i` such that the
109/// byte before `i` is a letter or `.` and every byte from `i` onward is `.`/`0`, matching
110/// Ruby's leftmost-match semantics for an anchored-at-end pattern.
111fn strip_trailing_padding(version: &str) -> &str {
112    let bytes = version.as_bytes();
113    for i in 1..bytes.len() {
114        let prev = bytes[i - 1];
115        if (prev.is_ascii_alphabetic() || prev == b'.')
116            && bytes[i..].iter().all(|&b| b == b'.' || b == b'0')
117        {
118            return &version[..i];
119        }
120    }
121    version
122}

124/// Strips a run of `.`/`0` bytes anchored at the string start or right after a `.`, and
125/// immediately followed by a letter, mirroring RubyGems' `Gem::Version#canonical_segments`
126/// regex `/(?<=\.|\A)[0.]+(?=[a-zA-Z])/` — applied only to prerelease versions (those
127/// containing a letter), and only the leftmost such run (Ruby uses `sub!`, not `gsub!`, so a
128/// second zero-padding run elsewhere in the string is left untouched). E.g.
129/// `"1.pre.0.beta1"` -> `"1.pre.beta1"`, so a padding zero segment right before a prerelease
130/// tag does not become its own token (#331).
131fn strip_padding_before_tag(version: &str) -> String {
132    let bytes = version.as_bytes();
133    for i in 0..bytes.len() {
134        if i > 0 && bytes[i - 1] != b'.' {
135            continue;
136        }
137        if bytes[i] != b'.' && bytes[i] != b'0' {
138            continue;
139        }
140        let end = i + bytes[i..]
141            .iter()
142            .take_while(|b| **b == b'.' || **b == b'0')
143            .count();
144        if bytes.get(end).is_some_and(u8::is_ascii_alphabetic) {
145            return format!("{}{}", &version[..i], &version[end..]);
146        }
147    }
148    version.to_string()
149}

151/// Canonicalizes `version` into its string form for comparison, following RubyGems'
152/// `Gem::Version#canonical_segments` two-regex algorithm (see
153/// `strip_trailing_padding`/`strip_padding_before_tag`): the string is first trimmed and a
154/// hyphen expands to `.pre.` (mirroring `Gem::Version#initialize`'s `strip` then
155/// `gsub!("-", ".pre.")`), then redundant `.`/`0` padding is stripped from the end and, for
156/// prerelease versions, from immediately before a tag. This makes `"1.2.3-0.beta1"` and
157/// `"1.2.3-beta1"` canonicalize identically, matching Ruby's `canonical_segments` (#331)
158/// rather than leaving the padding `0` as its own token the way tokenizing the raw string
159/// would.
160fn canonicalize(version: &str) -> String {
161    let expanded = version.trim().replace('-', ".pre.");
162    let trimmed = strip_trailing_padding(&expanded).to_string();
163    if expanded.bytes().any(|b| b.is_ascii_alphabetic()) {
164        strip_padding_before_tag(&trimmed)
165    } else {
166        trimmed
167    }
168}

170/// Ports `Gem::Version#canonical_segments`, via the literal string-level port
171/// `canonicalize` composed with `tokenize`.
172fn canonical_segments(version: &str) -> Vec<Token> {
173    tokenize(&canonicalize(version))
174}

176/// Compares two version strings, prerelease-aware.
177///
178/// Ports `Gem::Version#<=>` over the canonical segment lists produced by
179/// `canonical_segments`: shared positions compare numerically or
180/// alphabetically depending on token kind (an alpha segment always sorts
181/// below a numeric one at the same position), and once one side runs out of
182/// segments, the other side's first decisive trailing segment settles the
183/// result — an extra alpha segment means that side is a prerelease of the
184/// other (sorts lower), an extra nonzero numeric segment means it is more
185/// precise (sorts higher), and trailing zero segments are equivalent to not
186/// being there at all.
187///
188/// # Examples
189///
190/// ```
191/// use deps_bundler::version::compare_versions;
192/// use std::cmp::Ordering;
193///
194/// assert_eq!(compare_versions("1.0.1", "1.0.0"), Ordering::Greater);
195/// assert_eq!(compare_versions("1.0.0", "1.0.0"), Ordering::Equal);
196///
197/// // A dot-notation prerelease tag sorts below its own base release.
198/// assert_eq!(compare_versions("1.0.0", "1.0.0.pre1"), Ordering::Greater);
199///
200/// // RubyGems rewrites a hyphenated suffix as an implicit prerelease tag,
201/// // so it sorts below the release too, not above it.
202/// assert_eq!(compare_versions("1.0.0", "1.0.0-1"), Ordering::Greater);
203/// ```
204pub fn compare_versions(a: &str, b: &str) -> Ordering {
205    let lhs = canonical_segments(a);
206    let rhs = canonical_segments(b);
207
208    if lhs == rhs {
209        return Ordering::Equal;
210    }
211
212    let limit = lhs.len().min(rhs.len());
213    for i in 0..limit {
214        let ordering = match (&lhs[i], &rhs[i]) {
215            (Token::Numeric(x), Token::Numeric(y)) => cmp_digits(x, y),
216            (Token::Alpha(x), Token::Alpha(y)) => x.cmp(y),
217            (Token::Alpha(_), Token::Numeric(_)) => Ordering::Less,
218            (Token::Numeric(_), Token::Alpha(_)) => Ordering::Greater,
219        };
220        if ordering != Ordering::Equal {
221            return ordering;
222        }
223    }
224
225    if lhs.len() <= rhs.len() {
226        for token in &rhs[limit..] {
227            match token {
228                Token::Alpha(_) => return Ordering::Greater,
229                Token::Numeric(n) if !is_zero_digits(n) => return Ordering::Less,
230                _ => {}
231            }
232        }
233    } else {
234        for token in &lhs[limit..] {
235            match token {
236                Token::Alpha(_) => return Ordering::Less,
237                Token::Numeric(n) if !is_zero_digits(n) => return Ordering::Greater,
238                _ => {}
239            }
240        }
241    }
242    Ordering::Equal
243}

245/// Checks if a version matches the given requirement.
246pub fn version_matches_requirement(version: &str, requirement: &str) -> bool {
247    let req = requirement.trim();
248
249    if req == "*" {
250        return true;
251    }
252
253    // Pessimistic operator (~>)
254    if req.starts_with("~>") {
255        let req_ver = req.trim_start_matches("~>").trim();
256        return matches_pessimistic(version, req_ver);
257    }
258
259    // Greater than or equal
260    if req.starts_with(">=") {
261        let req_ver = req.trim_start_matches(">=").trim();
262        return compare_versions(version, req_ver) != Ordering::Less;
263    }
264
265    // Greater than
266    if req.starts_with('>') && !req.starts_with(">=") {
267        let req_ver = req.trim_start_matches('>').trim();
268        return compare_versions(version, req_ver) == Ordering::Greater;
269    }
270
271    // Less than or equal
272    if req.starts_with("<=") {
273        let req_ver = req.trim_start_matches("<=").trim();
274        return compare_versions(version, req_ver) != Ordering::Greater;
275    }
276
277    // Less than
278    if req.starts_with('<') && !req.starts_with("<=") {
279        let req_ver = req.trim_start_matches('<').trim();
280        return compare_versions(version, req_ver) == Ordering::Less;
281    }
282
283    // Not equal
284    if req.starts_with("!=") {
285        let req_ver = req.trim_start_matches("!=").trim();
286        return compare_versions(version, req_ver) != Ordering::Equal;
287    }
288
289    // Exact match. `compare_versions` tolerates and skips bytes outside its token alphabet
290    // (digits/letters), so a malformed operand like `= 1.0.0!!!` must be rejected explicitly
291    // via `is_valid_rubygems_version` first — otherwise it would silently compare as if the
292    // garbage suffix were absent and match `1.0.0`, where RubyGems itself raises
293    // `BadRequirementError` (#345 M1: this branch used to fail closed for free under raw
294    // string equality, which the switch to canonical comparison no longer does on its own).
295    if let Some(req_ver) = req.strip_prefix('=') {
296        let req_ver = req_ver.trim();
297        return is_valid_rubygems_version(req_ver)
298            && compare_versions(version, req_ver) == Ordering::Equal;
299    }
300
301    // Default (no operator): RubyGems compiles a bare version requirement to `=`
302    // (`Gem::Requirement.create("1.6.0")` -> `["=", ...]`), so it gets the same canonical
303    // equality and the same malformed-operand rejection as the explicit `=` branch above,
304    // not a raw string/prefix match (#345).
305    is_valid_rubygems_version(req) && compare_versions(version, req) == Ordering::Equal
306}

308/// Joins a token slice back into a dot-separated version string.
309fn join_tokens(tokens: &[Token]) -> String {
310    tokens
311        .iter()
312        .map(|t| match t {
313            Token::Numeric(s) | Token::Alpha(s) => s.as_str(),
314        })
315        .collect::<Vec<_>>()
316        .join(".")
317}

319/// Increments a non-negative decimal digit string by one.
320///
321/// Works on the raw digit string rather than a fixed-width integer, so an
322/// arbitrarily long digit run increments correctly instead of overflowing
323/// (#327 C3 — the same silent-drop failure class M8 removed from the
324/// tokenizer, previously reintroduced here via `parse::<u64>()`).
325fn increment_digits(digits: &str) -> String {
326    let mut bytes = digits.as_bytes().to_vec();
327    let mut i = bytes.len();
328    loop {
329        if i == 0 {
330            bytes.insert(0, b'1');
331            break;
332        }
333        i -= 1;
334        if bytes[i] == b'9' {
335            bytes[i] = b'0';
336        } else {
337            bytes[i] += 1;
338            break;
339        }
340    }
341    String::from_utf8(bytes).expect("incrementing ASCII digits stays valid UTF-8")
342}

344/// Ports `Gem::Version#release`: the numeric prefix before the first
345/// prerelease tag, as a dot-joined string. A version with no prerelease tag
346/// is returned unchanged (re-joined from its own tokens).
347fn release_string(version: &str) -> String {
348    let tokens = tokenize(version);
349    let end = tokens
350        .iter()
351        .position(|t| matches!(t, Token::Alpha(_)))
352        .unwrap_or(tokens.len());
353    join_tokens(&tokens[..end])
354}

356/// Ports `Gem::Version#bump`: the exclusive upper bound for a `~>`
357/// requirement, as a dot-joined string.
358///
359/// Truncates at the first prerelease tag (same as [`release_string`]), drops
360/// one more trailing segment if more than one remains, then increments the
361/// new last segment — e.g. `bump_string("3.7.0")` is `"3.8"` and
362/// `bump_string("3.7")` is `"4"`. Returns `None` only if the requirement has
363/// no numeric segment to increment at all (a malformed requirement).
364fn bump_string(requirement: &str) -> Option<String> {
365    let tokens = tokenize(requirement);
366    let end = tokens
367        .iter()
368        .position(|t| matches!(t, Token::Alpha(_)))
369        .unwrap_or(tokens.len());
370    let mut segments = tokens[..end].to_vec();
371    if segments.len() > 1 {
372        segments.pop();
373    }
374    match segments.last_mut()? {
375        Token::Numeric(s) => *s = increment_digits(s),
376        Token::Alpha(_) => unreachable!("segments before the first alpha token contain no alpha"),
377    }
378    Some(join_tokens(&segments))
379}

381/// Matches RubyGems' `Gem::Version::VERSION_PATTERN`
382/// (`[0-9]+(?>\.[0-9a-zA-Z]+)*(-[0-9A-Za-z-]+(\.[0-9A-Za-z-]+)*)?`, confirmed against a live
383/// RubyGems install rather than the packaged source alone): a leading digit run, followed by
384/// zero or more dot-separated alphanumeric segments, with an optional hyphenated suffix that
385/// may itself have any number of further dot-separated segments.
386static RUBYGEMS_VERSION_PATTERN: LazyLock<Regex> = LazyLock::new(|| {
387    Regex::new(r"^[0-9]+(?:\.[0-9a-zA-Z]+)*(?:-[0-9A-Za-z-]+(?:\.[0-9A-Za-z-]+)*)?$")
388        .expect("Invalid regex")
389});

391/// Checks whether `version` (the operand of a comparison operator in a Bundler requirement)
392/// is a syntactically valid RubyGems version, per `Gem::Version::VERSION_PATTERN`.
393///
394/// Not `Gem::Version.correct?`, which additionally accepts an empty or whitespace-only
395/// operand that `Gem::Requirement` itself rejects for every operator this gates. A malformed
396/// operand (e.g. `~> abc`, `< abc`, a bare `~>` with nothing after it) would
397/// otherwise make the corresponding comparison in [`version_matches_requirement`] evaluate
398/// `false` against every candidate, producing a misleading "no version satisfies
399/// requirement" diagnostic instead of flagging the requirement itself as invalid (#332).
400pub fn is_valid_rubygems_version(version: &str) -> bool {
401    RUBYGEMS_VERSION_PATTERN.is_match(version.trim())
402}

404/// Checks if a version matches a pessimistic requirement (`~>`).
405///
406/// Ports RubyGems' own pessimistic-operator check
407/// (`Gem::Requirement::OPS["~>"]`): `version >= requirement` and
408/// `version.release < requirement.bump`, via [`release_string`] and
409/// [`bump_string`] — e.g. `~> 3.7.0` is `>= 3.7.0, < 3.8`, and `~> 3.7` is
410/// `>= 3.7, < 4`.
411///
412/// Comparing against the version's *release* (not the raw version) means a
413/// prerelease of the ceiling correctly falls outside the range — e.g.
414/// `3.8.0.pre1` does not satisfy `~> 3.7.0`, since its release `3.8.0` is
415/// not less than the ceiling `3.8` (#327 C1). Building the ceiling via
416/// [`bump_string`] instead of bumping a dot-separated substring means a
417/// requirement with its own prerelease tag bumps correctly instead of
418/// ignoring the tag or silently matching nothing (#327 C2).
419fn matches_pessimistic(version: &str, requirement: &str) -> bool {
420    if compare_versions(version, requirement) == Ordering::Less {
421        return false;
422    }
423
424    let Some(ceiling) = bump_string(requirement) else {
425        return false;
426    };
427
428    compare_versions(&release_string(version), &ceiling) == Ordering::Less
429}

730        // Regression test for #345: `Gem::Requirement#satisfied_by?` compares `=`/`!=`
731        // operands via canonical `Gem::Version` equality, not raw string equality, so a
732        // trailing-zero segment must not defeat the match.
