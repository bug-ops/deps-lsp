13/// Whether every character of `name` is in RubyGems' gem-name charset
14/// (`Gem::Specification::VALID_NAME_PATTERN`): ASCII letters, digits, `.`, `-`, `_`.
15fn is_rubygems_name_charset(name: &str) -> bool {
16    name.chars()
17        .all(|c| c.is_ascii_alphanumeric() || matches!(c, '.' | '-' | '_'))
18}

20/// Extracts the version operand of `requirement` for the operators whose
21/// [`version_matches_requirement`] branch evaluates `false` against every candidate on a
22/// malformed operand ("fails closed") rather than `true` against every candidate ("fails
23/// open"): `~>`, `<`, `<=`, `=`, and the bare/no-operator pin. Mirrors
24/// `version_matches_requirement`'s own operator dispatch order (`<=` checked before `<`) so
25/// a requirement is classified identically in both places.
26///
27/// Returns `None` for `>`, `>=`, `!=`, and `*` — confirmed live against RubyGems (#332
28/// critique) that a malformed operand for these instead makes every candidate match except,
29/// for `!=` since #345's switch to canonical comparison, one candidate that happens to
30/// canonicalize identically to the operand's non-garbage prefix (e.g. `!= 1.0.0!!!` no longer
31/// matches `1.0.0` itself). Either way this never triggers the "no version satisfies
32/// requirement" false positive this gates — a single excluded candidate among the rest still
33/// leaves the requirement satisfiable — so none of these four need validation here.
34fn fail_closed_operand(requirement: &str) -> Option<&str> {
35    let req = requirement.trim();
36    if req == "*" || req.starts_with(">=") || req.starts_with('>') || req.starts_with("!=") {
37        return None;
38    }
39    if let Some(rest) = req.strip_prefix("~>") {
40        return Some(rest.trim());
41    }
42    if let Some(rest) = req.strip_prefix("<=") {
43        return Some(rest.trim());
44    }
45    if let Some(rest) = req.strip_prefix('<') {
46        return Some(rest.trim());
47    }
48    if let Some(rest) = req.strip_prefix('=') {
49        return Some(rest.trim());
50    }
51    Some(req)
52}

54/// Rubygems requirement matcher, compiled once per dependency by
55/// [`BundlerFormatter::compile_requirement`]. `version_matches_requirement` is a hand-rolled
56/// comparator with no external parser to fail on, so this always decides (`Some`).
57struct RubygemsMatcher(String);

59impl RequirementMatcher for RubygemsMatcher {
60    fn matches(&self, version: &ConcreteVersion) -> Option<bool> {
61        let version = version.as_str();
62        Some(version_matches_requirement(version, &self.0))
63    }

66/// A Bundler requirement written as `= X` or bare `X` (no operator) pins a single exact
67/// version. Returns the pinned version string, or `None` when `requirement` is a wildcard,
68/// empty, or a range/comparison operator (`~>`, `>=`, `>`, `<=`, `<`, `!=`) rather than a pin.
69fn exact_pin_version(requirement: &str) -> Option<&str> {
70    let req = requirement.trim();
71    if req.is_empty() || req == "*" {
72        return None;
73    }
74    if let Some(rest) = req.strip_prefix('=') {
75        let rest = rest.trim();
76        return (!rest.is_empty()).then_some(rest);
77    }
78    let is_range_op = ["~>", ">=", ">", "<=", "<", "!="]
79        .iter()
80        .any(|op| req.starts_with(op));
81    (!is_range_op).then_some(req)
82}

84/// RubyGems' `versions.json` endpoint omits yanked entries from the result entirely rather
85/// than flagging them — verified live: `rest-client` lists `1.6.9`/`1.6.14` but not the
86/// yanked `1.6.10`-`1.6.13`. Yanking is not confined to interior gaps, though: live
87/// verification against real yank records found gems (`puppetlabs-syntax`,
88/// `keeper_secrets_manager`, `tingee_ruby_sdk`) with every version *below* the current
89/// minimum yanked — e.g. `puppetlabs-syntax` yanked `6.0.0` and `7.2.0`, leaving `7.2.1` as
90/// the sole published version, so a pin on `7.2.0` sorts *below* `available`'s only entry.
91/// The top of the range is the only usable boundary, and only an accepted approximation of
92/// one: `available`'s observed maximum is ordinarily the newest real release, but if the
93/// newest release itself was later yanked with nothing published after it, a pin on it would
94/// sort *above* the new (lower) maximum — `available` alone cannot tell that case apart from a
95/// pin that was simply mistyped too high, and flagging the latter is the whole point of #252.
96/// This risk is accepted as an empirically rare residual (0/30 in the live sample that found
97/// the below-minimum counterexamples below) rather than modeled, since there is no signal in
98/// `available` to distinguish the two. So an exact pin (see [`exact_pin_version`]) is treated
99/// as possibly-yanked — and the unsatisfiable check suppressed — whenever it does not exceed
100/// `available`'s observed maximum, regardless of how far below the minimum it sorts; only a
101/// pin above the maximum is still flagged as unsatisfiable. This deliberately over-suppresses
102/// relative to the narrower "interior gap" theory this replaced (a below-minimum pin that
103/// really was never published now also goes unflagged), but under-suppression was the actual
104/// bug (#252) and the direction this errs in is the same one the whole mechanism exists to
105/// guarantee: never a false "unsatisfiable". Wider requirement shapes (`~>`, `>=`, ranges) are
106/// left alone: they typically span many versions, so a match existing only among yanked ones
107/// is a rare edge case rather than the common one exact pins hit.
108///
109/// Note on the boundary case (pin equal to the maximum): this gate compares via
110/// [`compare_versions`], and since #345 the actual scan in [`RubygemsMatcher`]
111/// (`version_matches_requirement`) does too for both the explicit `=` and bare-pin forms —
112/// both now agree that a pin of `"1.6"` canonically equals a published `"1.6.0"`. (Before
113/// #345, the string/prefix matcher used there disagreed with this gate at that boundary; the
114/// two no longer can.) A prerelease-tagged pin like `"2.0.0.rc1"` against a published
115/// `"2.0.0"` does not reach this path either: `compare_versions` correctly orders it below
116/// the stable release instead of tying, since #323's fix.
117fn exact_pin_could_be_yanked(requirement: &str, available: &[ConcreteVersion]) -> bool {
118    let Some(pin) = exact_pin_version(requirement) else {
119        return false;
120    };
121    let Some(max) = available
122        .iter()
123        .max_by(|a, b| compare_versions(a.as_str(), b.as_str()))
124    else {
125        return false;
126    };
127    !compare_versions(pin, max.as_str()).is_gt()
128}

133impl PackageNaming for BundlerFormatter {
134    /// Lints `name` against RubyGems' own gem-name rule (see `is_rubygems_name_charset` plus
135    /// its "must include at least one letter" check), so a structurally invalid gem name is
136    /// reported as "Invalid package name" instead of falling through to a registry lookup and
137    /// rendering the generic "Registry lookup failed" diagnostic (#402).
138    ///
139    /// # Errors
140    ///
141    /// Returns [`InvalidPackageName`] if `name` is empty, contains a character outside
142    /// RubyGems' `[a-zA-Z0-9.\-_]` charset, or contains no letter. The charset check runs
143    /// before the letter check (#402 critique M4) so a name that fails both — e.g. a
144    /// non-ASCII name with no ASCII letter at all — reports the charset violation rather than
145    /// the less specific "no letter" message.
146    fn validate_package_name(&self, name: &str) -> Result<(), InvalidPackageName> {
147        if name.is_empty() {
148            return Err(InvalidPackageName::new("name cannot be empty"));
149        }
150        if !is_rubygems_name_charset(name) {
151            return Err(InvalidPackageName::new(
152                "name must contain only ASCII letters, digits, '.', '-', or '_'",
153            ));
154        }
155        if !name.chars().any(|c| c.is_ascii_alphabetic()) {
156            return Err(InvalidPackageName::new(
157                "name must include at least one letter",
158            ));
159        }
160        Ok(())
161    }
162}

175impl RequirementResolution for BundlerFormatter {
176    fn version_satisfies_requirement(&self, version: &ConcreteVersion, requirement: &str) -> bool {
177        let version = version.as_str();
178        version_matches_requirement(version, requirement)
179    }
180
181    /// Compiles `requirement` into a `RubygemsMatcher` using the same
182    /// `version_matches_requirement` comparator as `version_satisfies_requirement` — Bundler
183    /// requirements have no separate "loose" vs. "precise" form to distinguish, and
184    /// `version_matches_requirement` never fails to parse, so this always decides (`Some`),
185    /// except when `fail_closed_operand` identifies `requirement` as one of the operator
186    /// shapes (`~>`, `<`, `<=`, `=`, bare) whose malformed-operand behavior fails closed
187    /// (matches no candidate) rather than open: with no up-front validation, that would
188    /// produce a misleading "no version satisfies requirement" diagnostic instead of
189    /// flagging the requirement itself as invalid, so this returns `None` for it instead,
190    /// matching the `is_valid_range`/`is_valid_requirement` precedent in Maven/Gradle/NuGet's
191    /// `compile_requirement` (#332). The "could this requirement be satisfied by a version
192    /// RubyGems hid" ambiguity is handled separately in
193    /// [`Self::requirement_is_undecidable_given_available`], which sees `available` and can
194    /// therefore decide it precisely instead of this method having to guess from
195    /// `requirement` alone.
196    fn compile_requirement(&self, requirement: &VersionReq) -> Option<Box<dyn RequirementMatcher>> {
197        compile_requirement_unless(
198            requirement.as_str(),
199            |r| fail_closed_operand(r).is_some_and(|operand| !is_valid_rubygems_version(operand)),
200            RubygemsMatcher,
201        )
202    }
203
204    /// See `exact_pin_could_be_yanked` for the RubyGems-specific rationale and heuristic.
205    fn requirement_is_undecidable_given_available(
206        &self,
207        requirement: &VersionReq,
208        available: &[ConcreteVersion],
209    ) -> bool {
210        exact_pin_could_be_yanked(requirement.as_str(), available)
211    }
212}
