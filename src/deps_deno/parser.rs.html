1//! `deno.json` / `deno.jsonc` parser using `jsonc-parser`'s AST (D6).
2//!
3//! Plain JSON is a strict subset of JSONC, so both `deno.json` and `deno.jsonc` share this
4//! single code path. Reading positions straight off the AST (rather than a text search) is
5//! required here specifically because Deno import maps routinely point several aliases at the
6//! identical specifier value — a find-first-occurrence text search would hand every such alias
7//! the same source position, corrupting hover/inlay-hint/code-action targeting. `deps-npm` and
8//! `deps-composer` since #613 also read positions off this same `jsonc-parser` AST (via
9//! `deps_core::json_ast`), for the equally AST-shaped duplicate-key and nested-value bugs their
10//! own manifests can hit — this module's `find_last_prop` reuse of that shared helper is the
11//! overlap between the two designs; the value-oriented parsing below (specifier scheme
12//! detection, escape guards) is Deno-specific and has no equivalent there.
13

44/// Parses a `deno.json`/`deno.jsonc` file and extracts all `imports` entries with
45/// positions.
46///
47/// # Errors
48///
49/// Returns an error if the content is not valid JSON/JSONC (malformed syntax, unclosed
50/// block comment, stray brace), or if the parsed AST nests deeper than
51/// [`deps_core::MAX_JSON_NESTING_DEPTH`] — the file is then not recognized as a Deno
52/// manifest and degrades gracefully (spec §6), matching every other ecosystem's
53/// parse-failure behavior.
54///
55/// # Examples
56///
57/// ```no_run
58/// use deps_deno::parser::parse_deno_json;
59/// use tower_lsp_server::ls_types::Uri;
60///
61/// let json = r#"{
62///   "imports": {
63///     "@std/fs": "jsr:@std/fs@^1.0"
64///   }
65/// }"#;
66/// let uri = Uri::from_file_path("/project/deno.json").unwrap();
67///
68/// let result = parse_deno_json(json, &uri).unwrap();
69/// assert_eq!(result.dependencies.len(), 1);
70/// assert_eq!(result.dependencies[0].name, "jsr:@std/fs");
71/// ```
72pub fn parse_deno_json(content: &str, uri: &Uri) -> Result<DenoParseResult> {
73    let ast = parse_to_ast(
74        content,
75        &CollectOptions::default(),
76        &ParseOptions::default(),
77    )
78    .map_err(|e| DepsError::ParseError {
79        file_type: "deno.json".into(),
80        source: e.to_string().into(),
81    })?;
82
83    if let Some(value) = &ast.value
84        && let Err(depth) = check_ast_nesting_depth(value, MAX_JSON_NESTING_DEPTH)
85    {
86        return Err(DepsError::ParseError {
87            file_type: "deno.json".into(),
88            source: json_depth_error_message(depth).into(),
89        });
90    }
91
92    let line_table = LineOffsetTable::new(content);
93    let mut dependencies = Vec::new();
94
95    if let Some(Value::Object(root)) = ast.value
96        && let Some(imports_prop) = find_last_prop(&root, "imports")
97        && let Value::Object(imports) = &imports_prop.value
98    {
99        collect_imports(imports, content, &line_table, &mut dependencies);
100    }
101
102    Ok(DenoParseResult {
103        dependencies,
104        uri: uri.clone(),
105    })
106}

108/// Computes `value`'s real nesting depth (`value` itself counts as depth 1, each further
109/// `Object`/`Array` descent adds one), rejecting once it exceeds `max_depth`.
110///
111/// Unlike a raw byte scan over the source text (`deps_core::check_json_nesting_depth`, the
112/// pattern `deps-npm`/`deps-composer` use ahead of their own strict-JSON `serde_json` parse),
113/// walking the already-parsed AST is immune to JSONC comment characters (`//`, `/* */`)
114/// miscounting as structural nesting — a comment's bracket characters never become AST nodes
115/// at all, so they cannot be mistaken for real nesting the way a text scanner would (impl-critic
116/// #618 S1). `jsonc-parser`'s own hardcoded 512-deep recursion cap already bounded
117/// `parse_to_ast` itself before this ever runs, so this walk exists to align `deno.json`
118/// parsing with `deps_core::MAX_JSON_NESTING_DEPTH` — the same, tighter bound every other
119/// workspace JSON call site enforces — not because `parse_to_ast`'s result is unsafe to walk.
120/// An explicit stack (rather than recursive descent) keeps the walk itself non-recursive
121/// regardless.
122///
123/// # Errors
124///
125/// Returns `Err(depth)` with the depth reached the instant nesting exceeds `max_depth`.
126fn check_ast_nesting_depth(root: &Value<'_>, max_depth: usize) -> std::result::Result<(), usize> {
127    let mut stack = vec![(root, 1_usize)];
128    while let Some((value, depth)) = stack.pop() {
129        // Only `Object`/`Array` nodes open a nesting level; a leaf (string/number/bool/null)
130        // is never itself checked or descended into, so it can never inflate the depth of the
131        // container that holds it.
132        match value {
133            Value::Object(obj) => {
134                if depth > max_depth {
135                    return Err(depth);
136                }
137                stack.extend(obj.properties.iter().map(|prop| (&prop.value, depth + 1)));
138            }
139            Value::Array(arr) => {
140                if depth > max_depth {
141                    return Err(depth);
142                }
143                stack.extend(arr.elements.iter().map(|elem| (elem, depth + 1)));
144            }
145            Value::StringLit(_)
146            | Value::NumberLit(_)
147            | Value::BooleanLit(_)
148            | Value::NullKeyword(_) => {}
149        }
150    }
151    Ok(())
152}

154/// Builds a [`DenoDependency`] for every entry in the `imports` object, applying
155/// last-alias-wins deduplication (S6) so a manifest with two entries for the same import
156/// alias produces exactly one dependency — positioned at the *last* occurrence, matching
157/// where its value actually lives once JSON parsing collapses the duplicate.
158fn collect_imports(
159    imports: &Object,
160    content: &str,
161    line_table: &LineOffsetTable,
162    out: &mut Vec<DenoDependency>,
163) {
164    let mut seen_aliases = HashSet::new();
165    let mut collected = Vec::new();
166
167    // Walk in reverse (last occurrence first) so the first alias we see for a given key
168    // is the one that wins; earlier (source-order) duplicates are then skipped.
169    for prop in imports.properties.iter().rev() {
170        let alias = prop.name.as_str();
171        if !seen_aliases.insert(alias.to_string()) {
172            continue;
173        }
174        let Value::StringLit(value_lit) = &prop.value else {
175            continue;
176        };
177        if let Some(dep) = build_dependency(value_lit, content, line_table) {
178            collected.push(dep);
179        }
180    }
181
182    // Restore source order for the surviving (deduplicated) entries.
183    collected.reverse();
184    out.extend(collected);
185}

187/// Builds a [`DenoDependency`] from one `imports` value's string literal, or `None` if the
188/// value is neither a recognized `jsr:`/`npm:` specifier nor a syntactically incomplete,
189/// still-being-typed one (D7 — e.g. `http://`, `file:`, a bare alias — silently skipped,
190/// same as an unparseable entry in any other ecosystem).
191fn build_dependency(
192    value_lit: &StringLit,
193    content: &str,
194    line_table: &LineOffsetTable,
195) -> Option<DenoDependency> {
196    let raw_value = value_lit.value.as_ref();
197
198    // S5 escape guard: `value_lit.range` covers the whole literal *including* its
199    // surrounding quotes. Byte-arithmetic from that range into offsets relative to the
200    // *unescaped* value is only sound when no unescaping happened — exactly when
201    // `value_lit.value` is `Cow::Borrowed` (a direct slice of the source). Deno specifiers
202    // never legitimately need escapes, so this is a correctness fail-safe: an escaped
203    // value only degrades to the inner-span fallback (no `version_range`) when it fully
204    // parses; a partial/in-progress escaped value (#310) is skipped entirely rather than
205    // guessing at an unsound byte range for it.
206    if let Cow::Owned(_) = &value_lit.value {
207        let parsed = parse_specifier(raw_value)?;
208        tracing::debug!(
209            "deno.json: import value contains escape sequences, skipping precise position tracking"
210        );
211        let inner_start = value_lit.range.start + 1;
212        let inner_end = value_lit.range.end.saturating_sub(1);
213        return Some(DenoDependency {
214            name: PackageName::new(parsed.name),
215            name_range: byte_range_to_lsp(content, line_table, inner_start, inner_end),
216            version_req: parsed.version_req.map(VersionReq::new),
217            version_range: None,
218            section: DenoDependencySection::Imports,
219        });
220    }
221
222    // The literal's inner (unquoted) text starts right after its opening quote.
223    let value_start = value_lit.range.start + 1;
224
225    if let Some(parsed) = parse_specifier(raw_value) {
226        let name_range = byte_range_to_lsp(
227            content,
228            line_table,
229            value_start + parsed.name_range.start,
230            value_start + parsed.name_range.end,
231        );
232        let version_range = parsed.version_range.map(|r| {
233            byte_range_to_lsp(
234                content,
235                line_table,
236                value_start + r.start,
237                value_start + r.end,
238            )
239        });
240
241        return Some(DenoDependency {
242            name: PackageName::new(parsed.name),
243            name_range,
244            version_req: parsed.version_req.map(VersionReq::new),
245            version_range,
246            section: DenoDependencySection::Imports,
247        });
248    }
249
250    // #310: a syntactically incomplete but still in-progress jsr:/npm: specifier ("jsr:",
251    // "jsr:@", "jsr:@std", "jsr:@std/", ...) — `parse_specifier` correctly rejects these
252    // (not yet a valid, routable name), but a `Dependency` must still exist here for
253    // `detect_completion_context` to have a range to fire `PackageName` completion
254    // against. This mirrors `deps-npm`'s parser, which always builds a `Dependency` from a
255    // `dependencies` object key regardless of scope completeness (`deps-npm/src/parser.rs`)
256    // — validity is deferred to `DenoFormatter::validate_package_name`'s diagnostic, not
257    // gated at parse time.
258    let partial_range = crate::specifier::partial_name_range(raw_value)?;
259    let name_range = byte_range_to_lsp(
260        content,
261        line_table,
262        value_start + partial_range.start,
263        value_start + partial_range.end,
264    );
265
266    Some(DenoDependency {
267        name: PackageName::new(&raw_value[partial_range]),
268        name_range,
269        version_req: None,
270        version_range: None,
271        section: DenoDependencySection::Imports,
272    })
273}

275/// Converts a `[start, end)` byte range in `content` to an LSP `Range`.
276fn byte_range_to_lsp(content: &str, table: &LineOffsetTable, start: usize, end: usize) -> Range {
277    Range::new(
278        table.byte_offset_to_position(content, start),
279        table.byte_offset_to_position(content, end),
280    )
281}

457    fn test_escaped_specifier_falls_back_to_inner_span_with_no_version_range() {
458        // S5 escape-guard fail-safe: a JSON `\uXXXX` escape inside the value (unescaping
459        // to the character '1' here) means `value_lit.value` is `Cow::Owned`, so
460        // byte-offset arithmetic into the raw *source* text is unsound for anything
461        // narrower than the whole literal. `name`/`version_req` still come from the
462        // already-unescaped value (jsonc-parser did that work); only position tracking
463        // degrades to the inner-span fallback with no `version_range`.
464        let json_escape = "\\u0031"; // a JSON \uXXXX escape unescaping to the character '1'
465        let raw_value = format!("jsr:@std/fs@{json_escape}.0"); // raw source text, escape included
466        let json = format!(r#"{{"imports": {{"@std/fs": "{raw_value}"}}}}"#);
467        let result = parse_deno_json(&json, &test_uri()).unwrap();
468        assert_eq!(result.dependencies.len(), 1);

472        assert_eq!(dep.version_req, Some("1.0".into()));

478        let value_start_byte = json.find(&format!("\"{raw_value}\"")).unwrap() + 1;

558        for value in ["jsr:", "jsr:@", "jsr:@std", "jsr:@std/"] {
559            let content = format!(r#"{{"imports": {{"@std/fs": "{value}"}}}}"#);
560            let result = parse_deno_json(&content, &test_uri()).unwrap();
561            assert_eq!(result.dependencies.len(), 1, "value: {value}");
562
563            let name_range = result.dependencies[0].name_range;
564            let cursor = name_range.end; // cursor right after the last typed character
565            let context = detect_completion_context(&result, cursor, &content);
566
567            match context {
568                CompletionContext::PackageName { prefix, .. } => {
569                    assert_eq!(prefix, value, "value: {value}");
570                }
571                other => panic!("value {value}: expected PackageName context, got {other:?}"),
572            }
573        }
574    }
