121    fn fallback_completion_prefix<'a>(
122        &self,
123        content: &'a str,
124        position: Position,
125    ) -> Option<&'a str> {
126        let line = deps_core::fallback_completion::line_at(content, position)?;
127        if !is_in_dependencies_section(content, position.line as usize) {
128            return None;
129        }
130        Some(extract_prefix(line, position.character))
131    }

133    fn completion_insert_text(&self, metadata: &dyn deps_core::Metadata) -> Option<String> {
134        let name = metadata.name();
135        let latest = metadata.latest_version().as_str();
136        // The key is quoted: an unquoted YAML plain scalar can't start with `@`
137        // (allowed by `is_safe_package_name` for npm/Deno-shaped names), which would
138        // otherwise emit invalid YAML instead of a dependency entry.
139        Some(format!("\"{name}\": ^{latest}"))
140    }

165/// pubspec.yaml's top-level (unindented) dependency-like keys.
166const SECTION_KEYS: &[&str] = &[
167    "dependencies:",
168    "dev_dependencies:",
169    "dependency_overrides:",
170];
171
172/// Checks if `line_number` of `content` is inside a pubspec.yaml dependency section.
173///
174/// Dart's `dependencies`, `dev_dependencies`, and `dependency_overrides` keys are
175/// top-level (unindented) YAML mappings; their entries stay part of the section until
176/// the next unindented key starts a new one. For `deps-lsp`'s raw-text fallback
177/// completion (parse-failure path).
178fn is_in_dependencies_section(content: &str, line_number: usize) -> bool {
179    let mut in_dependencies = false;
180
181    for (i, line) in content.lines().enumerate() {
182        if i > line_number {
183            break;
184        }
185
186        let trimmed = line.trim_start();
187        if trimmed.is_empty() || trimmed.starts_with('#') {
188            continue;
189        }
190
191        // Top-level (unindented) key: starts a new section, or leaves the current one.
192        if trimmed.len() == line.len() {
193            in_dependencies = SECTION_KEYS.iter().any(|key| trimmed.starts_with(key));
194        }
195    }
196
197    in_dependencies
198}

200/// Extracts the fallback-completion prefix on `line` up to `character` — a bare YAML
201/// key, with no manifest-syntax wrapper to strip.
202fn extract_prefix(line: &str, character: u32) -> &str {
203    deps_core::fallback_completion::raw_prefix(line, character)
204}
