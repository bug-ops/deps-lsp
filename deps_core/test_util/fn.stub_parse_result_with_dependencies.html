pub fn stub_parse_result_with_dependencies(count: usize) -> Box<dyn ParseResult>
Expand description

Builds a crate::ParseResult fixture with count synthetic dependencies.

Names them "dep-0", "dep-1", … — for tests exercising dependency-count-ceiling logic (#796) without needing a real ecosystem crate’s parser.

§Examples

use deps_core::ParseResult;
use deps_core::test_util::stub_parse_result_with_dependencies;

let parsed = stub_parse_result_with_dependencies(3);
assert_eq!(parsed.dependencies().len(), 3);
